function Protect-ImpactIQLogText {
    param([AllowEmptyString()][string]$Text)

    $Text = [regex]::Replace($Text, '(?i)\b(Bearer|MwcToken)\s+\S+', '$1 [REDACTED]')
    $Text = [regex]::Replace($Text, '\beyJ[\w-]+\.[\w-]+\.[\w-]+\b', '[REDACTED]')
    $Text = [regex]::Replace($Text, '(?i)(["'']?(?:access[_-]?token|refresh[_-]?token|id[_-]?token|client[_-]?secret|password|pwd|handoffCode|codeVerifier|authorization)["'']?\s*[:=]\s*)(?:"(?:\\.|[^"\\])*"|''(?:''''|[^''])*''|[^;\r\n,}]+)', '$1[REDACTED]')
    [regex]::Replace($Text, '(?i)([?&](?:_hc|code|code_verifier|access_token|token|sig)=)[^&\s"'']+', '$1[REDACTED]')
}

function Write-ImpactIQDiagnostic {
    param(
        [ValidateSet('INFO', 'WARN', 'ERROR')][string]$Level,
        [AllowEmptyString()][string]$Message
    )

    $text = Protect-ImpactIQLogText $Message
    $script:ImpactIQDiagnostics.Writer.WriteLine(('[{0:O}] [{1}] {2}' -f [DateTime]::UtcNow, $Level, $text))
}

function Get-ImpactIQDiagnosticLevel {
    param([AllowEmptyString()][string]$Text)

    if ($Text -match '(?i)\b(?:errors?|fatal|exception|failed|failure)\b|\[ERROR\]') { return 'ERROR' }
    if ($Text -match '(?i)\bwarn(?:ing)?s?\b|\[WARN\]') { return 'WARN' }
}

function Write-ImpactIQRunRecord {
    param(
        [Parameter(Mandatory)][AllowNull()][object]$Record,
        [switch]$DiscardOutput
    )

    if ($Record -is [Management.Automation.ErrorRecord]) {
        # Redirection can clone an ErrorRecord while retaining the same exception.
        if ($script:ImpactIQDiagnostics.SeenExceptions.Add($Record.Exception)) {
            $message = $Record.Exception.Message
            if ($Record.InvocationInfo.ScriptLineNumber) {
                $message += " (line $($Record.InvocationInfo.ScriptLineNumber), $($Record.FullyQualifiedErrorId))"
            }
            Write-ImpactIQDiagnostic -Level ERROR -Message $message
        }
        return
    }
    if ($Record -is [Management.Automation.WarningRecord]) {
        Write-ImpactIQDiagnostic -Level WARN -Message $Record.Message
        return
    }
    if ($Record -is [Management.Automation.InformationRecord]) {
        $data = $Record.MessageData
        $text = [string]$data
        $level = Get-ImpactIQDiagnosticLevel $text
        if (-not $level -and $data -is [Management.Automation.HostInformationMessage]) {
            if ($data.ForegroundColor -eq [ConsoleColor]::Red) { $level = 'ERROR' }
            elseif ($data.ForegroundColor -eq [ConsoleColor]::Yellow) { $level = 'WARN' }
        }
        if ($level) {
            Write-ImpactIQDiagnostic -Level $level -Message $text
        }
        elseif (-not $DiscardOutput) {
            if ($data -is [Management.Automation.HostInformationMessage]) {
                $parameters = @{ Object = $data.Message; NoNewline = $data.NoNewLine }
                if ($null -ne $data.ForegroundColor) { $parameters.ForegroundColor = $data.ForegroundColor }
                if ($null -ne $data.BackgroundColor) { $parameters.BackgroundColor = $data.BackgroundColor }
                Microsoft.PowerShell.Utility\Write-Host @parameters
            }
            else {
                Microsoft.PowerShell.Utility\Write-Information -MessageData $data -Tags $Record.Tags -InformationAction Continue
            }
        }
        return
    }
    if ($Record -is [string]) {
        $level = Get-ImpactIQDiagnosticLevel $Record
        if ($level) {
            Write-ImpactIQDiagnostic -Level $level -Message $Record
            return
        }
    }
    if (-not $DiscardOutput) { Write-Output -NoEnumerate $Record }
}

function Write-ImpactIQContinuingError {
    param(
        [Parameter(Mandatory)][Management.Automation.ErrorRecord]$Record,
        [string]$Context = 'Extraction step'
    )

    $script:ImpactIQRunFailed = $true
    Write-ImpactIQRunRecord -Record $Record
    Write-ImpactIQDiagnostic -Level WARN -Message "$Context could not complete; continuing with the remaining work."
}

function Invoke-ImpactIQLoggedRun {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)][Alias('LogPath')][string]$ImpactIQLogPath,
        [Parameter(Mandatory)][Alias('Action')][scriptblock]$ImpactIQAction,
        [switch]$KeepOpen,
        [switch]$StartSession
    )

    $impactIQPreviousErrorPreference = $ErrorActionPreference
    $impactIQPreviousWarningPreference = $WarningPreference
    $impactIQCloseWriter = $false
    $ErrorActionPreference = 'SilentlyContinue'
    try {
        if ($StartSession -or -not ($script:ImpactIQDiagnostics -and $script:ImpactIQDiagnostics.SessionOpen)) {
            if ($script:ImpactIQDiagnostics -and $script:ImpactIQDiagnostics.Writer) {
                $script:ImpactIQDiagnostics.Writer.Dispose()
            }
            $script:ImpactIQRunFailed = $false
            $script:ImpactIQLogUnavailable = $false
            $script:ImpactIQDiagnostics = @{
                Writer = $null
                LogPath = $ImpactIQLogPath
                SessionOpen = $true
                SeenExceptions = [Collections.Generic.HashSet[Exception]]::new()
            }
            foreach ($impactIQRecord in @($global:Error)) { $null = $script:ImpactIQDiagnostics.SeenExceptions.Add($impactIQRecord.Exception) }
        }
        if ($script:ImpactIQDiagnostics.LogPath -ne $ImpactIQLogPath) {
            throw 'The diagnostic log path cannot change during an extraction session.'
        }
        if (-not $script:ImpactIQDiagnostics.Writer) {
            $null = [IO.Directory]::CreateDirectory((Split-Path -Path $ImpactIQLogPath -Parent))
            $script:ImpactIQDiagnostics.Writer = [IO.StreamWriter]::new($ImpactIQLogPath, $true, [Text.UTF8Encoding]::new($false))
            $script:ImpactIQDiagnostics.Writer.AutoFlush = $true
            Write-ImpactIQDiagnostic -Level INFO -Message 'ImpactIQ run started.'
        }
        $ErrorActionPreference = 'SilentlyContinue'
        $WarningPreference = 'Continue'
        try {
            . $ImpactIQAction 2>&1 3>&1 6>&1 | ForEach-Object { Write-ImpactIQRunRecord -Record $_ }
        }
        catch {
            $script:ImpactIQRunFailed = $true
            Write-ImpactIQRunRecord -Record $_
        }
        finally {
            $impactIQErrorHistory = @($global:Error)
            [array]::Reverse($impactIQErrorHistory)
            foreach ($impactIQRecord in $impactIQErrorHistory) { Write-ImpactIQRunRecord -Record $impactIQRecord }
            if (-not $KeepOpen -and $script:ImpactIQRunFailed) {
                Write-ImpactIQDiagnostic -Level WARN -Message 'The run encountered errors. Some steps or outputs may be incomplete.'
            }
        }
    }
    catch {
        $script:ImpactIQRunFailed = $true
        $script:ImpactIQLogUnavailable = $true
        $impactIQCloseWriter = $true
        $impactIQFallbackPath = Join-Path ([IO.Path]::GetTempPath()) 'ImpactIQ-errorlog.txt'
        $impactIQLogMessage = Protect-ImpactIQLogText ("Cannot write diagnostics to '$ImpactIQLogPath': " + $_.Exception.Message)
        try {
            [IO.File]::AppendAllText($impactIQFallbackPath, ('[{0:O}] [ERROR] {1}{2}' -f [DateTime]::UtcNow, $impactIQLogMessage, [Environment]::NewLine), [Text.UTF8Encoding]::new($false))
        }
        catch [IO.IOException] { $script:ImpactIQLogUnavailable = $true }
        catch [UnauthorizedAccessException] { $script:ImpactIQLogUnavailable = $true }
    }
    finally {
        $script:ImpactIQDiagnostics.SessionOpen = [bool]$KeepOpen
        if ((-not $KeepOpen -or $impactIQCloseWriter) -and $script:ImpactIQDiagnostics.Writer) {
            try { $script:ImpactIQDiagnostics.Writer.Dispose() }
            catch [IO.IOException] { $script:ImpactIQRunFailed = $true; $script:ImpactIQLogUnavailable = $true }
            finally { $script:ImpactIQDiagnostics.Writer = $null }
        }
        $ErrorActionPreference = $impactIQPreviousErrorPreference
        $WarningPreference = $impactIQPreviousWarningPreference
    }
}

function Invoke-ImpactIQNativeCommand {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)][string]$FilePath,
        [string[]]$ArgumentList = @(),
        [switch]$DiscardOutput
    )

    $ErrorActionPreference = 'Continue'
    try { $null = Get-Command -Name $FilePath -CommandType Application -ErrorAction Stop }
    catch [Management.Automation.CommandNotFoundException] {
        $global:LASTEXITCODE = 1
        Write-ImpactIQContinuingError -Record $_ -Context "Native command '$FilePath'"
        return
    }
    & $FilePath @ArgumentList 2>&1 | ForEach-Object {
        Write-ImpactIQRunRecord -Record $_ -DiscardOutput:$DiscardOutput
    }
    $exitCode = $LASTEXITCODE
    $global:LASTEXITCODE = $exitCode
    if ($exitCode -ne 0) {
        Write-ImpactIQDiagnostic -Level ERROR -Message ("Native command '{0}' exited with code {1}." -f [IO.Path]::GetFileName($FilePath), $exitCode)
    }
}

function Invoke-ImpactIQNativeProcess {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)][string]$FilePath,
        [string[]]$ArgumentList = @(),
        [string]$WorkingDirectory = (Get-Location).ProviderPath,
        [switch]$PassThru
    )

    $startInfo = [Diagnostics.ProcessStartInfo]::new()
    $startInfo.FileName = $FilePath
    $startInfo.Arguments = $ArgumentList -join ' '
    $startInfo.WorkingDirectory = $WorkingDirectory
    $startInfo.UseShellExecute = $false
    $startInfo.CreateNoWindow = $true
    $startInfo.RedirectStandardOutput = $true
    $startInfo.RedirectStandardError = $true
    $process = [Diagnostics.Process]::new()
    $process.StartInfo = $startInfo
    $returnedProcess = $false
    try {
        $null = $process.Start()
        $stdout = $process.StandardOutput.ReadToEndAsync()
        $stderr = $process.StandardError.ReadToEndAsync()
        $process.WaitForExit()
        foreach ($line in ($stdout.GetAwaiter().GetResult() -split '\r?\n')) {
            if (-not $line) { continue }
            $level = Get-ImpactIQDiagnosticLevel $line
            if ($level) { Write-ImpactIQDiagnostic -Level $level -Message $line }
            else { Microsoft.PowerShell.Utility\Write-Host $line }
        }
        foreach ($line in ($stderr.GetAwaiter().GetResult() -split '\r?\n')) {
            if (-not $line) { continue }
            $level = Get-ImpactIQDiagnosticLevel $line
            if (-not $level) { $level = 'ERROR' }
            Write-ImpactIQDiagnostic -Level $level -Message $line
        }
        $global:LASTEXITCODE = $process.ExitCode
        if ($process.ExitCode -ne 0) {
            Write-ImpactIQDiagnostic -Level ERROR -Message ("Native process '{0}' exited with code {1}." -f [IO.Path]::GetFileName($FilePath), $process.ExitCode)
        }
        if ($PassThru) { $returnedProcess = $true; $process }
    }
    finally {
        if (-not $returnedProcess) { $process.Dispose() }
    }
}

# Read-Host With 60 Second Timeout
# ================================
function Read-HostWithTimeout {
    param(
        [string]$Prompt,
        [int]$TimeoutSeconds = 60
    )

    Write-Host -NoNewline "$Prompt "
    Write-Host "(waiting up to $TimeoutSeconds seconds):" -ForegroundColor DarkGray

    $inputBuffer = ""
    $endTime = (Get-Date).AddSeconds($TimeoutSeconds)

    while ((Get-Date) -lt $endTime) {
        if ([Console]::KeyAvailable) {
            $key = [Console]::ReadKey($true)
            # Enter pressed - stop reading
            if ($key.Key -eq "Enter") {
                Write-Host ""
                return $inputBuffer
            }
            # Backspace
            elseif ($key.Key -eq "Backspace") {
                if ($inputBuffer.Length -gt 0) {
                    $inputBuffer = $inputBuffer.Substring(0, $inputBuffer.Length - 1)
                    Write-Host -NoNewline "`b `b"
                }
            }
            else {
                $inputBuffer += $key.KeyChar
                Write-Host -NoNewline $key.KeyChar
            }
        }
        Start-Sleep -Milliseconds 100
    }

    Write-Host "Timeout reached - defaulting to Public" -ForegroundColor Yellow
    return ""
}

function Show-EnvironmentSelectionDialog {
    param(
        [int]$TimeoutSeconds = 60
    )

    Add-Type -AssemblyName System.Windows.Forms
    Add-Type -AssemblyName System.Drawing

    $form.FormBorderStyle = 'FixedDialog'

    $label.Text = "Select your Power BI cloud environment:"

    $label.Location = New-Object System.Drawing.Point(20, 20)
    $form.Controls.Add($label)

    # ListBox for environment selection
    $listBox = New-Object System.Windows.Forms.ListBox
    $listBox.Location = New-Object System.Drawing.Point(20, 50)

    $okButton = New-Object System.Windows.Forms.Button
    $okButton.Text = "OK"
    $okButton.Size = New-Object System.Drawing.Size(75, 30)
    $okButton.Location = New-Object System.Drawing.Point(205, 240)
    $okButton.Add_Click({
        $form.Tag = $listBox.SelectedItem
        $form.Close()
    })
    $form.Controls.Add($okButton)

    # Cancel button
    $cancelButton = New-Object System.Windows.Forms.Button
    $cancelButton.Text = "Cancel"
    $cancelButton.Size = New-Object System.Drawing.Size(75, 30)
    $cancelButton.Location = New-Object System.Drawing.Point(285, 240)
    $cancelButton.Add_Click({
        $form.Tag = 'Cancelled'
        $form.Close()
    })

    $timer = New-Object System.Windows.Forms.Timer

    $timer.Add_Tick({
        $timer.Stop()
        $form.Tag = 'Timeout'
        $form.Close()
    })

    $form.Add_FormClosing({
        param($sender, $e)
        $timer.Stop()
        $timer.Dispose()
        
        # If Tag is not set (form closed without button click), mark as cancelled
        if (-not $form.Tag) {
            $form.Tag = 'Cancelled'
        }
    })

    # Handle double-click on list item (same as OK button)
    $listBox.Add_DoubleClick({
        $form.Tag = $listBox.SelectedItem
        $form.Close()
    })

    # Set default button and show
    $form.AcceptButton = $okButton
    $form.CancelButton = $cancelButton
    $timer.Start()
    [void]$form.ShowDialog()

    return $form.Tag
}

. Invoke-ImpactIQLoggedRun -LogPath (Join-Path $global:ConfigFolder 'Error Log.txt') -KeepOpen -Action {
trap {
    Write-ImpactIQContinuingError -Record $_
    continue
}

if (-not (Get-Variable -Name LoginEnvironment -Scope Script -ErrorAction SilentlyContinue)) {

    $selectedEnv = Show-EnvironmentSelectionDialog -TimeoutSeconds 60

    # Handle timeout, cancellation, or no selection
    if ($selectedEnv -eq 'Timeout' -or $selectedEnv -eq 'Cancelled' -or [string]::IsNullOrWhiteSpace($selectedEnv)) {
        Write-Host "No environment selected or timeout reached - defaulting to Public" -ForegroundColor Yellow
        $LoginEnvironment = 'Public'
    }
    else {
        # Extract the environment name from the display text
        # The format is "EnvironmentName" or "EnvironmentName (Description)"
        $envName = $selectedEnv -replace ' \(.*\)', ''
        
        switch ($envName.Trim()) {
            'Public'       { $LoginEnvironment = 'Public' ; break }
            'Germany'      { $LoginEnvironment = 'Germany' ; break }
            'USGovHigh'    { $LoginEnvironment = 'USGovHigh' ; break }
            'USGovMil'     { $LoginEnvironment = 'USGovMil' ; break }
            'USGov'        { $LoginEnvironment = 'USGov' ; break }
            'China'        { $LoginEnvironment = 'China' ; break }
            default {
                Write-Warning "Unrecognized environment '$selectedEnv'. Defaulting to 'Public'."
                $LoginEnvironment = 'Public'
            }
        }
        
        Write-Host "[INFO] Selected environment: $LoginEnvironment" -ForegroundColor Green
    }
}

# If explicitly 'Public', set to $null for Connect-PowerBIServiceAccount default
if ($LoginEnvironment -eq 'Public') { 
    $LoginEnvironment = $null 
}


# If LoginEnvironment is explicitly set to Public, use $null so Connect-PowerBIServiceAccount uses default 'Public' endpoint.
if ($LoginEnvironment -eq 'Public') { $LoginEnvironment = $null }
}

function Get-PowerBIEndpoints {
    param(
        [string]$Environment
    )
    
    # Default endpoints for Public cloud
    $endpoints = @{

# Connect to the Power BI Service
function Connect-PowerBI {
    [CmdletBinding()]
    param(
        [string]$Environment,
        [int]$MaxAttempts = 2
    )

    $previousErrorActionPreference = $ErrorActionPreference

    try {
        $ErrorActionPreference = 'Stop'

        for ($attempt = 1; $attempt -le $MaxAttempts; $attempt++) {
            try {
                Write-Host "[INFO] Connecting to Power BI using environment: $EnvironmentForEndpoints" -ForegroundColor Cyan

                if ($Environment) {
                    Connect-PowerBIServiceAccount -Environment $Environment -ErrorAction Stop -WarningAction Continue | Out-Null
                }
                else {
                    Connect-PowerBIServiceAccount -ErrorAction Stop -WarningAction Continue | Out-Null
                }

                $token = Get-PowerBIAccessToken -ErrorAction Stop -WarningAction Continue
                if ($token -and $token.Authorization) {
                    Write-Host "[INFO] Connected to Power BI." -ForegroundColor Green
                    return
                }

                throw "Power BI sign-in completed without returning an access token."
            }
            catch {
                Disconnect-PowerBIServiceAccount -ErrorAction SilentlyContinue | Out-Null

                if ($attempt -ge $MaxAttempts) {
                    Write-Host "[ERROR] Power BI sign-in failed: $($_.Exception.Message)" -ForegroundColor Red
                    throw
                }

                Write-Warning "Power BI sign-in attempt $attempt failed: $($_.Exception.Message). Retrying..."
                Start-Sleep -Seconds 2
            }
        }
    }
    finally {
        $ErrorActionPreference = $previousErrorActionPreference
    }
}

function Get-FabricAccessToken {
    param(
        [switch]$ForceLogin
    )

    $azEnvironment = switch ($EnvironmentForEndpoints) {
        'China' { 'AzureChinaCloud' }
        'USGov' { 'AzureUSGovernment' }
        'USGovHigh' { 'AzureUSGovernment' }
        'USGovMil' { 'AzureUSGovernment' }
        default { 'AzureCloud' }
    }

    $azContext = Get-AzContext -ErrorAction SilentlyContinue
    if ($ForceLogin -or -not $azContext -or $azContext.Environment.Name -ne $azEnvironment) {
        Write-Host "[INFO] Connecting to Microsoft Fabric..." -ForegroundColor Cyan
        if (Get-Command Update-AzConfig -ErrorAction SilentlyContinue) {
            Update-AzConfig -LoginExperienceV2 Off -Scope Process -ErrorAction SilentlyContinue | Out-Null
        }
        Connect-AzAccount -Environment $azEnvironment -Scope Process -SkipContextPopulation -ErrorAction Stop -WarningAction Continue | Out-Null
    }

    try {
        $tokenValue = (Get-AzAccessToken -ResourceUrl $global:PowerBIEndpoints.FabricApiPrefix -ErrorAction Stop).Token
    }
    catch {
        if ($ForceLogin) {
            throw
        }

        return Get-FabricAccessToken -ForceLogin
    }

    if ($tokenValue -is [System.Security.SecureString]) {
        $tokenPointer = [Runtime.InteropServices.Marshal]::SecureStringToBSTR($tokenValue)
    } else {
        return [string]$tokenValue
    }

    try {
        return [Runtime.InteropServices.Marshal]::PtrToStringBSTR($tokenPointer)
    }
    finally {
        [Runtime.InteropServices.Marshal]::ZeroFreeBSTR($tokenPointer)
    }
}

# Function to get the current access token — MSAL auto-refreshes silently via the session cache
function Get-CurrentAccessToken {
    try {
        return (Get-PowerBIAccessToken -ErrorAction Stop -WarningAction Continue).Authorization -replace 'Bearer ', ''
    }
    catch {
        # MSAL cache may be corrupt or refresh token revoked — re-connect and retry
        Write-Host "[WARN] Token fetch failed, re-connecting..."
        try {
            Connect-PowerBI -Environment $LoginEnvironment -MaxAttempts 2 -ErrorAction Stop
            return (Get-PowerBIAccessToken -ErrorAction Stop -WarningAction Continue).Authorization -replace 'Bearer ', ''
        }
        catch {
            Write-Host "[ERROR] Re-connect also failed: $($_.Exception.Message)"
            throw "Cannot obtain Power BI access token."
        }
    }
}

function Invoke-FabricPagedGet {
    param(
        [Parameter(Mandatory = $true)]
        [string]$Url,
        [string]$AccessToken
    )

    $results = @()
    $nextUrl = $Url

    while ($nextUrl) {
        if (-not $AccessToken) {
            $AccessToken = Get-CurrentAccessToken
        }
        $headers = @{
            Authorization = "Bearer $AccessToken"
            'Content-Type' = 'application/json'
        }
        $response = Invoke-RestMethod -Uri $nextUrl -Headers $headers -Method GET -ErrorAction Stop

        if ($response.value) {
            $results += @($response.value)
        }

        $nextUrl = if ($response.continuationUri) {
            $response.continuationUri
        } else {
            $null
        }
    }

    return $results
}

# Function to rename properties in objects and handle duplicates
function Rename-Properties {
    param ($object, $renameMap)
    $newObject = New-Object PSObject
    foreach ($originalName in $renameMap.Keys) {
        $newPropertyName = $renameMap[$originalName]
        $propertyValue = if ($object.PSObject.Properties[$originalName]) { $object.$originalName } else { $null }
        if ($newObject.PSObject.Properties[$newPropertyName]) { $newPropertyName += "_duplicate" }
        $newObject | Add-Member -MemberType NoteProperty -Name $newPropertyName -Value $propertyValue
    }
    foreach ($property in $object.PSObject.Properties) {
        if (-not $renameMap.ContainsKey($property.Name)) {
            $newObject | Add-Member -MemberType NoteProperty -Name $property.Name -Value $property.Value
        }
    }
    return $newObject
}

function Show-WorkspacePicker {
    param(
        [Parameter(Mandatory)]
        [array]$Workspaces, # array of PSCustomObjects with id + name
        [switch]$RequireSelection
    )

    Add-Type -AssemblyName System.Windows.Forms
    Add-Type -AssemblyName System.Drawing

    # Form
    $form               = New-Object System.Windows.Forms.Form
    $form.Text          = "Select Workspaces"
    $form.StartPosition = 'CenterScreen'
    $form.Size          = New-Object System.Drawing.Size(520,580)
	$form.TopMost       = $true

    # Instruction label
    $lbl                = New-Object System.Windows.Forms.Label
    $lbl.Text           = "Select the workspaces to run against:"
    $lbl.AutoSize       = $true
    $lbl.Location       = New-Object System.Drawing.Point(12,12)
    $form.Controls.Add($lbl)

    # Search label
    $lblSearch          = New-Object System.Windows.Forms.Label
    $lblSearch.Text     = "Search:"
    $lblSearch.AutoSize = $true
    $lblSearch.Location = New-Object System.Drawing.Point(12,40)
    $form.Controls.Add($lblSearch)

    # Search TextBox
    $txtSearch          = New-Object System.Windows.Forms.TextBox
    $txtSearch.Location = New-Object System.Drawing.Point(70,37)
    $txtSearch.Size     = New-Object System.Drawing.Size(422,23)
    $form.Controls.Add($txtSearch)

    # CheckedListBox
    $clb                = New-Object System.Windows.Forms.CheckedListBox
    $clb.Location       = New-Object System.Drawing.Point(12,70)
	$clb.Size           = New-Object System.Drawing.Size(480,380)
    $clb.CheckOnClick   = $true
    $clb.Sorted         = $true

    # Store all workspace items for filtering
    $allWorkspaceItems = @()
    foreach ($ws in $Workspaces) {
        $display = "{0} ({1})" -f $ws.name, $ws.id
        $allWorkspaceItems += [pscustomobject]@{ Display=$display; Id=$ws.id; Name=$ws.name }
    }

    # Maintain persistent checked state across searches
    $persistentCheckedIds = @{}

    # Initial population
    foreach ($item in $allWorkspaceItems) {
        [void]$clb.Items.Add($item)
    }
    $clb.DisplayMember = 'Display'
    $form.Controls.Add($clb)

	# Track when items are checked/unchecked
    $clb.Add_ItemCheck({
        param($sender, $e)
        $item = $clb.Items[$e.Index]
        if ($e.NewValue -eq 'Checked') {
            $persistentCheckedIds[$item.Id] = $true
        } elseif ($e.NewValue -eq 'Unchecked') {
            $persistentCheckedIds.Remove($item.Id)
        }
    })

	# Search filter logic
    $txtSearch.Add_TextChanged({
        $searchText = $txtSearch.Text
        $clb.BeginUpdate()
        
        $clb.Items.Clear()
        
        # Escape search text once for reuse in loop
        $escapedSearchText = [regex]::Escape($searchText)
        
        # Filter and re-add items
        foreach ($item in $allWorkspaceItems) {
            if ([string]::IsNullOrWhiteSpace($searchText) -or 
                $item.Display -imatch $escapedSearchText) {
                $index = $clb.Items.Add($item)
                # Restore checked state from persistent store
                if ($persistentCheckedIds.ContainsKey($item.Id)) {
                    $clb.SetItemChecked($index, $true)
                }
            }
        }
        
        $clb.EndUpdate()
    })

    # "Select All" checkbox
    $chkAll            = New-Object System.Windows.Forms.CheckBox
    $chkAll.Text       = "Select All"
    $chkAll.AutoSize   = $true
    $chkAll.Location   = New-Object System.Drawing.Point(12,460)
    $chkAll.Add_CheckedChanged({
        for ($i=0; $i -lt $clb.Items.Count; $i++) {
            $clb.SetItemChecked($i, $chkAll.Checked)
        }
    })
    $form.Controls.Add($chkAll)

    # "Clear All" button
    $btnClearAll       = New-Object System.Windows.Forms.Button
    $btnClearAll.Text  = "Clear All"
    $btnClearAll.Width = 80
    $btnClearAll.Location = New-Object System.Drawing.Point(90,458)
    $btnClearAll.Add_Click({
        # Uncheck all items - ItemCheck events will update persistent store
        for ($i=0; $i -lt $clb.Items.Count; $i++) {
            $clb.SetItemChecked($i, $false)
        }
        $chkAll.Checked = $false
    })
    $form.Controls.Add($btnClearAll)

    # "Include My Workspace" checkbox
    $chkMy             = New-Object System.Windows.Forms.CheckBox
    $chkMy.Text        = "Include 'My Workspace'"
    $chkMy.AutoSize    = $true
    $chkMy.Location    = New-Object System.Drawing.Point(184,460)
    $chkMy.Checked     = $false
    $form.Controls.Add($chkMy)

    # OK button
    $okBtn             = New-Object System.Windows.Forms.Button
    $okBtn.Text        = "OK"
    $okBtn.Width       = 100
    $okBtn.Location    = New-Object System.Drawing.Point(286,500)
    $okBtn.Add_Click({
        $form.Tag = 'OK'
        $form.Close()
    })
    $form.Controls.Add($okBtn)

    # Cancel button
    $cancelBtn         = New-Object System.Windows.Forms.Button
    $cancelBtn.Text    = "Cancel"
    $cancelBtn.Width   = 100
    $cancelBtn.Location= New-Object System.Drawing.Point(392,500)
    $cancelBtn.Add_Click({
        $form.Tag = 'Cancel'
        $form.Close()
    })
    $form.Controls.Add($cancelBtn)

    # Timer (auto-close after 10 minutes)
    $timer = New-Object System.Windows.Forms.Timer
    $timer.Interval = 600000
    $timer.Add_Tick({
        $timer.Stop()
        $form.Tag = 'Timeout'
        $form.Close()
    })
    $timer.Start()

    [void]$form.ShowDialog()

    # Gather selections from persistent store (to include items not currently visible due to search)
    $selectedIds = @($persistentCheckedIds.Keys)

    # Treat the Cancel button AND closing the window (the X) as a cancel so the
    # caller returns to the run-mode menu. Only an explicit OK proceeds, and the
    # timer still auto-continues on Timeout.
    if ($form.Tag -ne 'OK' -and $form.Tag -ne 'Timeout') {
        throw "User cancelled workspace selection."
    }

    $timedOut = ($form.Tag -eq 'Timeout')

    # Include My Workspace only if checked… except on timeout, then force include
    $includeMy = if ($timedOut) { $true } else { $chkMy.Checked }

    # If nothing selected from workspace list:
    # - If "Include My Workspace" is checked, process only My Workspace (empty array for regular workspaces)
    # - Otherwise, default to all workspaces (backwards compatibility with timeout/cancel behavior)
    if ($selectedIds.Count -eq 0 -and -not $includeMy -and -not $RequireSelection) {
        $selectedIds = $Workspaces.id
    }

    return [pscustomobject]@{
        SelectedWorkspaceIds = $selectedIds
        IncludeMyWorkspace   = $includeMy
        TimedOut             = $timedOut
    }
}

function Get-FabricDataAgentInventory {
    param(
        [AllowEmptyCollection()][object[]]$Workspaces
    )

    $ErrorActionPreference = 'Stop'
    $agents = @()
    $failures = @()
    if ($Workspaces.Count -eq 0) { return @() }
    $accessToken = Get-FabricAccessToken
    foreach ($workspace in $Workspaces) {
        try {
            $url = "$($global:PowerBIEndpoints.FabricApiPrefix)/v1/workspaces/$($workspace.id)/items?type=DataAgent"
            foreach ($agent in (Invoke-FabricPagedGet -Url $url -AccessToken $accessToken)) {
                $agents += [PSCustomObject]@{
                    WorkspaceId = $workspace.id
                    WorkspaceName = $workspace.name
                    Agent = $agent
                }
            }
        }
        catch {
            $detail = if ($_.ErrorDetails.Message) { $_.ErrorDetails.Message } else { $_.Exception.GetBaseException().Message }
            $failures += [PSCustomObject]@{
                WorkspaceId = $workspace.id
                Detail = $detail
            }
        }
    }

    if ($failures.Count -eq $Workspaces.Count) {
        throw "Data Agent discovery failed in all $($Workspaces.Count) workspace(s). Check Fabric sign-in and item read permissions. First failure: $($failures[0].Detail)"
    }
    if ($failures.Count -gt 0) {
        Write-Host "[WARN] Data Agent discovery skipped $($failures.Count) of $($Workspaces.Count) workspace(s); results may be incomplete. First failure: $($failures[0].Detail)" -ForegroundColor Yellow
    }
    return $agents
}

function Get-FabricDataAgentToken {
    $ErrorActionPreference = 'Stop'
    $resource = $global:PowerBIEndpoints.FabricApiPrefix
    $context = Get-AzContext -ErrorAction Stop
    $azCommand = Get-Command az -ErrorAction SilentlyContinue

    # The CLI client can carry delegated scopes that the Azure PowerShell client lacks.
    if ($azCommand) {
        try {
            $cloud = Invoke-ImpactIQNativeCommand -FilePath 'az' -ArgumentList @('cloud', 'show', '--query', 'name', '-o', 'tsv')
            if ($LASTEXITCODE -ne 0 -or $cloud -ne $context.Environment.Name) {
                throw "Azure CLI cloud does not match the current Fabric sign-in."
            }
            $tenantId = $context.Tenant.Id
            $token = Invoke-ImpactIQNativeCommand -FilePath 'az' -ArgumentList @('account', 'get-access-token', '--resource', $resource, '--tenant', $tenantId, '--query', 'accessToken', '-o', 'tsv')
            if ($LASTEXITCODE -ne 0 -or [string]::IsNullOrWhiteSpace($token)) {
                Write-Host "[INFO] Signing in to Azure CLI for Data Agent definitions..." -ForegroundColor Cyan
                Invoke-ImpactIQNativeCommand -FilePath 'az' -ArgumentList @('login', '--tenant', $tenantId, '--allow-no-subscriptions', '--output', 'none')
                if ($LASTEXITCODE -ne 0) { throw "Azure CLI sign-in failed." }
                $token = Invoke-ImpactIQNativeCommand -FilePath 'az' -ArgumentList @('account', 'get-access-token', '--resource', $resource, '--tenant', $tenantId, '--query', 'accessToken', '-o', 'tsv')
            }
            if ($LASTEXITCODE -ne 0 -or [string]::IsNullOrWhiteSpace($token)) {
                throw "Azure CLI did not return a Fabric access token."
            }
            return $token.Trim()
        }
        catch {
            Write-Host "[WARN] Azure CLI token acquisition failed: $($_.Exception.Message). Trying the existing Fabric sign-in." -ForegroundColor Yellow
        }
    }
    else {
        Write-Host "[WARN] Azure CLI is not installed. Using the existing Fabric sign-in; if definitions return 403, install Azure CLI and grant the required Fabric permissions." -ForegroundColor Yellow
    }

    return Get-FabricAccessToken
}

function Get-FabricDataAgentDefinition {
    param(
        [string]$WorkspaceId,
        [string]$DataAgentId,
        [string]$AccessToken,
        [int]$TimeoutSeconds = 300
    )

    $ErrorActionPreference = 'Stop'
    $headers = @{ Authorization = "Bearer $AccessToken"; 'Content-Type' = 'application/json' }
    $definitionUrl = "$($global:PowerBIEndpoints.FabricApiPrefix)/v1/workspaces/$WorkspaceId/dataAgents/$DataAgentId/getDefinition"
    $deadline = (Get-Date).AddSeconds($TimeoutSeconds)

    try {
        $response = Invoke-WebRequest -Uri $definitionUrl -Method POST -Headers $headers -Body '{}' -UseBasicParsing -TimeoutSec $TimeoutSeconds
        if ([int]$response.StatusCode -eq 200) {
            $definition = ($response.Content | ConvertFrom-Json).definition
        }
        elseif ([int]$response.StatusCode -eq 202) {
            $operationUrl = [string]($response.Headers['Location'] | Select-Object -First 1)
            if (-not $operationUrl) { throw "Definition operation did not return a Location header." }
            do {
                $retrySeconds = 2
                if ($response.Headers['Retry-After']) {
                    $retrySeconds = [Math]::Max(1, [int]($response.Headers['Retry-After'] | Select-Object -First 1))
                }
                $remainingSeconds = ($deadline - (Get-Date)).TotalSeconds
                if ($remainingSeconds -le $retrySeconds) { throw "Definition operation timed out after $TimeoutSeconds seconds." }
                Start-Sleep -Seconds $retrySeconds
                $requestTimeout = [Math]::Max(1, [int][Math]::Ceiling(($deadline - (Get-Date)).TotalSeconds))
                $response = Invoke-WebRequest -Uri $operationUrl -Method GET -Headers $headers -UseBasicParsing -TimeoutSec $requestTimeout
                $operation = $response.Content | ConvertFrom-Json
                if ($operation.status -in @('Failed', 'Canceled', 'Cancelled')) {
                    throw "Definition operation $($operation.status): $($operation.error | ConvertTo-Json -Compress)"
                }
                if ($operation.status -notin @('NotStarted', 'Running', 'Succeeded')) {
                    throw "Unexpected definition operation status '$($operation.status)'."
                }
            } while ($operation.status -ne 'Succeeded')

            $remainingSeconds = ($deadline - (Get-Date)).TotalSeconds
            if ($remainingSeconds -le 0) { throw "Definition operation timed out after $TimeoutSeconds seconds." }
            $resultUrl = $operationUrl.TrimEnd('/') + '/result'
            $response = Invoke-WebRequest -Uri $resultUrl -Method GET -Headers $headers -UseBasicParsing -TimeoutSec ([Math]::Max(1, [int][Math]::Ceiling($remainingSeconds)))
            $definition = ($response.Content | ConvertFrom-Json).definition
        }
        else {
            throw "Unexpected getDefinition HTTP status $($response.StatusCode)."
        }
        if (-not $definition.parts) { throw "Definition contains no parts." }
        return $definition
    }
    catch {
        Write-Host "[WARN] Could not retrieve definition for Data Agent $DataAgentId in workspace $WorkspaceId : $($_.Exception.Message)" -ForegroundColor Yellow
        return $null
    }
}

function Get-FabricDataAgentModelTargets {
    param(
        [Parameter(Mandatory)]$Definition
    )

    $ErrorActionPreference = 'Stop'
    foreach ($part in $Definition.parts) {
        $path = $part.path -replace '\\', '/'
        if ($path -notmatch '^Files/Config/[^/]+/(?:.*/)?datasource\.json$') { continue }
        try {
            if ($part.payloadType -ne 'InlineBase64') { throw "Unsupported datasource payload type." }
            $source = [Text.Encoding]::UTF8.GetString([Convert]::FromBase64String($part.payload)) | ConvertFrom-Json
            if ($source.type -eq 'semantic_model') {
                if (-not $source.artifactId -or -not $source.workspaceId) {
                    throw "Semantic model source is missing artifactId or workspaceId."
                }
                [PSCustomObject]@{ DatasetId = $source.artifactId; WorkspaceId = $source.workspaceId }
            }
        }
        catch {
            throw "Cannot resolve connected models from '$path': $($_.Exception.Message)"
        }
    }
}

function Get-FabricAppDaxSource {
    [CmdletBinding()]
    param([Parameter(Mandatory)][AllowEmptyString()][string]$Source)

    $ErrorActionPreference = 'Stop'
    $tokens = [Collections.Generic.List[object]]::new()
    $queries = [Collections.Generic.List[object]]::new()
    $references = [Collections.Generic.HashSet[string]]::new([StringComparer]::Ordinal)
    $state = @{ Position = 0; Depth = 0; LexMatch = $null }
    $lex = [regex]::new('\G(?:(?<space>\s+)|(?<comment>//[^\r\n]*|/\*[\s\S]*?\*/)|(?<string>"(?:\\[\s\S]|[^"\\\r\n])*"|''(?:\\[\s\S]|[^''\\\r\n])*'')|(?<word>[a-zA-Z_$][\w$]*)|(?<number>\d+(?:\.\d+)?)|(?<punct>=>|\+\+|--|&&|\|\||\?\?|[^\s]))', 'Compiled')
    $regexLiteral = [regex]::new('\G/(?:\\[^\r\n]|\[(?:\\[^\r\n]|[^\]\\\r\n])*\]|[^/\\\[\r\n])+/[a-z]*', 'Compiled')
    $templateChunk = [regex]::new('\G(?:[^`\\$]+|\\(?:\r\n|[\s\S])|\$(?!\{))+', 'Compiled')
    $escapePattern = [regex]::new('\\(?:\r\n|[\s\S])', 'Compiled')

    function ConvertFrom-AppJsString([string]$Text) {
        if ($Text.IndexOf('\') -lt 0) { return $Text }
        $builder = [Text.StringBuilder]::new()
        $offset = 0
        foreach ($escape in $escapePattern.Matches($Text)) {
            if ($escape.Index -lt $offset) { continue }
            $null = $builder.Append($Text.Substring($offset, $escape.Index - $offset))
            $code = $escape.Value.Substring(1)
            $offset = $escape.Index + $escape.Length
            switch -CaseSensitive ($code) {
                'n' { $null = $builder.Append("`n") }
                'r' { $null = $builder.Append("`r") }
                't' { $null = $builder.Append("`t") }
                'b' { $null = $builder.Append([char]8) }
                'f' { $null = $builder.Append([char]12) }
                'v' { $null = $builder.Append([char]11) }
                '0' {
                    if ($offset -lt $Text.Length -and [char]::IsDigit($Text[$offset])) { throw 'Legacy octal escapes are unsupported.' }
                    $null = $builder.Append([char]0)
                }
                { $_ -in @("`n", "`r", "`r`n") } { }
                { $_ -in @('x', 'u') } {
                    $width = if ($code -eq 'x') { 2 } else { 4 }
                    if ($code -eq 'u' -and $offset -lt $Text.Length -and $Text[$offset] -eq '{') {
                        $end = $Text.IndexOf('}', $offset)
                        if ($end -lt 0) { throw 'Invalid Unicode escape.' }
                        $hex = $Text.Substring($offset + 1, $end - $offset - 1)
                        if ($hex -notmatch '^[0-9a-fA-F]{1,6}$') { throw 'Invalid Unicode escape.' }
                        $value = [Convert]::ToInt32($hex, 16)
                        if ($value -le 65535) { $null = $builder.Append([char]$value) }
                        else { $null = $builder.Append([char]::ConvertFromUtf32($value)) }
                        $offset = $end + 1
                    }
                    else {
                        if ($offset + $width -gt $Text.Length) { throw 'Incomplete hexadecimal escape.' }
                        $hex = $Text.Substring($offset, $width)
                        if ($hex -notmatch '^[0-9a-fA-F]+$') { throw 'Invalid hexadecimal escape.' }
                        $null = $builder.Append([char][Convert]::ToInt32($hex, 16))
                        $offset += $width
                    }
                }
                { $_ -match '^[1-9]$' } { throw 'Legacy numeric escapes are unsupported.' }
                default { $null = $builder.Append($code) }
            }
        }
        $null = $builder.Append($Text.Substring($offset))
        $builder.ToString()
    }

    function Add-AppJsLiteral([int]$Start, [int]$End, [string]$Text, [string[]]$Expressions) {
        if ($Text -match '^(?:\s+|(?://|--)[^\r\n]*(?:\r?\n|$)|/\*[\s\S]*?\*/)*(?:EVALUATE|DEFINE)\b') {
            $prefixStart = [Math]::Max(0, $Start - 300)
            $prefix = $Source.Substring($prefixStart, $Start - $prefixStart)
            $owner = [regex]::Match($prefix, '(?:function\s+(?<name>[\w$]+)\([^{}]*\)\s*\{\s*return\s*|(?:const|let|var)\s+(?<name>[\w$]+)\s*=\s*)$')
            $symbol = if ($owner.Success) { $owner.Groups['name'].Value } else { 'query_' + ($queries.Count + 1) }
            $queries.Add([PSCustomObject]@{
                symbol = $symbol
                evidence = $(if ($Expressions.Count) { 'DynamicTemplate' } else { 'StaticCandidate' })
                text = $Text.Trim()
                expressions = @($Expressions)
                start = $Start
                end = $End
                line = 0
                column = 0
                literalSource = $Source.Substring($Start, $End - $Start)
                builderSource = $null
                enclosingExpression = $null
            })
        }
        if (-not $Expressions.Count -and $Text -match '^(?:\.{0,2}/|https://)[^\s]+\.m?js(?:\?[^\s]*)?$') { $null = $references.Add($Text) }
    }

    function Read-AppJsCode([bool]$UntilBrace = $false) {
        $state.Depth++
        if ($state.Depth -gt 128) { throw 'JavaScript nesting limit exceeded.' }
        $braceDepth = 0
        $regexAllowed = $true
        try {
            while ($state.Position -lt $Source.Length) {
                $start = $state.Position
                # Restart only when template/regex scanning has moved beyond the previous match.
                $match = if ($null -ne $state.LexMatch -and
                    $state.LexMatch.Index + $state.LexMatch.Length -eq $start) {
                    $state.LexMatch.NextMatch()
                }
                else { $lex.Match($Source, $start) }
                $state.LexMatch = $match
                if (-not $match.Success) { throw 'Could not scan JavaScript token.' }
                $value = $match.Value
                $state.Position += $match.Length
                if ($match.Groups['space'].Success -or $match.Groups['comment'].Success) { continue }
                if ($value -eq '}' -and $UntilBrace -and $braceDepth -eq 0) { return }
                if ($value -eq '{') { $braceDepth++ }
                if ($value -eq '}') { $braceDepth-- }
                if ($match.Groups['string'].Success) {
                    $text = ConvertFrom-AppJsString $value.Substring(1, $value.Length - 2)
                    $tokens.Add([PSCustomObject]@{ Kind = 'string'; Value = $text; Start = $start; End = $state.Position })
                    Add-AppJsLiteral $start $state.Position $text @()
                    $regexAllowed = $false
                    continue
                }
                if ($value -eq '"' -or $value -eq "'") { throw 'Unterminated JavaScript string.' }
                if ($value -eq '`') {
                    $builder = [Text.StringBuilder]::new()
                    $expressions = [Collections.Generic.List[string]]::new()
                    $closed = $false
                    while ($state.Position -lt $Source.Length) {
                        $chunk = $templateChunk.Match($Source, $state.Position)
                        if ($chunk.Success) {
                            $normalized = $chunk.Value.Replace("`r`n", "`n").Replace("`r", "`n")
                            $null = $builder.Append((ConvertFrom-AppJsString $normalized))
                            $state.Position += $chunk.Length
                        }
                        elseif ($Source[$state.Position] -eq '`') { $state.Position++; $closed = $true; break }
                        elseif ($state.Position + 1 -lt $Source.Length -and
                            $Source[$state.Position] -eq '$' -and $Source[$state.Position + 1] -eq '{') {
                            $state.Position += 2
                            $expressionStart = $state.Position
                            Read-AppJsCode $true
                            $expression = $Source.Substring($expressionStart, $state.Position - $expressionStart - 1)
                            $expressions.Add($expression)
                            $null = $builder.Append('${' + $expression + '}')
                        }
                        else { throw 'Unsupported JavaScript template.' }
                    }
                    if (-not $closed) { throw 'Unterminated JavaScript template.' }
                    $text = $builder.ToString()
                    $tokens.Add([PSCustomObject]@{ Kind = 'template'; Value = $text; Start = $start; End = $state.Position })
                    Add-AppJsLiteral $start $state.Position $text $expressions.ToArray()
                    $regexAllowed = $false
                    continue
                }
                if ($value -eq '/' -and $regexAllowed) {
                    $literal = $regexLiteral.Match($Source, $start)
                    if (-not $literal.Success) { throw 'Unterminated or unsupported JavaScript regular expression.' }
                    $state.Position = $start + $literal.Length
                    $tokens.Add([PSCustomObject]@{ Kind = 'regex'; Value = ''; Start = $start; End = $state.Position })
                    $regexAllowed = $false
                    continue
                }
                $kind = if ($match.Groups['word'].Success) { 'word' } else { 'punct' }
                $tokens.Add([PSCustomObject]@{ Kind = $kind; Value = $value; Start = $start; End = $state.Position })
                $regexAllowed = $value -in @('(', '[', '{', ',', ':', ';', '=', '=>', '!', '?', '&&', '||', '??', '+', '-', '*', '/', '%', '&', '|', '^', '~', '<', '>', 'return', 'throw', 'case', 'typeof', 'void', 'delete', 'yield')
            }
            if ($UntilBrace) { throw 'Unterminated JavaScript template expression.' }
        }
        finally { $state.Depth-- }
    }

    function Read-AppJsObject([ref]$Index, [int]$Depth = 0) {
        if ($Depth -gt 8 -or $Index.Value -ge $tokens.Count -or $tokens[$Index.Value].Value -ne '{') { throw 'Unsupported model configuration.' }
        $Index.Value++
        $result = [Collections.Generic.Dictionary[string,object]]::new([StringComparer]::Ordinal)
        while ($Index.Value -lt $tokens.Count -and $tokens[$Index.Value].Value -ne '}') {
            $key = $tokens[$Index.Value]
            if ($key.Kind -notin @('word', 'string') -or $result.ContainsKey($key.Value)) { throw 'Nonliteral or duplicate configuration property.' }
            $Index.Value++
            if ($Index.Value -ge $tokens.Count -or $tokens[$Index.Value].Value -ne ':') { throw 'Unsupported configuration property.' }
            $Index.Value++
            if ($Index.Value -ge $tokens.Count) { throw 'Incomplete configuration.' }
            if ($tokens[$Index.Value].Kind -eq 'string') { $result[$key.Value] = $tokens[$Index.Value].Value; $Index.Value++ }
            elseif ($tokens[$Index.Value].Value -eq '{') { $result[$key.Value] = Read-AppJsObject $Index ($Depth + 1) }
            else { throw 'Nonliteral model configuration.' }
            if ($Index.Value -lt $tokens.Count -and $tokens[$Index.Value].Value -eq ',') { $Index.Value++ }
            elseif ($Index.Value -ge $tokens.Count -or $tokens[$Index.Value].Value -ne '}') { throw 'Unsupported configuration expression.' }
        }
        if ($Index.Value -ge $tokens.Count) { throw 'Unterminated configuration object.' }
        $Index.Value++
        return ,$result
    }

            elseif ($tokens[$index].Value -eq $closingToken) {

                if ($bodyDepth -eq 0) { $bodyClose = $index; break }
            }
        }
        if ($bodyClose -lt 0) { continue }

            Start = $tokens[$tokenIndex].Start
            End = $tokens[$bodyClose].End
            OpenIndex = $bodyOpen
            CloseIndex = $bodyClose
            ParameterOpen = $parameterOpen
            Parameters = $parameters.ToArray()
        })
    }

        # Preserve the original shortest-range rule, including first-match ties and inclusive ends.
        $rank = [long]($range.End - $range.Start) * ($functionRanges.Count + 1) + $rangeIndex
        $rangesByRank[$rank] = $range
        foreach ($position in @([int]$range.Start, [int]($range.End + 1))) {
            if (-not $scopeEvents.ContainsKey($position)) {
                $scopeEvents[$position] = @{
                    Add = [Collections.Generic.List[long]]::new()
                    Remove = [Collections.Generic.List[long]]::new()
                }
            }
        }
        $scopeEvents[[int]$range.Start].Add.Add($rank)
        $scopeEvents[[int]($range.End + 1)].Remove.Add($rank)
    }
    $activeRanges = [Collections.Generic.SortedSet[long]]::new()
    $scopePositions = [Collections.Generic.List[int]]::new()
    $scopeOwners = [Collections.Generic.List[object]]::new()
    foreach ($event in $scopeEvents.GetEnumerator()) {
        foreach ($rank in $event.Value.Remove) { $null = $activeRanges.Remove($rank) }
        foreach ($rank in $event.Value.Add) {
            $range = $rangesByRank[$rank]
            $parent = '-1'
            foreach ($candidateRank in $activeRanges) {
                $candidate = $rangesByRank[$candidateRank]
                if ($candidate.End -ge $range.End) {
                    $parent = [string]$candidate.Start
                    break
                }
            }
            $functionParents[[string]$range.Start] = $parent
        }
        foreach ($rank in $event.Value.Add) { $null = $activeRanges.Add($rank) }
        $scopePositions.Add($event.Key)
        $scopeOwners.Add($(if ($activeRanges.Count) { $rangesByRank[$activeRanges.Min] } else { $null }))
    }
    $scopeStarts = $scopePositions.ToArray()

    function Get-AppDaxFunctionAt([int]$Position) {
        $index = [Array]::BinarySearch($scopeStarts, $Position)
        if ($index -lt 0) { $index = (-bnot $index) - 1 }
        if ($index -ge 0) { $scopeOwners[$index] }
    }

    function Resolve-AppDaxAlias($Token) {
        if ($null -eq $Token) { return $null }
        if ($Token.Kind -eq 'string' -and $Token.Value -match '^[a-zA-Z_$][\w$]*$') { return $Token.Value }
        if ($Token.Kind -eq 'word' -and $aliasVariables.ContainsKey($Token.Value)) { return $aliasVariables[$Token.Value] }
        $null
    }

    $dependenciesByFunction = @{}
    function Get-AppDaxDependencies($Range) {
        if ($null -eq $Range) { return @{} }
        $key = [string]$Range.Start
        if ($dependenciesByFunction.ContainsKey($key)) { return $dependenciesByFunction[$key] }
        $dependencies = @{}
        for ($index = $Range.OpenIndex + 1; $index -lt $Range.CloseIndex - 2; $index++) {
            if ($tokens[$index].Value -notin @('const', 'let', 'var')) { continue }
            $cursor = $index + 1
            while ($cursor -lt $Range.CloseIndex) {
                if ($tokens[$cursor].Kind -ne 'word' -or $tokens[$cursor + 1].Value -ne '=') { break }
                $name = $tokens[$cursor].Value
                $expressionStart = $cursor + 2
                $roundDepth = $squareDepth = $curlyDepth = 0
                $expressionEnd = $expressionStart
                while ($expressionEnd -lt $Range.CloseIndex) {
                    $value = $tokens[$expressionEnd].Value
                    if ($value -eq '(') { $roundDepth++ }
                    elseif ($value -eq ')') { $roundDepth-- }
                    elseif ($value -eq '[') { $squareDepth++ }
                    elseif ($value -eq ']') { $squareDepth-- }
                    elseif ($value -eq '{') { $curlyDepth++ }
                    elseif ($value -eq '}') { $curlyDepth-- }
                    if ($roundDepth -eq 0 -and $squareDepth -eq 0 -and $curlyDepth -eq 0 -and $value -in @(',', ';')) { break }
                    $expressionEnd++
                }
                $identifiers = [Collections.Generic.HashSet[string]]::new([StringComparer]::Ordinal)
                for ($dependencyIndex = $expressionStart; $dependencyIndex -lt $expressionEnd; $dependencyIndex++) {
                    if ($tokens[$dependencyIndex].Kind -eq 'word') { $null = $identifiers.Add($tokens[$dependencyIndex].Value) }
                }
                $dependencies[$name] = @($identifiers)
                if ($expressionEnd -ge $Range.CloseIndex -or $tokens[$expressionEnd].Value -eq ';') {
                    $index = $expressionEnd
                    break
                }
                $cursor = $expressionEnd + 1
            }
        }
        $dependenciesByFunction[$key] = $dependencies
        $dependencies
    }

    function Add-AppDaxExpressionRoute([string]$Alias, [int]$StartIndex, [int]$EndIndex, $Range) {
        if (-not $Alias -or $StartIndex -ge $EndIndex) { return }
        $dependencies = Get-AppDaxDependencies $Range
        $pendingSymbols = [Collections.Generic.Stack[string]]::new()
        $seenSymbols = [Collections.Generic.HashSet[string]]::new([StringComparer]::Ordinal)
        for ($index = $StartIndex; $index -lt $EndIndex; $index++) {
            if ($tokens[$index].Kind -eq 'word') { $pendingSymbols.Push($tokens[$index].Value) }
            foreach ($query in $queriesByToken[$index]) {
                $null = $queryAliases[[string]$query.start].Add($Alias)
            }
        }
        while ($pendingSymbols.Count -gt 0) {
            $symbol = $pendingSymbols.Pop()
            if (-not $seenSymbols.Add($symbol)) { continue }
            if ($queryOwners.ContainsKey($symbol)) {
                foreach ($query in $queryOwners[$symbol]) { $null = $queryAliases[[string]$query.start].Add($Alias) }
            }
            if ($dependencies.ContainsKey($symbol)) {
                foreach ($dependency in $dependencies[$symbol]) { $pendingSymbols.Push($dependency) }
            }
        }
    }

    $wrapperParameterRoutes = @{}
    $parameterizedModelRoutes = @{}
    foreach ($range in $functionRanges) {

        for ($index = $range.OpenIndex + 1; $index -lt $range.CloseIndex - 5; $index++) {

            }
            if ((-not $alias -and $aliasParameterIndex -lt 0) -or $queryStart -lt 0) { continue }
            $roundDepth = $squareDepth = $curlyDepth = 0
            $queryEnd = $queryStart
            while ($queryEnd -lt $range.CloseIndex) {
                $value = $tokens[$queryEnd].Value
                if ($value -eq '(') { $roundDepth++ }
                elseif ($value -eq ')') {
                    if ($roundDepth -eq 0 -and $squareDepth -eq 0 -and $curlyDepth -eq 0) { break }
                    $roundDepth--
                }
                elseif ($value -eq '[') { $squareDepth++ }
                elseif ($value -eq ']') { $squareDepth-- }
                elseif ($value -eq '{') { $curlyDepth++ }
                elseif ($value -eq '}') {
                    if ($roundDepth -eq 0 -and $squareDepth -eq 0 -and $curlyDepth -eq 0) { break }
                    $curlyDepth--
                }
                if ($roundDepth -eq 0 -and $squareDepth -eq 0 -and $curlyDepth -eq 0 -and $value -eq ',') { break }
                $queryEnd++
            }
            if ($alias) {
                Add-AppDaxExpressionRoute $alias $queryStart $queryEnd $range
            }
            else {
                if (-not $parameterizedModelRoutes.ContainsKey($range.Name)) {
                    $parameterizedModelRoutes[$range.Name] = [Collections.Generic.List[object]]::new()
                }
                $parameterizedModelRoutes[$range.Name].Add([PSCustomObject]@{
                    ParameterIndex = $aliasParameterIndex
                    QueryStart = $queryStart
                    QueryEnd = $queryEnd
                    QueryParameterIndex = $(if ($queryEnd -eq $queryStart + 1 -and $tokens[$queryStart].Kind -eq 'word') {
                        [Array]::IndexOf([string[]]$range.Parameters, [string]$tokens[$queryStart].Value)
                    } else { -1 })
                    Range = $range
                })
            }
            if ($alias -and $queryEnd -eq $queryStart + 1 -and $tokens[$queryStart].Kind -eq 'word') {
                $parameterIndex = [Array]::IndexOf([string[]]$range.Parameters, [string]$tokens[$queryStart].Value)
                if ($parameterIndex -ge 0) {
                    if (-not $wrapperParameterRoutes.ContainsKey($range.Name)) { $wrapperParameterRoutes[$range.Name] = @{} }
                    $wrapperParameterRoutes[$range.Name][$parameterIndex] = $alias
                }
            }
        }
    }

    foreach ($wrapperName in $parameterizedModelRoutes.Keys) {
        for ($index = 0; $index -lt $tokens.Count - 2; $index++) {
            if ($tokens[$index].Value -ne $wrapperName -or $tokens[$index + 1].Value -ne '(' -or
                ($index -gt 0 -and $tokens[$index - 1].Value -eq 'function')) {
                continue
            }
            $arguments = [Collections.Generic.List[object]]::new()
            $argumentStart = $index + 2
            $roundDepth = $squareDepth = $curlyDepth = 0
            for ($cursor = $argumentStart; $cursor -lt $tokens.Count; $cursor++) {
                $value = $tokens[$cursor].Value
                if ($value -eq '(') { $roundDepth++ }
                elseif ($value -eq ')') {
                    if ($roundDepth -eq 0 -and $squareDepth -eq 0 -and $curlyDepth -eq 0) {
                        $arguments.Add([PSCustomObject]@{ Start = $argumentStart; End = $cursor })
                        break
                    }
                    else { $roundDepth-- }
                }
                elseif ($value -eq '[') { $squareDepth++ }
                elseif ($value -eq ']') { $squareDepth-- }
                elseif ($value -eq '{') { $curlyDepth++ }
                elseif ($value -eq '}') { $curlyDepth-- }
                elseif ($value -eq ',' -and $roundDepth -eq 0 -and $squareDepth -eq 0 -and $curlyDepth -eq 0) {
                    $arguments.Add([PSCustomObject]@{ Start = $argumentStart; End = $cursor })
                    $argumentStart = $cursor + 1
                }
            }
            $callerRange = Get-AppDaxFunctionAt $tokens[$index].Start
            foreach ($route in $parameterizedModelRoutes[$wrapperName]) {
                if ($route.ParameterIndex -ge $arguments.Count) { continue }
                $aliasArgument = $arguments[$route.ParameterIndex]
                if ($aliasArgument.End -ne $aliasArgument.Start + 1) { continue }
                $alias = Resolve-AppDaxAlias $tokens[$aliasArgument.Start]
                if (-not $alias) { continue }
                if ($route.QueryParameterIndex -ge 0 -and $route.QueryParameterIndex -lt $arguments.Count) {
                    $queryArgument = $arguments[$route.QueryParameterIndex]
                    Add-AppDaxExpressionRoute $alias $queryArgument.Start $queryArgument.End $callerRange
                }
                else {
                    Add-AppDaxExpressionRoute $alias $route.QueryStart $route.QueryEnd $route.Range
                }
            }
        }
    }

    foreach ($wrapperName in $wrapperParameterRoutes.Keys) {
        for ($index = 0; $index -lt $tokens.Count - 2; $index++) {
            if ($tokens[$index].Value -ne $wrapperName -or $tokens[$index + 1].Value -ne '(' -or
                ($index -gt 0 -and $tokens[$index - 1].Value -eq 'function')) {
                continue
            }
            $range = Get-AppDaxFunctionAt $tokens[$index].Start
            $argumentIndex = 0
            $argumentStart = $index + 2
            $roundDepth = $squareDepth = $curlyDepth = 0
            for ($cursor = $argumentStart; $cursor -lt $tokens.Count; $cursor++) {
                $value = $tokens[$cursor].Value
                if ($value -eq '(') { $roundDepth++ }
                elseif ($value -eq ')') {
                    if ($roundDepth -eq 0 -and $squareDepth -eq 0 -and $curlyDepth -eq 0) {
                        if ($wrapperParameterRoutes[$wrapperName].ContainsKey($argumentIndex)) {
                            Add-AppDaxExpressionRoute $wrapperParameterRoutes[$wrapperName][$argumentIndex] $argumentStart $cursor $range
                        }
                        break
                    }
                    $roundDepth--
                }
                elseif ($value -eq '[') { $squareDepth++ }
                elseif ($value -eq ']') { $squareDepth-- }
                elseif ($value -eq '{') { $curlyDepth++ }
                elseif ($value -eq '}') { $curlyDepth-- }
                elseif ($value -eq ',' -and $roundDepth -eq 0 -and $squareDepth -eq 0 -and $curlyDepth -eq 0) {
                    if ($wrapperParameterRoutes[$wrapperName].ContainsKey($argumentIndex)) {
                        Add-AppDaxExpressionRoute $wrapperParameterRoutes[$wrapperName][$argumentIndex] $argumentStart $cursor $range
                    }
                    $argumentIndex++
                    $argumentStart = $cursor + 1
                }
            }
        }
    }

    function Add-AppDaxConnectionRoutes {
        # Track query and connection parameters together through Rayfin hooks,
        # including declared defaults and explicit per-call overrides.
        $routeFunctions = @{}
        $routeParents = @{}
        $routeParameters = @{}
        $routeSummaries = @{}
        $routeDeclarations = [Collections.Generic.Dictionary[string,object]]::new([StringComparer]::Ordinal)
        $routeBuilders = [Collections.Generic.Dictionary[string,object]]::new([StringComparer]::Ordinal)
        $routeBuilderQueries = @{}
        $routeReturns = @{}
        $routeShadowedParameters = @{}
        $routeExpressions = [Collections.Generic.Dictionary[string,object]]::new([StringComparer]::Ordinal)
        $routePendingExpressions = [Collections.Generic.Queue[object]]::new()
        $routePendingValues = [Collections.Generic.Queue[object]]::new()
        $routeScopeCache = @{}

        function Get-AppRouteScope([int]$Position) {
            if (-not $routeScopeCache.ContainsKey($Position)) {
                $owner = Get-AppDaxFunctionAt $Position
                $routeScopeCache[$Position] = if ($owner) { [string]$owner.Start } else { '-1' }
            }
            $routeScopeCache[$Position]
        }

        function Get-AppRouteArguments([int]$Opening) {
            $argumentStart = $Opening + 1
            $round = $square = $curly = 0
            for ($cursor = $argumentStart; $cursor -lt $tokens.Count; $cursor++) {
                $value = $tokens[$cursor].Value
                if ($value -eq '(') { $round++ }
                elseif ($value -eq ')') {
                    if ($round -eq 0 -and $square -eq 0 -and $curly -eq 0) {
                        [PSCustomObject]@{ Start = $argumentStart; End = $cursor }
                        return
                    }
                    $round--
                }
                elseif ($value -eq '[') { $square++ }
                elseif ($value -eq ']') { $square-- }
                elseif ($value -eq '{') { $curly++ }
                elseif ($value -eq '}') { $curly-- }
                elseif ($value -eq ',' -and $round -eq 0 -and $square -eq 0 -and $curly -eq 0) {
                    [PSCustomObject]@{ Start = $argumentStart; End = $cursor }
                    $argumentStart = $cursor + 1
                }
            }
        }

        function Get-AppRouteValueEnd([int]$Start) {
            $depth = 0
            for ($cursor = $Start; $cursor -lt $tokens.Count; $cursor++) {
                if ($tokens[$cursor].Kind -ne 'punct') { continue }
                $value = $tokens[$cursor].Value
                if ($depth -eq 0 -and $value -in @(',', ';', ')', ']', '}')) { return $cursor }
                if ($value -in @('(', '[', '{')) { $depth++ }
                elseif ($value -in @(')', ']', '}')) { $depth-- }
            }
            $tokens.Count
        }

        foreach ($function in $functionRanges) {
            $key = [string]$function.Start
            $routeFunctions[$key] = $function
            $parent = $functionParents[$key]
            $routeParents[$key] = $parent
            $builderKey = $parent + [char]0 + $function.Name
            if (-not $routeBuilders.ContainsKey($builderKey)) {
                $routeBuilders[$builderKey] = [Collections.Generic.List[string]]::new()
            }
            $routeBuilders[$builderKey].Add($key)
            $routeSummaries[$key] = [Collections.Generic.HashSet[string]]::new([StringComparer]::Ordinal)
            $parameters = [Collections.Generic.List[object]]::new()
            $supported = $true
            foreach ($argument in @(Get-AppRouteArguments $function.ParameterOpen)) {
                if ($argument.Start -eq $argument.End) { continue }
                if ($tokens[$argument.Start].Kind -ne 'word' -or
                    ($argument.End -gt $argument.Start + 1 -and $tokens[$argument.Start + 1].Value -ne '=')) {
                    $supported = $false
                    break
                }
                $default = if ($argument.End -gt $argument.Start + 1) {
                    [PSCustomObject]@{ Start = $argument.Start + 2; End = $argument.End }
                } else { $null }
                $parameters.Add([PSCustomObject]@{ Name = $tokens[$argument.Start].Value; Default = $default })
            }
            if ($supported) {
                $routeParameters[$key] = $parameters.ToArray()
            }
            else {
                # Local expressions are still usable, but unsupported bindings must shadow outer symbols.
                $routeShadowedParameters[$key] = [Collections.Generic.HashSet[string]]::new([StringComparer]::Ordinal)
                foreach ($name in $function.Parameters) { $null = $routeShadowedParameters[$key].Add($name) }
            }
        }

        for ($index = 0; $index -lt $tokens.Count - 1; $index++) {
            if ($tokens[$index].Kind -ne 'word' -or $tokens[$index].Value -cne 'return') { continue }
            $scope = Get-AppRouteScope $tokens[$index].Start
            if (-not $routeFunctions.ContainsKey($scope)) { continue }
            if (-not $routeReturns.ContainsKey($scope)) {
                $routeReturns[$scope] = [Collections.Generic.List[object]]::new()
            }
            $routeReturns[$scope].Add([PSCustomObject]@{
                Start = $index + 1
                End = Get-AppRouteValueEnd ($index + 1)
            })
        }

        for ($index = 1; $index -lt $tokens.Count - 2; $index++) {
            if ($tokens[$index].Kind -ne 'word' -or $tokens[$index + 1].Value -ne '=' -or
                $tokens[$index - 1].Value -cnotin @('const', 'let', 'var', ',')) { continue }
            $scope = Get-AppRouteScope $tokens[$index].Start
            if ($scope -ne '-1' -and $index -lt $routeFunctions[$scope].OpenIndex) { continue }
            $routeDeclarations[$scope + [char]0 + $tokens[$index].Value] = [PSCustomObject]@{
                Start = $index + 2
                End = Get-AppRouteValueEnd ($index + 2)
            }
        }

        function Find-AppRouteSymbol([string]$Name, [string]$Scope, $Table) {
            while ($true) {
                $key = $Scope + [char]0 + $Name
                if ($Table.ContainsKey($key)) {
                    return [PSCustomObject]@{ Scope = $Scope; Value = $Table[$key] }
                }
                if ($Scope -eq '-1') { return }
                $Scope = $routeParents[$Scope]
            }
        }

        foreach ($query in $orderedQueries) {
            $owner = Get-AppRouteScope $query.start
            if ($owner -eq '-1') { continue }
            if (-not $routeBuilderQueries.ContainsKey($owner)) {
                $routeBuilderQueries[$owner] = [Collections.Generic.HashSet[string]]::new([StringComparer]::Ordinal)
            }
            $null = $routeBuilderQueries[$owner].Add('query|' + $query.start)
        }

        function Get-AppRouteExpression([int]$Start, [int]$End, [string]$Scope, [string]$Kind) {
            $key = $Scope + [char]0 + $Kind + [char]0 + $Start + ':' + $End
            if (-not $routeExpressions.ContainsKey($key)) {
                $node = [PSCustomObject]@{
                    Key = $key
                    Start = $Start
                    End = $End
                    Scope = $Scope
                    Kind = $Kind
                    Values = [Collections.Generic.HashSet[string]]::new([StringComparer]::Ordinal)
                    Dependents = [Collections.Generic.Dictionary[string,bool]]::new([StringComparer]::Ordinal)
                }
                $routeExpressions[$key] = $node
                $routePendingExpressions.Enqueue($node)
            }
            $routeExpressions[$key]
        }

        function Add-AppRouteValue($Node, [string]$Value) {
            if ($Node.Values.Add($Value)) {
                $routePendingValues.Enqueue([PSCustomObject]@{ Node = $Node; Value = $Value })
            }
        }

        function Add-AppRouteDependency($Parent, $Child, [bool]$QueriesOnly = $false) {
            if (-not $Child.Dependents.ContainsKey($Parent.Key)) {
                $Child.Dependents[$Parent.Key] = $QueriesOnly
            }
            elseif (-not $QueriesOnly) {
                $Child.Dependents[$Parent.Key] = $false
            }
            foreach ($value in $Child.Values) {
                if (-not $Child.Dependents[$Parent.Key] -or $value.StartsWith('query|')) {
                    Add-AppRouteValue $Parent $value
                }
            }
        }

        function Get-AppRouteValues([int]$Start, [int]$End, [string]$Scope, [string]$Kind) {
            if ($Start -ge $End) { return }
            $root = Get-AppRouteExpression $Start $End $Scope $Kind
            # Build each expression once; cyclic dependencies converge by propagating only new facts.
            while ($routePendingExpressions.Count) {
                $node = $routePendingExpressions.Dequeue()
                if ($node.Kind -eq 'alias') {
                    if ($node.End -ne $node.Start + 1) { continue }
                    if ($tokens[$node.Start].Kind -eq 'string') {
                        if ($tokens[$node.Start].Value -match '^[a-zA-Z_$][\w$]*$') {
                            Add-AppRouteValue $node ('alias|' + $tokens[$node.Start].Value)
                        }
                        continue
                    }
                }
                for ($cursor = $node.Start; $cursor -lt $node.End; $cursor++) {
                    $token = $tokens[$cursor]
                    if ($node.Kind -eq 'query' -and $token.Kind -in @('string', 'template')) {
                        foreach ($query in $queriesByToken[$cursor]) {
                            Add-AppRouteValue $node ('query|' + $query.start)
                        }
                    }
                    if ($token.Kind -ne 'word' -or
                        ($cursor -gt $node.Start -and $tokens[$cursor - 1].Value -in @('.', '?.'))) { continue }
                    if ($routeShadowedParameters.ContainsKey($node.Scope) -and $routeShadowedParameters[$node.Scope].Contains($token.Value)) { continue }
                    $parameterIndex = -1
                    if ($routeParameters.ContainsKey($node.Scope)) {
                        for ($p = 0; $p -lt $routeParameters[$node.Scope].Count; $p++) {
                            if ($routeParameters[$node.Scope][$p].Name -ceq $token.Value) { $parameterIndex = $p; break }
                        }
                    }
                    if ($parameterIndex -ge 0) {
                        Add-AppRouteValue $node ('parameter|' + $parameterIndex)
                        continue
                    }
                    $declaration = Find-AppRouteSymbol $token.Value $node.Scope $routeDeclarations
                    if ($declaration) {
                        $child = Get-AppRouteExpression $declaration.Value.Start $declaration.Value.End $declaration.Scope $node.Kind
                        Add-AppRouteDependency $node $child
                        continue
                    }
                    if ($node.Kind -eq 'query' -and $cursor + 1 -lt $node.End -and $tokens[$cursor + 1].Value -eq '(') {
                        $builder = Find-AppRouteSymbol $token.Value $node.Scope $routeBuilders
                        if ($builder -and $builder.Value.Count -eq 1) {
                            $builderScope = $builder.Value[0]
                            if ($routeBuilderQueries.ContainsKey($builderScope)) {
                                foreach ($value in $routeBuilderQueries[$builderScope]) { Add-AppRouteValue $node $value }
                            }
                            foreach ($returnRange in $routeReturns[$builderScope]) {
                                if ($returnRange.Start -ge $returnRange.End) { continue }
                                $child = Get-AppRouteExpression $returnRange.Start $returnRange.End $builderScope 'query'
                                Add-AppRouteDependency $node $child $true
                            }
                        }
                    }
                }
            }
            while ($routePendingValues.Count) {
                $entry = $routePendingValues.Dequeue()
                foreach ($dependent in $entry.Node.Dependents.GetEnumerator()) {
                    if (-not $dependent.Value -or $entry.Value.StartsWith('query|')) {
                        Add-AppRouteValue $routeExpressions[$dependent.Key] $entry.Value
                    }
                }
            }
            foreach ($value in $root.Values) { $value }
        }

        for ($index = 0; $index -lt $tokens.Count - 4; $index++) {
            if ($tokens[$index].Kind -ne 'punct' -or $tokens[$index].Value -ne '{') { continue }
            $scope = Get-AppRouteScope $tokens[$index].Start
            if (-not $routeSummaries.ContainsKey($scope)) { continue }
            $properties = [Collections.Generic.Dictionary[string,object]]::new([StringComparer]::Ordinal)
            $cursor = $index + 1
            $closed = $false
            while ($cursor -lt $tokens.Count) {
                if ($tokens[$cursor].Value -eq '}') { $closed = $true; break }
                if ($tokens[$cursor].Kind -notin @('word', 'string')) { break }
                $name = $tokens[$cursor].Value
                if ($cursor + 1 -lt $tokens.Count -and $tokens[$cursor + 1].Value -eq ':') {
                    $start = $cursor + 2
                    $end = Get-AppRouteValueEnd $start
                }
                else { $start = $cursor; $end = $cursor + 1 }
                if ($properties.ContainsKey($name)) { break }
                $properties[$name] = [PSCustomObject]@{ Start = $start; End = $end }
                $cursor = $end
                if ($cursor -lt $tokens.Count -and $tokens[$cursor].Value -eq ',') { $cursor++ }
                elseif ($cursor -ge $tokens.Count -or $tokens[$cursor].Value -ne '}') { break }
            }
            if (-not $closed -or -not $properties.ContainsKey('connection') -or -not $properties.ContainsKey('query')) { continue }
            $aliasRange = $properties['connection']
            $queryRange = $properties['query']
            $aliases = @(Get-AppRouteValues $aliasRange.Start $aliasRange.End $scope 'alias')
            $queryValues = @(Get-AppRouteValues $queryRange.Start $queryRange.End $scope 'query')
            foreach ($alias in $aliases) {
                foreach ($queryValue in $queryValues) {
                    if ($alias.StartsWith('alias|') -and $queryValue.StartsWith('query|')) {
                        $null = $queryAliases[$queryValue.Substring(6)].Add($alias.Substring(6))
                    }
                    else {
                        $null = $routeSummaries[$scope].Add($alias + [char]0 + $queryValue)
                    }
                }
            }
        }

        $callsByCallee = @{}
        $wrapperNames = [Collections.Generic.HashSet[string]]::new([StringComparer]::Ordinal)
        foreach ($key in $routeParameters.Keys) { $null = $wrapperNames.Add($routeFunctions[$key].Name) }
        for ($index = 0; $index -lt $tokens.Count - 2; $index++) {
            $token = $tokens[$index]
            if ($token.Kind -ne 'word' -or -not $wrapperNames.Contains($token.Value) -or
                $tokens[$index + 1].Value -ne '(' -or
                ($index -gt 0 -and $tokens[$index - 1].Value -cin @('function', '.', '?.'))) { continue }
            $scope = Get-AppRouteScope $token.Start
            $builder = Find-AppRouteSymbol $token.Value $scope $routeBuilders
            if ($builder -and $builder.Value.Count -eq 1 -and $routeSummaries.ContainsKey($builder.Value[0])) {
                $callee = $builder.Value[0]
                if (-not $callsByCallee.ContainsKey($callee)) {
                    $callsByCallee[$callee] = [Collections.Generic.List[object]]::new()
                }
                $callsByCallee[$callee].Add([PSCustomObject]@{
                    Caller = $scope
                    Callee = $callee
                    Arguments = @(Get-AppRouteArguments ($index + 1))
                })
            }
        }

        function Resolve-AppRouteArgument([string]$Value, $Call, [string]$Kind) {
            if (-not $Value.StartsWith('parameter|')) { return $Value }
            $index = [int]$Value.Substring(10)
            if ($index -lt $Call.Arguments.Count) {
                $argument = $Call.Arguments[$index]
                if ($argument.Start -lt $argument.End -and
                    -not ($argument.End -eq $argument.Start + 1 -and $tokens[$argument.Start].Value -ceq 'undefined')) {
                    Get-AppRouteValues $argument.Start $argument.End $Call.Caller $Kind
                    return
                }
            }
            $default = $routeParameters[$Call.Callee][$index].Default
            if ($null -ne $default) {
                Get-AppRouteValues $default.Start $default.End $routeParents[$Call.Callee] $Kind
            }
        }

        $pendingRoutes = [Collections.Generic.Queue[object]]::new()
        foreach ($callee in $routeSummaries.Keys) {
            foreach ($route in $routeSummaries[$callee]) {
                $pendingRoutes.Enqueue([PSCustomObject]@{ Callee = $callee; Route = $route })
            }
        }
        while ($pendingRoutes.Count) {
            $entry = $pendingRoutes.Dequeue()
            $pair = $entry.Route.Split([char]0)
            foreach ($call in $callsByCallee[$entry.Callee]) {
                $aliases = @(Resolve-AppRouteArgument $pair[0] $call 'alias')
                $queryValues = @(Resolve-AppRouteArgument $pair[1] $call 'query')
                foreach ($alias in $aliases) {
                    foreach ($queryValue in $queryValues) {
                        if ($alias.StartsWith('alias|') -and $queryValue.StartsWith('query|')) {
                            $null = $queryAliases[$queryValue.Substring(6)].Add($alias.Substring(6))
                        }
                        elseif ($routeSummaries.ContainsKey($call.Caller)) {
                            $route = $alias + [char]0 + $queryValue
                            if ($routeSummaries[$call.Caller].Add($route)) {
                                $pendingRoutes.Enqueue([PSCustomObject]@{ Callee = $call.Caller; Route = $route })
                            }
                        }
                    }
                }
            }
        }
    }

    Add-AppDaxConnectionRoutes

    $ownerAliasCache = [Collections.Generic.Dictionary[string,object]]::new([StringComparer]::Ordinal)
    $routeMatchCache = [Collections.Generic.Dictionary[string,object]]::new([StringComparer]::Ordinal)
    foreach ($query in $orderedQueries) {
        $queryPrefixStart = [Math]::Max(0, $query.start - 500)
        $queryPrefix = $Source.Substring($queryPrefixStart, $query.start - $queryPrefixStart)
        $modelObjectMatches = [regex]::Matches($queryPrefix, 'model\s*:\s*["''](?<route>[a-zA-Z_$][\w$]*)["'']\s*,\s*dax\s*:\s*$')
        if ($modelObjectMatches.Count -gt 0) {
            $null = $queryAliases[[string]$query.start].Add(
                $modelObjectMatches[$modelObjectMatches.Count - 1].Groups['route'].Value
            )
        }
        $wrapperMatches = [regex]::Matches($queryPrefix, '(?<![\w$])[a-zA-Z_$][\w$]*\s*\(\s*["''](?<route>[a-zA-Z_$][\w$]*)["'']\s*,')
        if ($wrapperMatches.Count -gt 0) {
            $null = $queryAliases[[string]$query.start].Add(
                $wrapperMatches[$wrapperMatches.Count - 1].Groups['route'].Value
            )
        }
        $owner = $queryOwnerByStart[[string]$query.start]
        if ($owner -and $owner -notlike 'query_*') {
            if (-not $ownerAliasCache.ContainsKey($owner)) {
                $ownerAliases = [Collections.Generic.HashSet[string]]::new([StringComparer]::Ordinal)
                $ownerPattern = '(?<![\w$])' + [regex]::Escape($owner) + '(?![\w$])'
                foreach ($callMatch in [regex]::Matches($Source, $ownerPattern + '\s*\(\s*["''](?<route>[a-zA-Z_$][\w$]*)["'']\s*,')) {
                    $null = $ownerAliases.Add($callMatch.Groups['route'].Value)
                }
                foreach ($ownerCall in [regex]::Matches($Source, $ownerPattern + '\s*\(')) {
                    $assignmentStart = [Math]::Max(0, $ownerCall.Index - 600)
                    $assignmentPrefix = $Source.Substring($assignmentStart, $ownerCall.Index - $assignmentStart)
                    $assignmentMatches = [regex]::Matches($assignmentPrefix, '(?<symbol>[a-zA-Z_$][\w$]*)\s*=')
                    if ($assignmentMatches.Count -eq 0) { continue }
                    $symbol = $assignmentMatches[$assignmentMatches.Count - 1].Groups['symbol'].Value
                    if (-not $routeMatchCache.ContainsKey($symbol)) {
                        $symbolPattern = [regex]::Escape($symbol)
                        $routePattern = 'connection\s*:\s*["''](?<route>[a-zA-Z_$][\w$]*)["'']\s*,\s*query\s*:\s*' +
                            $symbolPattern + '(?![\w$])'
                        $routeMatchCache[$symbol] = [regex]::Matches($Source, $routePattern)
                    }
                    foreach ($routeMatch in $routeMatchCache[$symbol]) {
                        $null = $ownerAliases.Add($routeMatch.Groups['route'].Value)
                    }
                }
                $ownerAliasCache[$owner] = $ownerAliases
            }
            foreach ($alias in $ownerAliasCache[$owner]) {
                $null = $queryAliases[[string]$query.start].Add($alias)
            }
        }
        $aliases = @($queryAliases[[string]$query.start] | Sort-Object)
        $query | Add-Member -NotePropertyName connectionAliases -NotePropertyValue $aliases
    }
    [PSCustomObject]@{
        queries = $orderedQueries
        scriptReferences = @($references)
        modelMappings = @($mappings.ToArray())
    }
}

function Initialize-FabricAppDaxParser {
    [CmdletBinding()]
    param(
        [string]$RuntimePath = (Join-Path $env:LOCALAPPDATA 'ImpactIQ/AppDaxParser'),
        [ValidateSet('PowerShell', 'Acorn')][string]$Engine = 'PowerShell'
    )

    $ErrorActionPreference = 'Stop'
    if ($Engine -eq 'PowerShell') { return [PSCustomObject]@{ Engine = 'PowerShell' } }
    $node = Get-Command node -ErrorAction Stop
    $nodeVersion = Invoke-ImpactIQNativeCommand -FilePath $node.Source -ArgumentList @('--version')
    if ($LASTEXITCODE -ne 0 -or $nodeVersion -notmatch '^v(\d+)\.' -or [int]$Matches[1] -lt 18) {
        throw 'Fabric app query extraction requires Node.js 18 or newer.'
    }
    $modulePath = Join-Path $RuntimePath 'node_modules'
    $metadataPath = Join-Path $modulePath 'acorn/package.json'
    $installedVersion = if (Test-Path -LiteralPath $metadataPath) {
        (Get-Content -LiteralPath $metadataPath -Raw -ErrorAction Stop | ConvertFrom-Json).version
    }
    if ($installedVersion -ne '8.15.0') {
        $npm = Get-Command npm.cmd -ErrorAction Stop
        Invoke-ImpactIQNativeCommand -FilePath $npm.Source -ArgumentList @('install', '--prefix', $RuntimePath, '--ignore-scripts', '--no-audit', '--no-fund', '--package-lock=false', 'acorn@8.15.0') | Out-Host
        if ($LASTEXITCODE -ne 0) { throw 'Could not install the pinned Acorn parser.' }
    }
    $parserSource = @'
const fs = require('node:fs');
const path = require('node:path');
const { createHash } = require('node:crypto');
const { parse } = require('acorn');

function extractDax(source) {
    const tree = parse(source, { ecmaVersion: 'latest', sourceType: 'module', locations: true });
    const queries = [];
    const functions = new Map();
    const scriptReferences = new Set();
    const modelMappings = [];
    const propertyName = property => property.type === 'Property' && !property.computed
        ? (property.key.name ?? property.key.value) : undefined;
    const literalProperty = (object, name) => {
        const properties = object.properties.filter(property => propertyName(property) === name);
        return properties.length === 1 && properties[0].value.type === 'Literal'
            && typeof properties[0].value.value === 'string' ? properties[0].value.value : undefined;
    };
    const guid = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;

    function visit(node, ancestors = []) {
        if (!node || typeof node.type !== 'string') return;
        if (propertyName(node) === 'semanticModels' && node.value.type === 'ObjectExpression') {
            for (const connection of node.value.properties) {
                const alias = propertyName(connection);
                if (typeof alias !== 'string' || connection.value.type !== 'ObjectExpression') continue;
                const workspaceId = literalProperty(connection.value, 'workspaceId');
                const modelId = literalProperty(connection.value, 'itemId');
                if (!guid.test(workspaceId) || !guid.test(modelId)) continue;
                modelMappings.push({ connectionAlias: alias, modelWorkspaceId: workspaceId, modelId,
                    evidence: 'StaticConfiguration', line: connection.loc.start.line, column: connection.loc.start.column + 1 });
            }
        }
        if (node.type === 'FunctionDeclaration' && node.id) {
            const declarations = functions.get(node.id.name) || [];
            declarations.push(node);
            functions.set(node.id.name, declarations);
        }
        let text;
        let expressions = [];
        if (node.type === 'Literal' && typeof node.value === 'string') {
            text = node.value;
            if (/^(?:\.{0,2}\/|https:\/\/)[^\s]+\.m?js(?:\?[^\s]*)?$/i.test(text)) {
                scriptReferences.add(text);
            }
        } else if (node.type === 'TemplateLiteral') {
            expressions = node.expressions.map(expression => source.slice(expression.start, expression.end));
            text = node.quasis.map((part, index) =>
                (part.value.cooked ?? part.value.raw) +
                (index < expressions.length ? '${' + expressions[index] + '}' : '')
            ).join('');
        }
        if (text && /^(?:\s+|(?:\/\/|--)[^\r\n]*(?:\r?\n|$)|\/\*[\s\S]*?\*\/)*(?:EVALUATE|DEFINE)\b/i.test(text)) {
            const owner = [...ancestors].reverse().find(parent =>
                parent.type === 'FunctionDeclaration' || parent.type === 'VariableDeclarator'
            );
            const builder = [...ancestors].reverse().find(parent =>
                ['FunctionDeclaration', 'FunctionExpression', 'ArrowFunctionExpression'].includes(parent.type)
            );
            queries.push({
                symbol: owner?.id?.name || `query_${queries.length + 1}`,
                evidence: expressions.length ? 'DynamicTemplate' : 'StaticCandidate',
                text: text.trim(),
                expressions,
                line: node.loc.start.line,
                column: node.loc.start.column + 1,
                start: node.start,
                end: node.end,
                literalSource: source.slice(node.start, node.end),
                builderSource: builder ? source.slice(builder.start, builder.end) : null,
                enclosingExpression: source.slice((builder || owner || node).start, (builder || owner || node).end)
            });
        }
        for (const value of Object.values(node)) {
            if (Array.isArray(value)) {
                for (const child of value) visit(child, [...ancestors, node]);
            } else if (value && typeof value.type === 'string') {
                visit(value, [...ancestors, node]);
            }
        }
    }

    visit(tree);
    const helpers = [];
    const included = new Set();
    const pending = queries.flatMap(query => query.expressions);
    while (pending.length) {
        const expression = pending.pop();
        const expressionTree = parse('(' + expression + ')', { ecmaVersion: 'latest' });
        function findCalls(node) {
            if (!node || typeof node.type !== 'string') return;
            if (node.type === 'CallExpression' && node.callee.type === 'Identifier') {
                const symbol = node.callee.name;
                const declarations = functions.get(symbol) || [];
                if (!included.has(symbol) && declarations.length === 1) {
                    included.add(symbol);
                    const declaration = declarations[0];
                    const helperSource = source.slice(declaration.start, declaration.end);
                    helpers.push({ symbol, source: helperSource, line: declaration.loc.start.line });
                    pending.push(helperSource);
                }
            }
            for (const value of Object.values(node)) {
                if (Array.isArray(value)) value.forEach(findCalls);
                else if (value && typeof value.type === 'string') findCalls(value);
            }
        }
        findCalls(expressionTree);
    }
    return { queries, helpers, scriptReferences: [...scriptReferences], modelMappings };
}

function exportDax(inputPath, outputPath) {
    const bytes = fs.readFileSync(inputPath);
    const result = extractDax(bytes.toString('utf8'));
    const catalog = {
        sourceFile: path.basename(inputPath),
        sha256: createHash('sha256').update(bytes).digest('hex'),
        extractedAtUtc: new Date().toISOString(),
        coverage: 'Static inspection of the supplied JavaScript only; not proof of executed queries or complete app coverage.',
        ...result
    };
    fs.mkdirSync(outputPath, { recursive: true });
    fs.writeFileSync(path.join(outputPath, 'queries.json'), JSON.stringify(catalog, null, 2) + '\n');
    const sections = [
        '# Fabric App DAX',
        '',
        catalog.coverage,
        '',
        'Dynamic templates retain JavaScript ${...} expressions and are not executable DAX. Static candidates are literal bodies; surrounding JavaScript can still modify them.',
        ''
    ];
    for (const [index, query] of result.queries.entries()) {
        const stem = `${String(index + 1).padStart(2, '0')}-${query.symbol.replace(/[^a-zA-Z0-9_-]/g, '_')}`;
        const extension = query.evidence === 'DynamicTemplate' ? '.dax-template.txt' : '.dax';
        fs.writeFileSync(path.join(outputPath, stem + extension), query.text + '\n');
        sections.push(`## ${index + 1}. ${query.symbol} (${query.evidence})`, '', '```dax', query.text, '```', '');
    }
    if (result.helpers.length) {
        sections.push('## Referenced JavaScript Helpers', '', '```javascript', ...result.helpers.map(helper => helper.source), '```', '');
    }
    fs.writeFileSync(path.join(outputPath, 'queries.md'), sections.join('\n'));
    return catalog;
}

module.exports = { extractDax, exportDax };

if (require.main === module) {
    const [inputPath, outputPath] = process.argv.slice(2);
    if (!inputPath || !outputPath) {
        console.error('Usage: node "Extract Fabric App DAX.cjs" <bundle.js> <output-directory>');
        process.exitCode = 1;
    } else {
        const catalog = exportDax(inputPath, outputPath);
        console.log(JSON.stringify({ queries: catalog.queries.length, helpers: catalog.helpers.length, sha256: catalog.sha256 }));
    }
}
'@
    $null = New-Item -ItemType Directory -Path $RuntimePath -Force

}

function Assert-FabricAppDaxUri {
    param([Parameter(Mandatory)][uri]$Uri, [Parameter(Mandatory)][string]$Suffix)
    if ($Uri.Scheme -ne 'https' -or $Uri.Port -ne 443 -or $Uri.UserInfo -or
        $Uri.Query -or $Uri.Fragment -or -not $Uri.DnsSafeHost.EndsWith($Suffix, [StringComparison]::OrdinalIgnoreCase)) {
        throw 'Unexpected Fabric discovery URL.'
    }
}

function Assert-FabricAppDaxHostingUri {
    param([Parameter(Mandatory)][uri]$Uri)

    Assert-FabricAppDaxUri $Uri '.fabricapps.net'
    $label = '[a-z0-9](?:[a-z0-9-]*[a-z0-9])?'
    if ($Uri.DnsSafeHost -notmatch "^$label\.webapp(?:\.$label)*\.fabricapps\.net$") {
        throw 'Unexpected Fabric hosting URL.'
    }
}

function Get-FabricAppDaxToken {
    param([Parameter(Mandatory)][string]$ResourceUrl)
    try {
        $token = (Get-AzAccessToken -ResourceUrl $ResourceUrl -ErrorAction Stop -WarningAction Continue).Token
        if ($token -is [Security.SecureString]) {
            return [Net.NetworkCredential]::new('', $token).Password
        }
        return [string]$token
    }
    catch { throw 'Azure authentication unavailable. Run Connect-AzAccount for the app tenant and retry.' }
}

function Invoke-FabricAppDaxRequest {
    param(
        [Parameter(Mandatory)][uri]$Uri,
        [string]$Method = 'GET',
        [hashtable]$Headers = @{},
        [string]$Body,
        $Session
    )
    $parameters = @{
        Uri = $Uri
        Method = $Method
        Headers = $Headers
        UseBasicParsing = $true
        MaximumRedirection = 0
        TimeoutSec = 60
        ErrorAction = 'Stop'
    }
    if ($Body) { $parameters.Body = $Body; $parameters.ContentType = 'application/json' }
    if ($Session) { $parameters.WebSession = $Session }
    try {
        $response = Invoke-WebRequest @parameters
        if ($response.StatusCode -ne 200) { throw 'Unexpected response status.' }
        if ($response.RawContentLength -gt 20MB) { throw 'Response exceeds the 20 MB asset limit.' }
        return $response
    }
    catch {
        $status = if ($_.Exception.Response) { [int]$_.Exception.Response.StatusCode } else { 'unavailable' }
        throw "Request failed (HTTP $status). Redirects, oversized responses, and non-200 responses are not accepted."
    }
}

function Resolve-FabricAppModelNames {
    [CmdletBinding()]
    param(
        [AllowEmptyCollection()][object[]]$Mappings = @(),
        [hashtable]$Cache = @{},
        [string]$Environment = 'Public',
        [AllowEmptyCollection()][object[]]$Datasets = @(),
        [AllowEmptyCollection()][object[]]$Workspaces = @()
    )

    $datasetLookup = @{}
    foreach ($dataset in $Datasets) {
        $datasetId = [string]$dataset.DatasetId
        $datasetWorkspaceId = [string]$dataset.WorkspaceId
        if ($datasetId -and $datasetWorkspaceId -and $dataset.DatasetName) {
            $datasetLookup["$datasetWorkspaceId/$datasetId"] = [string]$dataset.DatasetName
        }
    }
    $workspaceLookup = @{}
    foreach ($workspace in $Workspaces) {
        $workspaceId = [string]$workspace.WorkspaceId
        if ($workspaceId -and $workspace.WorkspaceName) {
            $workspaceLookup[$workspaceId] = [string]$workspace.WorkspaceName
        }
    }

    $headers = $null
    try {
        foreach ($mapping in $Mappings) {
            $modelGuid = [guid]::Empty
            $workspaceGuid = [guid]::Empty
            $validIds = [guid]::TryParse([string]$mapping.modelId, [ref]$modelGuid) -and
                [guid]::TryParse([string]$mapping.modelWorkspaceId, [ref]$workspaceGuid)
            $result = [ordered]@{
                connectionAlias = $mapping.connectionAlias
                modelId = $mapping.modelId
                modelWorkspaceId = $mapping.modelWorkspaceId
                evidence = $mapping.evidence
                modelName = ''
                modelWorkspaceName = ''
            }
            if ($validIds) {
                $workspaceId = $workspaceGuid.ToString()
                $modelId = $modelGuid.ToString()
                if ($workspaceLookup.ContainsKey($workspaceId)) {
                    $result.modelWorkspaceName = $workspaceLookup[$workspaceId]
                }
                $datasetKey = "$workspaceId/$modelId"
                if ($datasetLookup.ContainsKey($datasetKey)) {
                    $result.modelName = $datasetLookup[$datasetKey]
                }
            }
            if ($Environment -ne 'Public' -or -not $validIds) {
                [PSCustomObject]$result
                continue
            }
            foreach ($kind in @('Workspace', 'Model')) {
                if (($kind -eq 'Workspace' -and $result.modelWorkspaceName) -or
                    ($kind -eq 'Model' -and $result.modelName)) {
                    continue
                }
                $key = if ($kind -eq 'Workspace') { "Workspace/$workspaceGuid" } else { "Model/$workspaceGuid/$modelGuid" }
                if (-not $Cache.ContainsKey($key)) {
                    $entry = @{ Name = '' }
                    try {
                        if (-not $headers) {
                            $headers = @{ Authorization = ('Bearer ' + (Get-FabricAppDaxToken 'https://api.fabric.microsoft.com')) }
                        }
                        $uri = "https://api.fabric.microsoft.com/v1/workspaces/$workspaceGuid"
                        $expectedId = $workspaceGuid
                        if ($kind -eq 'Model') { $uri += "/items/$modelGuid"; $expectedId = $modelGuid }
                        $metadata = (Invoke-FabricAppDaxRequest -Uri $uri -Headers $headers).Content | ConvertFrom-Json
                        if ([string]$metadata.id -eq $expectedId.ToString() -and
                            -not [string]::IsNullOrWhiteSpace([string]$metadata.displayName) -and
                            ($kind -eq 'Workspace' -or $metadata.type -eq 'SemanticModel')) {
                            $entry.Name = [string]$metadata.displayName
                        }
                    }
                    catch {
                        Write-Warning "Could not resolve Fabric app $kind name for workspace $workspaceGuid and model $modelGuid. $($_.Exception.Message)"
                    }
                    $Cache[$key] = $entry
                }
                $entry = $Cache[$key]
                if ($kind -eq 'Workspace') {
                    $result.modelWorkspaceName = $entry.Name
                }
                else {
                    $result.modelName = $entry.Name
                }
            }
            [PSCustomObject]$result
        }
    }
    finally { $headers = $null }
}

function Get-FabricAppDaxCapture {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)][guid]$WorkspaceId,
        [string]$WorkspaceName,
        [Parameter(Mandatory)][guid]$AppId,
        [string]$AppName,
        [Parameter(Mandatory)]$ParserRuntime,
        [ValidateRange(1, 256)][int]$MaxAssets = 64,
        [string]$Environment = 'Public',
        [AllowEmptyCollection()][object[]]$Datasets = @(),
        [AllowEmptyCollection()][object[]]$Workspaces = @(),
        [hashtable]$ModelNameCache = @{}
    )

    $ErrorActionPreference = 'Stop'
    $queries = [Collections.Generic.List[object]]::new()
    $modelMappings = [Collections.Generic.List[object]]::new()
    $parserEngine = if ($ParserRuntime.Engine -eq 'Acorn') { 'Acorn' } else { 'PowerShell' }
    $temporaryFolder = $null
    $assetCount = 0
    $failedAssets = 0
    $captureError = ''
    $capture = [PSCustomObject][ordered]@{
        WorkspaceId = $WorkspaceId.ToString()
        WorkspaceName = $WorkspaceName
        AppId = $AppId.ToString()
        AppName = $AppName
        CaptureStatus = 'Started'
        CaptureError = ''
        AssetsInspected = 0
        SkippedAssets = 0
    }
    $stage = 'Discovery'
    $fabricToken = $powerBiToken = $mwc = $handoff = $session = $headers = $null
    $previousNodePath = $env:NODE_PATH
    if ($Environment -ne 'Public') {
        $capture.CaptureStatus = 'Failed'
        $capture.CaptureError = 'Fabric App extraction is only supported in the Public cloud.'
        Write-Host "[ERROR] App query extraction failed for $WorkspaceName ~ $AppName`: the preview hosting flow has only been verified in the Public cloud." -ForegroundColor Red
        return [PSCustomObject]@{ Queries = @(); Models = @(); Capture = $capture; Error = 'Fabric App extraction is only supported in the Public cloud.' }
    }
    try {
        if ($parserEngine -eq 'Acorn') {
            $parserPath = $ParserRuntime.ParserPath
            if (-not (Test-Path -LiteralPath $parserPath)) { throw 'JavaScript parser script is missing.' }
            $env:NODE_PATH = $ParserRuntime.ModulePath
        }
        $fabricToken = Get-FabricAppDaxToken 'https://api.fabric.microsoft.com'
        $headers = @{ Authorization = "Bearer $fabricToken" }
        $itemEndpoint = "https://api.fabric.microsoft.com/v1/workspaces/$WorkspaceId/appBackends/$AppId"
        $deployment = (Invoke-FabricAppDaxRequest -Uri "$itemEndpoint/__private/deploy/status" -Headers $headers).Content | ConvertFrom-Json
        $extended = (Invoke-FabricAppDaxRequest -Uri "$itemEndpoint/__private/extended-properties" -Headers $headers).Content | ConvertFrom-Json
        $hosting = [uri]$deployment.hostingUrl
        $workload = [uri]$extended.extendedProperties.BaaSEndpoint
        Assert-FabricAppDaxHostingUri $hosting
        Assert-FabricAppDaxUri $workload '.pbidedicated.windows.net'
        if ($hosting.AbsolutePath -ne '/' -or -not $workload.AbsolutePath.TrimEnd('/').EndsWith("/workspaces/$WorkspaceId/appbackends/$AppId", [StringComparison]::OrdinalIgnoreCase)) {
            throw 'Deployment URLs do not match the requested app.'
        }
        $origin = $hosting.GetLeftPart([UriPartial]::Authority)

        $stage = 'Workload authentication'
        $powerBiToken = Get-FabricAppDaxToken 'https://analysis.windows.net/powerbi/api'
        $headers = @{ Authorization = "Bearer $powerBiToken" }
        $clusterInfo = (Invoke-FabricAppDaxRequest -Uri 'https://api.powerbi.com/metadata/cluster' -Headers $headers).Content | ConvertFrom-Json
        $cluster = [uri]$clusterInfo.backendUrl
        Assert-FabricAppDaxUri $cluster '.analysis.windows.net'
        $headers['X-PowerBI-ResourceKey'] = $WorkspaceId.ToString()
        $claims = @{
            workspaceObjectId = $WorkspaceId.ToString()
            workloadType = 'BaaS'
            artifacts = @(@{ artifactType = 'AppBackend'; artifactObjectId = $AppId.ToString() })
        } | ConvertTo-Json -Depth 4 -Compress
        $mwc = (Invoke-FabricAppDaxRequest -Uri ($cluster.GetLeftPart([UriPartial]::Authority) + '/metadata/v201606/generatemwctokenv2') -Method POST -Headers $headers -Body $claims).Content | ConvertFrom-Json
        if (-not $mwc.Token -or $mwc.TargetUriHost -ne $workload.DnsSafeHost) { throw 'Workload token target mismatch.' }

        $headers = @{ Authorization = ('MwcToken ' + $mwc.Token); 'x-ms-workload-resource-moniker' = $AppId.ToString() }
        $runtimeSettings = $null
        try {
            $runtimeSettings = (Invoke-FabricAppDaxRequest -Uri ($workload.AbsoluteUri.TrimEnd('/') + '/api/projectRuntimeSettings') -Headers $headers).Content | ConvertFrom-Json
            foreach ($connector in $runtimeSettings.serviceSettings.connectors.PSObject.Properties) {
                if ($connector.Value.connector -cne 'fabric-semanticmodel') { continue }
                $modelGuid = [guid]::Empty
                $modelWorkspaceGuid = [guid]::Empty
                if (-not [guid]::TryParse([string]$connector.Value.config.itemId, [ref]$modelGuid) -or
                    -not [guid]::TryParse([string]$connector.Value.config.workspaceId, [ref]$modelWorkspaceGuid) -or
                    $modelGuid -eq [guid]::Empty -or $modelWorkspaceGuid -eq [guid]::Empty) {
                    continue
                }
                $modelMappings.Add([PSCustomObject]@{
                    connectionAlias = $connector.Name
                    modelId = $modelGuid.ToString()
                    modelWorkspaceId = $modelWorkspaceGuid.ToString()
                    evidence = 'BackendConnectorConfiguration'
                })
            }
        }
        catch {
        }
        finally { $runtimeSettings = $null }

        $stage = 'Hosting authentication'
        $random = [Security.Cryptography.RandomNumberGenerator]::Create()
        $sha = [Security.Cryptography.SHA256]::Create()
        try {
            $bytes = New-Object byte[] 32
            $random.GetBytes($bytes)
            $verifier = [Convert]::ToBase64String($bytes).TrimEnd('=').Replace('+', '-').Replace('/', '_')
            $challenge = [Convert]::ToBase64String($sha.ComputeHash([Text.Encoding]::ASCII.GetBytes($verifier))).TrimEnd('=').Replace('+', '-').Replace('/', '_')
        }
        finally { $random.Dispose(); $sha.Dispose() }
        $state = [guid]::NewGuid().ToString('N')
        $body = @{ returnOrigin = $origin; codeChallenge = $challenge; codeChallengeMethod = 'S256'; state = $state } | ConvertTo-Json -Compress
        $headers = @{ Authorization = ('MwcToken ' + $mwc.Token); 'x-ms-workload-resource-moniker' = $AppId.ToString() }
        $handoff = (Invoke-FabricAppDaxRequest -Uri ($workload.AbsoluteUri.TrimEnd('/') + '/api/auth/v1/brokered/authorize') -Method POST -Headers $headers -Body $body).Content | ConvertFrom-Json
        if (-not $handoff.handoffCode -or $handoff.state -cne $state) { throw 'Hosting handoff state mismatch.' }
        $session = New-Object Microsoft.PowerShell.Commands.WebRequestSession
        $headers = @{ Accept = 'application/json'; 'x-rayfin-sh-code-verifier' = $verifier; 'x-ms-workload-resource-moniker' = $AppId.ToString() }
        $null = Invoke-FabricAppDaxRequest -Uri ($origin + '/?_hc=' + [uri]::EscapeDataString($handoff.handoffCode)) -Headers $headers -Session $session
        $headers = @{ Accept = 'text/html' }

        $stage = 'Asset discovery'
        $html = (Invoke-FabricAppDaxRequest -Uri "$origin/" -Headers $headers -Session $session).Content
        if ($html -match '<title>Sign in required</title>') { throw 'Hosting session did not serve the app.' }
        $pending = [Collections.Generic.Queue[uri]]::new()
        $seen = [Collections.Generic.HashSet[string]]::new([StringComparer]::Ordinal)
        $scripts = [regex]::Matches($html, '(?is)<script\b(?<attributes>[^>]*)>(?<body>.*?)</script\s*>')
        foreach ($script in $scripts) {
            $source = [regex]::Match($script.Groups['attributes'].Value, '\bsrc\s*=\s*["''](?<url>[^"'']+)["'']', 'IgnoreCase')
            if ($source.Success) {
                $asset = [uri]::new([uri]("$origin/"), [Net.WebUtility]::HtmlDecode($source.Groups['url'].Value))
                $pending.Enqueue($asset)
            }
            elseif ($script.Groups['body'].Value.Trim()) { $failedAssets++ }
        }
        if ($scripts.Count -eq 0) { $failedAssets++ }
        if ($html -match '(?i)<base\b') { $failedAssets++ }
        while ($pending.Count -gt 0 -and $assetCount -lt $MaxAssets) {
            $asset = $pending.Dequeue()
            if (-not $seen.Add($asset.AbsoluteUri)) { continue }
            if ($asset.GetLeftPart([UriPartial]::Authority) -ne $origin -or $asset.UserInfo -or $asset.Query -or $asset.Fragment) {
                $failedAssets++
                continue
            }
            $assetCount++
            try {
                $response = Invoke-FabricAppDaxRequest -Uri $asset -Session $session
                if ($response.Headers['Content-Type'] -match 'text/html') { throw 'Expected JavaScript, not HTML.' }
                $content = if ($response.Content -is [byte[]]) { [Text.Encoding]::UTF8.GetString($response.Content) } else { [string]$response.Content }
                if ($parserEngine -eq 'PowerShell') {
                    $catalog = Get-FabricAppDaxSource -Source $content
                }
                else {
                    if (-not $temporaryFolder) {
                        $temporaryFolder = Join-Path ([IO.Path]::GetTempPath()) ('ImpactIQ-AppDax-' + [guid]::NewGuid().ToString('N'))
                    }
                    $assetFolder = Join-Path $temporaryFolder ("asset-{0:D3}" -f $assetCount)
                    $null = New-Item -ItemType Directory -Path $assetFolder -Force
                    $bundlePath = Join-Path $assetFolder 'bundle.js'
                    [IO.File]::WriteAllText($bundlePath, $content, [Text.UTF8Encoding]::new($false))
                    Invoke-ImpactIQNativeCommand -FilePath $ParserRuntime.NodePath -ArgumentList @($parserPath, $bundlePath, $assetFolder) -DiscardOutput
                    if ($LASTEXITCODE -ne 0) { throw 'JavaScript parsing failed.' }
                    $catalog = Get-Content -LiteralPath (Join-Path $assetFolder 'queries.json') -Raw | ConvertFrom-Json
                }
                foreach ($mapping in $catalog.modelMappings) {
                    $backendMappings = @($modelMappings | Where-Object {
                        $_.evidence -eq 'BackendConnectorConfiguration' -and $_.connectionAlias -ceq $mapping.connectionAlias
                    })
                    if ($backendMappings.Count -eq 0) { $modelMappings.Add($mapping) }
                }
                foreach ($row in @(ConvertTo-FabricAppQueryRows -Catalog $catalog -WorkspaceId $WorkspaceId -WorkspaceName $WorkspaceName -AppId $AppId -AppName $AppName -SourceUrl $asset.AbsoluteUri)) {
                    $queries.Add($row)
                }
                foreach ($reference in $catalog.scriptReferences) { $pending.Enqueue([uri]::new($asset, $reference)) }
            }
            catch { $failedAssets++ }
        }
        if ($failedAssets -gt 0 -or $pending.Count -gt 0) {
            $captureError = 'Some scripts were skipped or unavailable, or the asset limit was reached. Inline scripts and base URLs need manual review.'
        }
        elseif ($queries.Count -eq 0) {
            $captureError = 'No DAX literals were found in inspected scripts; this does not prove absence of queries.'
        }
    }
    catch {
        $captureError = "$stage failed. Check sign-in, app permissions, prerequisites, and preview API availability."
        Write-Host "[WARN] Fabric App '$AppName': $captureError" -ForegroundColor Yellow
    }
    finally {
        $env:NODE_PATH = $previousNodePath
        $fabricToken = $powerBiToken = $mwc = $handoff = $session = $headers = $verifier = $null
        if ($temporaryFolder -and (Test-Path -LiteralPath $temporaryFolder)) {
            try { Remove-Item -LiteralPath $temporaryFolder -Recurse -Force -ErrorAction Stop }
            catch {
                $captureError = 'Temporary app data could not be removed. Remove the temporary directory reported in the console.'
                Write-Warning "Temporary app data cleanup failed: $temporaryFolder" -WarningAction Continue
            }
        }
    }
    $configuredModels = @($modelMappings |
        Where-Object { $_.connectionAlias -and $_.modelId -and $_.modelWorkspaceId } |
        Select-Object connectionAlias, modelId, modelWorkspaceId, evidence -Unique)
    $configuredModels = @(Resolve-FabricAppModelNames -Mappings $configuredModels -Cache $ModelNameCache -Environment $Environment -Datasets $Datasets -Workspaces $Workspaces)
    $configuredAliases = @($configuredModels.connectionAlias | Select-Object -Unique)
    $queryRows = @($queries.ToArray())
    foreach ($queryRow in $queryRows) {
        $queryAlias = [string]$queryRow.ConnectionAlias
        if (-not $queryAlias -or $configuredAliases -ccontains $queryAlias) { continue }
        $aliasMatches = @($configuredAliases | Where-Object { $_ -ieq $queryAlias })
        if ($aliasMatches.Count -eq 1) {
            $queryRow.ConnectionAlias = $aliasMatches[0]
        }
    }
    if ($configuredAliases.Count -gt 0) {
        $queryRows = @($queryRows |
            Group-Object {
                $text = @($_.PSObject.Properties |
                    Where-Object { $_.Name -match '^QueryText(\d+)?$' } |
                    Sort-Object {
                        if ($_.Name -eq 'QueryText') { 1 }
                        else { [int]$_.Name.Substring(9) }
                    } |
                    ForEach-Object { [string]$_.Value }) -join ''
                ([string]$_.SourceUrl) + [char]0 + ([string]$_.QuerySymbol) + [char]0 + $text
            } |
            ForEach-Object {
                $validRows = @($_.Group | Where-Object {
                    $_.ConnectionAlias -and $configuredAliases -ccontains $_.ConnectionAlias
                })
                if ($validRows.Count -gt 0) {
                    $validRows
                }
                else {
                    $row = $_.Group[0]
                    $row.ConnectionAlias = ''
                    $row
                }
            })
    }
    Set-FabricAppQueryModelMapping -Queries $queryRows -Mappings $configuredModels
    $models = @($configuredModels |
        ForEach-Object {
            $mapping = $_
            [PSCustomObject][ordered]@{
                AppWorkspaceId = $WorkspaceId.ToString()
                AppWorkspaceName = $WorkspaceName
                AppId = $AppId.ToString()
                AppName = $AppName
                ConnectionAlias = $mapping.connectionAlias
                ModelWorkspaceId = $mapping.modelWorkspaceId
                ModelWorkspaceName = $mapping.modelWorkspaceName
                ModelId = $mapping.modelId
                ModelName = $mapping.modelName
                Evidence = $mapping.evidence
            }
        })
    if (-not $captureError -and $models.Count -eq 0) {
        $captureError = 'No semantic-model targets were discovered in backend connector settings or app scripts; query model fields remain blank.'
        Write-Host "[WARN] Fabric App '$AppName': $captureError" -ForegroundColor Yellow
    }
    if (-not $captureError) {
        Write-Host "App query extraction completed for $WorkspaceName ~ $AppName"
    }
    $capture.CaptureStatus = if ($captureError) { 'Partial' } else { 'StaticAssetsInspected' }
    $capture.CaptureError = $captureError
    $capture.AssetsInspected = $assetCount
    $capture.SkippedAssets = $failedAssets
    [PSCustomObject]@{ Queries = $queryRows; Models = $models; Capture = $capture; Error = $captureError }
}

function Set-FabricAppQueryModelMapping {
    [CmdletBinding()]
    param(
        [AllowEmptyCollection()][object[]]$Queries = @(),
        [AllowEmptyCollection()][object[]]$Mappings = @(),
        [AllowEmptyCollection()][object[]]$Datasets = @(),
        [AllowEmptyCollection()][object[]]$Workspaces = @()
    )

    $configuredModels = @($Mappings | Where-Object { $_.connectionAlias -and $_.modelId -and $_.modelWorkspaceId } |
        Sort-Object connectionAlias, modelId, modelWorkspaceId -Unique)
    foreach ($query in $Queries) {
        $mapping = $null
        if ($query.ConnectionAlias) {
            $matches = @($configuredModels | Where-Object { $_.connectionAlias -ceq $query.ConnectionAlias })
            if ($matches.Count -eq 1) {
                $mapping = $matches[0]
            }
        }
        elseif ($configuredModels.Count -eq 1) {
            $mapping = $configuredModels[0]
        }
        $modelName = ''
        $modelWorkspaceName = ''
        if ($mapping) {
            $modelName = [string]$mapping.modelName
            $modelWorkspaceName = [string]$mapping.modelWorkspaceName
        }
        $mappingEvidence = ''
        if ($mapping) {
            $mappingEvidence = if ($query.ConnectionAlias) { 'ConnectionAlias' } else { 'SingleConfiguredModel' }
        }
        $query.ConnectionAlias = if ($mapping) { $mapping.connectionAlias } else { $query.ConnectionAlias }
        $query.ModelId = if ($mapping) { $mapping.modelId } else { '' }
        $query.ModelName = $modelName
        $query.ModelWorkspaceId = if ($mapping) { $mapping.modelWorkspaceId } else { '' }
        $query.ModelWorkspaceName = $modelWorkspaceName
        $query | Add-Member -NotePropertyName ModelMappingEvidence -NotePropertyValue $mappingEvidence -Force
    }
}

function ConvertTo-FabricAppQueryRows {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)]$Catalog,
        [Parameter(Mandatory)][string]$WorkspaceId,
        [string]$WorkspaceName,
        [Parameter(Mandatory)][string]$AppId,
        [string]$AppName,
        [string]$SourceUrl,
        [string]$CatalogPath
    )

    foreach ($query in $Catalog.queries) {
        $connectionAliases = @($query.connectionAliases | Where-Object { $_ } | Select-Object -Unique)
        $aliasesForRows = if ($connectionAliases.Count -gt 0) { $connectionAliases } else { @('') }
        foreach ($connectionAlias in $aliasesForRows) {
            $record = [ordered]@{
                WorkspaceId = $WorkspaceId
                WorkspaceName = $WorkspaceName
                AppId = $AppId
                AppName = $AppName
                QuerySymbol = $query.symbol
                Evidence = $query.evidence
                QueryText = ''
                SourceUrl = $SourceUrl
                ConnectionAlias = $connectionAlias
                ModelId = ''
                ModelName = ''
                ModelWorkspaceId = ''
                ModelWorkspaceName = ''
                ModelMappingEvidence = ''
                TemplateExpressions = @($query.expressions | Where-Object { $null -ne $_ })
            }
            $text = [string]$query.text
            $offset = 0
            $part = 1
            while ($offset -lt $text.Length) {
                $length = [Math]::Min(32000, $text.Length - $offset)
                $nextOffset = $offset + $length
                if ($nextOffset -lt $text.Length -and
                    [char]::IsHighSurrogate($text[$nextOffset - 1]) -and
                    [char]::IsLowSurrogate($text[$nextOffset])) {
                    $length--
                }
                $column = if ($part -eq 1) { 'QueryText' } else { "QueryText$part" }
                $record[$column] = $text.Substring($offset, $length)
                $offset += $length
                $part++
            }
            [PSCustomObject]$record
        }
    }
}

function Export-FabricAppQueryWorkbook {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)][string]$Path,
        [AllowEmptyCollection()][object[]]$Queries = @(),
        [AllowEmptyCollection()][object[]]$Models = @(),
        [AllowEmptyCollection()][object[]]$Captures = @(),
        [AllowEmptyCollection()][object[]]$Dependencies = @(),
        [AllowEmptyCollection()][object[]]$DependencyStatus = @(),
        [switch]$IncludeDependencies
    )

    Import-Module ImportExcel -ErrorAction Stop
    $unmappedQueries = @($Queries | Where-Object { -not $_.ModelId -or -not $_.ModelWorkspaceId })
    Write-Host "[INFO] Fabric App model mapping: $($Queries.Count - $unmappedQueries.Count) of $($Queries.Count) captured query rows resolved."
    if ($unmappedQueries.Count -gt 0) {
        Write-Host "[WARN] $($unmappedQueries.Count) query row(s) remain unresolved after connection tracing. Export will continue with their model fields blank." -ForegroundColor Yellow
        foreach ($query in $unmappedQueries) {
            Write-Host "  $($query.WorkspaceName) ~ $($query.AppName) ~ $($query.QuerySymbol) (connection: '$($query.ConnectionAlias)')" -ForegroundColor Yellow
        }
    }
    $appRows = @($Captures | ForEach-Object {
        [PSCustomObject][ordered]@{
            WorkspaceId = $_.WorkspaceId
            WorkspaceName = $_.WorkspaceName
            AppId = $_.AppId
            AppName = $_.AppName
            CaptureStatus = $_.CaptureStatus
            CaptureError = $_.CaptureError
            AssetsInspected = $_.AssetsInspected
            SkippedAssets = $_.SkippedAssets
        }
    })
    $package = Open-ExcelPackage -Path $Path -Create -ErrorAction Stop
    try {
        $definitions = @(
            @{
                Name = 'FabricAppModels'
                Rows = $Models
                Headers = @('AppWorkspaceId', 'AppWorkspaceName', 'AppId', 'AppName', 'ConnectionAlias', 'ModelWorkspaceId', 'ModelWorkspaceName', 'ModelId', 'ModelName', 'Evidence')
            },
            @{
                Name = 'FabricApps'
                Rows = $appRows
                Headers = @('WorkspaceId', 'WorkspaceName', 'AppId', 'AppName', 'CaptureStatus', 'CaptureError', 'AssetsInspected', 'SkippedAssets')
            }
        )
        if ($IncludeDependencies) {
            $definitions = @(
                @{
                    Name = 'FabricAppQueries'
                    Rows = $Queries
                    Headers = @('WorkspaceId', 'WorkspaceName', 'AppId', 'AppName', 'QuerySymbol', 'Evidence', 'QueryText', 'SourceUrl', 'ConnectionAlias', 'ModelId', 'ModelName', 'ModelWorkspaceId', 'ModelWorkspaceName', 'ModelMappingEvidence')
                }
            ) + $definitions + @(
                @{
                    Name = 'QueryDependencies'
                    Rows = $Dependencies
                    Headers = @('QueryId', 'WorkspaceId', 'WorkspaceName', 'AppId', 'AppName', 'QuerySymbol', 'ConnectionAlias', 'ModelWorkspaceId', 'ModelId', 'ModelName', 'ObjectName', 'ObjectType', 'ObjectTable', 'DependsOn', 'DependsOnType', 'DependsOnTable', 'DependencyKind', 'DependencyDepth', 'Resolution', 'Detail')
                },
                @{
                    Name = 'QueryDependencyStatus'
                    Rows = $DependencyStatus
                    Headers = @('QueryId', 'AppId', 'AppName', 'QuerySymbol', 'ModelId', 'Status', 'DirectObjectCount', 'UnresolvedCount', 'Coverage')
                }
            )
        }
        foreach ($definition in $definitions) {
            if ($definition.Rows.Count -gt 1048575) { throw 'App query export exceeds the Excel row limit.' }
            $sheet = $package.Workbook.Worksheets[$definition.Name]
            if ($sheet) { $package.Workbook.Worksheets.Delete($definition.Name) }
            $sheet = $package.Workbook.Worksheets.Add($definition.Name)
            $headers = [System.Collections.Generic.List[string]]::new()
            foreach ($header in $definition.Headers) { $headers.Add($header) }
            foreach ($record in $definition.Rows) {
                foreach ($property in $record.PSObject.Properties.Name) {
                    if ($property -eq 'TemplateExpressions') { continue }
                    if (-not $headers.Contains($property)) { $headers.Add($property) }
                }
            }
            if ($headers.Count -gt 16384) { throw 'App query export exceeds the Excel column limit.' }
            for ($column = 0; $column -lt $headers.Count; $column++) {
                $sheet.Cells[1, ($column + 1)].Value = $headers[$column]
                $sheet.Column($column + 1).Width = if ($headers[$column] -like 'QueryText*') { 85 } else { 25 }
            }
            $rowNumber = 2
            foreach ($record in $definition.Rows) {
                for ($column = 0; $column -lt $headers.Count; $column++) {
                    $value = $record.($headers[$column])
                    if ($value -is [string] -and $value.Length -gt 32767) {
                        throw "Value in $($headers[$column]) exceeds Excel's cell limit."
                    }
                    $sheet.Cells[$rowNumber, ($column + 1)].Value = $value
                }
                $sheet.Row($rowNumber).Height = 60
                $rowNumber++
            }
            $sheet.Cells[1, 1, 1, $headers.Count].Style.Font.Bold = $true
            $sheet.Cells[1, 1, [Math]::Max(1, $rowNumber - 1), $headers.Count].AutoFilter = $true
            $sheet.Cells.Style.WrapText = $true
            $sheet.View.FreezePanes(2, 1)
        }
        $legacySheets = @('AppQueries', 'AppQueryCaptures')
        if (-not $IncludeDependencies) {
            $legacySheets += @('FabricAppQueries', 'QueryDependencies', 'QueryDependencyStatus')
        }
        foreach ($legacySheet in $legacySheets) {
            if ($package.Workbook.Worksheets[$legacySheet]) { $package.Workbook.Worksheets.Delete($legacySheet) }
        }
        $package.Save()
    }
    finally {
        $package.Dispose()
    }
}

function Get-FabricAppDaxReferences {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)][AllowEmptyString()][string]$Text,
        [string[]]$TemplateExpressions = @(),
        [string]$DefaultTable = '',
        [switch]$IsTemplate
    )

    $TemplateExpressions = @($TemplateExpressions | Where-Object { -not [string]::IsNullOrEmpty($_) })
    $dynamicMarker = '__ImpactIQDynamicExpression__'
    while ($Text.Contains($dynamicMarker)) { $dynamicMarker += '_' }
    foreach ($expression in $TemplateExpressions) {
        $Text = $Text.Replace(('${' + $expression + '}'), $dynamicMarker)
    }
    if (($IsTemplate -or $TemplateExpressions.Count) -and $Text.Contains('${')) {
        throw [FormatException]::new('A JavaScript interpolation could not be separated from the DAX. Its runtime dependencies remain unresolved.')
    }
    $lexer = [regex]::new('\G(?:(?<space>\s+)|(?<comment>//[^\r\n]*|--[^\r\n]*|/\*[\s\S]*?\*/)|(?<string>"(?:[^"]|"")*")|(?<table>''(?:[^'']|'''')*'')|(?<field>\[(?:[^\]]|\]\])*\])|(?<word>[\p{L}_][\p{L}\p{N}_.]*)|(?<number>\d+(?:\.\d+)?)|(?<punct>[^\s]))')
    $tokens = [Collections.Generic.List[object]]::new()
    $position = 0
    while ($position -lt $Text.Length) {
        $match = $lexer.Match($Text, $position)
        if (-not $match.Success) { throw [FormatException]::new("Invalid DAX token at position $position.") }
        $position += $match.Length
        if ($match.Groups['space'].Success -or $match.Groups['comment'].Success) { continue }
        foreach ($kind in @('string', 'table', 'field', 'word', 'number', 'punct')) {
            if ($match.Groups[$kind].Success) { break }
        }
        $value = $match.Value
        if ($kind -eq 'punct' -and $value -in @('"', "'", '[')) {
            throw [FormatException]::new("Unterminated DAX identifier or string at position $($match.Index).")
        }
        if ($value -eq '/' -and $position -lt $Text.Length -and $Text[$position] -eq '*') {
            throw [FormatException]::new('Unterminated DAX comment.')
        }
        if ($kind -eq 'string') { $value = $value.Substring(1, $value.Length - 2).Replace('""', '"') }
        elseif ($kind -eq 'table') { $value = $value.Substring(1, $value.Length - 2).Replace("''", "'") }
        elseif ($kind -eq 'field') { $value = $value.Substring(1, $value.Length - 2).Replace(']]', ']') }
        $tokens.Add([PSCustomObject]@{ Kind = $kind; Value = $value })
    }
    $variables = @{}
    $localMeasures = @{}
    $definitions = @{}
    for ($index = 0; $index -lt $tokens.Count - 1; $index++) {
        if ($tokens[$index].Kind -ne 'word') { continue }
        if ($tokens[$index].Value -eq 'VAR' -and $tokens[$index + 1].Kind -eq 'word') {
            $variables[$tokens[$index + 1].Value] = $true
        }
        if ($tokens[$index].Value -in @('COLUMN', 'TABLE', 'FUNCTION') -and
            $index -gt 0 -and $tokens[$index - 1].Value -ne '(') {
            throw [FormatException]::new('Query-scoped COLUMN/TABLE/FUNCTION definitions require engine validation; they are not treated as persisted model objects.')
        }
        if ($tokens[$index].Value -eq 'MEASURE' -and $index + 2 -lt $tokens.Count -and $tokens[$index + 2].Kind -eq 'field') {
            $name = $tokens[$index + 2].Value
            $localMeasures[$name] = $true
            $localMeasures[($tokens[$index + 1].Value + [char]0 + $name)] = $true
            $definitions[$index + 1] = $true
            $definitions[$index + 2] = $true
        }
    }
    $depths = [int[]]::new($tokens.Count)
    $depth = 0
    for ($index = 0; $index -lt $tokens.Count; $index++) {
        $depths[$index] = $depth
        if ($tokens[$index].Kind -ne 'punct') { continue }
        if ($tokens[$index].Value -in @('(', '{')) { $depth++ }
        elseif ($tokens[$index].Value -in @(')', '}')) { $depth-- }
    }
    $variableRanges = @{}
    for ($index = 0; $index -lt $tokens.Count - 3; $index++) {
        if ($tokens[$index].Kind -ne 'word' -or $tokens[$index].Value -ne 'VAR' -or $tokens[$index + 2].Value -ne '=') { continue }
        $end = $index + 3
        while ($end -lt $tokens.Count) {
            if ($depths[$end] -lt $depths[$index] -or
                ($depths[$end] -eq $depths[$index] -and
                    (($tokens[$end].Kind -eq 'word' -and $tokens[$end].Value -in @('VAR', 'RETURN', 'EVALUATE')) -or
                    ($tokens[$end].Kind -eq 'punct' -and $tokens[$end].Value -in @(')', '}'))))) { break }
            $end++
        }
        $name = $tokens[$index + 1].Value
        if (-not $variableRanges.ContainsKey($name)) { $variableRanges[$name] = [Collections.Generic.List[object]]::new() }
        $variableRanges[$name].Add(@{ Start = $index + 3; End = $end })
    }
    function Get-DaxReachableRanges([int]$Start, [int]$End) {
        $ranges = [Collections.Generic.List[object]]::new()
        $pending = [Collections.Generic.Queue[object]]::new()
        $pending.Enqueue(@{ Start = $Start; End = $End })
        $seen = @{}
        while ($pending.Count) {
            $range = $pending.Dequeue()
            $ranges.Add($range)
            for ($cursor = $range.Start; $cursor -lt $range.End; $cursor++) {
                $name = $tokens[$cursor].Value
                if ($tokens[$cursor].Kind -ne 'word' -or $seen.ContainsKey($name) -or -not $variableRanges.ContainsKey($name)) { continue }
                $seen[$name] = $true
                if ($variableRanges[$name].Count -eq 1) { $pending.Enqueue($variableRanges[$name][0]) }
            }
        }
        $ranges.ToArray()
    }
    $aliases = @{}
    $aliasRangeCache = @{}
    $outputRanges = @()
    $frames = [Collections.Generic.List[object]]::new()
    $references = [Collections.Generic.List[object]]::new()
    $reserved = @{}
    foreach ($word in @('DEFINE', 'EVALUATE', 'VAR', 'MEASURE', 'RETURN', 'ORDER', 'BY', 'ASC', 'DESC', 'IN', 'NOT', 'AND', 'OR', 'TRUE', 'FALSE', 'INTEGER', 'STRING', 'DOUBLE', 'BOOLEAN', 'DATETIME', 'CURRENCY', 'VARIANT', 'BLANK', 'DAY', 'WEEK', 'MONTH', 'QUARTER', 'YEAR', 'SECOND', 'MINUTE', 'HOUR', 'SKIP', 'DENSE', 'BOTH', 'NONE', 'ONEWAY', 'ONEWAY_LEFTFILTERSRIGHT', 'ONEWAY_RIGHTFILTERSLEFT', 'DEFAULT', 'FIRST', 'LAST', 'KEEP', 'IGNORE', 'STRICT', 'RELAXED', 'EXTENDING', 'PRECISE', 'BLANKS', 'ANCHORED', 'START', 'AT', 'ABS', 'REL')) {
        $reserved[$word] = $true
    }
    $ordering = $false
    for ($index = 0; $index -lt $tokens.Count; $index++) {
        $token = $tokens[$index]
        $previous = if ($index -gt 0) { $tokens[$index - 1] } else { $null }
        $next = if ($index + 1 -lt $tokens.Count) { $tokens[$index + 1] } else { $null }
        if ($token.Kind -eq 'word' -and $token.Value -eq 'EVALUATE') {
            $ordering = $false
            $end = $index + 1
            while ($end -lt $tokens.Count -and -not ($depths[$end] -eq 0 -and $tokens[$end].Kind -eq 'word' -and $tokens[$end].Value -in @('ORDER', 'EVALUATE'))) { $end++ }
            $outputRanges = @(Get-DaxReachableRanges ($index + 1) $end)
        }
        if ($token.Kind -eq 'word' -and $token.Value -eq 'BY' -and $previous.Value -eq 'ORDER') { $ordering = $true }
        if ($token.Kind -eq 'punct' -and $token.Value -eq '(') {
            $newFrame = @{ Function = [string]$previous.Value; Argument = 0; Braces = 0; Table = ''; InputVariable = ''; Start = $index; End = -1 }
            $frames.Add($newFrame)
            if ($newFrame.Function -eq 'GENERATESERIES') {
                if (-not $aliases.ContainsKey('Value')) { $aliases['Value'] = [Collections.Generic.List[object]]::new() }
                $aliases['Value'].Add($newFrame)
            }
            continue
        }
        if ($token.Kind -eq 'punct' -and $token.Value -eq ')') {
            if ($frames.Count -eq 0) { throw [FormatException]::new('Unbalanced DAX parentheses.') }
            $frames[$frames.Count - 1].End = $index
            $frames.RemoveAt($frames.Count - 1)
            continue
        }
        $frame = if ($frames.Count) { $frames[$frames.Count - 1] } else { $null }
        if ($frame -and $token.Kind -eq 'punct') {
            if ($token.Value -eq '{') { $frame.Braces++ }
            elseif ($token.Value -eq '}') { $frame.Braces-- }
            elseif ($token.Value -eq ',' -and $frame.Braces -eq 0) { $frame.Argument++ }
        }
        if ($frame -and $frame.Argument -eq 0 -and $previous.Value -eq '(' -and
            $token.Kind -in @('table', 'word') -and $next.Value -ne '(') {
            if ($variables.ContainsKey($token.Value)) { $frame.InputVariable = $token.Value }
            else { $frame.Table = $token.Value }
        }
        if ($frame -and $token.Kind -eq 'string' -and $frame.Braces -eq 0 -and $previous.Value -in @('(', ',')) {
            $isAlias = ($frame.Function -in @('ROW', 'DATATABLE') -and $frame.Argument % 2 -eq 0) -or
                ($frame.Function -in @('SELECTCOLUMNS', 'ADDCOLUMNS') -and $frame.Argument % 2 -eq 1) -or
                $frame.Function -in @('SUMMARIZE', 'SUMMARIZECOLUMNS', 'GROUPBY', 'ROLLUPADDISSUBTOTAL')
            if ($isAlias) {
                if (-not $aliases.ContainsKey($token.Value)) { $aliases[$token.Value] = [Collections.Generic.List[object]]::new() }
                $aliases[$token.Value].Add($frame)
            }
        }
        if ($definitions.ContainsKey($index)) { continue }
        if ($next.Kind -ne 'field' -and ($token.Kind -eq 'table' -or ($token.Kind -eq 'word' -and
            -not $reserved.ContainsKey($token.Value) -and -not $variables.ContainsKey($token.Value) -and
            $next.Value -ne '(' -and $previous.Value -ne 'VAR'))) {
            $requiredTable = $previous.Kind -eq 'word' -and $previous.Value -eq 'EVALUATE'
            if ($frame -and $previous.Kind -eq 'punct' -and $previous.Value -in @('(', ',')) {
                $requiredTable = $requiredTable -or
                    ($frame.Argument -eq 0 -and $frame.Function -in @('FILTER', 'SUMX', 'AVERAGEX', 'MINX', 'MAXX', 'COUNTX', 'COUNTAX', 'PRODUCTX', 'CONCATENATEX', 'ADDCOLUMNS', 'SELECTCOLUMNS', 'RANKX', 'COUNTROWS', 'VALUES', 'DISTINCT', 'ALL', 'ALLNOBLANKROW', 'ALLSELECTED', 'ALLEXCEPT', 'REMOVEFILTERS', 'SUMMARIZE', 'GROUPBY', 'CALCULATETABLE', 'RELATEDTABLE', 'TREATAS')) -or
                    ($frame.Function -in @('UNION', 'INTERSECT', 'EXCEPT', 'CROSSJOIN', 'GENERATE', 'GENERATEALL', 'NATURALINNERJOIN', 'NATURALLEFTOUTERJOIN')) -or
                    ($frame.Function -eq 'TOPN' -and $frame.Argument -eq 1)
            }
            if ($TemplateExpressions.Count -and $token.Kind -eq 'word' -and
                $token.Value -eq $dynamicMarker -and -not $requiredTable) { continue }
            $references.Add([PSCustomObject]@{
                Kind = $(if ($token.Kind -eq 'table' -or $requiredTable) { 'Table' } else { 'PossibleTable' })
                Table = $token.Value; Name = ''; ContextTable = ''; Alias = $false; Ordering = $ordering; GeneratedValue = $false
                Dynamic = $TemplateExpressions.Count -gt 0 -and $token.Value.Contains($dynamicMarker)
            })
        }
        if ($token.Kind -ne 'field') { continue }
        $qualified = $previous.Kind -in @('table', 'word') -and -not $reserved.ContainsKey($previous.Value)
        $table = if ($qualified) { $previous.Value } else { '' }
        if (($qualified -and $localMeasures.ContainsKey(($table + [char]0 + $token.Value))) -or
            (-not $qualified -and $localMeasures.ContainsKey($token.Value))) { continue }
        $context = $DefaultTable
        for ($frameIndex = $frames.Count - 1; $frameIndex -ge 0; $frameIndex--) {
            $candidate = $frames[$frameIndex]
            if ($candidate.Table -and $candidate.Function -in @('FILTER', 'SUMX', 'AVERAGEX', 'MINX', 'MAXX', 'COUNTX', 'COUNTAX', 'PRODUCTX', 'CONCATENATEX', 'ADDCOLUMNS', 'SELECTCOLUMNS', 'RANKX', 'GENERATE')) {
                $context = $candidate.Table
                break
            }
        }
        $visibleAlias = $false
        if (-not $qualified -and $aliases.ContainsKey($token.Value)) {
            $visibleRanges = if ($ordering) { $outputRanges } else {
                foreach ($activeFrame in $frames) {
                    @{ Start = $activeFrame.Start; End = $index }
                    $variable = $activeFrame.InputVariable
                    if ($variable -and $variableRanges.ContainsKey($variable) -and $variableRanges[$variable].Count -eq 1) {
                        if (-not $aliasRangeCache.ContainsKey($variable)) {
                            $range = $variableRanges[$variable][0]
                            $aliasRangeCache[$variable] = @(Get-DaxReachableRanges $range.Start $range.End)
                        }
                        $aliasRangeCache[$variable]
                    }
                }
            }
            foreach ($definitionFrame in $aliases[$token.Value]) {
                if ($definitionFrame.End -lt 0 -or $definitionFrame.End -ge $index) { continue }
                foreach ($range in $visibleRanges) {
                    if ($definitionFrame.Start -ge $range.Start -and $definitionFrame.End -le $range.End) { $visibleAlias = $true; break }
                }
                if ($visibleAlias) { break }
            }
        }
        $references.Add([PSCustomObject]@{
            Kind = 'Field'; Table = $table; Name = $token.Value; ContextTable = $context
            Alias = $visibleAlias; Ordering = $ordering; GeneratedValue = $false
            Dynamic = $TemplateExpressions.Count -gt 0 -and ($table.Contains($dynamicMarker) -or $token.Value.Contains($dynamicMarker))
        })
    }
    if ($frames.Count) { throw [FormatException]::new('Unbalanced DAX parentheses.') }
    $references.ToArray()
}

function Get-FabricAppQueryDependencies {
    [CmdletBinding()]
    param(
        [AllowEmptyCollection()][object[]]$Queries = @(),
        [AllowEmptyCollection()][object[]]$ModelObjects = @()
    )

    $ErrorActionPreference = 'Stop'
    function Get-ObjectKey([string]$Type, [string]$Table, [string]$Name) {
        ($Type + [char]0 + $Table + [char]0 + $Name).ToLowerInvariant()
    }
    function Get-ObjectType([string]$Type) {
        if ($Type -match 'Column$') { return 'Column' }
        $Type
    }
    $models = @{}
    foreach ($row in $ModelObjects) {
        $type = Get-ObjectType ([string]$row.Type)
        if ($type -notin @('Table', 'Column', 'Measure') -or -not $row.ModelID) { continue }
        $id = [string]$row.ModelID
        if (-not $models.ContainsKey($id)) { $models[$id] = @{ Objects = @{}; Fields = @{}; Names = @{} } }
        $model = $models[$id]
        $key = Get-ObjectKey $type $row.Table $row.Name
        if ($model.Objects.ContainsKey($key)) { continue }
        $object = [PSCustomObject]@{ Key = $key; Type = $type; Table = [string]$row.Table; Name = [string]$row.Name }
        $model.Objects[$key] = $object
        if ($type -in @('Column', 'Measure')) { $model.Fields[([string]$row.Table + [char]0 + [string]$row.Name)] = $key }
        $nameKey = $type + [char]0 + [string]$row.Name
        if (-not $model.Names.ContainsKey($nameKey)) { $model.Names[$nameKey] = [Collections.Generic.HashSet[string]]::new([StringComparer]::OrdinalIgnoreCase) }
        $null = $model.Names[$nameKey].Add($key)
    }
    function Resolve-QueryReference($Reference, $Model) {
        if ($Reference.Dynamic) { return }
        $keys = @()
        if ($Reference.Kind -in @('Table', 'PossibleTable')) {
            $key = Get-ObjectKey 'Table' $Reference.Table $Reference.Table
            if ($Model.Objects.ContainsKey($key)) { return $Model.Objects[$key] }
            return
        }
        if (-not $Reference.Table -and $Reference.Alias) { return }
        if ($Reference.Table) {
            $key = $Reference.Table + [char]0 + $Reference.Name
            if ($Model.Fields.ContainsKey($key)) { return $Model.Objects[$Model.Fields[$key]] }
            return
        }
        $nameKey = 'Measure' + [char]0 + $Reference.Name
        if ($Model.Names.ContainsKey($nameKey)) { $keys += @($Model.Names[$nameKey]) }
        if ($Reference.ContextTable) {
            $columnKey = Get-ObjectKey 'Column' $Reference.ContextTable $Reference.Name
            if ($Model.Objects.ContainsKey($columnKey)) { $keys += $columnKey }
        }
        if ($keys.Count -eq 1) { return $Model.Objects[$keys[0]] }
    }
    $edges = [Collections.Generic.List[object]]::new()
    $statuses = [Collections.Generic.List[object]]::new()
    $queryRows = [Collections.Generic.List[object]]::new()
    $sha = [Security.Cryptography.SHA256]::Create()
    try {
        foreach ($query in $Queries) {
            $text = @($query.PSObject.Properties | Where-Object { $_.Name -match '^QueryText(\d+)?$' } |
                Sort-Object { if ($_.Name -eq 'QueryText') { 1 } else { [int]$_.Name.Substring(9) } } |
                ForEach-Object { [string]$_.Value }) -join ''
            $identity = @($query.WorkspaceId, $query.AppId, $query.SourceUrl, $query.QuerySymbol, $query.ConnectionAlias, $query.ModelWorkspaceId, $query.ModelId, $text) | ConvertTo-Json -Compress
            $queryId = [BitConverter]::ToString($sha.ComputeHash([Text.Encoding]::UTF8.GetBytes($identity))).Replace('-', '').ToLowerInvariant()
            $exportQuery = $query | Select-Object * -ExcludeProperty TemplateExpressions
            $exportQuery | Add-Member -NotePropertyName QueryId -NotePropertyValue $queryId -Force
            $queryRows.Add($exportQuery)
            $seenEdges = @{}
            $directObjects = @{}
            $queryState = @{ Unresolved = 0 }
            $model = $models[[string]$query.ModelId]
            function Add-QueryEdge($Parent, $Target, [string]$Kind, [int]$Depth, [string]$Detail = '') {
                $targetKey = Get-ObjectKey $Target.Type $Target.Table $Target.Name
                $edgeKey = $Parent.Key + [char]1 + $targetKey + [char]1 + $Kind
                if ($seenEdges.ContainsKey($edgeKey)) { return }
                $seenEdges[$edgeKey] = $true
                $resolved = -not $Detail
                if (-not $resolved) { $queryState.Unresolved++ }
                $edges.Add([PSCustomObject][ordered]@{
                    QueryId = $queryId; WorkspaceId = $query.WorkspaceId; WorkspaceName = $query.WorkspaceName
                    AppId = $query.AppId; AppName = $query.AppName; QuerySymbol = $query.QuerySymbol
                    ConnectionAlias = $query.ConnectionAlias; ModelWorkspaceId = $query.ModelWorkspaceId
                    ModelId = $query.ModelId; ModelName = $query.ModelName
                    ObjectName = $Parent.Name; ObjectType = $Parent.Type; ObjectTable = $Parent.Table
                    DependsOn = $Target.Name; DependsOnType = $Target.Type; DependsOnTable = $Target.Table
                    DependencyKind = $Kind; DependencyDepth = $Depth
                    Resolution = $(if ($resolved) { 'Resolved' } else { 'Unresolved' }); Detail = $Detail
                })
                if ($resolved -and $Kind -eq 'Direct query reference') { $directObjects[$targetKey] = $true }
            }
            function Add-QueryReferences($Parent, [object[]]$References, [string]$Kind, [int]$Depth, [string]$ResolutionError = '') {
                foreach ($reference in $References) {
                    $target = if (-not $ResolutionError) { Resolve-QueryReference $reference $model }
                    if ($target) {
                        Add-QueryEdge $Parent $target $Kind $Depth
                    }
                    elseif ($reference.Kind -eq 'Field' -and -not $reference.Table -and ($reference.Alias -or $reference.GeneratedValue)) {
                        continue
                    }
                    else {
                        $detail = if ($ResolutionError) { $ResolutionError }
                            elseif ($reference.Kind -eq 'Invalid') { $reference.Error }
                            else { 'Reference is missing, ambiguous, or runtime-generated; no model object was guessed.' }
                        Add-QueryEdge $Parent ([PSCustomObject]@{
                            Type = $(if ($reference.Kind -in @('Table', 'PossibleTable')) { 'Table' } else { 'Unknown' })
                            Table = $reference.Table; Name = $(if ($reference.Name) { $reference.Name } else { $reference.Table })
                        }) $Kind $Depth $detail
                    }
                }
            }
            $root = [PSCustomObject]@{ Key = $queryId; Name = $query.QuerySymbol; Type = 'Query'; Table = '' }
            $status = 'ResolvedStaticReferences'
            $resolutionError = ''
            if (-not $query.ModelId -or -not $query.ModelWorkspaceId) {
                $status = 'MissingModelMapping'
                $resolutionError = 'The query has no unambiguous semantic-model mapping.'
            }
            elseif (-not $model) {
                $status = 'MissingModelMetadata'
                $resolutionError = 'No model-object metadata was extracted for this model ID. Check model access and model-detail extraction.'
            }
            if ($resolutionError) {
                try {
                    $references = @(Get-FabricAppDaxReferences -Text $text -TemplateExpressions @($query.TemplateExpressions) -IsTemplate:($query.Evidence -eq 'DynamicTemplate'))
                    Add-QueryReferences $root $references 'Direct query reference' 1 $resolutionError
                }
                catch [FormatException] {
                    Add-QueryEdge $root ([PSCustomObject]@{ Type = 'Unknown'; Table = ''; Name = '' }) 'Direct query reference' 1 $_.Exception.Message
                }
                if (-not $queryState.Unresolved) {
                    Add-QueryEdge $root ([PSCustomObject]@{ Type = 'Unknown'; Table = ''; Name = '' }) 'Direct query reference' 1 $resolutionError
                }
            }
            else {
                try {
                    $references = @(Get-FabricAppDaxReferences -Text $text -TemplateExpressions @($query.TemplateExpressions) -IsTemplate:($query.Evidence -eq 'DynamicTemplate'))
                    Add-QueryReferences $root $references 'Direct query reference' 1
                }
                catch [FormatException] {
                    Add-QueryEdge $root ([PSCustomObject]@{ Type = 'Unknown'; Table = ''; Name = '' }) 'Direct query reference' 1 $_.Exception.Message
                }
                if ($queryState.Unresolved) { $status = 'UnresolvedReferences' }
                elseif ($query.Evidence -eq 'DynamicTemplate') { $status = 'PartialDynamicTemplate' }
            }
            $statuses.Add([PSCustomObject][ordered]@{
                QueryId = $queryId; AppId = $query.AppId; AppName = $query.AppName; QuerySymbol = $query.QuerySymbol; ModelId = $query.ModelId
                Status = $status; DirectObjectCount = $directObjects.Count; UnresolvedCount = $queryState.Unresolved
                Coverage = 'Direct metadata-matched query references only. Measure/formula dependencies and inferred home tables are excluded. Runtime JavaScript and unvisited assets are not proven by this extract.'
            })
        }
    }
    finally { $sha.Dispose() }
    [PSCustomObject]@{ Queries = $queryRows.ToArray(); Dependencies = $edges.ToArray(); Status = $statuses.ToArray() }
}

function Export-FabricAppQueryDependencyDetail {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)][string]$BaseFolderPath,
        [AllowEmptyCollection()][object[]]$Queries = @(),
        [AllowEmptyCollection()][object[]]$Models = @(),
        [AllowEmptyCollection()][object[]]$Captures = @(),
        [AllowEmptyCollection()][object[]]$ModelObjects = @()
    )

    $ErrorActionPreference = 'Stop'
    $result = Get-FabricAppQueryDependencies -Queries $Queries -ModelObjects $ModelObjects
    $path = Join-Path $BaseFolderPath 'Fabric App Detail.xlsx'
    $temporaryPath = Join-Path $BaseFolderPath ('AppQueryDependencies-' + [guid]::NewGuid().ToString('N') + '.xlsx')
    try {
        Export-FabricAppQueryWorkbook -Path $temporaryPath -Queries $result.Queries -Models $Models -Captures $Captures -Dependencies $result.Dependencies -DependencyStatus $result.Status -IncludeDependencies -ErrorAction Stop
        Move-Item -LiteralPath $temporaryPath -Destination $path -Force -ErrorAction Stop
    }
    finally {
        if (Test-Path -LiteralPath $temporaryPath) { Remove-Item -LiteralPath $temporaryPath -Force -ErrorAction Stop }
    }
    $incomplete = @($result.Status | Where-Object { $_.Status -ne 'ResolvedStaticReferences' })
    if ($incomplete.Count) {
        Write-Warning "$($incomplete.Count) app query row(s) have partial or unresolved dependencies. Review QueryDependencyStatus and unresolved QueryDependencies rows in $path." -WarningAction Continue
    }
    Write-Host "[INFO] Exported $($result.Dependencies.Count) app query dependency edges to $path."
}

function Export-FabricAppDaxEnvironment {
    [CmdletBinding()]
    param(
        [AllowEmptyCollection()][object[]]$Items = @(),
        [Parameter(Mandatory)][string]$BaseFolderPath,
        [string]$Environment = 'Public',
        [AllowEmptyCollection()][object[]]$Datasets = @(),
        [AllowEmptyCollection()][object[]]$Workspaces = @(),
        [ValidateSet('PowerShell', 'Acorn')][string]$ParserEngine = 'PowerShell',
        [AllowEmptyCollection()][object[]]$ModelObjects = @()
    )

    $ErrorActionPreference = 'Stop'
    $environmentWorkbook = Join-Path $BaseFolderPath 'Power BI Environment Detail.xlsx'
    if (-not (Test-Path -LiteralPath $environmentWorkbook -PathType Leaf)) {
        throw 'The environment workbook must exist before app queries can be appended.'
    }
    $appItems = @($Items | Where-Object { $_.FabricItemType -eq 'AppBackend' })
    $appQueryRows = @()
    $appModelRows = @()
    $appQueryCaptures = @()
    $modelNameCache = @{}
    $appDaxRuntime = @{}
    $appDaxReady = $true
    if ($appItems.Count -eq 0) {
    }
    if ($appItems.Count -gt 0 -and $Environment -eq 'Public') {
        try { $appDaxRuntime = Initialize-FabricAppDaxParser -Engine $ParserEngine -ErrorAction Stop }
        catch {
            $appDaxReady = $false
            Write-Warning "Fabric App parser initialization failed: $($_.Exception.Message)" -WarningAction Continue
        }
    }
    foreach ($appItem in $appItems) {
        if (-not $appDaxReady) {
            $appQueryCaptures += [PSCustomObject]@{
                WorkspaceId = $appItem.WorkspaceId
                WorkspaceName = $appItem.WorkspaceName
                AppId = $appItem.FabricItemID
                AppName = $appItem.FabricItemName
                CaptureStatus = 'Failed'
                CaptureError = 'Fabric App parser initialization failed. Check the console error.'
                AssetsInspected = 0
                SkippedAssets = 0
            }
            continue
        }
        Write-Host "[INFO] Extracting Fabric app queries: $($appItem.WorkspaceName) ~ $($appItem.FabricItemName)"
        $appCapture = Get-FabricAppDaxCapture -WorkspaceId $appItem.WorkspaceId -WorkspaceName $appItem.WorkspaceName -AppId $appItem.FabricItemID -AppName $appItem.FabricItemName -ParserRuntime $appDaxRuntime -Environment $Environment -Datasets $Datasets -Workspaces $Workspaces -ModelNameCache $modelNameCache
        $appQueryRows += @($appCapture.Queries)
        $appModelRows += @($appCapture.Models)
        $appQueryCaptures += $appCapture.Capture
    }
    Export-FabricAppQueryWorkbook -Path $environmentWorkbook -Queries $appQueryRows -Models $appModelRows -Captures $appQueryCaptures -ErrorAction Stop
    Export-FabricAppQueryDependencyDetail -BaseFolderPath $BaseFolderPath -Queries $appQueryRows -Models $appModelRows -Captures $appQueryCaptures -ModelObjects $ModelObjects -ErrorAction Stop
    Write-Host "[INFO] Exported $($appQueryRows.Count) queries and $($appModelRows.Count) configured models to $environmentWorkbook."
}

function Export-EmptyDetailWorkbook {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)][string]$Path,
        [Parameter(Mandatory)][System.Collections.IDictionary]$Worksheets
    )

    $ErrorActionPreference = 'Stop'
    if (Test-Path -LiteralPath $Path) {
        Remove-Item -LiteralPath $Path -Force -ErrorAction Stop
    }

    $package = Open-ExcelPackage -Path $Path -Create -ErrorAction Stop
    try {
        foreach ($worksheetName in $Worksheets.Keys) {
            $sheet = $package.Workbook.Worksheets.Add($worksheetName)
            $headers = $Worksheets[$worksheetName]
            for ($column = 0; $column -lt $headers.Count; $column++) {
                $sheet.Cells[1, ($column + 1)].Value = $headers[$column]
            }
        }
        $package.Save()
    }
    finally {
        $package.Dispose()
    }
    Write-Output "Empty detail workbook saved to $Path"
}

            -Headers @{ Authorization = "Bearer $accessToken" } `
            -Method GET `
            -OutFile $OutFilePath
        # Treat as success only if a non-empty file was actually written
        if ((Test-Path $OutFilePath) -and ((Get-Item $OutFilePath).Length -gt 0)) {
            return $true
        }
        return $false
    } catch {
        return $false
    }
}

# Function to Export a Report's PBIR Definition via the Fabric Definition API,
# then zip the returned parts and rename the archive to .pbix.
# NOTE: A zipped PBIR definition renamed to .pbix is NOT an openable Power BI
# Desktop file by design - it is produced this way intentionally.
# Used ONLY as a fallback when the standard LiveConnect export fails.
# Written for Windows PowerShell 5.1 compatibility.
function Export-ReportDefinitionAsPbix {
    param (
        [string]$WorkspaceId,
        [string]$ReportId,
        [string]$OutFilePath
    )

    Add-Type -AssemblyName System.IO.Compression.FileSystem -ErrorAction SilentlyContinue

    $accessToken = Get-CurrentAccessToken
    $headers = @{ Authorization = "Bearer $accessToken"; 'Content-Type' = 'application/json' }
    $defUrl  = "$($global:PowerBIEndpoints.FabricApiPrefix)/v1/workspaces/$WorkspaceId/reports/$ReportId/getDefinition"

    try {
        $resp = Invoke-WebRequest -Uri $defUrl -Method POST -Headers $headers -Body '{}' -UseBasicParsing -ErrorAction Stop
    } catch {
        Write-Warning "getDefinition failed for report $ReportId : $($_.Exception.Message)"
        return $false
    }

    $definition = $null
    $statusCode = [int]$resp.StatusCode

    if ($statusCode -eq 200) {
        $definition = $resp.Content | ConvertFrom-Json
    }
    elseif ($statusCode -eq 202) {
        # Long-running operation: poll the operation until it completes, then fetch the result
        $opUrl = $resp.Headers['Location']
        if ($opUrl -is [array]) { $opUrl = $opUrl[0] }
        $retry = 2
        if ($resp.Headers['Retry-After']) { $retry = [int]($resp.Headers['Retry-After'] | Select-Object -First 1) }

        while ($true) {
            Start-Sleep -Seconds $retry
            try {
                $op = Invoke-WebRequest -Uri $opUrl -Method GET -Headers $headers -UseBasicParsing -ErrorAction Stop
            } catch {
                Write-Warning "Polling definition op failed for report $ReportId : $($_.Exception.Message)"
                return $false
            }
            $status = ($op.Content | ConvertFrom-Json).status
            if ($status -eq 'Succeeded') { break }
            if ($status -eq 'Failed')    { Write-Warning "Definition operation failed for report $ReportId"; return $false }
            if ($op.Headers['Retry-After']) { $retry = [int]($op.Headers['Retry-After'] | Select-Object -First 1) }
        }

        $resultUrl = ($opUrl.TrimEnd('/')) + '/result'
        try {
            $result = Invoke-WebRequest -Uri $resultUrl -Method GET -Headers $headers -UseBasicParsing -ErrorAction Stop
        } catch {
            Write-Warning "Fetching definition result failed for report $ReportId : $($_.Exception.Message)"
            return $false
        }
        $definition = $result.Content | ConvertFrom-Json
    }
    else {
        Write-Warning "Unexpected status $statusCode from getDefinition for report $ReportId"
        return $false
    }

    if (-not $definition.definition.parts) {
        Write-Warning "No definition parts returned for report $ReportId"
        return $false
    }

    # Stage the parts into a temp folder.
    # NOTE: getDefinition returns part paths WITHOUT a "Report/" prefix
    # (e.g. "definition/report.json"), but the PBIX/PBIR layout expected by the
    # Report Detail extractor is "Report/definition/...". Nest everything under
    # a "Report" folder so the zipped .pbix matches that structure.
    $stageDir = Join-Path $env:TEMP ("ReportDef_" + $ReportId)
    if (Test-Path $stageDir) { Remove-Item $stageDir -Recurse -Force }
    New-Item -ItemType Directory -Path $stageDir -Force | Out-Null
    $reportRoot = Join-Path $stageDir 'Report'
    New-Item -ItemType Directory -Path $reportRoot -Force | Out-Null

    foreach ($part in $definition.definition.parts) {
        $partPath = Join-Path $reportRoot $part.path
        $partDir  = Split-Path $partPath -Parent
        if (-not (Test-Path $partDir)) { New-Item -ItemType Directory -Path $partDir -Force | Out-Null }
        if ($part.payloadType -eq 'InlineBase64') {
            [System.IO.File]::WriteAllBytes($partPath, [System.Convert]::FromBase64String($part.payload))
        } else {
            [System.IO.File]::WriteAllText($partPath, [string]$part.payload)
        }
    }

    # If getDefinition returned the legacy (non-PBIR) layout, there is NO
    # "definition" folder - instead a root-level "report.json" carries the
    # report layout. Rename it to "Layout" (no extension) AND re-encode it as
    # UTF-16 LE (no BOM) to match the classic PBIX layout the downstream
    # extractor expects. getDefinition returns report.json as UTF-8, but a real
    # PBIX "Report/Layout" is UTF-16 LE; without this conversion the parser
    # reads garbage and fails.
    $defFolder = Join-Path $reportRoot 'definition'
    if (-not (Test-Path $defFolder)) {
        $reportJson = Join-Path $reportRoot 'report.json'
        if (Test-Path $reportJson) {
            $layoutPath = Join-Path $reportRoot 'Layout'
            if (Test-Path $layoutPath) { Remove-Item $layoutPath -Force }
            $layoutText = [System.IO.File]::ReadAllText($reportJson, [System.Text.Encoding]::UTF8)
            $utf16NoBom = New-Object System.Text.UnicodeEncoding($false, $false)  # LE, no BOM
            [System.IO.File]::WriteAllText($layoutPath, $layoutText, $utf16NoBom)
            Remove-Item $reportJson -Force
        }
    }

    # getDefinition does NOT return a "Connections" file, but the downstream
    # Report Detail extractor expects one at the ROOT (alongside the "Report"
    # folder) to populate the Connections sheet + ModelID/ReportID. Synthesize
    # it from definition.pbir's datasetReference so the layout matches a PBIX.
    $pbirPath = Join-Path $reportRoot 'definition.pbir'
    if (Test-Path $pbirPath) {
        try {
            $pbir  = Get-Content -Path $pbirPath -Raw | ConvertFrom-Json
            $dsRef = $pbir.datasetReference
            $connString = $null
            $datasetId  = ''

            if ($dsRef.byConnection -and $dsRef.byConnection.connectionString) {
                $connString = $dsRef.byConnection.connectionString
                $m = [regex]::Match($connString, 'semanticmodelid=([0-9a-fA-F-]+)')
                if ($m.Success) { $datasetId = $m.Groups[1].Value }
            }
            elseif ($dsRef.byPath -and $dsRef.byPath.path) {
                $connString = 'byPath:' + $dsRef.byPath.path
            }

            if ($connString) {
                $connObj = [pscustomobject]@{
                    Version     = '3.0'
                    Connections = @(
                        [pscustomobject]@{
                            Name             = 'EntityDataSource'
                            ConnectionString = $connString
                            ConnectionType   = 'pbiServiceLive'
                        }
                    )
                    RemoteArtifacts = @(
                        [pscustomobject]@{
                            DatasetId = $datasetId
                            ReportId  = $ReportId
                        }
                    )
                }
                # Root-level file literally named "Connections" (no extension) to match PBIX
                $connJson = $connObj | ConvertTo-Json -Depth 6 -Compress
                [System.IO.File]::WriteAllText((Join-Path $stageDir 'Connections'), $connJson)
            }
        } catch {
            Write-Warning "Could not synthesize Connections file for report $ReportId : $($_.Exception.Message)"
        }
    }

    # Zip the staged folder, then rename the archive to .pbix
    if (Test-Path $OutFilePath) { Remove-Item $OutFilePath -Force }
    $tempZip = Join-Path $env:TEMP ("ReportDef_" + $ReportId + ".zip")
    if (Test-Path $tempZip) { Remove-Item $tempZip -Force }

    [System.IO.Compression.ZipFile]::CreateFromDirectory($stageDir, $tempZip)
    Move-Item -Path $tempZip -Destination $OutFilePath -Force

    Remove-Item $stageDir -Recurse -Force -ErrorAction SilentlyContinue
    return $true
}

	# Function: get first free drive letter (Z down to D)
	function Get-FreeDriveLetter {
	    $used = Get-PSDrive -PSProvider FileSystem | ForEach-Object { $_.Name }
	    foreach ($letter in [char[]](90..68)) {  # Z..Y..X.. down to D
	        if ($used -notcontains $letter) { return "${letter}:" }
	    }
	    throw "No free drive letters available."
	}

	foreach ($workspace in $workspacesInfo) {
	    $workspaceName = $workspace.WorkspaceName
	    $workspaceId = $workspace.WorkspaceId
	
	    # Clean up workspace name
	    $cleanWorkspaceName = $workspaceName -replace '\[', '(' -replace '\]', ')'
	    $cleanWorkspaceName = $cleanWorkspaceName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
	    $cleanWorkspaceName = $cleanWorkspaceName.TrimStart()
	
	    # Fetch reports from the existing list
	    $reports = $reportsInfo | Where-Object { $_.WorkspaceId -eq $workspaceId }
	
	    foreach ($report in $reports) {
	        $reportName = $report.ReportName
	        $reportId = $report.ReportId
	
	        # Clean up report name
	        $cleanReportName = $reportName -replace '\[', '(' -replace '\]', ')'
	        $cleanReportName = $cleanReportName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
	        $cleanReportName = $cleanReportName.TrimStart()
	
	        # Determine file extension
	        $fileExtension = if ($report.WebUrl -like "*/rdlreports/*") { "rdl" } else { "pbix" }
	        $filename = "$cleanWorkspaceName ~ $cleanReportName.$fileExtension"
	        $filepath = Join-Path -Path $newDateFolder -ChildPath $filename
	        $extractFolder = Join-Path -Path $tempExtractFolder -ChildPath "$cleanWorkspaceName ~ $cleanReportName"
	
	        # Delete file if it already exists
	        if (Test-Path $filepath) {
	            Remove-Item $filepath -Force
	        }
	
	        Write-Host "Exporting $cleanWorkspaceName ~ $cleanReportName"
	
	        # Export based on Workspace Type
	        if ($workspace.WorkspaceIsOnDedicatedCapacity -eq $false) {
	            # Pro Workspace → IncludeModel (real .pbix, used for model extraction below)
	            $includeModelExported = Export-ReportUsingAPI -GroupId $workspaceId -ReportId $reportId -OutFilePath $filepath -DownloadType "IncludeModel"
	            if (-not $includeModelExported) {
	                Write-Warning "IncludeModel export failed for $cleanWorkspaceName ~ $cleanReportName; falling back to getDefinition."
	                Export-ReportDefinitionAsPbix -WorkspaceId $workspaceId -ReportId $reportId -OutFilePath $filepath | Out-Null
	            }
	        } elseif ($fileExtension -eq 'pbix') {
	            # Non-Pro PBIX report (Premium / Fabric / etc.).
	            # If the report carries a sensitivity label (only visible via the Fabric
	            # List Items API), prefer getDefinition FIRST so the protected definition is
	            # captured directly, falling back to LiveConnect only if that fails.
	            # Otherwise keep the default LiveConnect-first behaviour with getDefinition as
	            # the fallback (a zipped PBIR definition renamed to .pbix - intentional).
	            if ($global:ReportsWithSensitivityLabel -and $global:ReportsWithSensitivityLabel.ContainsKey($reportId)) {
	                $defExported = Export-ReportDefinitionAsPbix -WorkspaceId $workspaceId -ReportId $reportId -OutFilePath $filepath
	                if (-not $defExported) {
	                    Write-Warning "getDefinition export failed for $cleanWorkspaceName ~ $cleanReportName; falling back to LiveConnect."
	                    $liveExported = Export-ReportUsingAPI -GroupId $workspaceId -ReportId $reportId -OutFilePath $filepath -DownloadType "LiveConnect"
	                    if (-not $liveExported) {
	                        Write-Warning "LiveConnect export also failed for $cleanWorkspaceName ~ $cleanReportName."
	                    }
	                }
	            } else {
	                # No sensitivity label → LiveConnect first, getDefinition fallback.
	                $liveExported = Export-ReportUsingAPI -GroupId $workspaceId -ReportId $reportId -OutFilePath $filepath -DownloadType "LiveConnect"
	                if (-not $liveExported) {
	                    Write-Warning "LiveConnect export failed for $cleanWorkspaceName ~ $cleanReportName; falling back to getDefinition."
	                    $defExported = Export-ReportDefinitionAsPbix -WorkspaceId $workspaceId -ReportId $reportId -OutFilePath $filepath
	                    if (-not $defExported) {
	                        Write-Warning "getDefinition export also failed for $cleanWorkspaceName ~ $cleanReportName."
	                    }
	                }
	            }
	        } else {
	            # Non-Pro paginated (RDL) report → LiveConnect export
	            $null = Export-ReportUsingAPI -GroupId $workspaceId -ReportId $reportId -OutFilePath $filepath -DownloadType "LiveConnect"
	        }
			
		# Only process model extraction for Pro Workspaces
		if ($workspace.WorkspaceIsOnDedicatedCapacity -eq $false) {
		    Write-Host "Extracting $cleanWorkspaceName ~ $cleanReportName"
		
		    $driveLetter = Get-FreeDriveLetter
		
		    # ✅ Make sure the extract folder exists before subst
		    if (-not (Test-Path $extractFolder)) {
		        New-Item -ItemType Directory -Path $extractFolder -Force | Out-Null
		    }
		
		    if (Test-Path $driveLetter) { Invoke-ImpactIQNativeCommand -FilePath 'subst.exe' -ArgumentList @($driveLetter, '/D') -DiscardOutput }
		
		    Invoke-ImpactIQNativeCommand -FilePath 'subst.exe' -ArgumentList @($driveLetter, $extractFolder) -DiscardOutput
		
		    if (-not (Test-Path $driveLetter)) {
		        Write-Error "Failed to map $extractFolder to $driveLetter"
		        continue
		    }
		
		    try {
		        # Extract into subst drive
				Write-Host "If available, extracting Model from $cleanWorkspaceName ~ $cleanReportName"
		        Invoke-ImpactIQNativeCommand -FilePath $pbiToolsPath -ArgumentList @('extract', $filepath, '-extractFolder', $driveLetter, '-modelSerialization', 'Raw') -DiscardOutput
		
		        Invoke-ImpactIQNativeCommand -FilePath $pbiToolsPath -ArgumentList @('generate-bim', $driveLetter, '-transforms', 'RemovePBIDataSourceVersion') -DiscardOutput
		
		        $bimFilesGenerated = Get-ChildItem -Path $extractFolder -Filter *.bim
		        foreach ($bimFile in $bimFilesGenerated) {
		            $newBimName = "$cleanWorkspaceName ~ $cleanReportName.bim"
		            $newBimPath = Join-Path -Path $tempExtractFolder -ChildPath $newBimName
		
		            Move-Item -Path $bimFile.FullName -Destination $newBimPath -Force
		        }
		    }
		    finally {
		        # ✅ Always clean up subst drive, even if extract/generate fails
		        Invoke-ImpactIQNativeCommand -FilePath 'subst.exe' -ArgumentList @($driveLetter, '/D') -DiscardOutput
		    }
		}
	}
}

# Function to download and extract Fabric (Gen2) dataflows
function Export-FabricDataflow {
    param (
        [string]$WorkspaceId,
        [string]$DataflowId,
        [string]$WorkspaceName,
        [string]$DataflowName,
        [string]$OutputPath,
        [string]$AccessToken
    )
    
    # Clean up names
    $cleanWorkspaceName = $WorkspaceName -replace '\[', '(' -replace '\]', ')'
    $cleanWorkspaceName = $cleanWorkspaceName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
    $cleanWorkspaceName = $cleanWorkspaceName.TrimStart()
    
    $cleanDataflowName = $DataflowName -replace '\[', '(' -replace '\]', ')'
    $cleanDataflowName = $cleanDataflowName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
    $cleanDataflowName = $cleanDataflowName.TrimStart()
    
    $fabricApiBaseUri = "$($global:PowerBIEndpoints.FabricApiPrefix)/v1/workspaces"
    $endpoint = "$fabricApiBaseUri/$WorkspaceId/dataflows/$DataflowId/getDefinition"
    
    $headers = @{
        "Authorization" = "Bearer $AccessToken"
        "Content-Type"  = "application/json"
    }
    
    try {
        $response = Invoke-RestMethod -Uri $endpoint -Method POST -Headers $headers -Body "{}"
        
        if (-not $response.definition.parts) {
            return $null
        }
        
        # Create temp directory for extraction
        $tempDir = Join-Path $env:TEMP "FabricDataflow_$DataflowId"
        if (Test-Path $tempDir) {
            Remove-Item -Path $tempDir -Recurse -Force
        }
        New-Item -Path $tempDir -ItemType Directory -Force | Out-Null
        
        # Extract files
        foreach ($part in $response.definition.parts) {
            $fileName = $part.path
            $filePath = Join-Path $tempDir $fileName
            $payloadType = $part.payloadType
            $payload = $part.payload
            
            if ($payloadType -eq "InlineBase64") {
                $bytes = [System.Convert]::FromBase64String($payload)
                $fileDir = Split-Path $filePath
                if (-not (Test-Path $fileDir)) {
                    New-Item -ItemType Directory -Path $fileDir -Force | Out-Null
                }
                [System.IO.File]::WriteAllBytes($filePath, $bytes)
            }
        }
        
        # Find the .pq file
        $pqFile = Get-ChildItem -Path $tempDir -Filter "*.pq" -Recurse | Select-Object -First 1
        
        if ($pqFile) {
            # Read the .pq file content
            $pqContent = Get-Content -Path $pqFile.FullName -Raw -Encoding UTF8
            
            # Save as .pqt
            $outputFileName = "$cleanWorkspaceName ~ $cleanDataflowName.pq"
            $outputFilePath = Join-Path $OutputPath $outputFileName
            $pqContent | Out-File -FilePath $outputFilePath -Encoding UTF8
                        
            # Clean up temp directory
            Remove-Item -Path $tempDir -Recurse -Force
            
            return @{
                Content = $pqContent
                FilePath = $outputFilePath
            }
        } else {
            Remove-Item -Path $tempDir -Recurse -Force
            return $null
        }
        
    } catch {
        return $null
    }
}

# Function to parse .pq content and extract queries
function Parse-FabricDataflowContent {
    param (
        [string]$Content,
        [string]$DataflowId,
        [string]$DataflowName,
        [string]$WorkspaceName,
        [datetime]$ReportDate
    )
    
    $queries = @()
    
    # Clean up names
    $cleanWorkspaceName = $WorkspaceName -replace '\[', '(' -replace '\]', ')'
    $cleanWorkspaceName = $cleanWorkspaceName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
    $cleanWorkspaceName = $cleanWorkspaceName.TrimStart()
    
    $cleanDataflowName = $DataflowName -replace '\[', '(' -replace '\]', ')'
    $cleanDataflowName = $cleanDataflowName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
    $cleanDataflowName = $cleanDataflowName.TrimStart()
    
    # Split content by "section Section1;" to get the queries section
    $sections = $Content -split 'section Section1;', 2
    
    if ($sections.Count -gt 1) {
        $queriesSection = $sections[1]
        
        # Use regex to find all queries:
        # Supports both shared QueryName = ... and shared #"Query Name With Spaces" = ...
        $pattern = '(?s)(?:\[[^\]]*\]\s*)?shared\s+(?:#"(.*?)"|([A-Za-z_]\w*))\s*=\s*(.*?)(?=(?:\[[^\]]*\]\s*)?shared\s+(?:#"(?:.*?)"|[A-Za-z_]\w*)\s*=|$)'
        $matches = [regex]::Matches($queriesSection, $pattern)
        
        foreach ($match in $matches) {
            # Group 1 = hash-quoted name, Group 2 = unquoted name, Group 3 = expression
            if ($match.Groups[1].Success) {
                $queryName = $match.Groups[1].Value
            } else {
                $queryName = $match.Groups[2].Value
            }
            $queryExpression = $match.Groups[3].Value.Trim()

            # Remove trailing semicolons
            $queryExpression = $queryExpression -replace ';\s*$', ''
            $queryExpression = $queryExpression.Trim()

            # Skip if empty or just whitespace
            if ([string]::IsNullOrWhiteSpace($queryName) -or [string]::IsNullOrWhiteSpace($queryExpression)) {
                continue
            }

            $queries += [PSCustomObject]@{
                "Dataflow ID" = $DataflowId
                "Dataflow Name" = $DataflowName
                "Query Name" = $queryName
                "Query" = $queryExpression
                "Report Date" = $ReportDate
                "Workspace Name - Dataflow Name" = "$cleanWorkspaceName ~ $cleanDataflowName"
            }
        }
    }
    
    return $queries
}

function Export-FabricDataAgentDetail {
    param(
        [string]$BaseFolderPath,
        [string]$ReportDate,
        [AllowEmptyCollection()][object[]]$Workspaces,
        [hashtable]$DatasetNames
    )

    $ErrorActionPreference = 'Stop'
    Write-Output "Data Agent Detail Extraction Process Started"
    $agentsInfo = @()
    $sourcesInfo = @()
    $definitionToken = if ($global:RunMode -eq 'Agents') { $script:DataAgentDefinitionToken } else { $null }
    $tokenAttempted = [bool]$definitionToken
    $sourceTypes = @{
        semantic_model = "Semantic Model"
        lakehouse = "Lakehouse"
        lakehouse_tables = "Lakehouse (Tables)"
        data_warehouse = "Data Warehouse"
        kusto = "KQL Database"
        graph = "Graph (GQL)"
        mirrored_database = "Mirrored Database"
        mirrored_azure_databricks = "Mirrored Azure Databricks"
        unknown = "Unknown"
    }

    $inventory = @()
    if ($global:RunMode -ne 'Agents') {
        $inventoryWorkspaces = @($Workspaces | Where-Object { $_.WorkspaceId -ne "My Workspace" } |
            ForEach-Object { [PSCustomObject]@{ id = $_.WorkspaceId; name = $_.WorkspaceName } })
        $inventory = @(Get-FabricDataAgentInventory -Workspaces $inventoryWorkspaces)
    }

    foreach ($workspace in $Workspaces) {
        $workspaceId = $workspace.WorkspaceId
        $workspaceName = $workspace.WorkspaceName
        if ($workspaceId -eq "My Workspace") { continue }
        $agents = @()
        if ($global:RunMode -eq 'Agents') {
            $agents = @($script:SelectedDataAgents.Keys | Where-Object { $_.StartsWith("$workspaceId|") } |
                ForEach-Object { $script:SelectedDataAgents[$_] })
        }
        else {
            $agents = @($inventory | Where-Object { $_.WorkspaceId -eq $workspaceId } | ForEach-Object { $_.Agent })
        }

        foreach ($agent in $agents) {
            if (-not $tokenAttempted) {
                $tokenAttempted = $true
                try { $definitionToken = Get-FabricDataAgentToken }
                catch {
                    Write-Host "[WARN] Fabric authentication failed for Data Agent definitions: $($_.Exception.Message). Agents will still be listed." -ForegroundColor Yellow
                }
            }
            $definition = $null
            $agentKey = "$workspaceId|$($agent.id)"
            if ($global:RunMode -eq 'Agents' -and $script:DataAgentDefinitions.ContainsKey($agentKey)) {
                $definition = $script:DataAgentDefinitions[$agentKey]
            }
            elseif ($definitionToken) {
                $definition = Get-FabricDataAgentDefinition -WorkspaceId $workspaceId -DataAgentId $agent.id -AccessToken $definitionToken
            }
            $cleanWorkspaceName = ($workspaceName -replace '\[', '(' -replace '\]', ')' -replace '[^a-zA-Z0-9\(\)&,.-]', ' ').TrimStart()
            $cleanAgentName = ($agent.displayName -replace '\[', '(' -replace '\]', ')' -replace '[^a-zA-Z0-9\(\)&,.-]', ' ').TrimStart()
            $combinedName = "$cleanWorkspaceName ~ $cleanAgentName"
            $schemaVersion = ""
            $aiInstructions = ""
            $sourceCount = 0
            $definitionRetrieved = $false

            if ($definition) {
                $definitionRetrieved = $true
                foreach ($part in $definition.parts) {
                    try {
                        if ($part.payloadType -ne 'InlineBase64') {
                            throw "Unsupported definition payload type '$($part.payloadType)'."
                        }
                        $relativePath = $part.path -replace '\\', '/'
                        if ($relativePath -notmatch '^Files/Config/(data_agent\.json|[^/]+/stage_config\.json|[^/]+/(?:.*/)?datasource\.json)$') {
                            continue
                        }
                        $partJson = [Text.Encoding]::UTF8.GetString([Convert]::FromBase64String($part.payload)) | ConvertFrom-Json
                        if ($relativePath -eq 'Files/Config/data_agent.json') {
                            $schemaVersion = $partJson.'$schema'
                        }
                        elseif ($relativePath -eq 'Files/Config/draft/stage_config.json') {
                            $aiInstructions = $partJson.aiInstructions
                        }
                        elseif ($relativePath.EndsWith('/datasource.json')) {
                            $stage = $relativePath.Split('/')[2]
                            $sourceType = [string]$partJson.type
                            $friendlyType = if ($sourceTypes.ContainsKey($sourceType)) { $sourceTypes[$sourceType] } else { $sourceType }
                            $resolvedName = $partJson.displayName
                            if ($partJson.artifactId -and $DatasetNames.ContainsKey($partJson.artifactId)) {
                                $resolvedName = $DatasetNames[$partJson.artifactId]
                            }
                            $sourceWorkspaceName = ($Workspaces | Where-Object { $_.WorkspaceId -eq $partJson.workspaceId } | Select-Object -First 1).WorkspaceName
                            if (-not $sourceWorkspaceName) { $sourceWorkspaceName = "Unknown Workspace" }
                            $sourcesInfo += [PSCustomObject][ordered]@{
                                "Data Agent ID" = $agent.id
                                "Data Agent Name" = $agent.displayName
                                "Workspace ID" = $workspaceId
                                "Workspace Name" = $workspaceName
                                "Stage" = $stage
                                "Source Type" = $friendlyType
                                "Source Type (Raw)" = $sourceType
                                "Source Display Name" = $partJson.displayName
                                "Connected Item ID" = $partJson.artifactId
                                "Connected Item Name" = $resolvedName
                                "Source Workspace ID" = $partJson.workspaceId
                                "Source Workspace Name" = $sourceWorkspaceName
                                "Source Instructions" = $partJson.dataSourceInstructions
                                "Source Description" = $partJson.userDescription
                                "Selected Top-Level Elements" = @($partJson.elements | Where-Object { $null -ne $_ }).Count
                                "Report Date" = $ReportDate
                                "Workspace Name - Data Agent Name" = $combinedName
                            }
                            if ($stage -eq 'draft') { $sourceCount++ }
                        }
                    }
                    catch {
                        $definitionRetrieved = $false
                        Write-Host "[WARN] Data Agent '$($agent.displayName)' part '$($part.path)' could not be extracted: $($_.Exception.Message)" -ForegroundColor Yellow
                    }
                }
            }

            $agentsInfo += [PSCustomObject][ordered]@{
                "Data Agent ID" = $agent.id
                "Data Agent Name" = $agent.displayName
                "Description" = $agent.description
                "Workspace ID" = $workspaceId
                "Workspace Name" = $workspaceName
                "Schema Version" = $schemaVersion
                "AI Instructions" = $aiInstructions
                "Connected Source Count" = $sourceCount
                "Definition Retrieved" = $definitionRetrieved
                "Sensitivity Label ID" = $agent.sensitivityLabel.id
                "Report Date" = $ReportDate
                "Workspace Name - Data Agent Name" = $combinedName
            }
        }
    }

    Write-Host "[INFO] Data Agent extraction complete. Agents found: $($agentsInfo.Count); connected sources: $($sourcesInfo.Count)." -ForegroundColor Cyan
    if ($agentsInfo.Count -eq 0) {
        $agentsInfo += [PSCustomObject][ordered]@{
            "Data Agent ID" = ""; "Data Agent Name" = ""; "Description" = ""; "Workspace ID" = ""; "Workspace Name" = ""
            "Schema Version" = ""; "AI Instructions" = ""; "Connected Source Count" = ""; "Definition Retrieved" = ""; "Sensitivity Label ID" = ""
            "Report Date" = ""; "Workspace Name - Data Agent Name" = ""
        }
    }
    if ($sourcesInfo.Count -eq 0) {
        $sourcesInfo += [PSCustomObject][ordered]@{
            "Data Agent ID" = ""; "Data Agent Name" = ""; "Workspace ID" = ""; "Workspace Name" = ""; "Stage" = ""
            "Source Type" = ""; "Source Type (Raw)" = ""; "Source Display Name" = ""; "Connected Item ID" = ""
            "Connected Item Name" = ""; "Source Workspace ID" = ""; "Source Workspace Name" = ""; "Source Instructions" = ""
            "Source Description" = ""; "Selected Top-Level Elements" = ""; "Report Date" = ""; "Workspace Name - Data Agent Name" = ""
        }
    }
    $outputPath = Join-Path $BaseFolderPath "Data Agent Detail.xlsx"
    if (Test-Path -LiteralPath $outputPath) { Remove-Item -LiteralPath $outputPath -Force }
    $agentsInfo | Export-Excel -Path $outputPath -WorksheetName "Data Agents" -AutoSize -ErrorAction Stop
    $sourcesInfo | Export-Excel -Path $outputPath -WorksheetName "Data Agent Sources" -AutoSize -Append -ErrorAction Stop
    Write-Output "Data Agent Detail Extraction Process Completed. Excel output saved to $outputPath"
}
