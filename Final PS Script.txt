function Write-ImpactIQDiagnostic {
    param(
        [ValidateSet('INFO', 'WARN', 'ERROR')][string]$Level,
        [AllowEmptyString()][string]$Message
    )

    $script:ImpactIQDiagnostics.Writer.WriteLine(('[{0:O}] [{1}] {2}' -f [DateTime]::UtcNow, $Level, $Message))
}

function Get-ImpactIQDiagnosticLevel {
    param([AllowEmptyString()][string]$Text)

    if ($Text -match '(?i)\b(?:errors?|fatal|exception|failed|failure)\b|\[ERROR\]') { return 'ERROR' }
    if ($Text -match '(?i)\bwarn(?:ing)?s?\b|\[WARN\]') { return 'WARN' }
}

function Write-ImpactIQRunRecord {
    param(
        [Parameter(Mandatory)][AllowNull()][object]$Record,
        [switch]$DiscardOutput
    )

    if ($Record -is [Management.Automation.ErrorRecord]) {
        # Redirection can clone an ErrorRecord while retaining the same exception.
        if ($script:ImpactIQDiagnostics.SeenExceptions.Add($Record.Exception)) {
            $message = $Record.Exception.Message
            if ($Record.InvocationInfo.ScriptLineNumber) {
                $message += " (line $($Record.InvocationInfo.ScriptLineNumber), $($Record.FullyQualifiedErrorId))"
            }
            Write-ImpactIQDiagnostic -Level ERROR -Message $message
        }
        return
    }
    if ($Record -is [Management.Automation.WarningRecord]) {
        Write-ImpactIQDiagnostic -Level WARN -Message $Record.Message
        return
    }
    if ($Record -is [Management.Automation.InformationRecord]) {
        $data = $Record.MessageData
        $text = [string]$data
        $level = Get-ImpactIQDiagnosticLevel $text
        if (-not $level -and $data -is [Management.Automation.HostInformationMessage]) {
            if ($data.ForegroundColor -eq [ConsoleColor]::Red) { $level = 'ERROR' }
            elseif ($data.ForegroundColor -eq [ConsoleColor]::Yellow) { $level = 'WARN' }
        }
        if ($level) {
            Write-ImpactIQDiagnostic -Level $level -Message $text
        }
        elseif (-not $DiscardOutput) {
            if ($data -is [Management.Automation.HostInformationMessage]) {
                $parameters = @{ Object = $data.Message; NoNewline = $data.NoNewLine }
                if ($null -ne $data.ForegroundColor) { $parameters.ForegroundColor = $data.ForegroundColor }
                if ($null -ne $data.BackgroundColor) { $parameters.BackgroundColor = $data.BackgroundColor }
                Microsoft.PowerShell.Utility\Write-Host @parameters
            }
            else {
                Microsoft.PowerShell.Utility\Write-Information -MessageData $data -Tags $Record.Tags -InformationAction Continue
            }
        }
        return
    }
    if ($Record -is [string]) {
        $level = Get-ImpactIQDiagnosticLevel $Record
        if ($level) {
            Write-ImpactIQDiagnostic -Level $level -Message $Record
            return
        }
    }
    if (-not $DiscardOutput) { Write-Output -NoEnumerate $Record }
}

function Write-ImpactIQContinuingError {
    param(
        [Parameter(Mandatory)][Management.Automation.ErrorRecord]$Record,
        [string]$Context = 'Extraction step'
    )

    $script:ImpactIQRunFailed = $true
    Write-ImpactIQRunRecord -Record $Record
    Write-ImpactIQDiagnostic -Level WARN -Message "$Context could not complete; continuing with the remaining work."
}

function Invoke-ImpactIQLoggedRun {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)][Alias('LogPath')][string]$ImpactIQLogPath,
        [Parameter(Mandatory)][Alias('Action')][scriptblock]$ImpactIQAction,
        [switch]$KeepOpen,
        [switch]$StartSession
    )

    $impactIQPreviousErrorPreference = $ErrorActionPreference
    $impactIQPreviousWarningPreference = $WarningPreference
    $impactIQCloseWriter = $false
    $ErrorActionPreference = 'SilentlyContinue'
    try {
        if ($StartSession -or -not ($script:ImpactIQDiagnostics -and $script:ImpactIQDiagnostics.SessionOpen)) {
            if ($script:ImpactIQDiagnostics -and $script:ImpactIQDiagnostics.Writer) {
                $script:ImpactIQDiagnostics.Writer.Dispose()
            }
            $script:ImpactIQRunFailed = $false
            $script:ImpactIQLogUnavailable = $false
            $script:ImpactIQDiagnostics = @{
                Writer = $null
                LogPath = $ImpactIQLogPath
                SessionOpen = $true
                SeenExceptions = [Collections.Generic.HashSet[Exception]]::new()
            }
            foreach ($impactIQRecord in @($global:Error)) { $null = $script:ImpactIQDiagnostics.SeenExceptions.Add($impactIQRecord.Exception) }
        }
        if ($script:ImpactIQDiagnostics.LogPath -ne $ImpactIQLogPath) {
            throw 'The diagnostic log path cannot change during an extraction session.'
        }
        if (-not $script:ImpactIQDiagnostics.Writer) {
            $null = [IO.Directory]::CreateDirectory((Split-Path -Path $ImpactIQLogPath -Parent))
            $script:ImpactIQDiagnostics.Writer = [IO.StreamWriter]::new($ImpactIQLogPath, $true, [Text.UTF8Encoding]::new($false))
            $script:ImpactIQDiagnostics.Writer.AutoFlush = $true
            Write-ImpactIQDiagnostic -Level INFO -Message 'ImpactIQ run started.'
        }
        $ErrorActionPreference = 'SilentlyContinue'
        $WarningPreference = 'Continue'
        try {
            . $ImpactIQAction 2>&1 3>&1 6>&1 | ForEach-Object { Write-ImpactIQRunRecord -Record $_ }
        }
        catch {
            $script:ImpactIQRunFailed = $true
            Write-ImpactIQRunRecord -Record $_
        }
        finally {
            $impactIQErrorHistory = @($global:Error)
            [array]::Reverse($impactIQErrorHistory)
            foreach ($impactIQRecord in $impactIQErrorHistory) { Write-ImpactIQRunRecord -Record $impactIQRecord }
            if (-not $KeepOpen -and $script:ImpactIQRunFailed) {
                Write-ImpactIQDiagnostic -Level WARN -Message 'The run encountered errors. Some steps or outputs may be incomplete.'
            }
        }
    }
    catch {
        $script:ImpactIQRunFailed = $true
        $script:ImpactIQLogUnavailable = $true
        $impactIQCloseWriter = $true
        $impactIQFallbackPath = Join-Path ([IO.Path]::GetTempPath()) 'ImpactIQ-errorlog.txt'
        $impactIQLogMessage = "Cannot write diagnostics to '$ImpactIQLogPath': " + $_.Exception.Message
        try {
            [IO.File]::AppendAllText($impactIQFallbackPath, ('[{0:O}] [ERROR] {1}{2}' -f [DateTime]::UtcNow, $impactIQLogMessage, [Environment]::NewLine), [Text.UTF8Encoding]::new($false))
        }
        catch [IO.IOException] { $script:ImpactIQLogUnavailable = $true }
        catch [UnauthorizedAccessException] { $script:ImpactIQLogUnavailable = $true }
    }
    finally {
        $script:ImpactIQDiagnostics.SessionOpen = [bool]$KeepOpen
        if ((-not $KeepOpen -or $impactIQCloseWriter) -and $script:ImpactIQDiagnostics.Writer) {
            try { $script:ImpactIQDiagnostics.Writer.Dispose() }
            catch [IO.IOException] { $script:ImpactIQRunFailed = $true; $script:ImpactIQLogUnavailable = $true }
            finally { $script:ImpactIQDiagnostics.Writer = $null }
        }
        $ErrorActionPreference = $impactIQPreviousErrorPreference
        $WarningPreference = $impactIQPreviousWarningPreference
    }
}

function Invoke-ImpactIQNativeCommand {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)][string]$FilePath,
        [string[]]$ArgumentList = @(),
        [switch]$DiscardOutput
    )

    $ErrorActionPreference = 'Continue'
    try { $null = Get-Command -Name $FilePath -CommandType Application -ErrorAction Stop }
    catch [Management.Automation.CommandNotFoundException] {
        $global:LASTEXITCODE = 1
        Write-ImpactIQContinuingError -Record $_ -Context "Native command '$FilePath'"
        return
    }
    & $FilePath @ArgumentList 2>&1 | ForEach-Object {
        Write-ImpactIQRunRecord -Record $_ -DiscardOutput:$DiscardOutput
    }
    $exitCode = $LASTEXITCODE
    $global:LASTEXITCODE = $exitCode
    if ($exitCode -ne 0) {
        Write-ImpactIQDiagnostic -Level ERROR -Message ("Native command '{0}' exited with code {1}." -f [IO.Path]::GetFileName($FilePath), $exitCode)
    }
}

function Invoke-ImpactIQNativeProcess {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)][string]$FilePath,
        [string[]]$ArgumentList = @(),
        [string]$WorkingDirectory = (Get-Location).ProviderPath,
        [switch]$PassThru,
        [switch]$StartOnly
    )

    $startInfo = [Diagnostics.ProcessStartInfo]::new()
    $startInfo.FileName = $FilePath
    $startInfo.Arguments = $ArgumentList -join ' '
    $startInfo.WorkingDirectory = $WorkingDirectory
    $startInfo.UseShellExecute = $false
    $startInfo.CreateNoWindow = $true
    $startInfo.RedirectStandardOutput = $true
    $startInfo.RedirectStandardError = $true
    $process = [Diagnostics.Process]::new()
    $process.StartInfo = $startInfo
    $returnedProcess = $false
    $started = $false
    try {
        $null = $process.Start()
        $started = $true
        $run = [PSCustomObject]@{
            Process = $process
            FilePath = $FilePath
            StandardOutput = $process.StandardOutput.ReadToEndAsync()
            StandardError = $process.StandardError.ReadToEndAsync()
        }
        if ($StartOnly) {
            $returnedProcess = $true
            return $run
        }
        $process.WaitForExit()
        Write-ImpactIQNativeProcessResult -Run $run
        if ($PassThru) { $returnedProcess = $true; $process }
    }
    finally {
        if (-not $returnedProcess) {
            try {
                if ($StartOnly -and $started -and -not $process.HasExited) {
                    Stop-Process -Id $process.Id -Force -ErrorAction Stop
                    $process.WaitForExit()
                }
            }
            finally { $process.Dispose() }
        }
    }
}

function Write-ImpactIQNativeProcessResult {
    param([Parameter(Mandatory)]$Run)

    foreach ($line in ($Run.StandardOutput.GetAwaiter().GetResult() -split '\r?\n')) {
        if (-not $line) { continue }
        $level = Get-ImpactIQDiagnosticLevel $line
        if ($level) { Write-ImpactIQDiagnostic -Level $level -Message $line }
        else { Microsoft.PowerShell.Utility\Write-Host $line }
    }
    foreach ($line in ($Run.StandardError.GetAwaiter().GetResult() -split '\r?\n')) {
        if (-not $line) { continue }
        $level = Get-ImpactIQDiagnosticLevel $line
        if (-not $level) { $level = 'ERROR' }
        Write-ImpactIQDiagnostic -Level $level -Message $line
    }
    $global:LASTEXITCODE = $Run.Process.ExitCode
    if ($Run.Process.ExitCode -ne 0) {
        Write-ImpactIQDiagnostic -Level ERROR -Message ("Native process '{0}' exited with code {1}." -f [IO.Path]::GetFileName($Run.FilePath), $Run.Process.ExitCode)
    }
}

function Get-ImpactIQToolVersion {
    param([Parameter(Mandatory)][string]$Version)

    $match = [regex]::Match($Version.Trim(), '^v?(?<core>\d+\.\d+(?:\.\d+){0,2})(?<suffix>-[0-9A-Za-z.-]+)?(?:\+[0-9A-Za-z.-]+)?$')
    if (-not $match.Success) { throw "Cannot compare tool version '$Version'." }
    $number = [version]$match.Groups['core'].Value
    $normalized = '{0}.{1}.{2}' -f $number.Major, $number.Minor, [Math]::Max(0, $number.Build)
    if ($number.Revision -gt 0) { $normalized += '.' + $number.Revision }
    $normalized + $match.Groups['suffix'].Value
}

function Invoke-ImpactIQToolUpdate {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)][string]$Name,
        [Parameter(Mandatory)][string]$Version,
        [Parameter(Mandatory)][uri]$DownloadUrl,
        [Parameter(Mandatory)][string]$FolderPath,
        [Parameter(Mandatory)][string]$ExecutableName
    )

    $ErrorActionPreference = 'Stop'
    if (-not $DownloadUrl.IsAbsoluteUri -or $DownloadUrl.Scheme -ne 'https') {
        throw "Tool download URL must use HTTPS: $DownloadUrl"
    }
    Write-Host "[INFO] Checking manifest-pinned $Name release..."
    $targetVersion = Get-ImpactIQToolVersion -Version $Version
    $executablePath = Join-Path $FolderPath $ExecutableName
    if (Test-Path -LiteralPath $executablePath -PathType Leaf) {
        try {
            # FileVersion contains build numbers; ProductVersion matches the release tag.
            $installedVersion = Get-ImpactIQToolVersion -Version (Get-Item -LiteralPath $executablePath -ErrorAction Stop).VersionInfo.ProductVersion
        }
        catch {
            $installedVersion = $null
            Write-Warning "Cannot read the installed $Name release version: $($_.Exception.Message). Attempting an update."
        }
        if ($installedVersion -ceq $targetVersion) {
            Write-Host "[INFO] $Name $targetVersion is already installed; skipping download and extraction."
            return
        }
    }

    $temporaryPath = Join-Path ([IO.Path]::GetTempPath()) ('ImpactIQ-Tool-' + [guid]::NewGuid().ToString('N'))
    try {
        $null = [IO.Directory]::CreateDirectory($temporaryPath)
        $archivePath = Join-Path $temporaryPath 'download.zip'
        $extractPath = Join-Path $temporaryPath 'extracted'
        Write-Host "[INFO] Downloading $Name $targetVersion..."
        Invoke-WebRequest -Uri $DownloadUrl.AbsoluteUri -OutFile $archivePath -UseBasicParsing -TimeoutSec 300 -ErrorAction Stop
        Add-Type -AssemblyName System.IO.Compression.FileSystem -ErrorAction Stop
        [IO.Compression.ZipFile]::ExtractToDirectory($archivePath, $extractPath)
        $executables = @(Get-ChildItem -LiteralPath $extractPath -Filter $ExecutableName -File -Recurse -ErrorAction Stop)
        if ($executables.Count -ne 1) { throw "$Name archive must contain exactly one $ExecutableName." }
        $downloadedVersion = Get-ImpactIQToolVersion -Version $executables[0].VersionInfo.ProductVersion
        if ($downloadedVersion -cne $targetVersion) {
            throw "$Name archive version '$downloadedVersion' does not match manifest version '$targetVersion'."
        }
        $null = [IO.Directory]::CreateDirectory($FolderPath)
        Copy-Item -Path (Join-Path $executables[0].DirectoryName '*') -Destination $FolderPath -Recurse -Force -ErrorAction Stop
        $installedVersion = Get-ImpactIQToolVersion -Version (Get-Item -LiteralPath $executablePath -ErrorAction Stop).VersionInfo.ProductVersion
        if ($installedVersion -cne $targetVersion) { throw "$Name installed version does not match the downloaded release." }
        Write-Host "[INFO] Updated $Name to $targetVersion."
    }
    finally {
        if (Test-Path -LiteralPath $temporaryPath) { Remove-Item -LiteralPath $temporaryPath -Recurse -Force -ErrorAction Stop }
    }
}

function Initialize-ImpactIQExecutionPolicy {
    [CmdletBinding()]
    param()

    $ErrorActionPreference = 'Stop'
    if ((Get-ExecutionPolicy -ErrorAction Stop) -eq 'Bypass') { return }
    foreach ($scope in @('MachinePolicy', 'UserPolicy')) {
        $policy = Get-ExecutionPolicy -Scope $scope -ErrorAction Stop
        if ($policy -ne 'Undefined') {
            Write-ImpactIQDiagnostic -Level WARN -Message "$scope enforces execution policy '$policy'. Process Bypass cannot override Group Policy; continuing under the enforced policy."
            return
        }
    }
    Set-ExecutionPolicy -ExecutionPolicy Bypass -Scope Process -Force -ErrorAction Stop
    if ((Get-ExecutionPolicy -ErrorAction Stop) -ne 'Bypass') {
        throw 'Process-scoped Bypass could not be enabled. Check the effective execution policy.'
    }
}

function Initialize-ImpactIQModules {
    [CmdletBinding()]
    param()

    $ErrorActionPreference = 'Stop'
    $installationReady = $false
    $previousSecurityProtocol = $null
    $requiredModules = @('ImportExcel', 'Az.Accounts', 'MicrosoftPowerBIMgmt.Profile')
    try {
        foreach ($module in $requiredModules) {
            try {
                $moduleInfo = Import-Module -Name $module -PassThru -ErrorAction Stop
                if (-not $moduleInfo) { throw "Import did not confirm that $module was loaded." }
                continue
            }
            catch {
                Write-ImpactIQDiagnostic -Level WARN -Message "Required module '$module' could not be imported: $($_.Exception.Message). Attempting one CurrentUser installation from PSGallery."
            }

            try {
                if (-not $installationReady) {
                    Get-Command Install-Module -ErrorAction Stop | Out-Null
                    $previousSecurityProtocol = [Net.ServicePointManager]::SecurityProtocol
                    [Net.ServicePointManager]::SecurityProtocol = $previousSecurityProtocol -bor [Net.SecurityProtocolType]::Tls12
                    if (-not (Get-PackageProvider -Name NuGet -ListAvailable -ErrorAction SilentlyContinue |
                              Where-Object { $_.Version -ge [version]'2.8.5.201' })) {
                        Install-PackageProvider -Name NuGet -MinimumVersion 2.8.5.201 -Scope CurrentUser -Force -ErrorAction Stop | Out-Null
                    }
                    $installationReady = $true
                }
                Install-Module -Name $module -Repository PSGallery -Scope CurrentUser -Force -ErrorAction Stop
                $moduleInfo = Import-Module -Name $module -PassThru -Force -ErrorAction Stop
                if (-not $moduleInfo) { throw "Import did not confirm that $module was loaded." }
            }
            catch {
                throw "Required module '$module' could not be installed and imported after one retry. $($_.Exception.Message) Check PSGallery access and organizational policy; for loaded-assembly conflicts, retry in a fresh PowerShell process."
            }
        }
    }
    finally {
        if ($null -ne $previousSecurityProtocol) {
            [Net.ServicePointManager]::SecurityProtocol = $previousSecurityProtocol
        }
    }
}

# Install pinned tool versions from Config\tool-manifest.json when the installed version differs.
# A missing or invalid manifest is logged and the tools already installed are used.
$bundledTools = @()
try {
    $toolManifestPath = Join-Path $configFolder 'tool-manifest.json'
    if (-not (Test-Path -LiteralPath $toolManifestPath -PathType Leaf)) { throw "Tool manifest is missing: $toolManifestPath" }
    $toolManifest = Get-Content -LiteralPath $toolManifestPath -Raw -ErrorAction Stop | ConvertFrom-Json -ErrorAction Stop
    if (-not $toolManifest.tools) { throw "Tool manifest contains no tools: $toolManifestPath" }
    $bundledTools = @($toolManifest.tools)
}
catch { Write-ImpactIQContinuingError -Record $_ -Context 'Tool manifest' }
foreach ($tool in $bundledTools) {
    try {
        foreach ($field in @('name', 'version', 'url', 'destination', 'executable')) { if (-not $tool.$field) { throw "Tool manifest entry is missing '$field'." } }
        Invoke-ImpactIQToolUpdate -Name $tool.name -Version $tool.version -DownloadUrl $tool.url -FolderPath (Join-Path $configFolder $tool.destination) -ExecutableName $tool.executable
    }
    catch { Write-ImpactIQContinuingError -Record $_ -Context "$($tool.name) update" }
}

        $teTestScriptPath = Join-Path $env:TEMP "TabularEditor_PreflightCheck.cs"
        Set-Content -Path $teTestScriptPath -Value 'Info("Tabular Editor preflight OK");' -Encoding UTF8

        $teTestArgs = "`"$blankModelPath`" -S `"$teTestScriptPath`""
        $teProcess = Invoke-ImpactIQNativeProcess -FilePath $TabularEditor2Path -PassThru -ArgumentList $teTestArgs -ErrorAction Stop

        if ($teProcess.ExitCode -eq 0) {
            $tabularEditorWorks = $true
        }

        Remove-Item -Path $teTestScriptPath -Force -ErrorAction SilentlyContinue
    }
}
catch {
    $tabularEditorWorks = $false
}

# Read-Host With 60 Second Timeout
# ================================
function Read-HostWithTimeout {
    param(
        [string]$Prompt,
        [int]$TimeoutSeconds = 60
    )

    Write-Host -NoNewline "$Prompt "
    Write-Host "(waiting up to $TimeoutSeconds seconds):" -ForegroundColor DarkGray

    $inputBuffer = ""
    $endTime = (Get-Date).AddSeconds($TimeoutSeconds)

    while ((Get-Date) -lt $endTime) {
        if ([Console]::KeyAvailable) {
            $key = [Console]::ReadKey($true)
            # Enter pressed - stop reading
            if ($key.Key -eq "Enter") {
                Write-Host ""
                return $inputBuffer
            }
            # Backspace
            elseif ($key.Key -eq "Backspace") {
                if ($inputBuffer.Length -gt 0) {
                    $inputBuffer = $inputBuffer.Substring(0, $inputBuffer.Length - 1)
                    Write-Host -NoNewline "`b `b"
                }
            }
            else {
                $inputBuffer += $key.KeyChar
                Write-Host -NoNewline $key.KeyChar
            }
        }
        Start-Sleep -Milliseconds 100
    }

    Write-Host "Timeout reached - defaulting to Public" -ForegroundColor Yellow
    return ""
}

# Show Environment Selection Dialog

function Show-EnvironmentSelectionDialog {
    param(
        [int]$TimeoutSeconds = 60
    )

    Add-Type -AssemblyName System.Windows.Forms
    Add-Type -AssemblyName System.Drawing

    $form = New-Object System.Windows.Forms.Form
    $form.Text = "Select Power BI Environment"
    $form.StartPosition = 'CenterScreen'
    $form.Size = New-Object System.Drawing.Size(400, 320)
    $form.TopMost = $true
    $form.FormBorderStyle = 'FixedDialog'
    $form.MaximizeBox = $false

    # Label
    $label = New-Object System.Windows.Forms.Label
    $label.Text = "Select your Power BI cloud environment:"
    $label.AutoSize = $true
    $label.Location = New-Object System.Drawing.Point(20, 20)
    $form.Controls.Add($label)

    # ListBox for environment selection
    $listBox = New-Object System.Windows.Forms.ListBox
    $listBox.Location = New-Object System.Drawing.Point(20, 50)
    $listBox.Size = New-Object System.Drawing.Size(340, 150)
    $listBox.SelectionMode = 'One'
    
    # Add environment options
    $environments = @(
        'Public (Commercial)',
        'Germany',
        'USGov',
        'China',
        'USGovHigh',
        'USGovMil'
    )
    
    foreach ($env in $environments) {
        [void]$listBox.Items.Add($env)
    }
    
    # Set default selection to Public
    $listBox.SelectedIndex = 0
    $form.Controls.Add($listBox)

    # Timeout label
    $timeoutLabel = New-Object System.Windows.Forms.Label
    $timeoutLabel.Text = "Timeout in $TimeoutSeconds seconds (defaults to Public)"
    $timeoutLabel.AutoSize = $true
    $timeoutLabel.Location = New-Object System.Drawing.Point(20, 210)
    $timeoutLabel.ForeColor = [System.Drawing.Color]::Gray
    $form.Controls.Add($timeoutLabel)

    # OK button
    $okButton = New-Object System.Windows.Forms.Button
    $okButton.Text = "OK"
    $okButton.Size = New-Object System.Drawing.Size(75, 30)
    $okButton.Location = New-Object System.Drawing.Point(205, 240)
    $okButton.Add_Click({
        $form.Tag = $listBox.SelectedItem
        $form.Close()
    })
    $form.Controls.Add($okButton)

    # Cancel button
    $cancelButton = New-Object System.Windows.Forms.Button
    $cancelButton.Text = "Cancel"
    $cancelButton.Size = New-Object System.Drawing.Size(75, 30)
    $cancelButton.Location = New-Object System.Drawing.Point(285, 240)
    $cancelButton.Add_Click({
        $form.Tag = 'Cancelled'
        $form.Close()
    })
    $form.Controls.Add($cancelButton)

    # Timer for timeout
    $timer = New-Object System.Windows.Forms.Timer
    $timer.Interval = $TimeoutSeconds * 1000
    $timer.Add_Tick({
        $timer.Stop()
        $form.Tag = 'Timeout'
        $form.Close()
    })

    # Add FormClosing event handler
    $form.Add_FormClosing({
        param($sender, $e)
        $timer.Stop()
        $timer.Dispose()
        
        # If Tag is not set (form closed without button click), mark as cancelled
        if (-not $form.Tag) {
            $form.Tag = 'Cancelled'
        }
    })

    # Handle double-click on list item (same as OK button)
    $listBox.Add_DoubleClick({
        $form.Tag = $listBox.SelectedItem
        $form.Close()
    })

    # Set default button and show
    $form.AcceptButton = $okButton
    $form.CancelButton = $cancelButton

# Azure sign-in (Fabric). Power BI signs in separately through Connect-PowerBI.
function Connect-ImpactIQ {
    [CmdletBinding()]
    param()

    $ErrorActionPreference = 'Stop'
    $script:ImpactIQAzContext = $null
    $azEnvironment = switch ($EnvironmentForEndpoints) {
        'China' { 'AzureChinaCloud' }
        'USGov' { 'AzureUSGovernment' }
        'USGovHigh' { 'AzureUSGovernment' }
        'USGovMil' { 'AzureUSGovernment' }
        default { 'AzureCloud' }
    }
    Disable-AzContextAutosave -Scope Process -ErrorAction Stop | Out-Null
    if (Get-Command Update-AzConfig -ErrorAction SilentlyContinue) {
        Update-AzConfig -LoginExperienceV2 Off -Scope Process -ErrorAction Stop | Out-Null
    }
    Write-Host "[INFO] Connecting to Microsoft Fabric through Az.Accounts..." -ForegroundColor Cyan
    Connect-AzAccount -Environment $azEnvironment -Scope Process -SkipContextPopulation -ErrorAction Stop -WarningAction Continue | Out-Null
    $context = Get-AzContext -ErrorAction Stop
    if (-not $context -or $context.Environment.Name -ne $azEnvironment -or
        -not $context.Tenant.Id -or -not $context.Account.Id) {
        throw 'Azure sign-in did not return a valid account and tenant for the selected cloud.'
    }
    $script:ImpactIQAzContext = $context
}

# The Az.Accounts token for the Power BI audience only carries user_impersonation (Azure PowerShell client),
# which the XMLA endpoint rejects. Power BI/XMLA tokens must come from the Power BI Mgmt client instead.
function Connect-PowerBI {
    [CmdletBinding()]
    param(
        [int]$MaxAttempts = 2
    )

    $previousErrorActionPreference = $ErrorActionPreference
    try {
        $ErrorActionPreference = 'Stop'
        if (-not $script:ImpactIQAzContext) {
            throw 'Power BI sign-in requires a completed Azure sign-in first.'
        }
        for ($attempt = 1; $attempt -le $MaxAttempts; $attempt++) {
            try {
                Write-Host "[INFO] Connecting to Power BI using environment: $EnvironmentForEndpoints" -ForegroundColor Cyan
                Connect-PowerBIServiceAccount -Tenant $script:ImpactIQAzContext.Tenant.Id -Environment $EnvironmentForEndpoints -ErrorAction Stop -WarningAction Continue | Out-Null
                $authorization = [string](Get-PowerBIAccessToken -ErrorAction Stop -WarningAction Continue).Authorization
                if ([string]::IsNullOrWhiteSpace(($authorization -replace '^Bearer\s*', ''))) {
                    throw 'Power BI sign-in returned an empty access token.'
                }
                Write-Host "[INFO] Connected to Power BI." -ForegroundColor Green
                return
            }
            catch {
                Disconnect-PowerBIServiceAccount -ErrorAction SilentlyContinue | Out-Null
                if ($attempt -ge $MaxAttempts) {
                    Write-Host "[ERROR] Power BI sign-in failed: $($_.Exception.Message)" -ForegroundColor Red
                    throw
                }
                Write-Warning "Power BI sign-in attempt $attempt failed: $($_.Exception.Message). Retrying..."
                Start-Sleep -Seconds 2
            }
        }
    }
    finally {
        $ErrorActionPreference = $previousErrorActionPreference
    }
}

# MSAL renews the Power BI token silently; reconnect only Power BI if renewal fails.
function Get-ImpactIQPowerBIToken {
    [CmdletBinding()]
    param()

    $ErrorActionPreference = 'Stop'
    $readToken = {
        $value = ([string](Get-PowerBIAccessToken -ErrorAction Stop -WarningAction Continue).Authorization) -replace '^Bearer\s*', ''
        if ([string]::IsNullOrWhiteSpace($value)) { throw 'Power BI authentication returned an empty access token.' }
        $value
    }
    try { return (& $readToken) }
    catch {
        Write-Warning "Power BI token fetch failed ($($_.Exception.Message)); reconnecting Power BI."
        Connect-PowerBI -MaxAttempts 2 -ErrorAction Stop
        return (& $readToken)
    }
}

function Get-ImpactIQAccessToken {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)]
        [string]$ResourceUrl,
        # Fabric App hosting needs the Azure-issued Power BI-audience token; XMLA needs the Power BI client token.
        [switch]$Azure
    )

    $ErrorActionPreference = 'Stop'
    if ($ResourceUrl -notin @($global:PowerBIEndpoints.ResourceUrl, $global:PowerBIEndpoints.FabricApiPrefix)) {
        throw 'Token resource does not match the selected Power BI or Fabric cloud.'
    }
    if ($ResourceUrl -eq $global:PowerBIEndpoints.ResourceUrl -and -not $Azure) {
        return Get-ImpactIQPowerBIToken -ErrorAction Stop
    }
    if (-not $script:ImpactIQAzContext) {
        throw 'No Azure session is available. Restart the run and complete the Az.Accounts sign-in.'
    }
    try {
        $tokenValue = (Get-AzAccessToken -ResourceUrl $ResourceUrl -TenantId $script:ImpactIQAzContext.Tenant.Id -DefaultProfile $script:ImpactIQAzContext -ErrorAction Stop -WarningAction Continue).Token
        if ($tokenValue -is [Security.SecureString]) {
            $tokenPointer = [Runtime.InteropServices.Marshal]::SecureStringToBSTR($tokenValue)
            try { $tokenValue = [Runtime.InteropServices.Marshal]::PtrToStringBSTR($tokenPointer) }
            finally { [Runtime.InteropServices.Marshal]::ZeroFreeBSTR($tokenPointer) }
        }
        if ([string]::IsNullOrWhiteSpace([string]$tokenValue)) {
            throw "Azure authentication returned an empty token for $ResourceUrl."
        }
        return [string]$tokenValue
    }
    finally { $tokenValue = $null }
}

function Invoke-ImpactIQRestMethod {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)][uri]$Url,
        [ValidateSet('GET')][string]$Method = 'GET'
    )

    $ErrorActionPreference = 'Stop'
    if ($Url.Scheme -ne 'https' -or $Url.Port -ne 443 -or $Url.UserInfo -or $Url.Fragment) {
        throw 'Metadata API requests require an HTTPS URL without user information or a fragment.'
    }
    $powerBiHost = ([uri]$global:PowerBIEndpoints.ApiPrefix).DnsSafeHost
    $fabricHost = ([uri]$global:PowerBIEndpoints.FabricApiPrefix).DnsSafeHost
    $resource = if ($Url.DnsSafeHost -ieq $powerBiHost) { $global:PowerBIEndpoints.ResourceUrl }
        elseif ($Url.DnsSafeHost -ieq $fabricHost) { $global:PowerBIEndpoints.FabricApiPrefix }
        else { throw "Unsupported metadata API host: $($Url.DnsSafeHost)" }
    $headers = @{ Authorization = ('Bearer ' + (Get-ImpactIQAccessToken -ResourceUrl $resource -ErrorAction Stop)) }
    try {
        $response = Invoke-WebRequest -Uri $Url -Method $Method -Headers $headers -UseBasicParsing -MaximumRedirection 0 -TimeoutSec 100 -ErrorAction Stop
        if ($response.Content -is [byte[]]) { return [Text.Encoding]::UTF8.GetString($response.Content) }
        return [string]$response.Content
    }
    finally { $headers = $null }
}

function Invoke-FabricPagedGet {
    param(
        [Parameter(Mandatory = $true)]
        [string]$Url
    )

    $results = @()
    $nextUrl = $Url
    $seenUrls = [Collections.Generic.HashSet[string]]::new([StringComparer]::Ordinal)

    while ($nextUrl) {
        $uri = [uri]$nextUrl
        if ($uri.DnsSafeHost -ine ([uri]$global:PowerBIEndpoints.FabricApiPrefix).DnsSafeHost -or -not $seenUrls.Add($uri.AbsoluteUri)) {
            throw 'Unexpected or repeated Fabric pagination URL.'
        }
        $response = Invoke-ImpactIQRestMethod -Url $uri -Method GET -ErrorAction Stop | ConvertFrom-Json -ErrorAction Stop

        if ($response.value) {
            $results += @($response.value)
        }

        $nextUrl = if ($response.continuationUri) {
            $response.continuationUri
        } elseif ($response.continuationToken) {
            $builder = [UriBuilder]::new($uri)
            $queryParts = @($uri.Query.TrimStart('?') -split '&' |
                Where-Object { $_ -and $_ -notmatch '^continuationToken=' })
            $builder.Query = ($queryParts + ('continuationToken=' + [uri]::EscapeDataString([string]$response.continuationToken))) -join '&'
            $builder.Uri.AbsoluteUri
        } else {
            $null
        }
    }

    return $results
}

# Function to rename properties in objects and handle duplicates
function Rename-Properties {
    param ($object, $renameMap)
    $newObject = New-Object PSObject
    foreach ($originalName in $renameMap.Keys) {
        $newPropertyName = $renameMap[$originalName]
        $propertyValue = if ($object.PSObject.Properties[$originalName]) { $object.$originalName } else { $null }
        if ($newObject.PSObject.Properties[$newPropertyName]) { $newPropertyName += "_duplicate" }
        $newObject | Add-Member -MemberType NoteProperty -Name $newPropertyName -Value $propertyValue
    }
    foreach ($property in $object.PSObject.Properties) {
        if (-not $renameMap.ContainsKey($property.Name)) {
            $newObject | Add-Member -MemberType NoteProperty -Name $property.Name -Value $property.Value
        }
    }
    return $newObject
}

    param(

        [array]$Workspaces, # array of PSCustomObjects with id + name
        [switch]$RequireSelection
    )

    Add-Type -AssemblyName System.Windows.Forms
    Add-Type -AssemblyName System.Drawing

    $form.Controls.Add($lbl)

    # Search label
    $lblSearch          = New-Object System.Windows.Forms.Label
    $lblSearch.Text     = "Search:"
    $lblSearch.AutoSize = $true
    $lblSearch.Location = New-Object System.Drawing.Point(12,40)
    $form.Controls.Add($lblSearch)

    # Search TextBox
    $txtSearch          = New-Object System.Windows.Forms.TextBox
    $txtSearch.Location = New-Object System.Drawing.Point(70,37)
    $txtSearch.Size     = New-Object System.Drawing.Size(422,23)
    $form.Controls.Add($txtSearch)

    # CheckedListBox
    $clb                = New-Object System.Windows.Forms.CheckedListBox
    $clb.Location       = New-Object System.Drawing.Point(12,70)
	$clb.Size           = New-Object System.Drawing.Size(480,380)
    $clb.CheckOnClick   = $true
    $clb.Sorted         = $true

    # Store all workspace items for filtering
    $allWorkspaceItems = @()
    foreach ($ws in $Workspaces) {
        $display = "{0} ({1})" -f $ws.name, $ws.id
        $allWorkspaceItems += [pscustomobject]@{ Display=$display; Id=$ws.id; Name=$ws.name }
    }

    # Maintain persistent checked state across searches
    $persistentCheckedIds = @{}

    # Initial population
    foreach ($item in $allWorkspaceItems) {
        [void]$clb.Items.Add($item)
    }
    $clb.DisplayMember = 'Display'
    $form.Controls.Add($clb)

	# Track when items are checked/unchecked
    $clb.Add_ItemCheck({
        param($sender, $e)
        $item = $clb.Items[$e.Index]
        if ($e.NewValue -eq 'Checked') {
            $persistentCheckedIds[$item.Id] = $true
        } elseif ($e.NewValue -eq 'Unchecked') {
            $persistentCheckedIds.Remove($item.Id)
        }
    })

	# Search filter logic
    $txtSearch.Add_TextChanged({
        $searchText = $txtSearch.Text
        $clb.BeginUpdate()
        
        $clb.Items.Clear()
        
        # Escape search text once for reuse in loop
        $escapedSearchText = [regex]::Escape($searchText)
        
        # Filter and re-add items
        foreach ($item in $allWorkspaceItems) {
            if ([string]::IsNullOrWhiteSpace($searchText) -or 
                $item.Display -imatch $escapedSearchText) {
                $index = $clb.Items.Add($item)
                # Restore checked state from persistent store
                if ($persistentCheckedIds.ContainsKey($item.Id)) {
                    $clb.SetItemChecked($index, $true)
                }
            }
        }
        
        $clb.EndUpdate()
    })

    # "Select All" checkbox
    $chkAll            = New-Object System.Windows.Forms.CheckBox
    $chkAll.Text       = "Select All"
    $chkAll.AutoSize   = $true
    $chkAll.Location   = New-Object System.Drawing.Point(12,460)

            $clb.SetItemChecked($i, $chkAll.Checked)
        }
    })
    $form.Controls.Add($chkAll)

    # "Clear All" button
    $btnClearAll       = New-Object System.Windows.Forms.Button
    $btnClearAll.Text  = "Clear All"
    $btnClearAll.Width = 80
    $btnClearAll.Location = New-Object System.Drawing.Point(90,458)
    $btnClearAll.Add_Click({
        # Uncheck all items - ItemCheck events will update persistent store
        for ($i=0; $i -lt $clb.Items.Count; $i++) {
            $clb.SetItemChecked($i, $false)
        }
        $chkAll.Checked = $false
    })
    $form.Controls.Add($btnClearAll)

    # "Include My Workspace" checkbox
    $chkMy             = New-Object System.Windows.Forms.CheckBox
    $chkMy.Text        = "Include 'My Workspace'"
    $chkMy.AutoSize    = $true
    $chkMy.Location    = New-Object System.Drawing.Point(184,460)
    $chkMy.Checked     = $false
    $form.Controls.Add($chkMy)

    # OK button
    $okBtn             = New-Object System.Windows.Forms.Button
    $okBtn.Text        = "OK"
    $okBtn.Width       = 100
    $okBtn.Location    = New-Object System.Drawing.Point(286,500)
    $okBtn.Add_Click({
        $form.Tag = 'OK'
        $form.Close()
    })
    $form.Controls.Add($okBtn)

    # Cancel button
    $cancelBtn         = New-Object System.Windows.Forms.Button
    $cancelBtn.Text    = "Cancel"
    $cancelBtn.Width   = 100
    $cancelBtn.Location= New-Object System.Drawing.Point(392,500)
    $cancelBtn.Add_Click({
        $form.Tag = 'Cancel'
        $form.Close()
    })
    $form.Controls.Add($cancelBtn)

    # Timer (auto-close after 10 minutes)
    $timer = New-Object System.Windows.Forms.Timer
    $timer.Interval = 600000
    $timer.Add_Tick({
        $timer.Stop()
        $form.Tag = 'Timeout'
        $form.Close()
    })
    $timer.Start()

    [void]$form.ShowDialog()

    # Gather selections from persistent store (to include items not currently visible due to search)
    $selectedIds = @($persistentCheckedIds.Keys)

    # Treat the Cancel button AND closing the window (the X) as a cancel so the
    # caller returns to the run-mode menu. Only an explicit OK proceeds, and the
    # timer still auto-continues on Timeout.
    if ($form.Tag -ne 'OK' -and $form.Tag -ne 'Timeout') {
        throw "User cancelled workspace selection."
    }

    $timedOut = ($form.Tag -eq 'Timeout')

    # Include My Workspace only if checked… except on timeout, then force include
    $includeMy = if ($timedOut) { $true } else { $chkMy.Checked }

    # If nothing selected from workspace list:
    # - If "Include My Workspace" is checked, process only My Workspace (empty array for regular workspaces)
    # - Otherwise, default to all workspaces (backwards compatibility with timeout/cancel behavior)
    if ($selectedIds.Count -eq 0 -and -not $includeMy -and -not $RequireSelection) {
        $selectedIds = $Workspaces.id
    }

    return [pscustomobject]@{
        SelectedWorkspaceIds = $selectedIds
        IncludeMyWorkspace   = $includeMy
        TimedOut             = $timedOut
    }

# ================================
# Show Report Picker Dialog
# ================================
function Show-ReportPicker {
    param(
        [Parameter(Mandatory)]
        [array]$Reports
    )

    Add-Type -AssemblyName System.Windows.Forms
    Add-Type -AssemblyName System.Drawing

    $form               = New-Object System.Windows.Forms.Form
    $form.Text          = "Select Reports"
    $form.StartPosition = 'CenterScreen'
    $form.Size          = New-Object System.Drawing.Size(650,580)
    $form.TopMost       = $true

    # Instruction label
    $lbl                = New-Object System.Windows.Forms.Label
    $lbl.Text           = "Select the reports to run against (connected models will be included automatically):"
    $lbl.AutoSize       = $true
    $lbl.Location       = New-Object System.Drawing.Point(12,12)
    $form.Controls.Add($lbl)

    # Search label
    $lblSearch          = New-Object System.Windows.Forms.Label
    $lblSearch.Text     = "Search:"
    $lblSearch.AutoSize = $true
    $lblSearch.Location = New-Object System.Drawing.Point(12,40)
    $form.Controls.Add($lblSearch)

    # Search TextBox
    $txtSearch          = New-Object System.Windows.Forms.TextBox
    $txtSearch.Location = New-Object System.Drawing.Point(70,37)
    $txtSearch.Size     = New-Object System.Drawing.Size(552,23)
    $form.Controls.Add($txtSearch)

    # CheckedListBox
    $clb                = New-Object System.Windows.Forms.CheckedListBox
    $clb.Location       = New-Object System.Drawing.Point(12,70)
    $clb.Size           = New-Object System.Drawing.Size(610,380)
    $clb.CheckOnClick   = $true
    $clb.Sorted         = $true

    # Build display items
    $allReportItems = @()
    foreach ($rpt in $Reports) {
        $display = "{0} - {1} ({2})" -f $rpt.WorkspaceName, $rpt.ReportName, $rpt.ReportId
        $item = [pscustomobject]@{
            Display            = $display
            ReportId           = $rpt.ReportId
            ReportName         = $rpt.ReportName
            WorkspaceId        = $rpt.WorkspaceId
            WorkspaceName      = $rpt.WorkspaceName
            DatasetId          = $rpt.DatasetId
            DatasetWorkspaceId = $rpt.DatasetWorkspaceId
        }
        $item | Add-Member -MemberType ScriptMethod -Name ToString -Value { $this.Display } -Force
        $allReportItems += $item
    }

    # Persistent checked state across search filtering
    $persistentCheckedIds = @{}

    # Initial population
    foreach ($item in $allReportItems) {
        [void]$clb.Items.Add($item)
    }
    $clb.DisplayMember = 'Display'
    $form.Controls.Add($clb)

    # Track check/uncheck
    $clb.Add_ItemCheck({
        param($sender, $e)
        $item = $clb.Items[$e.Index]
        if ($e.NewValue -eq 'Checked') {
            $persistentCheckedIds[$item.ReportId] = $true
        } elseif ($e.NewValue -eq 'Unchecked') {
            $persistentCheckedIds.Remove($item.ReportId)
        }
    })

    # Search filter logic
    $txtSearch.Add_TextChanged({
        $searchText = $txtSearch.Text
        $clb.BeginUpdate()
        $clb.Items.Clear()
        $escapedSearchText = [regex]::Escape($searchText)
        foreach ($item in $allReportItems) {
            if ([string]::IsNullOrWhiteSpace($searchText) -or
                $item.Display -imatch $escapedSearchText) {
                $index = $clb.Items.Add($item)
                if ($persistentCheckedIds.ContainsKey($item.ReportId)) {
                    $clb.SetItemChecked($index, $true)
                }
            }
        }
        $clb.EndUpdate()
    })

    # "Select All" checkbox
    $chkAll            = New-Object System.Windows.Forms.CheckBox
    $chkAll.Text       = "Select All"
    $chkAll.AutoSize   = $true
    $chkAll.Location   = New-Object System.Drawing.Point(12,460)
    $chkAll.Add_CheckedChanged({
        for ($i=0; $i -lt $clb.Items.Count; $i++) {
            $clb.SetItemChecked($i, $chkAll.Checked)
        }
    })
    $form.Controls.Add($chkAll)

    # "Clear All" button
    $btnClearAll       = New-Object System.Windows.Forms.Button
    $btnClearAll.Text  = "Clear All"
    $btnClearAll.Width = 80
    $btnClearAll.Location = New-Object System.Drawing.Point(100,458)
    $btnClearAll.Add_Click({
        for ($i=0; $i -lt $clb.Items.Count; $i++) {
            $clb.SetItemChecked($i, $false)
        }
        $chkAll.Checked = $false
    })
    $form.Controls.Add($btnClearAll)

    # OK button
    $okBtn             = New-Object System.Windows.Forms.Button
    $okBtn.Text        = "OK"
    $okBtn.Width       = 100
    $okBtn.Location    = New-Object System.Drawing.Point(416,500)
    $okBtn.Add_Click({
        $form.Tag = 'OK'
        $form.Close()
    })
    $form.Controls.Add($okBtn)

    # Cancel button
    $cancelBtn         = New-Object System.Windows.Forms.Button
    $cancelBtn.Text    = "Cancel"
    $cancelBtn.Width   = 100
    $cancelBtn.Location= New-Object System.Drawing.Point(522,500)
    $cancelBtn.Add_Click({
        $form.Tag = 'Cancel'
        $form.Close()
    })
    $form.Controls.Add($cancelBtn)

    # Timer (auto-close after 2 minutes)
    $timer = New-Object System.Windows.Forms.Timer
    $timer.Interval = 120000
    $timer.Add_Tick({
        $timer.Stop()
        $form.Tag = 'Timeout'
        $form.Close()
    })
    $timer.Start()

    $form.AcceptButton = $okBtn
    $form.CancelButton = $cancelBtn
    [void]$form.ShowDialog()

    if ($form.Tag -eq 'Cancel') {
        return $null
    }

    # Gather selected reports from persistent store
    $selectedReportIds = @($persistentCheckedIds.Keys)
    $selectedReports = @($allReportItems | Where-Object { $selectedReportIds -contains $_.ReportId })

    return $selectedReports
}

function Get-FabricDataAgentInventory {
    param(
        [AllowEmptyCollection()][object[]]$Workspaces
    )

    $ErrorActionPreference = 'Stop'
    $agents = @()
    $failures = @()
    if ($Workspaces.Count -eq 0) { return @() }
    foreach ($workspace in $Workspaces) {
        try {
            $url = "$($global:PowerBIEndpoints.FabricApiPrefix)/v1/workspaces/$($workspace.id)/items?type=DataAgent"
            foreach ($agent in (Invoke-FabricPagedGet -Url $url)) {
                $agents += [PSCustomObject]@{
                    WorkspaceId = $workspace.id
                    WorkspaceName = $workspace.name
                    Agent = $agent
                }
            }
        }
        catch {
            $detail = if ($_.ErrorDetails.Message) { $_.ErrorDetails.Message } else { $_.Exception.GetBaseException().Message }
            $failures += [PSCustomObject]@{
                WorkspaceId = $workspace.id
                Detail = $detail
            }
        }
    }

    if ($failures.Count -eq $Workspaces.Count) {
        throw "Data Agent discovery failed in all $($Workspaces.Count) workspace(s). Check Fabric sign-in and item read permissions. First failure: $($failures[0].Detail)"
    }
    if ($failures.Count -gt 0) {
        Write-Host "[WARN] Data Agent discovery skipped $($failures.Count) of $($Workspaces.Count) workspace(s); results may be incomplete. First failure: $($failures[0].Detail)" -ForegroundColor Yellow
    }
    return $agents
}

function Get-FabricItemDefinition {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)][uri]$DefinitionUrl,
        [Parameter(Mandatory)][string]$AccessToken,
        [ValidateRange(1, 3600)][int]$TimeoutSeconds = 300
    )

    $ErrorActionPreference = 'Stop'
    $fabricOrigin = ([uri]$global:PowerBIEndpoints.FabricApiPrefix).GetLeftPart([UriPartial]::Authority)
    if ($DefinitionUrl.Scheme -ne 'https' -or $DefinitionUrl.UserInfo -or $DefinitionUrl.Query -or
        $DefinitionUrl.Fragment -or $DefinitionUrl.GetLeftPart([UriPartial]::Authority) -ine $fabricOrigin) {
        throw 'Definition requests must use the selected Fabric API origin.'
    }
    $headers = @{ Authorization = "Bearer $AccessToken"; 'Content-Type' = 'application/json' }
    $deadline = (Get-Date).AddSeconds($TimeoutSeconds)

    try {
        $response = Invoke-WebRequest -Uri $DefinitionUrl -Method POST -Headers $headers -Body '{}' -UseBasicParsing -MaximumRedirection 0 -TimeoutSec $TimeoutSeconds -ErrorAction Stop
        if ([int]$response.StatusCode -eq 202) {
            $operationUrl = [string]($response.Headers['Location'] | Select-Object -First 1)
            if (-not $operationUrl) { throw "Definition operation did not return a Location header." }
            $operationUri = [uri]::new($DefinitionUrl, $operationUrl)
            $regionalLocation = $fabricOrigin -ieq 'https://api.fabric.microsoft.com' -and
                $operationUri.DnsSafeHost.EndsWith('.analysis.windows.net', [StringComparison]::OrdinalIgnoreCase)
            if ($operationUri.Scheme -ne 'https' -or $operationUri.Port -ne 443 -or $operationUri.UserInfo -or $operationUri.Query -or
                $operationUri.Fragment -or ($operationUri.GetLeftPart([UriPartial]::Authority) -ine $fabricOrigin -and -not $regionalLocation) -or
                $operationUri.AbsolutePath -notmatch '^/v1/operations/[^/]+/?$') {
                throw 'Definition operation returned an unexpected Location URL.'
            }
            # Regional Location headers still identify operations available on the public API; keep tokens on the selected API origin.
            $operationUri = [uri]::new([uri]$fabricOrigin, $operationUri.AbsolutePath)
            do {
                $retrySeconds = 2
                if ($response.Headers['Retry-After']) {
                    $retrySeconds = [Math]::Max(1, [int]($response.Headers['Retry-After'] | Select-Object -First 1))
                }
                $remainingSeconds = ($deadline - (Get-Date)).TotalSeconds
                if ($remainingSeconds -le $retrySeconds) { throw "Definition operation timed out after $TimeoutSeconds seconds." }
                Start-Sleep -Seconds $retrySeconds
                $requestTimeout = [Math]::Max(1, [int][Math]::Ceiling(($deadline - (Get-Date)).TotalSeconds))
                $response = Invoke-WebRequest -Uri $operationUri -Method GET -Headers $headers -UseBasicParsing -MaximumRedirection 0 -TimeoutSec $requestTimeout -ErrorAction Stop
                if ([int]$response.StatusCode -ne 200) { throw "Unexpected definition polling HTTP status $($response.StatusCode)." }
                $operation = $response.Content | ConvertFrom-Json -ErrorAction Stop
                if ($operation.status -in @('Failed', 'Canceled', 'Cancelled')) {
                    throw "Definition operation $($operation.status): $($operation.error | ConvertTo-Json -Compress)"
                }
                if ($operation.status -notin @('NotStarted', 'Running', 'Succeeded')) {
                    throw "Unexpected definition operation status '$($operation.status)'."
                }
            } while ($operation.status -ne 'Succeeded')

            $remainingSeconds = ($deadline - (Get-Date)).TotalSeconds
            if ($remainingSeconds -le 0) { throw "Definition operation timed out after $TimeoutSeconds seconds." }
            $resultUrl = $operationUri.AbsoluteUri.TrimEnd('/') + '/result'
            $response = Invoke-WebRequest -Uri $resultUrl -Method GET -Headers $headers -UseBasicParsing -MaximumRedirection 0 -TimeoutSec ([Math]::Max(1, [int][Math]::Ceiling($remainingSeconds))) -ErrorAction Stop
        }
        if ([int]$response.StatusCode -ne 200) {
            throw "Unexpected getDefinition HTTP status $($response.StatusCode)."
        }
        $content = if ($response.Content -is [byte[]]) { [Text.Encoding]::UTF8.GetString($response.Content) } else { [string]$response.Content }
        $result = $content | ConvertFrom-Json -ErrorAction Stop
        if (-not $result.definition.parts) { throw "Definition contains no parts." }
        return $result
    }
    finally { $headers = $null }
}

function Get-FabricDataAgentDefinition {
    param(
        [string]$WorkspaceId,
        [string]$DataAgentId,
        [string]$AccessToken,
        [int]$TimeoutSeconds = 300
    )

    try {
        $url = "$($global:PowerBIEndpoints.FabricApiPrefix)/v1/workspaces/$WorkspaceId/dataAgents/$DataAgentId/getDefinition"
        return (Get-FabricItemDefinition -DefinitionUrl $url -AccessToken $AccessToken -TimeoutSeconds $TimeoutSeconds).definition
    }
    catch {
        $permissionHint = ''
        if ($_.Exception.Response -and [int]$_.Exception.Response.StatusCode -eq 403) {
            $permissionHint = ' Check delegated Item.ReadWrite.All consent for the Azure PowerShell application and read/write access to the agent. No alternate sign-in will be attempted.'
        }
        Write-Host "[WARN] Could not retrieve definition for Data Agent $DataAgentId in workspace $WorkspaceId : $($_.Exception.Message)$permissionHint" -ForegroundColor Yellow
        return $null
    }
}

function Expand-FabricAppFrontend {
    [CmdletBinding()]
    param([Parameter(Mandatory)][string]$FilesRoot)

    $ErrorActionPreference = 'Stop'
    $frontend = Join-Path $FilesRoot 'frontend'
    foreach ($name in @('static-app.zip', 'static.zip')) {
        $archivePath = Join-Path $frontend $name
        if (-not (Test-Path -LiteralPath $archivePath -PathType Leaf)) { continue }
        Add-Type -AssemblyName System.IO.Compression.FileSystem -ErrorAction Stop
        $destination = Join-Path $frontend ([IO.Path]::GetFileNameWithoutExtension($name))
        [IO.Compression.ZipFile]::ExtractToDirectory($archivePath, $destination)
        Remove-Item -LiteralPath $archivePath -Force -ErrorAction Stop
    }
}

function Get-FabricAppConfiguredModels {
    [CmdletBinding()]
    param([Parameter(Mandatory)][string]$BackupPath)

    $ErrorActionPreference = 'Stop'
    function Read-ConnectorScalar([string]$Value) {
        $value = $Value.Trim()
        $quoted = [regex]::Match($value, '^(?<value>"(?:\\.|[^"\\])*")\s*(?:#.*)?$')
        if ($quoted.Success) { return ($quoted.Groups['value'].Value | ConvertFrom-Json -ErrorAction Stop) }
        $quoted = [regex]::Match($value, '^''(?<value>(?:''''|[^''])*)''\s*(?:#.*)?$')
        if ($quoted.Success) { return $quoted.Groups['value'].Value.Replace("''", "'") }
        $value = ($value -replace '\s+#.*$', '').Trim()
        if (-not $value -or $value -match '^[&*!|>{}\[\],%@`"''#]|:\s') {
            throw 'Unsupported connector value. Expected a plain or quoted scalar.'
        }
        return $value
    }

    $path = Join-Path $BackupPath 'rayfin.yml'
    $lines = [IO.File]::ReadAllLines($path)
    $header = -1
    $headerValue = ''
    for ($i = 0; $i -lt $lines.Length; $i++) {
        if ($lines[$i] -cmatch '^(?:connectors|"connectors"|''connectors'')\s*:(?<value>.*)$') {
            if ($header -ge 0) { throw 'Duplicate connectors sections in the saved app configuration.' }
            $header = $i
            $headerValue = ($Matches.value -replace '\s+#.*$', '').Trim()
        }
    }
    if ($header -lt 0) { throw "No connectors section was found in '$path'." }
    if ($headerValue -and $headerValue -ne '{}' -and -not $headerValue.StartsWith('#')) {
        throw 'Unsupported connectors section. Expected an indented mapping.'
    }
    $connectors = [Collections.Generic.List[object]]::new()
    $aliases = [Collections.Generic.HashSet[string]]::new([StringComparer]::Ordinal)
    $aliasIndent = -1
    $current = $null
    $pairPattern = '^(?<indent> *)(?<key>"(?:\\.|[^"\\])*"|''(?:''''|[^''])*''|[^:#"'']+?)\s*:\s*(?<value>.*)$'
    for ($i = $header + 1; $i -lt $lines.Length; $i++) {
        $line = $lines[$i]
        if (-not $line.Trim() -or $line.TrimStart().StartsWith('#')) { continue }
        if ($line -match '^\S') { break }
        if ($line -match '^\s*\t' -or $headerValue -eq '{}') { throw 'Unsupported connector indentation or mapping structure.' }
        $indent = [regex]::Match($line, '^ *').Length
        $pair = [regex]::Match($line, $pairPattern)
        if (-not $pair.Success) {
            if ($current -and $indent -gt $current.PropertyIndent) { continue }
            throw "Unsupported connector mapping at line $($i + 1)."
        }
        $key = Read-ConnectorScalar $pair.Groups['key'].Value
        $value = $pair.Groups['value'].Value.Trim()
        if ($key -ceq '<<') { throw 'Connector merges and anchors are not supported by the local connection reader.' }
        if ($aliasIndent -lt 0) { $aliasIndent = $indent }
        if ($indent -eq $aliasIndent) {
            if (($value -and -not $value.StartsWith('#')) -or -not $aliases.Add($key)) {
                throw "Unsupported or duplicate connector '$key'. Expected an indented mapping."
            }
            $current = @{ Alias = $key; Type = ''; Config = @{}; PropertyIndent = -1; ConfigIndent = -1; ActiveProperty = ''; Seen = @{} }
            $connectors.Add($current)
            continue
        }
        if ($indent -lt $aliasIndent -or -not $current) { throw 'Unsupported connector indentation.' }
        if ($current.PropertyIndent -ge 0 -and $indent -lt $current.PropertyIndent) { throw 'Unsupported connector property indentation.' }
        if ($current.PropertyIndent -lt 0) { $current.PropertyIndent = $indent }
        if ($indent -eq $current.PropertyIndent) {
            $current.ActiveProperty = $key
            if ($key -cin @('connector', 'config')) {
                if ($current.Seen.ContainsKey($key)) { throw "Duplicate '$key' in connector '$($current.Alias)'." }
                $current.Seen[$key] = $true
            }
            if ($key -ceq 'connector') { $current.Type = Read-ConnectorScalar $value }
            elseif ($key -ceq 'config' -and $value -and $value -ne '{}' -and -not $value.StartsWith('#')) {
                throw "Unsupported config for connector '$($current.Alias)'. Expected an indented mapping."
            }
        }
        elseif ($indent -gt $current.PropertyIndent -and $current.ActiveProperty -ceq 'config') {
            if ($current.ConfigIndent -lt 0) { $current.ConfigIndent = $indent }
            if ($indent -eq $current.ConfigIndent -and $key -cin @('itemId', 'workspaceId')) {
                if ($current.Config.ContainsKey($key)) { throw "Duplicate '$key' in connector '$($current.Alias)'." }
                $current.Config[$key] = Read-ConnectorScalar $value
            }
        }
    }
    foreach ($connector in $connectors) {
        if (-not $connector.Type) { throw "Connector '$($connector.Alias)' has no explicit connector type." }
        if ($connector.Type -cne 'fabric-semanticmodel') { continue }
        $modelId = [guid]::Empty
        $workspaceId = [guid]::Empty
        if (-not [guid]::TryParse([string]$connector.Config.itemId, [ref]$modelId) -or
            -not [guid]::TryParse([string]$connector.Config.workspaceId, [ref]$workspaceId) -or
            $modelId -eq [guid]::Empty -or $workspaceId -eq [guid]::Empty) {
            throw "Saved semantic-model connector '$($connector.Alias)' must contain valid itemId and workspaceId values."
        }
        [PSCustomObject]@{
            connectionAlias = $connector.Alias; modelId = $modelId.ToString()
            modelWorkspaceId = $workspaceId.ToString(); evidence = 'SavedConnectorConfiguration'
        }
    }
}

function Get-FabricAppConnectionSnapshot {
    [CmdletBinding()]
    param(
        [AllowEmptyCollection()][object[]]$Items = @(),
        [Parameter(Mandatory)][hashtable]$BackupPaths
    )

    $snapshot = @{}
    foreach ($app in @($Items | Where-Object { $_.FabricItemType -eq 'AppBackend' })) {
        $key = "$($app.WorkspaceId)|$($app.FabricItemID)"
        if ($snapshot.ContainsKey($key)) { continue }
        $entry = [PSCustomObject]@{ Models = @(); Error = ''; BackupPath = [string]$BackupPaths[$key] }
        $snapshot[$key] = $entry
        if (-not $entry.BackupPath) {
            $entry.Error = 'The app definition backup did not complete in this run.'
            Write-Warning "Fabric App '$($app.FabricItemName)': $($entry.Error)" -WarningAction Continue
            continue
        }
        try {
            $entry.Models = @(Get-FabricAppConfiguredModels -BackupPath $entry.BackupPath -ErrorAction Stop)
        }
        catch {
            $entry.Error = $_.Exception.Message
            Write-Warning "Fabric App '$($app.FabricItemName)' model connection discovery failed: $($entry.Error)" -WarningAction Continue
        }
    }
    return $snapshot
}

function Export-FabricAppBackups {
    [CmdletBinding()]
    param(
        [AllowEmptyCollection()][object[]]$Items = @(),
        [Parameter(Mandatory)][string]$BaseFolderPath,
        [Parameter(Mandatory)][ValidatePattern('^\d{4}-\d{2}-\d{2}$')][string]$ReportDate,
        [hashtable]$BackupPaths = @{}
    )

    $ErrorActionPreference = 'Stop'
    $BackupPaths.Clear()
    $apps = @($Items | Where-Object { $_.FabricItemType -eq 'AppBackend' })
    $backupRoot = Join-Path (Join-Path $BaseFolderPath 'Fabric App Backups') $ReportDate
    if (Test-Path -LiteralPath $backupRoot) {
        Remove-Item -LiteralPath $backupRoot -Recurse -Force -ErrorAction Stop
    }
    $null = New-Item -ItemType Directory -Path $backupRoot -Force
    $seen = [Collections.Generic.HashSet[string]]::new([StringComparer]::OrdinalIgnoreCase)
    foreach ($app in $apps) {
        $stagePath = $null
        try {
            $workspaceId = ([guid]$app.WorkspaceId).ToString()
            $appId = ([guid]$app.FabricItemID).ToString()
            if (-not $seen.Add("$workspaceId|$appId")) { continue }
            $url = "$($global:PowerBIEndpoints.FabricApiPrefix)/v1/workspaces/$workspaceId/items/$appId/getDefinition"
            $result = Get-FabricItemDefinition -DefinitionUrl $url -AccessToken (Get-FabricAccessToken) -ErrorAction Stop
            $stagePath = Join-Path $backupRoot ('.partial-' + [guid]::NewGuid().ToString('N'))
            $filesRoot = [IO.Path]::GetFullPath($stagePath)
            $null = New-Item -ItemType Directory -Path $filesRoot
            $paths = [Collections.Generic.HashSet[string]]::new([StringComparer]::OrdinalIgnoreCase)
            foreach ($part in $result.definition.parts) {
                if ($part.payloadType -ne 'InlineBase64') { throw "Unsupported payload type '$($part.payloadType)' for '$($part.path)'." }
                $relativePath = ([string]$part.path).Replace('/', '\')
                if ([string]::IsNullOrWhiteSpace($relativePath) -or [IO.Path]::IsPathRooted($relativePath)) {
                    throw "Unsafe definition part path '$($part.path)'."
                }
                foreach ($segment in $relativePath.Split('\')) {
                    if (-not $segment -or $segment -in @('.', '..') -or $segment -ne $segment.TrimEnd([char[]]' .') -or
                        $segment.IndexOfAny([IO.Path]::GetInvalidFileNameChars()) -ge 0 -or
                        $segment -match '^(?i:CON|PRN|AUX|NUL|COM[1-9]|LPT[1-9])(?:\.|$)') {
                        throw "Unsafe definition part path '$($part.path)'."
                    }
                }
                $partPath = [IO.Path]::GetFullPath((Join-Path $filesRoot $relativePath))
                if (-not $partPath.StartsWith($filesRoot + '\', [StringComparison]::OrdinalIgnoreCase) -or -not $paths.Add($partPath)) {
                    throw "Unsafe or duplicate definition part path '$($part.path)'."
                }
                $bytes = [Convert]::FromBase64String([string]$part.payload)
                $null = New-Item -ItemType Directory -Path (Split-Path $partPath -Parent) -Force
                [IO.File]::WriteAllBytes($partPath, $bytes)
            }
            Expand-FabricAppFrontend -FilesRoot $filesRoot -ErrorAction Stop
            $destination = Get-FabricAppBackupPath -BaseFolderPath $BaseFolderPath -ReportDate $ReportDate -WorkspaceName $app.WorkspaceName -AppName $app.FabricItemName
            # Apps that share a workspace and name (or truncate to the same label) get a numbered suffix instead of overwriting each other.
            $baseDestination = $destination
            for ($copy = 2; Test-Path -LiteralPath $destination; $copy++) { $destination = "$baseDestination ($copy)" }
            [IO.Directory]::Move($stagePath, $destination)
            $stagePath = $null
            $BackupPaths["$workspaceId|$appId"] = $destination
            Write-Host "[INFO] Fabric App backup saved: $destination ($($paths.Count) definition parts)."
        }
        catch {
            Write-Warning "Fabric App backup failed for '$($app.WorkspaceName) ~ $($app.FabricItemName)' ($($app.FabricItemID)): $($_.Exception.Message)" -WarningAction Continue
        }
        finally {
            if ($stagePath -and (Test-Path -LiteralPath $stagePath)) {
                try { Remove-Item -LiteralPath $stagePath -Recurse -Force -ErrorAction Stop }
                catch { Write-Warning "Could not remove incomplete Fabric App backup '$stagePath': $($_.Exception.Message)" -WarningAction Continue }
            }
        }
    }
}

function Get-FabricDataAgentModelTargets {
    param(
        [Parameter(Mandatory)]$Definition
    )

    $ErrorActionPreference = 'Stop'
    foreach ($part in $Definition.parts) {
        $path = $part.path -replace '\\', '/'
        if ($path -notmatch '^Files/Config/[^/]+/(?:.*/)?datasource\.json$') { continue }
        try {
            if ($part.payloadType -ne 'InlineBase64') { throw "Unsupported datasource payload type." }
            $source = [Text.Encoding]::UTF8.GetString([Convert]::FromBase64String($part.payload)) | ConvertFrom-Json
            if ($source.type -eq 'semantic_model') {
                if (-not $source.artifactId -or -not $source.workspaceId) {
                    throw "Semantic model source is missing artifactId or workspaceId."
                }
                [PSCustomObject]@{ DatasetId = $source.artifactId; WorkspaceId = $source.workspaceId }
            }
        }
        catch {
            throw "Cannot resolve connected models from '$path': $($_.Exception.Message)"
        }
    }
}

function Resolve-FabricAppModelNames {
    [CmdletBinding()]
    param(
        [AllowEmptyCollection()][object[]]$Mappings = @(),
        [AllowEmptyCollection()][object[]]$Datasets = @(),
        [AllowEmptyCollection()][object[]]$Workspaces = @()
    )

    # Names come only from the run's dataset and workspace inventories; no extra API requests are made.
    $datasetLookup = @{}
    foreach ($dataset in $Datasets) {
        $datasetId = [string]$dataset.DatasetId
        $datasetWorkspaceId = [string]$dataset.WorkspaceId
        if ($datasetId -and $datasetWorkspaceId -and $dataset.DatasetName) {
            $datasetLookup["$datasetWorkspaceId/$datasetId"] = [string]$dataset.DatasetName
        }
    }
    $workspaceLookup = @{}
    foreach ($workspace in $Workspaces) {
        $workspaceId = [string]$workspace.WorkspaceId
        if ($workspaceId -and $workspace.WorkspaceName) {
            $workspaceLookup[$workspaceId] = [string]$workspace.WorkspaceName
        }
    }

    foreach ($mapping in $Mappings) {
        $modelGuid = [guid]::Empty
        $workspaceGuid = [guid]::Empty
        $validIds = [guid]::TryParse([string]$mapping.modelId, [ref]$modelGuid) -and
            [guid]::TryParse([string]$mapping.modelWorkspaceId, [ref]$workspaceGuid)
        $result = [ordered]@{
            connectionAlias = $mapping.connectionAlias
            modelId = $mapping.modelId
            modelWorkspaceId = $mapping.modelWorkspaceId
            evidence = $mapping.evidence
            modelName = ''
            modelWorkspaceName = ''
        }
        if ($validIds) {
            $workspaceId = $workspaceGuid.ToString()
            if ($workspaceLookup.ContainsKey($workspaceId)) {
                $result.modelWorkspaceName = $workspaceLookup[$workspaceId]
            }
            $datasetKey = "$workspaceId/$($modelGuid.ToString())"
            if ($datasetLookup.ContainsKey($datasetKey)) {
                $result.modelName = $datasetLookup[$datasetKey]
            }
        }
        [PSCustomObject]$result
    }
}

function Write-FabricAppLog {
    param([ValidateSet('INFO', 'WARN', 'ERROR')][string]$Level, [string]$Message)

    # Per-script details go to the run log only, so the console shows when each app starts and finishes.
    if ($script:ImpactIQDiagnostics -and $script:ImpactIQDiagnostics.Writer) { Write-ImpactIQDiagnostic -Level $Level -Message $Message }
}

function Get-FabricAppFrontendRoot {
    param([Parameter(Mandatory)][string]$BackupPath)

    $frontendRoot = @('static-app', 'static') | ForEach-Object {
        Join-Path (Join-Path $BackupPath 'frontend') $_
    } | Where-Object { Test-Path -LiteralPath $_ -PathType Container } | Select-Object -First 1
    if ($frontendRoot) { [IO.Path]::GetFullPath($frontendRoot) }
}

# Parses the saved frontend scripts of every requested app in one Tabular Editor run. Tabular Editor executes
# Config\Fabric App Extract Script.csx against the blank model, so no Node.js, module install or compiled helper is
# needed. Requests: objects with Key, FrontendRoot and MaxAssets. Returns a hashtable of per-app results keyed by Key.
function Invoke-FabricAppExtract {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)][object[]]$Requests,
        [Parameter(Mandatory)][string]$TabularEditorPath,
        [Parameter(Mandatory)][string]$BlankModelPath,
        [Parameter(Mandatory)][string]$ScriptPath,
        [Parameter(Mandatory)][string]$WorkingDirectory,
        # Reproduces the original PowerShell parser exactly (regression tests); the default also resolves minified names by scope.
        [switch]$StrictParity
    )

    $ErrorActionPreference = 'Stop'
    foreach ($path in @($TabularEditorPath, $BlankModelPath, $ScriptPath)) {
        if (-not (Test-Path -LiteralPath $path -PathType Leaf)) { throw "A file needed to parse Fabric apps is missing: $path" }
    }
    $folder = Join-Path ([IO.Path]::GetTempPath()) ('ImpactIQ-AppExtract-' + [guid]::NewGuid().ToString('N'))
    $null = New-Item -ItemType Directory -Path $folder
    $previousRequest = $env:IMPACTIQ_FABRIC_APP_EXTRACT_REQUEST
    try {
        $requestPath = Join-Path $folder 'request.json'
        $resultPath = Join-Path $folder 'result.json'
        $request = [ordered]@{
            outputPath = $resultPath
            strictParity = [bool]$StrictParity
            apps = @($Requests | ForEach-Object { [ordered]@{ key = [string]$_.Key; frontendRoot = [string]$_.FrontendRoot; maxAssets = [int]$_.MaxAssets } })
        }
        [IO.File]::WriteAllText($requestPath, (ConvertTo-Json -InputObject $request -Depth 5), [Text.UTF8Encoding]::new($false))
        $env:IMPACTIQ_FABRIC_APP_EXTRACT_REQUEST = $requestPath
        $process = Invoke-ImpactIQNativeProcess -FilePath $TabularEditorPath -WorkingDirectory $WorkingDirectory -PassThru -ArgumentList "`"$BlankModelPath`" -S `"$ScriptPath`"" -ErrorAction Stop
        try { $exitCode = $process.ExitCode }
        finally { $process.Dispose() }
        if ($exitCode -ne 0 -or -not (Test-Path -LiteralPath $resultPath -PathType Leaf)) {
            throw "Tabular Editor could not parse the saved Fabric app scripts (exit code $exitCode); its output is in the error log."
        }
        Add-Type -AssemblyName System.Web.Extensions
        $serializer = [Web.Script.Serialization.JavaScriptSerializer]::new()
        $serializer.MaxJsonLength = [int]::MaxValue
        $document = $serializer.DeserializeObject([IO.File]::ReadAllText($resultPath))
        $results = @{}
        foreach ($app in @($document['apps'])) { $results[[string]$app['key']] = $app }
        $results
    }
    finally {
        $env:IMPACTIQ_FABRIC_APP_EXTRACT_REQUEST = $previousRequest
        if (Test-Path -LiteralPath $folder) {
            try { Remove-Item -LiteralPath $folder -Recurse -Force -ErrorAction Stop }
            catch { Write-Warning "Temporary Fabric app parser data could not be removed: $folder" -WarningAction Continue }
        }
    }
}

function Get-FabricAppDaxCapture {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)][guid]$WorkspaceId,
        [string]$WorkspaceName,
        [Parameter(Mandatory)][guid]$AppId,
        [string]$AppName,
        [Parameter(Mandatory)][string]$BackupPath,
        [switch]$BackupFailed,
        [Parameter(Mandatory)]$Extract,
        [string]$Environment = 'Public',
        [AllowEmptyCollection()][object[]]$Datasets = @(),
        [AllowEmptyCollection()][object[]]$Workspaces = @(),
        [AllowEmptyCollection()][object[]]$ConfiguredModels = @(),
        [string]$ModelMappingError = ''
    )

    $ErrorActionPreference = 'Stop'
    $queries = [Collections.Generic.List[object]]::new()
    $modelMappings = [Collections.Generic.List[object]]::new()
    $assetCount = 0
    $failedAssets = 0
    $captureError = ''
    $capture = [PSCustomObject][ordered]@{
        WorkspaceId = $WorkspaceId.ToString()
        WorkspaceName = $WorkspaceName
        AppId = $AppId.ToString()
        AppName = $AppName
        CaptureStatus = 'Started'
        CaptureError = ''
        AssetsInspected = 0
        SkippedAssets = 0
    }
    $stage = 'Discovery'
    $configurationError = $ModelMappingError
    if ($Environment -ne 'Public') {
        $capture.CaptureStatus = 'Failed'
        $capture.CaptureError = 'Fabric App extraction is only supported in the Public cloud.'
        Write-Host "[ERROR] App query extraction failed for $WorkspaceName ~ $AppName`: the preview hosting flow has only been verified in the Public cloud." -ForegroundColor Red
        return [PSCustomObject]@{ Queries = @(); Models = @(); Capture = $capture; Error = 'Fabric App extraction is only supported in the Public cloud.' }
    }
    try {
        $stage = 'Saved backup discovery'
        if ($BackupFailed) { throw 'The app definition backup did not complete in this run. Existing files will not be used.' }
        if (-not (Test-Path -LiteralPath $BackupPath -PathType Container)) {
            throw "App backup is unavailable: $BackupPath. Hosted assets will not be downloaded."
        }
        foreach ($mapping in $ConfiguredModels) { $modelMappings.Add($mapping) }
        if ($configurationError) { Write-Warning "Fabric App '$AppName' model connections are incomplete: $configurationError" }
        if ($Extract['error']) { throw [string]$Extract['error'] }
        $assetCount = [int]$Extract['assetCount']
        $failedAssets = [int]$Extract['failedAssets']
        foreach ($message in @($Extract['messages'])) { Write-FabricAppLog INFO "Fabric App '$AppName': $message" }
        foreach ($warning in @($Extract['warnings'])) { Write-FabricAppLog WARN "Fabric App '$AppName' $warning" }
        Write-FabricAppLog INFO ("Fabric App '{0}' timing: {1} saved script(s), {2:N0} KB; local read {3:N1}s; parsing {4:N1}s; cross-file linking {5:N1}s; {6} script(s) skipped by the DAX prefilter." -f $AppName, $assetCount, ([double]$Extract['scriptChars'] / 1KB), [double]$Extract['readSeconds'], [double]$Extract['parseSeconds'], [double]$Extract['linkSeconds'], [int]$Extract['filteredAssets'])
        foreach ($asset in @($Extract['assets'])) {
            try {
                $constants = [Collections.Generic.Dictionary[string,string]]::new([StringComparer]::Ordinal)
                foreach ($name in $asset['constants'].Keys) { $constants[[string]$name] = [string]$asset['constants'][$name] }
                $catalog = [PSCustomObject]@{
                    queries = @(foreach ($query in @($asset['queries'])) {
                        [PSCustomObject]@{
                            symbol = [string]$query['symbol']
                            evidence = [string]$query['evidence']
                            text = [string]$query['text']
                            expressions = @($query['expressions'])
                            valueExpressions = @($query['valueExpressions'])
                            connectionAliases = @($query['connectionAliases'])
                            unexpandedSource = [string]$query['unexpandedSource']
                        }
                    })
                    constants = $constants
                    fieldCatalog = @($asset['fieldCatalog'])
                }
                foreach ($row in @(ConvertTo-FabricAppQueryRows -Catalog $catalog -WorkspaceId $WorkspaceId -WorkspaceName $WorkspaceName -AppId $AppId -AppName $AppName -SourceUrl ([uri][string]$asset['path']).AbsoluteUri)) {
                    $queries.Add($row)
                }
            }
            catch {
                $failedAssets++
                Write-FabricAppLog WARN "Fabric App '$AppName' script '$([IO.Path]::GetFileName([string]$asset['path']))' could not be inspected: $($_.Exception.Message)"
            }
        }
        if ($failedAssets -gt 0 -or $Extract['assetLimitReached']) {
            $captureError = 'Some saved scripts could not be inspected, or the asset limit was reached. Inline scripts and base URLs need manual review.'
        }
        elseif ($queries.Count -eq 0) {
            $captureError = 'No DAX literals were found in inspected scripts; this does not prove absence of queries.'
        }
    }
    catch {
        $captureError = "$stage failed: $($_.Exception.Message)"
        Write-Host "[WARN] Fabric App '$AppName': $captureError" -ForegroundColor Yellow
    }
    $captureError = (@($configurationError, $captureError) | Where-Object { $_ }) -join ' '
    $configuredModels = @($modelMappings |
        Where-Object { $_.connectionAlias -and $_.modelId -and $_.modelWorkspaceId } |
        Select-Object connectionAlias, modelId, modelWorkspaceId, evidence -Unique)
    $configuredModels = @(Resolve-FabricAppModelNames -Mappings $configuredModels -Datasets $Datasets -Workspaces $Workspaces)
    $configuredAliases = @($configuredModels.connectionAlias | Select-Object -Unique)
    $queryRows = @($queries.ToArray())
    foreach ($queryRow in $queryRows) {
        $queryAlias = [string]$queryRow.ConnectionAlias
        if (-not $queryAlias -or $configuredAliases -ccontains $queryAlias) { continue }
        $aliasMatches = @($configuredAliases | Where-Object { $_ -ieq $queryAlias })
        if ($aliasMatches.Count -eq 1) {
            $queryRow.ConnectionAlias = $aliasMatches[0]
        }
    }
    if ($configuredAliases.Count -gt 0) {
        $queryRows = @($queryRows |
            Group-Object {
                $text = @($_.PSObject.Properties |
                    Where-Object { $_.Name -match '^QueryText(\d+)?$' } |
                    Sort-Object {
                        if ($_.Name -eq 'QueryText') { 1 }
                        else { [int]$_.Name.Substring(9) }
                    } |
                    ForEach-Object { [string]$_.Value }) -join ''
                ([string]$_.SourceUrl) + [char]0 + ([string]$_.QuerySymbol) + [char]0 + $text
            } |
            ForEach-Object {
                $validRows = @($_.Group | Where-Object {
                    $_.ConnectionAlias -and $configuredAliases -ccontains $_.ConnectionAlias
                })
                if ($validRows.Count -gt 0) {
                    $validRows
                }
                else {
                    $row = $_.Group[0]
                    $row.ConnectionAlias = ''
                    $row
                }
            })
    }
    Set-FabricAppQueryModelMapping -Queries $queryRows -Mappings $configuredModels
    $models = @($configuredModels |
        ForEach-Object {
            $mapping = $_
            [PSCustomObject][ordered]@{
                AppWorkspaceId = $WorkspaceId.ToString()
                AppWorkspaceName = $WorkspaceName
                AppId = $AppId.ToString()
                AppName = $AppName
                ConnectionAlias = $mapping.connectionAlias
                ModelWorkspaceId = $mapping.modelWorkspaceId
                ModelWorkspaceName = $mapping.modelWorkspaceName
                ModelId = $mapping.modelId
                ModelName = $mapping.modelName
                Evidence = $mapping.evidence
            }
        })
    if (-not $captureError -and $models.Count -eq 0) {
        $captureError = 'No semantic-model targets were discovered in the saved app connector configuration; query model fields remain blank.'
        Write-Host "[WARN] Fabric App '$AppName': $captureError" -ForegroundColor Yellow
    }
    if (-not $captureError) {
        Write-Host "App query extraction completed for $WorkspaceName ~ $AppName"
    }
    $capture.CaptureStatus = if ($captureError) {
        if ($queries.Count -eq 0 -and $models.Count -eq 0) { 'Failed' } else { 'Partial' }
    } else { 'StaticAssetsInspected' }
    $capture.CaptureError = $captureError
    $capture.AssetsInspected = $assetCount
    $capture.SkippedAssets = $failedAssets
    [PSCustomObject]@{ Queries = $queryRows; Models = $models; Capture = $capture; Error = $captureError }
}

function Set-FabricAppQueryModelMapping {
    [CmdletBinding()]
    param(
        [AllowEmptyCollection()][object[]]$Queries = @(),
        [AllowEmptyCollection()][object[]]$Mappings = @(),
        [AllowEmptyCollection()][object[]]$Datasets = @(),
        [AllowEmptyCollection()][object[]]$Workspaces = @()
    )

    $configuredModels = @($Mappings | Where-Object { $_.connectionAlias -and $_.modelId -and $_.modelWorkspaceId } |
        Sort-Object connectionAlias, modelId, modelWorkspaceId -Unique)
    foreach ($query in $Queries) {
        $mapping = $null
        if ($query.ConnectionAlias) {
            $matches = @($configuredModels | Where-Object { $_.connectionAlias -ceq $query.ConnectionAlias })
            if ($matches.Count -eq 1) {
                $mapping = $matches[0]
            }
        }
        elseif ($configuredModels.Count -eq 1) {
            $mapping = $configuredModels[0]
        }
        $modelName = ''
        $modelWorkspaceName = ''
        if ($mapping) {
            $modelName = [string]$mapping.modelName
            $modelWorkspaceName = [string]$mapping.modelWorkspaceName
        }
        $mappingEvidence = ''
        if ($mapping) {
            $mappingEvidence = if ($query.ConnectionAlias) { 'ConnectionAlias' } else { 'SingleConfiguredModel' }
        }
        $query.ConnectionAlias = if ($mapping) { $mapping.connectionAlias } else { $query.ConnectionAlias }
        $query.ModelId = if ($mapping) { $mapping.modelId } else { '' }
        $query.ModelName = $modelName
        $query.ModelWorkspaceId = if ($mapping) { $mapping.modelWorkspaceId } else { '' }
        $query.ModelWorkspaceName = $modelWorkspaceName
        $query | Add-Member -NotePropertyName ModelMappingEvidence -NotePropertyValue $mappingEvidence -Force
    }
}

function ConvertTo-FabricAppQueryRows {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)]$Catalog,
        [Parameter(Mandatory)][string]$WorkspaceId,
        [string]$WorkspaceName,
        [Parameter(Mandatory)][string]$AppId,
        [string]$AppName,
        [string]$SourceUrl,
        [string]$CatalogPath
    )

    foreach ($query in $Catalog.queries) {
        $connectionAliases = @($query.connectionAliases | Where-Object { $_ } | Select-Object -Unique)
        $aliasesForRows = if ($connectionAliases.Count -gt 0) { $connectionAliases } else { @('') }
        foreach ($connectionAlias in $aliasesForRows) {
            $record = [ordered]@{
                WorkspaceId = $WorkspaceId
                WorkspaceName = $WorkspaceName
                AppId = $AppId
                AppName = $AppName
                QuerySymbol = $query.symbol
                Evidence = $query.evidence
                QueryText = ''
                SourceUrl = $SourceUrl
                ConnectionAlias = $connectionAlias
                ModelId = ''
                ModelName = ''
                ModelWorkspaceId = ''
                ModelWorkspaceName = ''
                ModelMappingEvidence = ''
                UnexpandedSource = $(if ($query.PSObject.Properties['unexpandedSource']) { [string]$query.unexpandedSource } else { '' })
                TemplateExpressions = @($query.expressions | Where-Object { $null -ne $_ })
                TemplateValueExpressions = @($(if ($query.PSObject.Properties['valueExpressions']) { $query.valueExpressions }) | Where-Object { $null -ne $_ })
                TemplateConstants = $(if ($Catalog.PSObject.Properties['constants'] -and $Catalog.constants) { $Catalog.constants } else { @{} })
                TemplateFieldCatalog = @($(if ($Catalog.PSObject.Properties['fieldCatalog']) { $Catalog.fieldCatalog }))
            }
            $text = [string]$query.text
            $offset = 0
            $part = 1
            while ($offset -lt $text.Length) {
                $length = [Math]::Min(32000, $text.Length - $offset)
                $nextOffset = $offset + $length
                if ($nextOffset -lt $text.Length -and
                    [char]::IsHighSurrogate($text[$nextOffset - 1]) -and
                    [char]::IsLowSurrogate($text[$nextOffset])) {
                    $length--
                }
                $column = if ($part -eq 1) { 'QueryText' } else { "QueryText$part" }
                $record[$column] = $text.Substring($offset, $length)
                $offset += $length
                $part++
            }
            [PSCustomObject]$record
        }
    }
}

function Export-FabricAppQueryWorkbook {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)][string]$Path,
        [AllowEmptyCollection()][object[]]$Queries = @(),
        [AllowEmptyCollection()][object[]]$Models = @(),
        [AllowEmptyCollection()][object[]]$Captures = @(),
        [AllowEmptyCollection()][object[]]$Dependencies = @(),
        [AllowEmptyCollection()][object[]]$DependencyStatus = @(),
        [switch]$IncludeDependencies
    )

    Import-Module ImportExcel -ErrorAction Stop
    $fragmentEvidence = @('DaxFragment', 'ConfigExpression')
    $fullQueries = @($Queries | Where-Object { $_.Evidence -notin $fragmentEvidence })
    $fragmentQueries = @($Queries | Where-Object { $_.Evidence -in $fragmentEvidence })
    $unmappedQueries = @($fullQueries | Where-Object { -not $_.ModelId -or -not $_.ModelWorkspaceId })
    Write-FabricAppLog INFO "Fabric App model mapping: $($fullQueries.Count - $unmappedQueries.Count) of $($fullQueries.Count) captured query rows resolved."
    if ($fragmentQueries.Count -gt 0) {
        $unmappedFragments = @($fragmentQueries | Where-Object { -not $_.ModelId -or -not $_.ModelWorkspaceId })
        Write-FabricAppLog INFO "Fabric App DAX fragments: $($fragmentQueries.Count - $unmappedFragments.Count) of $($fragmentQueries.Count) fragment rows have a connection-based model. Remaining fragments are matched by exact model-object names during dependency export, or stay unassigned."
    }
    if ($unmappedQueries.Count -gt 0) {
        Write-FabricAppLog WARN "$($unmappedQueries.Count) query row(s) remain unresolved after connection tracing. Export will continue with their model fields blank."
        foreach ($query in $unmappedQueries) {
            Write-FabricAppLog WARN "  $($query.WorkspaceName) ~ $($query.AppName) ~ $($query.QuerySymbol) (connection: '$($query.ConnectionAlias)')"
        }
    }
    $appRows = @($Captures | ForEach-Object {
        [PSCustomObject][ordered]@{
            WorkspaceId = $_.WorkspaceId
            WorkspaceName = $_.WorkspaceName
            AppId = $_.AppId
            AppName = $_.AppName
            CaptureStatus = $_.CaptureStatus
            CaptureError = $_.CaptureError
            AssetsInspected = $_.AssetsInspected
            SkippedAssets = $_.SkippedAssets
        }
    })
    $package = Open-ExcelPackage -Path $Path -Create -ErrorAction Stop
    try {
        $definitions = @(
            @{
                Name = 'FabricAppModels'
                Rows = $Models
                Headers = @('AppWorkspaceId', 'AppWorkspaceName', 'AppId', 'AppName', 'ConnectionAlias', 'ModelWorkspaceId', 'ModelWorkspaceName', 'ModelId', 'ModelName', 'Evidence')
            },
            @{
                Name = 'FabricApps'
                Rows = $appRows
                Headers = @('WorkspaceId', 'WorkspaceName', 'AppId', 'AppName', 'CaptureStatus', 'CaptureError', 'AssetsInspected', 'SkippedAssets')
            }
        )
        if ($IncludeDependencies) {
            $definitions = @(
                @{
                    Name = 'FabricAppQueries'
                    Rows = $Queries
                    Headers = @('WorkspaceId', 'WorkspaceName', 'AppId', 'AppName', 'QuerySymbol', 'Evidence', 'QueryText', 'SourceUrl', 'ConnectionAlias', 'ModelId', 'ModelName', 'ModelWorkspaceId', 'ModelWorkspaceName', 'ModelMappingEvidence', 'ModelSelection', 'RejectedModels', 'FittingModels', 'UnexpandedSource')
                }
            ) + $definitions + @(
                @{
                    Name = 'QueryDependencies'
                    Rows = $Dependencies
                    Headers = @('QueryId', 'WorkspaceId', 'WorkspaceName', 'AppId', 'AppName', 'QuerySymbol', 'ConnectionAlias', 'ModelWorkspaceId', 'ModelWorkspaceName', 'ModelId', 'ModelName', 'ObjectName', 'ObjectType', 'DependsOn', 'DependsOnType', 'DependsOnTable', 'DependencyKind', 'DependencyDepth', 'Resolution', 'ReasonCode', 'Detail', 'ExistsInOtherModels')
                },
                @{
                    Name = 'QueryDependencyStatus'
                    Rows = $DependencyStatus
                    Headers = @('QueryId', 'AppId', 'AppName', 'QuerySymbol', 'ModelId', 'Status', 'DirectObjectCount', 'UnresolvedCount', 'IncompleteCount', 'ModelSelection', 'FittingModels', 'Coverage')
                }
            )
        }
        foreach ($definition in $definitions) {
            if ($definition.Rows.Count -gt 1048575) { throw 'App query export exceeds the Excel row limit.' }
            $sheet = $package.Workbook.Worksheets[$definition.Name]
            if ($sheet) { $package.Workbook.Worksheets.Delete($definition.Name) }
            $sheet = $package.Workbook.Worksheets.Add($definition.Name)
            $headers = [System.Collections.Generic.List[string]]::new()
            foreach ($header in $definition.Headers) { $headers.Add($header) }
            foreach ($record in $definition.Rows) {
                foreach ($property in $record.PSObject.Properties.Name) {
                    if ($property -in @('TemplateExpressions', 'TemplateValueExpressions', 'TemplateConstants', 'TemplateFieldCatalog')) { continue }
                    if (-not $headers.Contains($property)) { $headers.Add($property) }
                }
            }
            if ($headers.Count -gt 16384) { throw 'App query export exceeds the Excel column limit.' }
            for ($column = 0; $column -lt $headers.Count; $column++) {
                $sheet.Cells[1, ($column + 1)].Value = $headers[$column]
                $sheet.Column($column + 1).Width = if ($headers[$column] -like 'QueryText*') { 85 } else { 25 }
            }
            $rowNumber = 2
            foreach ($record in $definition.Rows) {
                for ($column = 0; $column -lt $headers.Count; $column++) {
                    $value = $record.($headers[$column])
                    if ($value -is [string] -and $value.Length -gt 32767) {
                        throw "Value in $($headers[$column]) exceeds Excel's cell limit."
                    }
                    $sheet.Cells[$rowNumber, ($column + 1)].Value = $value
                }
                $sheet.Row($rowNumber).Height = 60
                $rowNumber++
            }
            $sheet.Cells[1, 1, 1, $headers.Count].Style.Font.Bold = $true
            $sheet.Cells[1, 1, [Math]::Max(1, $rowNumber - 1), $headers.Count].AutoFilter = $true
            $sheet.Cells.Style.WrapText = $true
            $sheet.View.FreezePanes(2, 1)
        }
        $legacySheets = @('AppQueries', 'AppQueryCaptures')
        if (-not $IncludeDependencies) {
            $legacySheets += @('FabricAppQueries', 'QueryDependencies', 'QueryDependencyStatus')
        }
        foreach ($legacySheet in $legacySheets) {
            if ($package.Workbook.Worksheets[$legacySheet]) { $package.Workbook.Worksheets.Delete($legacySheet) }
        }
        $package.Save()
    }
    finally {
        $package.Dispose()
    }
}

function Get-FabricAppDaxReferences {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)][AllowEmptyString()][string]$Text,
        [string[]]$TemplateExpressions = @(),
        [string]$DefaultTable = '',
        [switch]$IsTemplate,
        [switch]$IsFragment,
        [Collections.IDictionary]$Constants = @{},
        [string[]]$FieldCatalog = @(),
        [string[]]$TemplateValueExpressions = @()
    )

    $TemplateExpressions = @($TemplateExpressions | Where-Object { -not [string]::IsNullOrEmpty($_) })
    # Interpolations that only carry data (numbers, escaped text, value lists) read as a plain number to the DAX parser.
    $valueExpressionSet = [Collections.Generic.HashSet[string]]::new([StringComparer]::Ordinal)
    foreach ($valueExpression in @($TemplateValueExpressions)) {
        if ($valueExpression) { $null = $valueExpressionSet.Add($valueExpression); $null = $valueExpressionSet.Add($valueExpression.Trim()) }
    }
    $constantMap = [Collections.Generic.Dictionary[string,string]]::new([StringComparer]::Ordinal)
    foreach ($constantName in $Constants.Keys) { $constantMap[[string]$constantName] = [string]$Constants[$constantName] }
    $dynamicMarker = '__ImpactIQDynamicExpression__'
    while ($Text.Contains($dynamicMarker)) { $dynamicMarker += '_' }
    function Skip-AppJsString([string]$Value, [int]$Index) {
        $cursor = $Index + 1
        while ($cursor -lt $Value.Length) {
            if ($Value[$cursor] -eq '\') { $cursor += 2; continue }
            if ($Value[$cursor] -eq $Value[$Index]) { return $cursor }
            $cursor++
        }
        -1
    }
    function Skip-AppJsTemplate([string]$Value, [int]$Index) {
        $cursor = $Index + 1
        while ($cursor -lt $Value.Length) {
            if ($Value[$cursor] -eq '\') { $cursor += 2; continue }
            if ($Value[$cursor] -eq '`') { return $cursor }
            if ($Value[$cursor] -eq '$' -and $cursor + 1 -lt $Value.Length -and $Value[$cursor + 1] -eq '{') {
                $close = Skip-AppJsExpression $Value ($cursor + 2)
                if ($close -lt 0) { return -1 }
                $cursor = $close + 1
                continue
            }
            $cursor++
        }
        -1
    }
    function Skip-AppJsExpression([string]$Value, [int]$Index) {
        $depth = 1
        $cursor = $Index
        while ($cursor -lt $Value.Length) {
            $character = [string]$Value[$cursor]
            if ($character -eq '`') {
                $end = Skip-AppJsTemplate $Value $cursor
                if ($end -lt 0) { return -1 }
                $cursor = $end + 1
                continue
            }
            if ($character -eq '"' -or $character -eq "'") {
                $end = Skip-AppJsString $Value $cursor
                if ($end -lt 0) { return -1 }
                $cursor = $end + 1
                continue
            }
            if ($character -eq '{') { $depth++ }
            elseif ($character -eq '}') {
                $depth--
                if ($depth -eq 0) { return $cursor }
            }
            $cursor++
        }
        -1
    }
    function Expand-AppTemplateBody([string]$Body, [int]$Depth) {
        $builder = [Text.StringBuilder]::new()
        $cursor = 0
        while ($cursor -lt $Body.Length) {
            $open = $Body.IndexOf('${', $cursor, [StringComparison]::Ordinal)
            if ($open -lt 0) { $null = $builder.Append($Body.Substring($cursor)); break }
            $null = $builder.Append($Body.Substring($cursor, $open - $cursor))
            $close = Skip-AppJsExpression $Body ($open + 2)
            if ($close -lt 0) { $null = $builder.Append($Body.Substring($open)); break }
            $null = $builder.Append((Expand-AppExpression $Body.Substring($open + 2, $close - $open - 2) $Depth))
            $cursor = $close + 1
        }
        [regex]::Replace($builder.ToString(), '\\([`$\\"''])', '$1')
    }
    # A constant table name, or the DAX emitted by a top-level template/string branch (for example `flag ? `...` : ""`),
    # replaces the interpolation. Anything else stays an opaque runtime expression.
    function Expand-AppExpression([string]$Expression, [int]$Depth) {
        $name = $Expression.Trim()
        if ($valueExpressionSet.Contains($name)) { return ' 0 ' }
        if ($constantMap.ContainsKey($name)) { return $constantMap[$name] }
        $pieces = [Collections.Generic.List[string]]::new()
        $nesting = 0
        $cursor = 0
        while ($Depth -lt 4 -and $cursor -lt $Expression.Length) {
            $character = [string]$Expression[$cursor]
            if ($character -eq '`') {
                $end = Skip-AppJsTemplate $Expression $cursor
                if ($end -lt 0) { $pieces.Clear(); break }
                if ($nesting -eq 0) { $pieces.Add((Expand-AppTemplateBody $Expression.Substring($cursor + 1, $end - $cursor - 1) ($Depth + 1))) }
                $cursor = $end + 1
                continue
            }
            if ($character -eq '"' -or $character -eq "'") {
                $end = Skip-AppJsString $Expression $cursor
                if ($end -lt 0) { $pieces.Clear(); break }
                if ($nesting -eq 0) {
                    $body = [regex]::Replace($Expression.Substring($cursor + 1, $end - $cursor - 1), '\\(.)', '$1')
                    if ($body -cmatch '(?<![\w.])[A-Z][A-Z0-9.]{2,}\s*\(|(?<![\w''\]])\[[A-Za-z_][^\]]*\]') { $pieces.Add($body) }
                }
                $cursor = $end + 1
                continue
            }
            if ($character -in @('(', '[', '{')) { $nesting++ }
            elseif ($character -in @(')', ']', '}')) { $nesting-- }
            $cursor++
        }
        if ($pieces.Count -eq 0) { return $dynamicMarker }
        ' ' + ($pieces -join ' ') + ' '
    }
    # Longest first, so an expression that contains another interpolation is replaced before its inner part.
    foreach ($expression in @($TemplateExpressions | Sort-Object { $_.Length } -Descending)) {
        $Text = $Text.Replace(('${' + $expression + '}'), $(if ($valueExpressionSet.Contains($expression)) { ' 0 ' } else { Expand-AppExpression $expression 0 }))
    }
    if (($IsTemplate -or $TemplateExpressions.Count) -and $Text.Contains('${')) {
        throw [FormatException]::new('A JavaScript interpolation could not be separated from the DAX. Its runtime dependencies remain unresolved.')
    }
    $lexer = [regex]::new('\G(?:(?<space>\s+)|(?<comment>//[^\r\n]*|--[^\r\n]*|/\*[\s\S]*?\*/)|(?<string>"(?:[^"]|"")*")|(?<table>''(?:[^'']|'''')*'')|(?<field>\[(?:[^\]]|\]\])*\])|(?<word>[\p{L}_][\p{L}\p{N}_.]*)|(?<number>\d+(?:\.\d+)?)|(?<punct>[^\s]))')
    $tokens = [Collections.Generic.List[object]]::new()
    $position = 0
    while ($position -lt $Text.Length) {
        $match = $lexer.Match($Text, $position)
        if (-not $match.Success) {
            if ($IsFragment) { $position++; continue }
            throw [FormatException]::new("Invalid DAX token at position $position.")
        }
        $position += $match.Length
        if ($match.Groups['space'].Success -or $match.Groups['comment'].Success) { continue }
        foreach ($kind in @('string', 'table', 'field', 'word', 'number', 'punct')) {
            if ($match.Groups[$kind].Success) { break }
        }
        $value = $match.Value
        if ($kind -eq 'punct' -and $value -in @('"', "'", '[')) {
            if ($IsFragment) { continue }
            throw [FormatException]::new("Unterminated DAX identifier or string at position $($match.Index).")
        }
        if ($value -eq '/' -and $position -lt $Text.Length -and $Text[$position] -eq '*') {
            throw [FormatException]::new('Unterminated DAX comment.')
        }
        if ($kind -eq 'string') { $value = $value.Substring(1, $value.Length - 2).Replace('""', '"') }
        elseif ($kind -eq 'table') { $value = $value.Substring(1, $value.Length - 2).Replace("''", "'") }
        elseif ($kind -eq 'field') { $value = $value.Substring(1, $value.Length - 2).Replace(']]', ']') }
        $tokens.Add([PSCustomObject]@{ Kind = $kind; Value = $value })
    }
    $variables = @{}
    $localMeasures = @{}
    $definitions = @{}
    # An interpolation that could not be expanded may have declared variables, tables or measures this parser cannot see.
    $hasOpaqueFragment = $TemplateExpressions.Count -gt 0 -and $Text.Contains($dynamicMarker)
    for ($index = 0; $index -lt $tokens.Count - 1; $index++) {
        if ($tokens[$index].Kind -ne 'word') { continue }
        if ($tokens[$index].Value -eq 'VAR' -and $tokens[$index + 1].Kind -eq 'word') {
            $variables[$tokens[$index + 1].Value] = $true
        }
        if ($tokens[$index].Value -in @('COLUMN', 'TABLE', 'FUNCTION') -and
            $index -gt 0 -and $tokens[$index - 1].Value -ne '(') {
            throw [FormatException]::new('Query-scoped COLUMN/TABLE/FUNCTION definitions require engine validation; they are not treated as persisted model objects.')
        }
        if ($tokens[$index].Value -eq 'MEASURE' -and $index + 2 -lt $tokens.Count -and $tokens[$index + 2].Kind -eq 'field') {
            $name = $tokens[$index + 2].Value
            $localMeasures[$name] = $true
            $localMeasures[($tokens[$index + 1].Value + [char]0 + $name)] = $true
            $definitions[$index + 1] = $true
            $definitions[$index + 2] = $true
        }
    }
    $depths = [int[]]::new($tokens.Count)
    $depth = 0
    for ($index = 0; $index -lt $tokens.Count; $index++) {
        $depths[$index] = $depth
        if ($tokens[$index].Kind -ne 'punct') { continue }
        if ($tokens[$index].Value -in @('(', '{')) { $depth++ }
        elseif ($tokens[$index].Value -in @(')', '}')) { $depth-- }
    }
    $variableRanges = @{}
    for ($index = 0; $index -lt $tokens.Count - 3; $index++) {
        if ($tokens[$index].Kind -ne 'word' -or $tokens[$index].Value -ne 'VAR' -or $tokens[$index + 2].Value -ne '=') { continue }
        $end = $index + 3
        while ($end -lt $tokens.Count) {
            if ($depths[$end] -lt $depths[$index] -or
                ($depths[$end] -eq $depths[$index] -and
                    (($tokens[$end].Kind -eq 'word' -and $tokens[$end].Value -in @('VAR', 'RETURN', 'EVALUATE')) -or
                    ($tokens[$end].Kind -eq 'punct' -and $tokens[$end].Value -in @(')', '}'))))) { break }
            $end++
        }
        $name = $tokens[$index + 1].Value
        if (-not $variableRanges.ContainsKey($name)) { $variableRanges[$name] = [Collections.Generic.List[object]]::new() }
        $variableRanges[$name].Add(@{ Start = $index + 3; End = $end })
    }
    function Get-DaxReachableRanges([int]$Start, [int]$End) {
        $ranges = [Collections.Generic.List[object]]::new()
        $pending = [Collections.Generic.Queue[object]]::new()
        $pending.Enqueue(@{ Start = $Start; End = $End })
        $seen = @{}
        while ($pending.Count) {
            $range = $pending.Dequeue()
            $ranges.Add($range)
            for ($cursor = $range.Start; $cursor -lt $range.End; $cursor++) {
                $name = $tokens[$cursor].Value
                if ($tokens[$cursor].Kind -ne 'word' -or $seen.ContainsKey($name) -or -not $variableRanges.ContainsKey($name)) { continue }
                $seen[$name] = $true
                if ($variableRanges[$name].Count -eq 1) { $pending.Enqueue($variableRanges[$name][0]) }
            }
        }
        $ranges.ToArray()
    }
    $aliases = @{}
    $aliasRangeCache = @{}
    $outputRanges = @()
    $frames = [Collections.Generic.List[object]]::new()
    $references = [Collections.Generic.List[object]]::new()
    $reserved = @{}
    foreach ($word in @('DEFINE', 'EVALUATE', 'VAR', 'MEASURE', 'RETURN', 'ORDER', 'BY', 'ASC', 'DESC', 'IN', 'NOT', 'AND', 'OR', 'TRUE', 'FALSE', 'INTEGER', 'STRING', 'DOUBLE', 'BOOLEAN', 'DATETIME', 'CURRENCY', 'VARIANT', 'BLANK', 'DAY', 'WEEK', 'MONTH', 'QUARTER', 'YEAR', 'SECOND', 'MINUTE', 'HOUR', 'SKIP', 'DENSE', 'BOTH', 'NONE', 'ONEWAY', 'ONEWAY_LEFTFILTERSRIGHT', 'ONEWAY_RIGHTFILTERSLEFT', 'DEFAULT', 'FIRST', 'LAST', 'KEEP', 'IGNORE', 'STRICT', 'RELAXED', 'EXTENDING', 'PRECISE', 'BLANKS', 'ANCHORED', 'START', 'AT', 'ABS', 'REL')) {
        $reserved[$word] = $true
    }
    $ordering = $false
    for ($index = 0; $index -lt $tokens.Count; $index++) {
        $token = $tokens[$index]
        $previous = if ($index -gt 0) { $tokens[$index - 1] } else { $null }
        $next = if ($index + 1 -lt $tokens.Count) { $tokens[$index + 1] } else { $null }
        if ($token.Kind -eq 'word' -and $token.Value -eq 'EVALUATE') {
            $ordering = $false
            $end = $index + 1
            while ($end -lt $tokens.Count -and -not ($depths[$end] -eq 0 -and $tokens[$end].Kind -eq 'word' -and $tokens[$end].Value -in @('ORDER', 'EVALUATE'))) { $end++ }
            $outputRanges = @(Get-DaxReachableRanges ($index + 1) $end)
        }
        if ($token.Kind -eq 'word' -and $token.Value -eq 'BY' -and $previous.Value -eq 'ORDER') { $ordering = $true }
        if ($token.Kind -eq 'punct' -and $token.Value -eq '(') {
            $newFrame = @{ Function = [string]$previous.Value; Argument = 0; Braces = 0; Table = ''; InputVariable = ''; InputVariables = [Collections.Generic.List[string]]::new(); Start = $index; End = -1 }
            $frames.Add($newFrame)
            if ($newFrame.Function -eq 'GENERATESERIES') {
                if (-not $aliases.ContainsKey('Value')) { $aliases['Value'] = [Collections.Generic.List[object]]::new() }
                $aliases['Value'].Add($newFrame)
            }
            continue
        }
        if ($token.Kind -eq 'punct' -and $token.Value -eq ')') {
            if ($frames.Count -eq 0) {
                if ($IsFragment) { continue }
                throw [FormatException]::new('Unbalanced DAX parentheses.')
            }
            $frames[$frames.Count - 1].End = $index
            $frames.RemoveAt($frames.Count - 1)
            continue
        }
        $frame = if ($frames.Count) { $frames[$frames.Count - 1] } else { $null }
        if ($frame -and $token.Kind -eq 'punct') {
            if ($token.Value -eq '{') {
                $frame.Braces++
                # A table constructor `{ ... }` used as a table argument produces one implicit column named [Value].
                if ($previous -and $previous.Kind -eq 'punct' -and $previous.Value -in @('(', ',')) {
                    $constructorDepth = 0
                    $constructorEnd = -1
                    for ($scan = $index; $scan -lt $tokens.Count; $scan++) {
                        if ($tokens[$scan].Kind -ne 'punct') { continue }
                        if ($tokens[$scan].Value -eq '{') { $constructorDepth++ }
                        elseif ($tokens[$scan].Value -eq '}') { $constructorDepth--; if ($constructorDepth -eq 0) { $constructorEnd = $scan; break } }
                    }
                    if ($constructorEnd -gt $index) {
                        if (-not $aliases.ContainsKey('Value')) { $aliases['Value'] = [Collections.Generic.List[object]]::new() }
                        $aliases['Value'].Add(@{ Start = $index; End = $constructorEnd })
                    }
                }
            }
            elseif ($token.Value -eq '}') { $frame.Braces-- }
            elseif ($token.Value -eq ',' -and $frame.Braces -eq 0) { $frame.Argument++ }
        }
        if ($frame -and $frame.Argument -eq 0 -and $previous.Value -eq '(' -and
            $token.Kind -in @('table', 'word') -and $next.Value -ne '(') {
            if ($variables.ContainsKey($token.Value)) { $frame.InputVariable = $token.Value }
            else { $frame.Table = $token.Value }
        }
        # A variable anywhere in a call's first argument (for example CROSSJOIN(Var, ...)) supplies that call's columns.
        if ($token.Kind -eq 'word' -and $variables.ContainsKey($token.Value) -and $next.Value -ne '(' -and $previous.Value -ne 'VAR') {
            foreach ($inputFrame in $frames) {
                if ($inputFrame.Argument -eq 0 -and -not $inputFrame.InputVariables.Contains($token.Value)) { $inputFrame.InputVariables.Add($token.Value) }
            }
        }
        if ($frame -and $token.Kind -eq 'string' -and $frame.Braces -eq 0 -and $previous.Value -in @('(', ',')) {
            $isAlias = ($frame.Function -in @('ROW', 'DATATABLE') -and $frame.Argument % 2 -eq 0) -or
                ($frame.Function -in @('SELECTCOLUMNS', 'ADDCOLUMNS') -and $frame.Argument % 2 -eq 1) -or
                $frame.Function -in @('SUMMARIZE', 'SUMMARIZECOLUMNS', 'GROUPBY', 'ROLLUPADDISSUBTOTAL')
            if ($isAlias) {
                if (-not $aliases.ContainsKey($token.Value)) { $aliases[$token.Value] = [Collections.Generic.List[object]]::new() }
                $aliases[$token.Value].Add($frame)
            }
        }
        if ($definitions.ContainsKey($index)) { continue }
        if ($next.Kind -ne 'field' -and ($token.Kind -eq 'table' -or ($token.Kind -eq 'word' -and
            -not $reserved.ContainsKey($token.Value) -and -not $variables.ContainsKey($token.Value) -and
            $next.Value -ne '(' -and $previous.Value -ne 'VAR' -and $token.Value -cnotmatch '^__[A-Z][A-Z0-9_]*__$'))) {
            $requiredTable = $previous.Kind -eq 'word' -and $previous.Value -eq 'EVALUATE'
            if ($frame -and $previous.Kind -eq 'punct' -and $previous.Value -in @('(', ',')) {
                $requiredTable = $requiredTable -or
                    ($frame.Argument -eq 0 -and $frame.Function -in @('FILTER', 'SUMX', 'AVERAGEX', 'MINX', 'MAXX', 'COUNTX', 'COUNTAX', 'PRODUCTX', 'CONCATENATEX', 'ADDCOLUMNS', 'SELECTCOLUMNS', 'RANKX', 'COUNTROWS', 'VALUES', 'DISTINCT', 'ALL', 'ALLNOBLANKROW', 'ALLSELECTED', 'ALLEXCEPT', 'REMOVEFILTERS', 'SUMMARIZE', 'GROUPBY', 'CALCULATETABLE', 'RELATEDTABLE', 'TREATAS')) -or
                    ($frame.Function -in @('UNION', 'INTERSECT', 'EXCEPT', 'CROSSJOIN', 'GENERATE', 'GENERATEALL', 'NATURALINNERJOIN', 'NATURALLEFTOUTERJOIN')) -or
                    ($frame.Function -eq 'TOPN' -and $frame.Argument -eq 1)
            }
            if ($TemplateExpressions.Count -and $token.Kind -eq 'word' -and
                $token.Value -eq $dynamicMarker -and -not $requiredTable) { continue }
            # A fragment cannot declare its own VAR names, so only quoted names are table references.
            if ($IsFragment -and $token.Kind -ne 'table') { continue }
            $references.Add([PSCustomObject]@{
                Kind = $(if ($token.Kind -eq 'table' -or $requiredTable) { 'Table' } else { 'PossibleTable' })
                Table = $token.Value; Name = ''; ContextTable = ''; Alias = $false; Ordering = $ordering; GeneratedValue = $false
                Dynamic = $TemplateExpressions.Count -gt 0 -and $token.Value.Contains($dynamicMarker)
                LocalPossible = $hasOpaqueFragment -and $token.Kind -eq 'word'
            })
        }
        if ($token.Kind -ne 'field') { continue }
        $qualified = $previous.Kind -in @('table', 'word') -and -not $reserved.ContainsKey($previous.Value)
        $table = if ($qualified) { $previous.Value } else { '' }
        if (($qualified -and $localMeasures.ContainsKey(($table + [char]0 + $token.Value))) -or
            (-not $qualified -and $localMeasures.ContainsKey($token.Value))) { continue }
        $context = $DefaultTable
        for ($frameIndex = $frames.Count - 1; $frameIndex -ge 0; $frameIndex--) {
            $candidate = $frames[$frameIndex]
            if ($candidate.Table -and $candidate.Function -in @('FILTER', 'SUMX', 'AVERAGEX', 'MINX', 'MAXX', 'COUNTX', 'COUNTAX', 'PRODUCTX', 'CONCATENATEX', 'ADDCOLUMNS', 'SELECTCOLUMNS', 'RANKX', 'GENERATE')) {
                $context = $candidate.Table
                break
            }
        }
        $visibleAlias = $false
        if (-not $qualified -and $aliases.ContainsKey($token.Value)) {
            $visibleRanges = if ($ordering) { $outputRanges } else {
                foreach ($activeFrame in $frames) {
                    @{ Start = $activeFrame.Start; End = $index }
                    foreach ($variable in $activeFrame.InputVariables) {
                        if ($variableRanges.ContainsKey($variable) -and $variableRanges[$variable].Count -eq 1) {
                            if (-not $aliasRangeCache.ContainsKey($variable)) {
                                $range = $variableRanges[$variable][0]
                                $aliasRangeCache[$variable] = @(Get-DaxReachableRanges $range.Start $range.End)
                            }
                            $aliasRangeCache[$variable]
                        }
                    }
                }
            }
            foreach ($definitionFrame in $aliases[$token.Value]) {
                if ($definitionFrame.End -lt 0 -or $definitionFrame.End -ge $index) { continue }
                foreach ($range in $visibleRanges) {
                    if ($definitionFrame.Start -ge $range.Start -and $definitionFrame.End -le $range.End) { $visibleAlias = $true; break }
                }
                if ($visibleAlias) { break }
            }
        }
        $references.Add([PSCustomObject]@{
            Kind = 'Field'; Table = $table; Name = $token.Value; ContextTable = $context
            Alias = $visibleAlias; Ordering = $ordering; GeneratedValue = $false
            Dynamic = $TemplateExpressions.Count -gt 0 -and ($table.Contains($dynamicMarker) -or $token.Value.Contains($dynamicMarker))
            LocalPossible = $hasOpaqueFragment -and -not $table
        })
    }
    if ($frames.Count -and -not $IsFragment) { throw [FormatException]::new('Unbalanced DAX parentheses.') }
    # A column chosen at runtime from a static field catalog can only be a candidate; the dependency stage
    # keeps a candidate only when that exact table and field exist in the model.
    if ($FieldCatalog.Count) {
        foreach ($reference in @($references)) {
            if ($reference.Kind -ne 'Field' -or -not $reference.Table -or $reference.Table.Contains($dynamicMarker) -or $reference.Name -cne $dynamicMarker) { continue }
            foreach ($candidate in $FieldCatalog) {
                $references.Add([PSCustomObject]@{
                    Kind = 'Field'; Table = $reference.Table; Name = $candidate; ContextTable = ''; Alias = $false
                    Ordering = $false; GeneratedValue = $false; Dynamic = $false; Catalog = $true
                })
            }
        }
    }
    $references.ToArray()
}

function Get-FabricAppQueryDependencies {
    [CmdletBinding()]
    param(
        [AllowEmptyCollection()][object[]]$Queries = @(),
        [AllowEmptyCollection()][object[]]$ModelObjects = @(),
        [AllowEmptyCollection()][object[]]$AppModels = @(),
        [AllowEmptyCollection()][object[]]$Datasets = @()
    )

    $ErrorActionPreference = 'Stop'
    $fragmentEvidence = @('DaxFragment', 'ConfigExpression')
    function Get-ObjectKey([string]$Type, [string]$Table, [string]$Name) {
        ($Type + [char]0 + $Table + [char]0 + $Name).ToLowerInvariant()
    }
    function Get-ObjectType([string]$Type) {
        if ($Type -match 'Column$') { return 'Column' }
        $Type
    }
    function Get-ModelKey([string]$WorkspaceId, [string]$ModelId) {
        ($WorkspaceId + [char]0 + $ModelId).ToLowerInvariant()
    }
    function Get-AppKey([string]$WorkspaceId, [string]$AppId) {
        ($WorkspaceId + [char]0 + $AppId).ToLowerInvariant()
    }
    function Get-QueryModelKey($Query) {
        Get-ModelKey ([string]$Query.ModelWorkspaceId) ([string]$Query.ModelId)
    }
    $appScopes = @{}
    $configuredKeys = @{}
    $configuredIds = @{}
    $scopeMappings = if ($PSBoundParameters.ContainsKey('AppModels')) { $AppModels }
        else {
            # Direct callers without an app inventory may use only the explicit query targets, never other saved models.
            foreach ($query in $Queries) {
                if (-not $query.ModelId -or -not $query.ModelWorkspaceId) { continue }
                [PSCustomObject]@{
                    AppWorkspaceId = $query.WorkspaceId; AppId = $query.AppId; ConnectionAlias = $query.ConnectionAlias
                    ModelWorkspaceId = $query.ModelWorkspaceId; ModelId = $query.ModelId; ModelName = $query.ModelName
                }
            }
        }
    foreach ($mapping in $scopeMappings) {
        if (-not $mapping.ModelWorkspaceId -or -not $mapping.ModelId) { continue }
        $appWorkspaceId = [string]$mapping.AppWorkspaceId
        if (-not $appWorkspaceId) {
            $appWorkspaces = @($Queries | Where-Object { $_.AppId -eq $mapping.AppId } | Select-Object -ExpandProperty WorkspaceId -Unique)
            if ($appWorkspaces.Count -ne 1) {
                Write-Warning "App '$($mapping.AppId)' has no unambiguous app workspace ID; its model connection was excluded." -WarningAction Continue
                continue
            }
            $appWorkspaceId = [string]$appWorkspaces[0]
        }
        $appKey = Get-AppKey $appWorkspaceId ([string]$mapping.AppId)
        if (-not $appScopes.ContainsKey($appKey)) { $appScopes[$appKey] = @{ Mappings = [Collections.Generic.List[object]]::new(); Keys = @{} } }
        $modelKey = Get-ModelKey ([string]$mapping.ModelWorkspaceId) ([string]$mapping.ModelId)
        $appScopes[$appKey].Mappings.Add($mapping)
        $appScopes[$appKey].Keys[$modelKey] = $true
        $configuredKeys[$modelKey] = $true
        $configuredIds[[string]$mapping.ModelId] = $true
    }
    function Get-AppScope($Query) {
        $appScopes[(Get-AppKey ([string]$Query.WorkspaceId) ([string]$Query.AppId))]
    }
    function Get-ConfiguredAppModels($Query) {
        $scope = Get-AppScope $Query
        if ($scope) { $scope.Mappings.ToArray() }
    }
    function Test-ConfiguredQueryModel($Query) {
        $scope = Get-AppScope $Query
        $scope -and $scope.Keys.ContainsKey((Get-QueryModelKey $Query))
    }
    $datasetWorkspaces = @{}
    foreach ($dataset in $Datasets) {
        if (-not $dataset.DatasetId -or -not $dataset.WorkspaceId) { continue }
        $id = [string]$dataset.DatasetId
        if (-not $datasetWorkspaces.ContainsKey($id)) { $datasetWorkspaces[$id] = [Collections.Generic.HashSet[string]]::new([StringComparer]::OrdinalIgnoreCase) }
        $null = $datasetWorkspaces[$id].Add([string]$dataset.WorkspaceId)
    }
    $models = @{}
    $missingWorkspaceIds = @{}
    foreach ($row in $ModelObjects) {
        $type = Get-ObjectType ([string]$row.Type)
        if ($type -notin @('Table', 'Column', 'Measure') -or -not $row.ModelID) { continue }
        $modelId = [string]$row.ModelID
        if (-not $configuredIds.ContainsKey($modelId)) { continue }
        $workspaceId = [string]$row.ModelWorkspaceId
        if (-not $workspaceId) { $workspaceId = [string]$row.WorkspaceId }
        if (-not $workspaceId -and $datasetWorkspaces.ContainsKey($modelId) -and $datasetWorkspaces[$modelId].Count -eq 1) {
            $workspaceId = [string]@($datasetWorkspaces[$modelId])[0]
        }
        if (-not $workspaceId) {
            if (-not $missingWorkspaceIds.ContainsKey($modelId)) {
                $missingWorkspaceIds[$modelId] = $true
                Write-Warning "Model '$modelId' metadata has no unambiguous workspace ID; it cannot resolve app dependencies. Supply its exact dataset/workspace inventory." -WarningAction Continue
            }
            continue
        }
        $id = Get-ModelKey $workspaceId $modelId
        if (-not $configuredKeys.ContainsKey($id)) { continue }
        if (-not $models.ContainsKey($id)) { $models[$id] = @{ Objects = @{}; Fields = @{}; Names = @{}; ModelName = [string]$row.ModelName } }
        $model = $models[$id]
        $key = Get-ObjectKey $type $row.Table $row.Name
        if ($model.Objects.ContainsKey($key)) { continue }
        $object = [PSCustomObject]@{ Key = $key; Type = $type; Table = [string]$row.Table; Name = [string]$row.Name }
        $model.Objects[$key] = $object
        if ($type -in @('Column', 'Measure')) { $model.Fields[([string]$row.Table + [char]0 + [string]$row.Name)] = $key }
        $nameKey = $type + [char]0 + [string]$row.Name
        if (-not $model.Names.ContainsKey($nameKey)) { $model.Names[$nameKey] = [Collections.Generic.HashSet[string]]::new([StringComparer]::OrdinalIgnoreCase) }
        $null = $model.Names[$nameKey].Add($key)
    }
    function Resolve-QueryReference($Reference, $Model) {
        if ($Reference.Dynamic) { return }
        $keys = @()
        if ($Reference.Kind -in @('Table', 'PossibleTable')) {
            $key = Get-ObjectKey 'Table' $Reference.Table $Reference.Table
            if ($Model.Objects.ContainsKey($key)) { return $Model.Objects[$key] }
            return
        }
        if (-not $Reference.Table -and $Reference.Alias) { return }
        if ($Reference.Table) {
            $key = $Reference.Table + [char]0 + $Reference.Name
            if ($Model.Fields.ContainsKey($key)) { return $Model.Objects[$Model.Fields[$key]] }
            return
        }
        $nameKey = 'Measure' + [char]0 + $Reference.Name
        if ($Model.Names.ContainsKey($nameKey)) { $keys += @($Model.Names[$nameKey]) }
        if ($Reference.ContextTable) {
            $columnKey = Get-ObjectKey 'Column' $Reference.ContextTable $Reference.Name
            if ($Model.Objects.ContainsKey($columnKey)) { $keys += $columnKey }
        }
        if ($keys.Count -eq 1) { return $Model.Objects[$keys[0]] }
    }
    function Get-ModelLabel([string]$ModelId) {
        if ($models.ContainsKey($ModelId) -and $models[$ModelId].ModelName) { return $models[$ModelId].ModelName }
        $ModelId
    }
    # Why a reference could not be tied to a saved object, so a name mismatch is distinguishable from source that
    # this extract could not fully reconstruct.
    function Get-UnresolvedReason($Reference, $Model, [string]$Evidence = '') {
        if ($Reference.Dynamic) { return 'RuntimeIdentifier' }
        if ($Reference.LocalPossible) { return 'UnexpandedTemplate' }
        if ($Reference.Kind -in @('Table', 'PossibleTable')) { return 'TableMissing' }
        if ($Reference.Table) {
            if ($Model.Objects.ContainsKey((Get-ObjectKey 'Table' $Reference.Table $Reference.Table))) { return 'FieldMissing' }
            return 'TableMissing'
        }
        $candidateCount = 0
        if ($Model.Names.ContainsKey('Measure' + [char]0 + $Reference.Name)) { $candidateCount += $Model.Names['Measure' + [char]0 + $Reference.Name].Count }
        if ($Reference.ContextTable -and $Model.Objects.ContainsKey((Get-ObjectKey 'Column' $Reference.ContextTable $Reference.Name))) { $candidateCount++ }
        if ($candidateCount -gt 1) { return 'AmbiguousReference' }
        # A fragment has no enclosing query, so an unqualified name may be a row-context column or a generated column.
        if ($Evidence -ceq 'DaxFragment') { return 'FragmentContext' }
        'UnqualifiedNotFound'
    }
    function Get-ReasonDetail([string]$Reason, $Reference, [string]$ModelId, [string[]]$OtherModels) {
        $label = Get-ModelLabel $ModelId
        $message = switch ($Reason) {
            'TableMissing' { "Table '$($Reference.Table)' does not exist in model '$label'." }
            'FieldMissing' { "Table '$($Reference.Table)' exists in model '$label' but has no column or measure '$($Reference.Name)'." }
            'UnqualifiedNotFound' { "No measure '$($Reference.Name)' (or column in the row-context table) exists in model '$label'; it may be a query-local variable or generated column." }
            'FragmentContext' { "Unqualified name '$($Reference.Name)' is in a DAX fragment with no enclosing query, so it may be a row-context or generated column; it is not a confirmed miss in model '$label'." }
            'OtherModelBranch' { "'$(if ($Reference.Table) { $Reference.Table + '[' + $Reference.Name + ']' } else { $Reference.Name })' is not in candidate model '$label'. The query has no connection of its own; its candidate model assignment is inferred from partial object matches." }
            'AmbiguousReference' { "Unqualified name '$($Reference.Name)' matches more than one saved object in model '$label'; no object was guessed." }
            'RuntimeIdentifier' { 'The table or field name is chosen at runtime; no model object was guessed.' }
            'UnexpandedTemplate' { "'$(if ($Reference.Name) { $Reference.Name } else { $Reference.Table })' may be defined by a JavaScript-built fragment that could not be expanded." }
            default { 'Reference is missing, ambiguous, or runtime-generated; no model object was guessed.' }
        }
        if ($OtherModels.Count) { $message += " The exact reference exists in: $($OtherModels -join '; ')." }
        $message
    }
    # Alternative matches are limited to this app's exact configured workspace/model pairs.
    function Find-OtherModels($Reference, [string]$ExcludeModelId, $Query) {
        if ($Reference.Dynamic -or $Reference.LocalPossible) { return @() }
        $scope = Get-AppScope $Query
        if (-not $scope) { return @() }
        @(foreach ($id in $scope.Keys.Keys) {
            if ($id -eq $ExcludeModelId -or -not $models.ContainsKey($id)) { continue }
            if (Resolve-QueryReference $Reference $models[$id]) { Get-ModelLabel $id }
        }) | Select-Object -First 5
    }
    $edges = [Collections.Generic.List[object]]::new()
    $statuses = [Collections.Generic.List[object]]::new()
    $queryRows = [Collections.Generic.List[object]]::new()
    $sha = [Security.Cryptography.SHA256]::Create()
    $referenceCache = @{}
    function Get-QueryReferenceList($Query, [string]$Text) {
        $cacheKey = (@([string]$Query.SourceUrl, [string]$Query.Evidence, $Text) + @($Query.TemplateExpressions | ForEach-Object { [string]$_ }) + '|value|' + @($(if ($Query.PSObject.Properties['TemplateValueExpressions']) { $Query.TemplateValueExpressions }) | ForEach-Object { [string]$_ })) -join [string][char]0
        if ($referenceCache.ContainsKey($cacheKey)) { return $referenceCache[$cacheKey] }
        $constants = [Collections.Generic.Dictionary[string,string]]::new([StringComparer]::Ordinal)
        $constantSource = if ($Query.PSObject.Properties['TemplateConstants']) { $Query.TemplateConstants }
        if ($constantSource -is [Collections.IDictionary]) { foreach ($key in $constantSource.Keys) { $constants[[string]$key] = [string]$constantSource[$key] } }
        elseif ($constantSource) { foreach ($property in $constantSource.PSObject.Properties) { $constants[$property.Name] = [string]$property.Value } }
        $list = @(Get-FabricAppDaxReferences -Text $Text -TemplateExpressions @($Query.TemplateExpressions) -IsTemplate:($Query.Evidence -eq 'DynamicTemplate') -IsFragment:($Query.Evidence -in $fragmentEvidence) -Constants $constants -FieldCatalog @($(if ($Query.PSObject.Properties['TemplateFieldCatalog']) { $Query.TemplateFieldCatalog })) -TemplateValueExpressions @($(if ($Query.PSObject.Properties['TemplateValueExpressions']) { $Query.TemplateValueExpressions })))
        $referenceCache[$cacheKey] = $list
        $list
    }
    # References that name a specific table, column or measure, so they can vouch for (or against) a model.
    function Get-ExplicitReferences([object[]]$ReferenceList) {
        $explicit = @($ReferenceList | Where-Object {
            -not $_.Dynamic -and -not $_.LocalPossible -and -not ([string]$_.Table).StartsWith('__') -and
            ($_.Kind -eq 'Table' -or ($_.Kind -eq 'Field' -and ($_.Table -or -not $_.Alias)))
        })
        # A runtime-selected column still names its (quoted) table.
        $explicit += @($ReferenceList | Where-Object { $_.Kind -eq 'Field' -and $_.Dynamic -and $_.Table -and $_.Name -and -not $_.Table.Contains('__ImpactIQDynamicExpression__') -and -not ($_.PSObject.Properties['Catalog'] -and $_.Catalog) } |
            ForEach-Object { [PSCustomObject]@{ Kind = 'Table'; Table = $_.Table; Name = ''; ContextTable = ''; Alias = $false; Dynamic = $false } })
        @($explicit | Where-Object { -not ($_.PSObject.Properties['Catalog'] -and $_.Catalog) })
    }
    function Get-ModelFit([object[]]$Explicit, [string]$ModelId) {
        $matched = 0
        $missing = 0
        foreach ($reference in $Explicit) {
            if (Resolve-QueryReference $reference $models[$ModelId]) { $matched++ }
            elseif ($reference.Kind -eq 'Table' -or $reference.Table) { $missing++ }
        }
        [PSCustomObject]@{ Matched = $matched; Missing = $missing }
    }
    # A bracket-only name has no table that could be missing, so it fits only a model that actually contains it.
    function Test-ModelFit($Fit, [string]$Evidence) {
        $Fit.Missing -eq 0 -and ($Evidence -cne 'ConfigExpression' -or $Fit.Matched -gt 0)
    }
    function Add-UnlinkedStatus($Query, [string]$QueryId, [string]$Selection) {
        $statuses.Add([PSCustomObject][ordered]@{
            QueryId = $QueryId; AppId = $Query.AppId; AppName = $Query.AppName; QuerySymbol = $Query.QuerySymbol; ModelId = $Query.ModelId
            Status = 'UnlinkedConfigExpression'; DirectObjectCount = 0; UnresolvedCount = 0; IncompleteCount = 0
            ModelSelection = $Selection; FittingModels = ''
            Coverage = 'A bracketed name found in app scripts that matches no object in the app''s configured models with available metadata. No dependency was recorded.'
        })
    }
    function Get-FragmentHomeModel($Query, [object[]]$Explicit) {
        $configured = @((Get-ConfiguredAppModels $Query) | Where-Object { $models.ContainsKey((Get-ModelKey $_.ModelWorkspaceId $_.ModelId)) } |
            Group-Object { [string]$_.ModelWorkspaceId + '/' + [string]$_.ModelId })
        $fitting = @($configured | Where-Object {
            $fit = Get-ModelFit $Explicit (Get-ModelKey $_.Group[0].ModelWorkspaceId $_.Group[0].ModelId)
            (Test-ModelFit $fit ([string]$Query.Evidence)) -and $fit.Matched -gt 0
        })
        if ($fitting.Count -eq 1) { $fitting[0].Group[0] }
    }
    # A fragment has no connection of its own. It is assigned only when its exact table/field/measure names
    # exist in exactly one of the app's configured models and no explicit table reference is missing there.
    function Resolve-FragmentModel($Query, [string]$Text) {
        $candidates = @((Get-ConfiguredAppModels $Query) | Where-Object {
            $models.ContainsKey((Get-ModelKey $_.ModelWorkspaceId $_.ModelId)) -and
            (-not $Query.ConnectionAlias -or $_.ConnectionAlias -ceq $Query.ConnectionAlias)
        })
        if ($candidates.Count -eq 0) { return $Query }
        try { $fragmentReferences = Get-QueryReferenceList $Query $Text }
        catch [FormatException] { return $Query }
        $explicit = Get-ExplicitReferences $fragmentReferences
        $fits = @(foreach ($candidate in $candidates) {
            $fit = Get-ModelFit $explicit (Get-ModelKey $candidate.ModelWorkspaceId $candidate.ModelId)
            if ($fit.Matched -gt 0 -and $fit.Missing -eq 0) { $candidate }
        })
        $distinct = @($fits | Group-Object { [string]$_.ModelWorkspaceId + '/' + [string]$_.ModelId })
        if ($distinct.Count -ne 1) { return $Query }
        $chosen = $distinct[0].Group[0]
        $matchedQuery = $Query | Select-Object *
        foreach ($property in @(
            @('ConnectionAlias', $chosen.ConnectionAlias), @('ModelId', $chosen.ModelId), @('ModelName', $chosen.ModelName),
            @('ModelWorkspaceId', $chosen.ModelWorkspaceId), @('ModelWorkspaceName', $chosen.ModelWorkspaceName),
            @('ModelMappingEvidence', 'ObjectMatch'))) {
            $matchedQuery | Add-Member -NotePropertyName $property[0] -NotePropertyValue ([string]$property[1]) -Force
        }
        $matchedQuery
    }
    # A query with no connection of its own is listed against every configured model of its app that fits it, so a
    # dependency is never missed: all models that contain every object it names or, for a full query that no single
    # model fits because it branches between models, every model that contains any of them. Fragments use only the
    # first tier. A query that matches no configured model stays unmapped.
    function Expand-UnmappedQuery($Query, [string]$Text) {
        $candidates = @((Get-ConfiguredAppModels $Query) | Where-Object {
            $models.ContainsKey((Get-ModelKey $_.ModelWorkspaceId $_.ModelId)) -and
            (-not $Query.ConnectionAlias -or $_.ConnectionAlias -ceq $Query.ConnectionAlias)
        } | Group-Object { [string]$_.ModelWorkspaceId + '/' + [string]$_.ModelId } | ForEach-Object { $_.Group[0] })
        if ($candidates.Count -eq 0) { return $null }
        try { $explicit = Get-ExplicitReferences @(Get-QueryReferenceList $Query $Text) }
        catch [FormatException] { return $null }
        $evidence = [string]$Query.Evidence
        $scored = @(foreach ($candidate in $candidates) {
            [PSCustomObject]@{ Candidate = $candidate; Fit = (Get-ModelFit $explicit (Get-ModelKey $candidate.ModelWorkspaceId $candidate.ModelId)) }
        })
        $chosen = @($scored | Where-Object { (Test-ModelFit $_.Fit $evidence) -and $_.Fit.Matched -gt 0 })
        $partial = $false
        if ($chosen.Count -eq 0 -and $evidence -cnotin $fragmentEvidence) {
            $chosen = @($scored | Where-Object { $_.Fit.Matched -gt 0 })
            $partial = $true
        }
        if ($chosen.Count -eq 0) { return $null }
        $copies = @(foreach ($entry in $chosen) {
            $copy = $Query | Select-Object *
            foreach ($property in @(
                @('ConnectionAlias', $entry.Candidate.ConnectionAlias), @('ModelId', $entry.Candidate.ModelId), @('ModelName', $entry.Candidate.ModelName),
                @('ModelWorkspaceId', $entry.Candidate.ModelWorkspaceId), @('ModelWorkspaceName', $entry.Candidate.ModelWorkspaceName),
                @('ModelMappingEvidence', 'ObjectMatch'))) {
                $copy | Add-Member -NotePropertyName $property[0] -NotePropertyValue ([string]$property[1]) -Force
            }
            $copy
        })
        [PSCustomObject]@{ Queries = $copies; Partial = $partial }
    }
    try {
        $items = [Collections.Generic.List[object]]::new()
        foreach ($query in $Queries) {
            $text = @($query.PSObject.Properties | Where-Object { $_.Name -match '^QueryText(\d+)?$' } |
                Sort-Object { if ($_.Name -eq 'QueryText') { 1 } else { [int]$_.Name.Substring(9) } } |
                ForEach-Object { [string]$_.Value }) -join ''
            if ((-not $query.ModelId -or -not $query.ModelWorkspaceId) -and $query.Evidence -in $fragmentEvidence) {
                $query = Resolve-FragmentModel $query $text
            }
            $expansion = $null
            if (-not $query.ModelId -or -not $query.ModelWorkspaceId) { $expansion = Expand-UnmappedQuery $query $text }
            $variants = if ($expansion) { @($expansion.Queries) } else { @($query) }
            $groupKey = @($query.WorkspaceId, $query.AppId, $query.SourceUrl, $query.QuerySymbol, $query.Evidence, $text) | ConvertTo-Json -Compress
            foreach ($variant in $variants) {
                $items.Add([PSCustomObject]@{
                    Query = $variant; Text = $text; GroupKey = $groupKey; Drop = $false; Unlinked = $false
                    ModelKey = (Get-QueryModelKey $variant); Configured = [bool](Test-ConfiguredQueryModel $variant)
                    PartialMatch = [bool]($expansion -and $expansion.Partial)
                    ModelSelection = ''; RejectedModels = ''; FittingModels = ''
                })
            }
        }
        # One captured query can be routed to several models. A model that lacks tables or fields the query names
        # explicitly cannot be the one it runs against, so it is rejected when another candidate fits every reference.
        foreach ($group in @($items | Group-Object GroupKey -CaseSensitive)) {
            $candidates = @($group.Group | Where-Object { $_.Configured -and $models.ContainsKey($_.ModelKey) })
            if ($candidates.Count -eq 0) { continue }
            $evidence = [string]$candidates[0].Query.Evidence
            try { $explicit = Get-ExplicitReferences @(Get-QueryReferenceList $candidates[0].Query $candidates[0].Text) }
            catch [FormatException] { continue }
            $candidateIds = @($candidates | ForEach-Object { $_.ModelKey } | Select-Object -Unique)
            $fitting = @($candidateIds | Where-Object { Test-ModelFit (Get-ModelFit $explicit $_) $evidence })
            $rejected = @($candidateIds | Where-Object { $_ -notin $fitting })
            $selection = if ($fitting.Count -eq 0) { 'NoCandidateFits' }
                elseif ($candidateIds.Count -lt 2) { '' }
                elseif ($rejected.Count -eq 0) { 'AmbiguousModel' }
                elseif ($fitting.Count -eq 1) { 'ObjectMatch' }
                else { 'ObjectMatchMultiple' }
            $rejectedLabel = if ($fitting.Count) { (@($rejected | ForEach-Object { Get-ModelLabel $_ }) -join '; ') } else { '' }
            foreach ($candidate in $candidates) {
                $candidate.ModelSelection = $(if ($candidate.PartialMatch) { 'PartialMatch' } else { $selection })
                $candidate.RejectedModels = $rejectedLabel
                if ($fitting.Count -gt 0 -and $candidate.ModelKey -in $rejected) { $candidate.Drop = $true }
            }
            # A fragment's connection is inferred from nearby code, so when none of its candidates fit but exactly one
            # of the app's configured models does, that model is used. Full queries never have their connection replaced.
            if ($fitting.Count -eq 0 -and $evidence -in $fragmentEvidence) {
                $chosenModel = Get-FragmentHomeModel $candidates[0].Query $explicit
                if ($chosenModel) {
                    $reassigned = $candidates[0].Query | Select-Object *
                    foreach ($property in @(
                        @('ConnectionAlias', $chosenModel.ConnectionAlias), @('ModelId', $chosenModel.ModelId), @('ModelName', $chosenModel.ModelName),
                        @('ModelWorkspaceId', $chosenModel.ModelWorkspaceId), @('ModelWorkspaceName', $chosenModel.ModelWorkspaceName),
                        @('ModelMappingEvidence', 'ObjectMatch'))) {
                        $reassigned | Add-Member -NotePropertyName $property[0] -NotePropertyValue ([string]$property[1]) -Force
                    }
                    $candidates[0].Query = $reassigned
                    $candidates[0].ModelKey = Get-QueryModelKey $reassigned
                    $candidates[0].ModelSelection = 'ReassignedByObjectMatch'
                    $candidates[0].RejectedModels = (@($candidateIds | ForEach-Object { Get-ModelLabel $_ }) -join '; ')
                    foreach ($extra in @($candidates | Select-Object -Skip 1)) { $extra.Drop = $true }
                }
            }
        }
        # Unconnected models cannot turn a bracket-only string into evidence of an app dependency.
        $unlinkedKeys = @{}
        foreach ($item in $items) {
            if ($item.Drop -or $item.Query.Evidence -cne 'ConfigExpression' -or -not $item.Configured -or -not $models.ContainsKey($item.ModelKey)) { continue }
            try { $explicit = Get-ExplicitReferences @(Get-QueryReferenceList $item.Query $item.Text) }
            catch [FormatException] { continue }
            if (Test-ModelFit (Get-ModelFit $explicit $item.ModelKey) 'ConfigExpression') { continue }
            $known = $false
            foreach ($reference in $explicit) { if (@(Find-OtherModels $reference '' $item.Query).Count) { $known = $true; break } }
            if ($known) { continue }
            if ($unlinkedKeys.ContainsKey($item.GroupKey)) { $item.Drop = $true; continue }
            $unlinkedKeys[$item.GroupKey] = $true
            $item.Unlinked = $true
        }
        # Models that would satisfy every explicit reference when the assigned model does not.
        foreach ($item in $items) {
            if ($item.Drop -or $item.Unlinked -or -not $item.Configured -or -not $models.ContainsKey($item.ModelKey)) { continue }
            $itemEvidence = [string]$item.Query.Evidence
            try { $explicit = Get-ExplicitReferences @(Get-QueryReferenceList $item.Query $item.Text) }
            catch [FormatException] { continue }
            if (Test-ModelFit (Get-ModelFit $explicit $item.ModelKey) $itemEvidence) { continue }
            $scope = Get-AppScope $item.Query
            $item.FittingModels = (@($scope.Keys.Keys | Where-Object {
                $_ -ne $item.ModelKey -and $models.ContainsKey($_) -and ($fit = Get-ModelFit $explicit $_) -and $fit.Matched -gt 0 -and (Test-ModelFit $fit $itemEvidence)
            } | ForEach-Object { Get-ModelLabel $_ } | Select-Object -First 5) -join '; ')
        }
        foreach ($item in @($items | Where-Object { -not $_.Drop })) {
            $query = $item.Query
            $text = $item.Text
            $identity = @($query.WorkspaceId, $query.AppId, $query.SourceUrl, $query.QuerySymbol, $query.ConnectionAlias, $query.ModelWorkspaceId, $query.ModelId, $text) | ConvertTo-Json -Compress
            $queryId = [BitConverter]::ToString($sha.ComputeHash([Text.Encoding]::UTF8.GetBytes($identity))).Replace('-', '').ToLowerInvariant()
            $exportQuery = $query | Select-Object * -ExcludeProperty TemplateExpressions, TemplateValueExpressions, TemplateConstants, TemplateFieldCatalog
            $exportQuery | Add-Member -NotePropertyName QueryId -NotePropertyValue $queryId -Force
            $exportQuery | Add-Member -NotePropertyName ModelSelection -NotePropertyValue $item.ModelSelection -Force
            $exportQuery | Add-Member -NotePropertyName RejectedModels -NotePropertyValue $item.RejectedModels -Force
            $exportQuery | Add-Member -NotePropertyName FittingModels -NotePropertyValue $item.FittingModels -Force
            $queryRows.Add($exportQuery)
            $seenEdges = @{}
            $directObjects = @{}
            $queryState = @{ Unresolved = 0; Incomplete = 0 }
            $model = if ($item.Configured) { $models[$item.ModelKey] }
            function Add-QueryEdge($Parent, $Target, [string]$Kind, [int]$Depth, [string]$Detail = '', [string]$Reason = '', [string[]]$OtherModels = @()) {
                $targetKey = Get-ObjectKey $Target.Type $Target.Table $Target.Name
                $edgeKey = $Parent.Key + [char]1 + $targetKey + [char]1 + $Kind
                if ($seenEdges.ContainsKey($edgeKey)) { return }
                $seenEdges[$edgeKey] = $true
                $resolved = -not $Detail
                if (-not $resolved) {
                    $queryState.Unresolved++
                    # These cannot be judged against the model: the name or its definition exists only at runtime.
                    if ($Reason -in @('RuntimeIdentifier', 'UnexpandedTemplate', 'FragmentContext', 'OtherModelBranch')) { $queryState.Incomplete++ }
                    if ($item.PartialMatch) { return }
                }
                $edges.Add([PSCustomObject][ordered]@{
                    QueryId = $queryId; WorkspaceId = $query.WorkspaceId; WorkspaceName = $query.WorkspaceName
                    AppId = $query.AppId; AppName = $query.AppName; QuerySymbol = $query.QuerySymbol
                    ConnectionAlias = $query.ConnectionAlias; ModelWorkspaceId = $query.ModelWorkspaceId
                    ModelWorkspaceName = $query.ModelWorkspaceName
                    ModelId = $query.ModelId; ModelName = $query.ModelName
                    ObjectName = $Parent.Name; ObjectType = $Parent.Type
                    DependsOn = $Target.Name; DependsOnType = $Target.Type; DependsOnTable = $Target.Table
                    DependencyKind = $Kind; DependencyDepth = $Depth
                    Resolution = $(if ($resolved) { 'Resolved' } else { 'Unresolved' }); Detail = $Detail
                    ReasonCode = $Reason; ExistsInOtherModels = ($OtherModels -join '; ')
                })
                if ($resolved -and $Kind -eq 'Direct query reference') { $directObjects[$targetKey] = $true }
            }
            function Add-QueryReferences($Parent, [object[]]$References, [string]$Kind, [int]$Depth, [string]$ResolutionError = '', [string]$ResolutionCode = '') {
                foreach ($reference in $References) {
                    $target = if (-not $ResolutionError) { Resolve-QueryReference $reference $model }
                    if ($reference.PSObject.Properties['Catalog'] -and $reference.Catalog) {
                        if ($target) { Add-QueryEdge $Parent $target 'Catalog field reference' $Depth }
                        continue
                    }
                    if ($target) {
                        Add-QueryEdge $Parent $target $Kind $Depth
                    }
                    elseif ($reference.Kind -eq 'Field' -and -not $reference.Table -and ($reference.Alias -or $reference.GeneratedValue)) {
                        continue
                    }
                    else {
                        $reason = if ($ResolutionCode) { $ResolutionCode }
                            elseif ($reference.Kind -eq 'Invalid') { 'InvalidReference' }
                            else { Get-UnresolvedReason $reference $model ([string]$query.Evidence) }
                        if ($item.PartialMatch -and $reason -in @('TableMissing', 'FieldMissing', 'UnqualifiedNotFound')) { $reason = 'OtherModelBranch' }
                        $others = if ($reference.Kind -ne 'Invalid' -and $reason -in @('TableMissing', 'FieldMissing', 'UnqualifiedNotFound', 'FragmentContext', 'OtherModelBranch', 'NoModelMapping', 'MissingModelMetadata')) {
                            @(Find-OtherModels $reference $item.ModelKey $query)
                        } else { @() }
                        $detail = if ($ResolutionError) { $ResolutionError }
                            elseif ($reference.Kind -eq 'Invalid') { $reference.Error }
                            else { Get-ReasonDetail $reason $reference $item.ModelKey @() }
                        if ($others.Count) { $detail += " The exact reference exists in: $($others -join '; ')." }
                        Add-QueryEdge $Parent ([PSCustomObject]@{
                            Type = $(if ($reference.Kind -in @('Table', 'PossibleTable')) { 'Table' } else { 'Unknown' })
                            Table = $reference.Table; Name = $(if ($reference.Name) { $reference.Name } else { $reference.Table })
                        }) $Kind $Depth $detail $reason $others
                    }
                }
            }
            $root = [PSCustomObject]@{ Key = $queryId; Name = $query.QuerySymbol; Type = 'Query'; Table = '' }
            $status = 'ResolvedStaticReferences'
            $resolutionError = ''
            $resolutionCode = ''
            if ($item.Unlinked) {
                Add-UnlinkedStatus $query $queryId $item.ModelSelection
                continue
            }
            if (-not $query.ModelId -or -not $query.ModelWorkspaceId) {
                # A bare bracketed name that matches no backed-up model is not evidence of a query.
                if ($query.Evidence -eq 'ConfigExpression') {
                    $namesKnownObject = $false
                    try {
                        foreach ($reference in @(Get-QueryReferenceList $query $text)) {
                            if (@(Find-OtherModels $reference '' $query).Count) { $namesKnownObject = $true; break }
                        }
                    }
                    catch [FormatException] { }
                    if (-not $namesKnownObject) {
                        Add-UnlinkedStatus $query $queryId ''
                        continue
                    }
                }
                $status = 'MissingModelMapping'
                $resolutionCode = 'NoModelMapping'
                $resolutionError = 'The query has no unambiguous semantic-model mapping.'
            }
            elseif (-not $item.Configured) {
                $status = 'MissingModelMapping'
                $resolutionCode = 'ModelNotConfigured'
                $resolutionError = 'The query workspace/model ID pair is not among this app''s configured connections. No other workspace or same-named model was used.'
            }
            elseif (-not $model) {
                $status = 'MissingModelMetadata'
                $resolutionCode = 'MissingModelMetadata'
                $resolutionError = 'No model-object metadata was extracted for this exact workspace/model ID pair. Check model access and model-detail extraction.'
            }
            if ($resolutionError) {
                try {
                    $references = Get-QueryReferenceList $query $text
                    Add-QueryReferences $root $references 'Direct query reference' 1 $resolutionError $resolutionCode
                }
                catch [FormatException] {
                    Add-QueryEdge $root ([PSCustomObject]@{ Type = 'Unknown'; Table = ''; Name = '' }) 'Direct query reference' 1 $_.Exception.Message 'ParseError'
                }
                if (-not $queryState.Unresolved) {
                    Add-QueryEdge $root ([PSCustomObject]@{ Type = 'Unknown'; Table = ''; Name = '' }) 'Direct query reference' 1 $resolutionError $resolutionCode
                }
            }
            else {
                try {
                    $references = Get-QueryReferenceList $query $text
                    Add-QueryReferences $root $references 'Direct query reference' 1
                }
                catch [FormatException] {
                    Add-QueryEdge $root ([PSCustomObject]@{ Type = 'Unknown'; Table = ''; Name = '' }) 'Direct query reference' 1 $_.Exception.Message 'ParseError'
                }
                # References that exist only at runtime are incomplete evidence, not proof of a wrong model or name.
                if ($queryState.Unresolved -gt $queryState.Incomplete) { $status = 'UnresolvedReferences' }
                elseif ($query.Evidence -in $fragmentEvidence) { $status = 'PartialFragment' }
                elseif ($query.Evidence -eq 'DynamicTemplate') { $status = 'PartialDynamicTemplate' }
                elseif ($queryState.Unresolved) { $status = 'PartialComposition' }
            }
            $statuses.Add([PSCustomObject][ordered]@{
                QueryId = $queryId; AppId = $query.AppId; AppName = $query.AppName; QuerySymbol = $query.QuerySymbol; ModelId = $query.ModelId
                Status = $status; DirectObjectCount = $directObjects.Count; UnresolvedCount = $queryState.Unresolved
                IncompleteCount = $queryState.Incomplete; ModelSelection = $item.ModelSelection; FittingModels = $item.FittingModels
                Coverage = $(if ($item.PartialMatch) { 'Inferred candidate model: only metadata-matched dependencies are listed. Unmatched and runtime references are omitted from dependency rows but counted in UnresolvedCount and IncompleteCount. This is not a confirmed execution mapping.' }
                    else { 'Direct metadata-matched query references only. Measure/formula dependencies and inferred home tables are excluded. Runtime JavaScript and unvisited assets are not proven by this extract.' })
            })
        }
    }
    finally { $sha.Dispose() }
    [PSCustomObject]@{ Queries = $queryRows.ToArray(); Dependencies = $edges.ToArray(); Status = $statuses.ToArray() }
}

function Export-FabricAppQueryDependencyDetail {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)][string]$BaseFolderPath,
        [AllowEmptyCollection()][object[]]$Queries = @(),
        [AllowEmptyCollection()][object[]]$Models = @(),
        [AllowEmptyCollection()][object[]]$Captures = @(),
        [AllowEmptyCollection()][object[]]$ModelObjects = @(),
        [AllowEmptyCollection()][object[]]$Datasets = @()
    )

    $ErrorActionPreference = 'Stop'
    $result = Get-FabricAppQueryDependencies -Queries $Queries -ModelObjects $ModelObjects -AppModels $Models -Datasets $Datasets
    $path = Join-Path $BaseFolderPath 'Fabric App Detail.xlsx'
    $temporaryPath = Join-Path $BaseFolderPath ('AppQueryDependencies-' + [guid]::NewGuid().ToString('N') + '.xlsx')
    try {
        Export-FabricAppQueryWorkbook -Path $temporaryPath -Queries $result.Queries -Models $Models -Captures $Captures -Dependencies $result.Dependencies -DependencyStatus $result.Status -IncludeDependencies -ErrorAction Stop
        Move-Item -LiteralPath $temporaryPath -Destination $path -Force -ErrorAction Stop
    }
    finally {
        if (Test-Path -LiteralPath $temporaryPath) { Remove-Item -LiteralPath $temporaryPath -Force -ErrorAction Stop }
    }
    $incomplete = @($result.Status | Where-Object { $_.Status -ne 'ResolvedStaticReferences' })
    if ($incomplete.Count) {
        Write-FabricAppLog WARN "$($incomplete.Count) app query row(s) have partial or unresolved dependencies. Review QueryDependencyStatus and unresolved QueryDependencies rows in $path."
    }
    Write-FabricAppLog INFO "Exported $($result.Dependencies.Count) app query dependency edges to $path."
}

function Export-FabricAppDaxEnvironment {
    [CmdletBinding()]
    param(
        [AllowEmptyCollection()][object[]]$Items = @(),
        [Parameter(Mandatory)][string]$BaseFolderPath,
        [Parameter(Mandatory)][string]$ReportDate,
        [Parameter(Mandatory)][hashtable]$ConnectionCache,
        [string]$Environment = 'Public',
        [AllowEmptyCollection()][object[]]$Datasets = @(),
        [AllowEmptyCollection()][object[]]$Workspaces = @(),
        [Parameter(Mandatory)][string]$TabularEditorPath,
        [Parameter(Mandatory)][string]$BlankModelPath,
        [Parameter(Mandatory)][string]$ExtractScriptPath,
        [ValidateRange(1, 256)][int]$MaxAssets = 64,
        [AllowEmptyCollection()][object[]]$ModelObjects = @()
    )

    $ErrorActionPreference = 'Stop'
    $environmentWorkbook = Join-Path $BaseFolderPath 'Power BI Environment Detail.xlsx'
    if (-not (Test-Path -LiteralPath $environmentWorkbook -PathType Leaf)) {
        throw 'The environment workbook must exist before app queries can be appended.'
    }
    $appItems = @($Items | Where-Object { $_.FabricItemType -eq 'AppBackend' })
    $appQueryRows = @()
    $appModelRows = @()
    $appQueryCaptures = @()
    $appContexts = [Collections.Generic.List[object]]::new()
    foreach ($appItem in $appItems) {
        $backupPath = Get-FabricAppBackupPath -BaseFolderPath $BaseFolderPath -ReportDate $ReportDate -WorkspaceName $appItem.WorkspaceName -AppName $appItem.FabricItemName
        $key = "$($appItem.WorkspaceId)|$($appItem.FabricItemID)"
        $connections = $ConnectionCache[$key]
        $configuredModels = @()
        $mappingError = 'Model connection lookup was not completed for this app.'
        if ($connections) {
            $configuredModels = @($connections.Models)
            $mappingError = [string]$connections.Error
            if ($connections.BackupPath) { $backupPath = [string]$connections.BackupPath }
        }
        $backupFailed = -not ($connections -and $connections.BackupPath)
        $frontendRoot = $null
        if (-not $backupFailed -and (Test-Path -LiteralPath $backupPath -PathType Container)) {
            $frontendRoot = Get-FabricAppFrontendRoot -BackupPath $backupPath
        }
        $appContexts.Add([PSCustomObject]@{
            Item = $appItem; Key = $key; BackupPath = $backupPath; ConfiguredModels = $configuredModels
            MappingError = $mappingError; BackupFailed = $backupFailed; FrontendRoot = $frontendRoot
        })
    }
    # One Tabular Editor run parses every app's saved scripts.
    $extractResults = @{}
    $extractReady = $true
    $requests = @($appContexts | Where-Object { $_.FrontendRoot } | ForEach-Object {
        [PSCustomObject]@{ Key = $_.Key; FrontendRoot = $_.FrontendRoot; MaxAssets = $MaxAssets }
    })
    if ($Environment -eq 'Public' -and $requests.Count -gt 0) {
        Write-Host "[INFO] Fabric App query extraction started for $($requests.Count) app(s)."
        try {
            $extractResults = Invoke-FabricAppExtract -Requests $requests -TabularEditorPath $TabularEditorPath -BlankModelPath $BlankModelPath -ScriptPath $ExtractScriptPath -WorkingDirectory $BaseFolderPath -ErrorAction Stop
        }
        catch {
            $extractReady = $false
            Write-Warning "Fabric App script parsing failed: $($_.Exception.Message)" -WarningAction Continue
        }
    }
    foreach ($context in $appContexts) {
        $appItem = $context.Item
        if (-not $extractReady -and $context.FrontendRoot) {
            $appQueryCaptures += [PSCustomObject]@{
                WorkspaceId = $appItem.WorkspaceId
                WorkspaceName = $appItem.WorkspaceName
                AppId = $appItem.FabricItemID
                AppName = $appItem.FabricItemName
                CaptureStatus = 'Failed'
                CaptureError = 'Fabric App script parsing with Tabular Editor failed. Check the console error.'
                AssetsInspected = 0
                SkippedAssets = 0
            }
            continue
        }
        Write-Host "[INFO] Extracting Fabric app queries: $($appItem.WorkspaceName) ~ $($appItem.FabricItemName)"
        $extract = $extractResults[$context.Key]
        if (-not $extract) {
            $extract = @{ error = $(if ($context.FrontendRoot) { 'Tabular Editor returned no result for this app.' } else { "No extracted frontend exists in app backup '$($context.BackupPath)'." }) }
        }
        $appCapture = Get-FabricAppDaxCapture -WorkspaceId $appItem.WorkspaceId -WorkspaceName $appItem.WorkspaceName -AppId $appItem.FabricItemID -AppName $appItem.FabricItemName -BackupPath $context.BackupPath -BackupFailed:$context.BackupFailed -Extract $extract -Environment $Environment -Datasets $Datasets -Workspaces $Workspaces -ConfiguredModels $context.ConfiguredModels -ModelMappingError $context.MappingError
        $appQueryRows += @($appCapture.Queries)
        $appModelRows += @($appCapture.Models)
        $appQueryCaptures += $appCapture.Capture
    }
    Export-FabricAppQueryWorkbook -Path $environmentWorkbook -Queries $appQueryRows -Models $appModelRows -Captures $appQueryCaptures -ErrorAction Stop
    Export-FabricAppQueryDependencyDetail -BaseFolderPath $BaseFolderPath -Queries $appQueryRows -Models $appModelRows -Captures $appQueryCaptures -ModelObjects $ModelObjects -Datasets $Datasets -ErrorAction Stop
    Write-Host "[INFO] Exported $($appQueryRows.Count) queries and $($appModelRows.Count) configured models to $environmentWorkbook."
}

function Invoke-ImpactIQTabularEditorJobs {
    [CmdletBinding()]
    param(
        [AllowEmptyCollection()][object[]]$Jobs,
        [Parameter(Mandatory)][string]$TabularEditorPath,
        [Parameter(Mandatory)][string]$WorkingDirectory,
        [Parameter(Mandatory)][string]$Context
    )

    $ErrorActionPreference = 'Stop'
    function Remove-ImpactIQJobScript($Job) {
        if ($Job.PSObject.Properties['ScriptPath']) {
            try {
                if (Test-Path -LiteralPath $Job.ScriptPath) {
                    Remove-Item -LiteralPath $Job.ScriptPath -Force -ErrorAction Stop
                }
            }
            catch {
                Write-ImpactIQContinuingError -Record $_ -Context "Cleaning up $Context '$($Job.Name)'"
            }
        }
    }

    $active = [Collections.Generic.List[object]]::new()
    $activeOutputs = @{}
    $nextJob = 0
    try {
        while ($nextJob -lt $Jobs.Count -or $active.Count) {
            while ($nextJob -lt $Jobs.Count -and $active.Count -lt 4) {
                $job = $Jobs[$nextJob]
                $conflict = $false
                foreach ($outputPath in $job.Outputs) {
                    if ($activeOutputs.ContainsKey($outputPath)) { $conflict = $true; break }
                }
                if ($conflict) { break }
                $nextJob++
                try {
                    if ($job.PSObject.Properties['ScriptPath']) {
                        Set-Content -LiteralPath $job.ScriptPath -Value $job.ScriptContent -Encoding UTF8 -ErrorAction Stop
                    }
                    Write-Host $job.Message
                    $arguments = if ($job.Arguments -is [scriptblock]) { & $job.Arguments $job } else { $job.Arguments }
                    $run = Invoke-ImpactIQNativeProcess -FilePath $TabularEditorPath -WorkingDirectory $WorkingDirectory -ArgumentList $arguments -StartOnly -ErrorAction Stop
                    $active.Add([PSCustomObject]@{ Job = $job; Run = $run })
                    foreach ($outputPath in $job.Outputs) { $activeOutputs[$outputPath] = $true }
                }
                catch {
                    Write-ImpactIQContinuingError -Record $_ -Context "$Context '$($job.Name)'"
                    Remove-ImpactIQJobScript $job
                }
            }
            for ($index = $active.Count - 1; $index -ge 0; $index--) {
                $item = $active[$index]
                if (-not $item.Run.Process.HasExited) { continue }
                try {
                    $item.Run.Process.WaitForExit()
                    Write-ImpactIQNativeProcessResult -Run $item.Run
                    if ($item.Run.Process.ExitCode -ne 0) {
                        $script:ImpactIQRunFailed = $true
                        Write-ImpactIQDiagnostic -Level WARN -Message "$Context '$($item.Job.Name)' could not complete; continuing with the remaining models."
                    }
                    elseif ($item.Job.PSObject.Properties['RequiredOutputs']) {
                        foreach ($outputPath in $item.Job.RequiredOutputs) {
                            if (-not (Test-Path -LiteralPath $outputPath -PathType Leaf) -or
                                (Get-Item -LiteralPath $outputPath -ErrorAction Stop).Length -eq 0) {
                                throw "$Context '$($item.Job.Name)' did not create a non-empty model backup."
                            }
                        }
                    }
                }
                catch {
                    Write-ImpactIQContinuingError -Record $_ -Context "$Context '$($item.Job.Name)'"
                }
                finally {
                    $item.Run.Process.Dispose()
                    Remove-ImpactIQJobScript $item.Job
                    foreach ($outputPath in $item.Job.Outputs) { $activeOutputs.Remove($outputPath) }
                    $active.RemoveAt($index)
                }
            }
            if ($active.Count) { Start-Sleep -Milliseconds 50 }
        }
    }
    finally {
        foreach ($item in $active) {
            try {
                if (-not $item.Run.Process.HasExited) {
                    Stop-Process -Id $item.Run.Process.Id -Force -ErrorAction Stop
                    $item.Run.Process.WaitForExit()
                }
            }
            catch {
                Write-ImpactIQContinuingError -Record $_ -Context "Stopping $Context '$($item.Job.Name)'"
            }
            finally {
                $item.Run.Process.Dispose()
                Remove-ImpactIQJobScript $item.Job
            }
        }
    }
}

# Loop through filtered workspaces
foreach ($workspace in $workspacesInfo) {

Write-Output "Workspace [$($workspace.WorkspaceName)] Report & Model metadata extraction process started."

    # Fetch datasets
    $datasetsUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/groups/$($workspace.WorkspaceId)/datasets"
    try {
        $datasets = Invoke-ImpactIQRestMethod -Method GET -Url $datasetsUrl -ErrorAction SilentlyContinue | ConvertFrom-Json
    }
    catch {
        Write-ImpactIQContinuingError -Record $_ -Context "Workspace '$workspaceName' model metadata"
        continue
    }

    foreach ($dataset in $datasets.value) {
        if ($global:RunMode -in @('Agents', 'FabricApps') -and -not $global:SelectedDatasetIds.ContainsKey($dataset.id)) { continue }
        $renamedDataset = Rename-Properties -object $dataset -renameMap $datasetRenameMap
        $renamedDataset | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $workspace.WorkspaceId -Force
        $renamedDataset | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $workspace.WorkspaceName -Force
        
    # Store the DatasetId and DatasetName in the lookup table
        $datasetNameLookup[$dataset.id] = $dataset.name
        $datasetsInfo += $renamedDataset

        # Fetch dataset sources
        $datasourcesUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/groups/$($workspace.WorkspaceId)/datasets/$($dataset.id)/datasources"
        $datasources = Invoke-ImpactIQRestMethod -Method GET -Url $datasourcesUrl | ConvertFrom-Json

        foreach ($datasource in $datasources.value) {
            $renamedDatasource = Rename-Properties -object $datasource -renameMap $datasetDatasourceRenameMap
            $renamedDatasource | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $workspace.WorkspaceId -Force
            $renamedDatasource | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $workspace.WorkspaceName -Force
            $renamedDatasource | Add-Member -NotePropertyName "DatasetId" -NotePropertyValue $dataset.id -Force
            $renamedDatasource | Add-Member -NotePropertyName "DatasetName" -NotePropertyValue $dataset.name -Force
            if ($datasource.connectionDetails) {
                $renamedDatasource.DatasetDatasourceConnectionDetails = $datasource.connectionDetails | ConvertTo-Json -Compress
            }
            $datasetSourcesInfo += $renamedDatasource
        }
    }

    # Fetch reports
    $reportsUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/groups/$($workspace.WorkspaceId)/reports"
    $reports = $null
    if ($global:RunMode -notin @('Agents', 'FabricApps')) {
        $reports = Invoke-ImpactIQRestMethod -Method GET -Url $reportsUrl | ConvertFrom-Json
    }

	# Create a hash set to store Report IDs
	$reportIds = @{}

    foreach ($report in $reports.value) {
        $renamedReport = Rename-Properties -object $report -renameMap $reportRenameMap
        $renamedReport | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $workspace.WorkspaceId -Force
        $renamedReport | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $workspace.WorkspaceName -Force


        # Retrieve and add the correct DatasetName from the lookup table if DatasetId exists
        $datasetId = $report.datasetId
        if ($datasetId -and $datasetNameLookup.ContainsKey($datasetId)) {
            $renamedReport | Add-Member -NotePropertyName "DatasetName" -NotePropertyValue $datasetNameLookup[$datasetId] -Force
        } else {
            $renamedReport | Add-Member -NotePropertyName "DatasetName" -NotePropertyValue "Unknown Dataset" -Force
        }

        $reportsInfo += $renamedReport

        # Fetch report pages
        $pagesUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/groups/$($workspace.WorkspaceId)/reports/$($report.id)/pages"
        $pages = Invoke-ImpactIQRestMethod -Method GET -Url $pagesUrl | ConvertFrom-Json
        foreach ($page in $pages.value) {
            $renamedPage = Rename-Properties -object $page -renameMap $pageRenameMap
            $renamedPage | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $workspace.WorkspaceId -Force
            $renamedPage | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $workspace.WorkspaceName -Force
            $renamedPage | Add-Member -NotePropertyName "ReportId" -NotePropertyValue $report.id -Force
            $renamedPage | Add-Member -NotePropertyName "ReportName" -NotePropertyValue $report.name -Force
            $reportPagesInfo += $renamedPage

            # Store the report ID in the hash set
            $reportIds[$report.id] = $true
        }
    }
Write-Output "Workspace [$($workspace.WorkspaceName)] Report & Model metadata extraction process completed."
}

# Fetch Refresh Schedule for Datasets
foreach ($workspace in $workspacesInfo) {

Write-Output "Workspace [$($workspace.WorkspaceName)] Model Refresh Schedule extraction process started."

    foreach ($dataset in $datasetsInfo | Where-Object { $_.WorkspaceId -eq $workspace.WorkspaceId }) {
        $refreshScheduleUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/groups/$($workspace.WorkspaceId)/datasets/$($dataset.DatasetId)/refreshSchedule"
        
        try {
            $refreshScheduleResponse = Invoke-ImpactIQRestMethod -Method GET -Url $refreshScheduleUrl | ConvertFrom-Json
            
            # Get base properties that apply to all rows
            $renamedScheduleRecord = Rename-Properties -object $refreshScheduleResponse -renameMap $datasetRefreshScheduleRenameMap
            
            # Create separate rows for each day-time combination
            $days = if ($refreshScheduleResponse.days) { $refreshScheduleResponse.days } else { @($null) }
            $times = if ($refreshScheduleResponse.times) { $refreshScheduleResponse.times } else { @($null) }
            
            foreach ($day in $days) {
                foreach ($time in $times) {
                    $scheduleRow = $renamedScheduleRecord.PSObject.Copy()
                    $scheduleRow | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $workspace.WorkspaceId -Force
                    $scheduleRow | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $workspace.WorkspaceName -Force
                    $scheduleRow | Add-Member -NotePropertyName "DatasetId" -NotePropertyValue $dataset.DatasetId -Force
                    $scheduleRow | Add-Member -NotePropertyName "DatasetName" -NotePropertyValue $dataset.DatasetName -Force
                    $scheduleRow | Add-Member -NotePropertyName "DatasetRefreshScheduleDay" -NotePropertyValue $day -Force
                    $scheduleRow | Add-Member -NotePropertyName "DatasetRefreshScheduleTime" -NotePropertyValue $time -Force
                    
                    $datasetRefreshSchedule += $scheduleRow
                }
            }
        }
        catch {
            # Silently continue if refresh schedule is not available for this dataset
        }
    }
Write-Output "Workspace [$($workspace.WorkspaceName)] Model Refresh Schedule extraction process completed."
}


# Fetch Dataflows for Workspaces
foreach ($workspace in $workspacesInfo) {
    if ($global:RunMode -in @('Agents', 'FabricApps')) { continue }

Write-Output "Workspace [$($workspace.WorkspaceName)] Dataflow metadata extraction process started."

    # Fetch Gen 1 and Gen 2 dataflows using Power BI API
    $dataflowsUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/groups/$($workspace.WorkspaceId)/dataflows"
    $dataflowsResponse = Invoke-ImpactIQRestMethod -Method GET -Url $dataflowsUrl | ConvertFrom-Json

    # Ensure response is not null before looping
    if ($dataflowsResponse.value) {
        foreach ($dataflow in $dataflowsResponse.value) {
            $renamedDataflow = Rename-Properties -object $dataflow -renameMap $dataflowRenameMap
            $renamedDataflow | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $workspace.WorkspaceId -Force
            $renamedDataflow | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $workspace.WorkspaceName -Force

            # Store DataflowId and DataflowName in a lookup table
            if ($dataflow.objectId) {  
                $dataflowNameLookup[$dataflow.objectId] = $dataflow.name  
            }

            $dataflowsInfo += $renamedDataflow

            # Fetch Dataflow Datasources
            $dataflowSourcesUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/groups/$($workspace.WorkspaceId)/dataflows/$($dataflow.objectId)/datasources"
            $dataflowSourcesResponse = Invoke-ImpactIQRestMethod -Method GET -Url $dataflowSourcesUrl | ConvertFrom-Json

            # Ensure response is not null before looping
            if ($dataflowSourcesResponse.value) {
                foreach ($datasource in $dataflowSourcesResponse.value) {
                    $renamedDataflowDatasource = Rename-Properties -object $datasource -renameMap $dataflowDatasourceRenameMap

                    # Ensure required fields are included
                    $renamedDataflowDatasource | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $workspace.WorkspaceId -Force
                    $renamedDataflowDatasource | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $workspace.WorkspaceName -Force
                    $renamedDataflowDatasource | Add-Member -NotePropertyName "DataflowId" -NotePropertyValue $dataflow.objectId -Force

            		if ($datasource.connectionDetails) {
               		 $renamedDataflowDatasource.DataflowDatasourceConnectionDetails = $datasource.connectionDetails | ConvertTo-Json -Compress
            }
                    

                    if ($dataflowNameLookup.ContainsKey($dataflow.objectId)) {
                        $renamedDataflowDatasource | Add-Member -NotePropertyName "DataflowName" -NotePropertyValue $dataflowNameLookup[$dataflow.objectId] -Force
                    } else {
                        $renamedDataflowDatasource | Add-Member -NotePropertyName "DataflowName" -NotePropertyValue "Unknown Dataflow" -Force
                    }

                    # Store in collection
                    $dataflowSourcesInfo += $renamedDataflowDatasource
                }
            }
        }
    }

    # Fetch Gen 2 CICD dataflows using Fabric API
    try {
        $fabricDataflowsUrl = "$($global:PowerBIEndpoints.FabricApiPrefix)/v1/workspaces/$($workspace.WorkspaceId)/dataflows"
        $fabricHeaders = @{
            'Authorization' = "Bearer $(Get-FabricAccessToken)"
            'Content-Type' = 'application/json'
        }
        $fabricDataflowsResponse = Invoke-RestMethod -Uri $fabricDataflowsUrl -Headers $fabricHeaders -Method GET -ErrorAction Stop

        # Ensure response is not null before looping
        if ($fabricDataflowsResponse.value) {
            foreach ($fabricDataflow in $fabricDataflowsResponse.value) {
                # Map Fabric API response to match Power BI API structure using rename map
                $renamedFabricDataflow = Rename-Properties -object $fabricDataflow -renameMap $fabricDataflowRenameMap
                $renamedFabricDataflow | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $workspace.WorkspaceId -Force
                $renamedFabricDataflow | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $workspace.WorkspaceName -Force
                $renamedFabricDataflow | Add-Member -NotePropertyName "DataflowGeneration" -NotePropertyValue "Gen 2 CICD" -Force
                
                # Store DataflowId and DataflowName in a lookup table
                if ($fabricDataflow.id) {  
                    $dataflowNameLookup[$fabricDataflow.id] = $fabricDataflow.displayName  
                }

                $dataflowsInfo += $renamedFabricDataflow
            }
        }
    }
    catch {
        # Silently continue if Fabric API is not available or returns an error
        # This allows the script to work in environments without Fabric API access
    }

Write-Output "Workspace [$($workspace.WorkspaceName)] Dataflow metadata extraction process completed."
}

# Track which reports have a sensitivity label applied. This detail is ONLY exposed by
# the Fabric List Items API (not the Power BI reports API): a labelled report includes a
# "sensitivityLabel" object, an unlabelled one omits it entirely. When a report is
# labelled, the Report Backup step prefers getDefinition over LiveConnect. We only record
# presence here - the specific sensitivity and id itself is not needed.
$global:ReportsWithSensitivityLabel = @{}

# Loop through all workspaces to fetch Fabric Items (excluding Reports and Semantic Models)
foreach ($workspace in $workspacesInfo) {

Write-Output "Workspace [$($workspace.WorkspaceName)] Fabric Item metadata extraction process started."

    $workspaceId = $workspace.WorkspaceId
    $workspaceName = $workspace.WorkspaceName
    $fabricItemsUrl = "$($global:PowerBIEndpoints.FabricApiPrefix)/v1/workspaces/$workspaceId/items"

    try {
        $fabricItems = @(Invoke-FabricPagedGet -Url $fabricItemsUrl)
        if ($global:RunMode -eq 'Agents') {
            $fabricItems = @($fabricItems | Where-Object {
                ($_.type -eq 'DataAgent' -and $script:SelectedDataAgents.ContainsKey("$workspaceId|$($_.id)")) -or
                ($_.type -eq 'SemanticModel' -and $global:SelectedDatasetIds.ContainsKey($_.id))
            })
        }
        elseif ($global:RunMode -eq 'FabricApps') {
            $fabricItems = @($fabricItems | Where-Object {
                ($_.type -eq 'AppBackend' -and $script:SelectedFabricApps.ContainsKey("$workspaceId|$($_.id)")) -or
                ($_.type -eq 'SemanticModel' -and $global:SelectedDatasetIds.ContainsKey($_.id))
            })
        }

        # Record reports that carry a sensitivity label (presence only - the id is not needed).
        foreach ($fabricItem in $fabricItems) {
            if ($fabricItem.type -eq 'Report' -and $fabricItem.sensitivityLabel) {
                $global:ReportsWithSensitivityLabel[$fabricItem.id] = $true
            }

            $itemConnectionsUrl = "$($global:PowerBIEndpoints.FabricApiPrefix)/v1/workspaces/$workspaceId/items/$($fabricItem.id)/connections"
            try {
                foreach ($connection in (Invoke-FabricPagedGet -Url $itemConnectionsUrl)) {
                    $itemConnectionsInfo += [PSCustomObject]@{
                        WorkspaceId      = $workspaceId
                        WorkspaceName    = $workspaceName
                        FabricItemId     = $fabricItem.id
                        FabricItemName   = $fabricItem.displayName
                        FabricItemType   = $fabricItem.type
                        ConnectionId     = $connection.id
                        ConnectionName   = $connection.displayName
                        GatewayId        = $connection.gatewayId
                        ConnectivityType = $connection.connectivityType
                        ConnectionType   = $connection.connectionDetails.type
                        ConnectionPath   = $connection.connectionDetails.path
                    }
                }
            } catch {
                Write-Warning "Failed to fetch connections for Fabric item $($fabricItem.id) in workspace $workspaceId. $($_.Exception.Message)"
            }
        }

        $filteredItems = $fabricItems | Where-Object {
            $_.type -ne 'Report' -and $_.type -ne 'SemanticModel'
        }

        foreach ($item in $filteredItems) {
            $renamedItem = Rename-Properties -object $item -renameMap $fabricItemsRenameMap
            $renamedItem | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $workspaceId -Force
            $renamedItem | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $workspaceName -Force

            $fabricItemsInfo += $renamedItem
        }
    } catch {
        Write-Warning "Failed to fetch Fabric items for Workspace ID: $workspaceId. $_"
    }

Write-Output "Workspace [$($workspace.WorkspaceName)] Fabric Item metdadata extraction process completed."

}

    $myWorkspaceDatasetsUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/datasets"
    $myWorkspaceDatasets = Invoke-ImpactIQRestMethod -Method GET -Url $myWorkspaceDatasetsUrl | ConvertFrom-Json
    foreach ($dataset in $myWorkspaceDatasets.value) {
        $renamedDataset = Rename-Properties -object $dataset -renameMap $datasetRenameMap
        $renamedDataset | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $myWorkspaceId -Force
        $renamedDataset | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $myWorkspaceName -Force
        $datasetNameLookup[$dataset.id] = $dataset.name
        $datasetsInfo += $renamedDataset

        $datasourcesUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/datasets/$($dataset.id)/datasources"
        $datasources = Invoke-ImpactIQRestMethod -Method GET -Url $datasourcesUrl | ConvertFrom-Json
        foreach ($datasource in $datasources.value) {
            $renamedDatasource = Rename-Properties -object $datasource -renameMap $datasetDatasourceRenameMap
            $renamedDatasource | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $myWorkspaceId -Force
            $renamedDatasource | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $myWorkspaceName -Force
            $renamedDatasource | Add-Member -NotePropertyName "DatasetId" -NotePropertyValue $dataset.id -Force
            $renamedDatasource | Add-Member -NotePropertyName "DatasetName" -NotePropertyValue $dataset.name -Force
            if ($datasource.connectionDetails) {
                $renamedDatasource.DatasetDatasourceConnectionDetails = $datasource.connectionDetails | ConvertTo-Json -Compress
            }
            $datasetSourcesInfo += $renamedDatasource
        }
    }

    # Fetch reports from "My Workspace"
    $myWorkspaceReportsUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/reports"
    $myWorkspaceReports = Invoke-ImpactIQRestMethod -Method GET -Url $myWorkspaceReportsUrl | ConvertFrom-Json

    $sharedReportExists = $false
    foreach ($report in $myWorkspaceReports.value) {
        # Skip if already captured elsewhere (workspace/app)
        if ($appReportIds.ContainsKey($report.id) -or $reportIds.ContainsKey($report.id) -or $originalReportObjectIds.ContainsKey($report.id)) {
            continue
        }
        
        # Skip reports with appId - these are shared via apps, not actually in My Workspace
        if ($report.appId) {
            continue
        }

        if ($report.isOwnedByMe -eq $false) {
            $workspaceIdValue = "Shared Reports (No Workspace Access)"
            $workspaceNameValue = "Shared Reports (No Workspace Access)"
            $sharedReportExists = $true
        } else {
            $workspaceIdValue = $myWorkspaceId
            $workspaceNameValue = $myWorkspaceName
        }

        $renamedReport = Rename-Properties -object $report -renameMap $reportRenameMap
        $renamedReport | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $workspaceIdValue -Force
        $renamedReport | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $workspaceNameValue -Force

        $datasetId = $report.datasetId
        if ($datasetId -and $datasetNameLookup.ContainsKey($datasetId)) {
            $renamedReport | Add-Member -NotePropertyName "DatasetName" -NotePropertyValue $datasetNameLookup[$datasetId] -Force
        } else {
            $renamedReport | Add-Member -NotePropertyName "DatasetName" -NotePropertyValue "Unknown Dataset" -Force
        }
        $reportsInfo += $renamedReport

        $pagesUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/reports/$($report.id)/pages"
        $pages = Invoke-ImpactIQRestMethod -Method GET -Url $pagesUrl | ConvertFrom-Json
        foreach ($page in $pages.value) {
            $renamedPage = Rename-Properties -object $page -renameMap $pageRenameMap
            $renamedPage | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $workspaceIdValue -Force
            $renamedPage | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $workspaceNameValue -Force
            $renamedPage | Add-Member -NotePropertyValue "ReportId" -NotePropertyValue $report.id -Force
            $renamedPage | Add-Member -NotePropertyValue "ReportName" -NotePropertyValue $report.name -Force
            $reportPagesInfo += $renamedPage
        }
    }

    if ($sharedReportExists) {
        $sharedWorkspaceDetails = [PSCustomObject]@{
            WorkspaceId                    = "Shared Reports (No Workspace Access)"
            WorkspaceName                  = "Shared Reports (No Workspace Access)"
            WorkspaceType                  = "Workspace"
            WorkspaceIsReadOnly            = $false
            WorkspaceIsOnDedicatedCapacity = $false
        }
        $workspacesInfo += $sharedWorkspaceDetails
    }

    # Dataset refresh history for My Workspace
    foreach ($dataset in $datasetsInfo | Where-Object { $_.WorkspaceId -eq $myWorkspaceId }) {
        $refreshHistoryUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/datasets/$($dataset.DatasetId)/refreshes"
        $refreshHistoryResponse = Invoke-ImpactIQRestMethod -Method GET -Url $refreshHistoryUrl | ConvertFrom-Json
        foreach ($refresh in $refreshHistoryResponse.value) {
            $renamedRefreshRecord = Rename-Properties -object $refresh -renameMap $datasetRefreshRenameMap
            $renamedRefreshRecord | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $myWorkspaceId -Force
            $renamedRefreshRecord | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $myWorkspaceName -Force
            $renamedRefreshRecord | Add-Member -NotePropertyName "DatasetId" -NotePropertyValue $dataset.DatasetId -Force
            $renamedRefreshRecord | Add-Member -NotePropertyName "DatasetName" -NotePropertyValue $dataset.DatasetName -Force
            $datasetRefreshHistory += $renamedRefreshRecord
        }
    }

    # Dataset refresh schedule for My Workspace
    foreach ($dataset in $datasetsInfo | Where-Object { $_.WorkspaceId -eq $myWorkspaceId }) {
        $refreshScheduleUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/datasets/$($dataset.DatasetId)/refreshSchedule"
        
        try {
            $refreshScheduleResponse = Invoke-ImpactIQRestMethod -Method GET -Url $refreshScheduleUrl | ConvertFrom-Json
            
            # Get base properties that apply to all rows
            $renamedScheduleRecord = Rename-Properties -object $refreshScheduleResponse -renameMap $datasetRefreshScheduleRenameMap
            
            # Create separate rows for each day-time combination
            $days = if ($refreshScheduleResponse.days) { $refreshScheduleResponse.days } else { @($null) }
            $times = if ($refreshScheduleResponse.times) { $refreshScheduleResponse.times } else { @($null) }
            
            foreach ($day in $days) {
                foreach ($time in $times) {
                    $scheduleRow = $renamedScheduleRecord.PSObject.Copy()
                    $scheduleRow | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $myWorkspaceId -Force
                    $scheduleRow | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $myWorkspaceName -Force
                    $scheduleRow | Add-Member -NotePropertyName "DatasetId" -NotePropertyValue $dataset.DatasetId -Force
                    $scheduleRow | Add-Member -NotePropertyName "DatasetName" -NotePropertyValue $dataset.DatasetName -Force
                    $scheduleRow | Add-Member -NotePropertyName "DatasetRefreshScheduleDay" -NotePropertyValue $day -Force
                    $scheduleRow | Add-Member -NotePropertyName "DatasetRefreshScheduleTime" -NotePropertyValue $time -Force
                    
                    $datasetRefreshSchedule += $scheduleRow
                }
            }
        }
        catch {
            # Silently continue if refresh schedule is not available for this dataset
        }
    }

} else {
    Write-Host "Skipping 'My Workspace' (user did not select it and no timeout)."
}

function Export-EmptyDetailWorkbook {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)][string]$Path,
        [Parameter(Mandatory)][System.Collections.IDictionary]$Worksheets
    )

    $ErrorActionPreference = 'Stop'
    if (Test-Path -LiteralPath $Path) {
        Remove-Item -LiteralPath $Path -Force -ErrorAction Stop
    }

    $package = Open-ExcelPackage -Path $Path -Create -ErrorAction Stop
    try {
        foreach ($worksheetName in $Worksheets.Keys) {
            $sheet = $package.Workbook.Worksheets.Add($worksheetName)
            $headers = $Worksheets[$worksheetName]
            for ($column = 0; $column -lt $headers.Count; $column++) {
                $sheet.Cells[1, ($column + 1)].Value = $headers[$column]
            }
        }
        $package.Save()
    }
    finally {
        $package.Dispose()
    }
    Write-Output "Empty detail workbook saved to $Path"
}

function Invoke-ImpactIQModelDetailExtraction {
    [CmdletBinding()]
    param(
        [AllowEmptyCollection()][object[]]$Datasets,
        [Parameter(Mandatory)][string]$SourceFolderPath,
        [Parameter(Mandatory)][string]$OutputFolderPath,
        [Parameter(Mandatory)][string]$BaseFolderPath,
        [Parameter(Mandatory)][string]$TabularEditorPath,
        [Parameter(Mandatory)][string]$ModelDetailScriptPath,
        [Parameter(Mandatory)][string]$DependencyScriptPath,
        [string]$RunMode
    )

    $ErrorActionPreference = 'Stop'
    $jobs = [Collections.Generic.List[object]]::new()
    foreach ($dataset in $Datasets) {
        try {
            $workspaceName = ($dataset.WorkspaceName -replace '\[', '(' -replace '\]', ')' -replace '[^a-zA-Z0-9\(\)&,.-]', ' ').TrimStart()
            $datasetName = ($dataset.DatasetName -replace '\[', '(' -replace '\]', ')' -replace '[^a-zA-Z0-9\(\)&,.-]', ' ').TrimStart()
            $inputPath = Join-Path $SourceFolderPath "$workspaceName ~ $datasetName.bim"
            if ($RunMode -in @('Agents', 'FabricApps') -and -not (Test-Path -LiteralPath $inputPath)) {
                Write-Host "[WARN] Model backup unavailable for '$($dataset.DatasetName)'; skipping model detail." -ForegroundColor Yellow
                continue
            }
            # The C# scripts name their CSVs from the database, not the BIM filename.
            $databaseName = [string]((Get-Content -LiteralPath $inputPath -Raw -ErrorAction Stop | ConvertFrom-Json -ErrorAction Stop).name)
            if ([string]::IsNullOrWhiteSpace($databaseName)) { throw "Model backup '$inputPath' has no database name." }
            $jobs.Add([PSCustomObject]@{
                Name = "$workspaceName ~ $datasetName"
                Message = "**Extracting Model Detail and Measure Dependencies for $workspaceName ~ $datasetName**"
                Arguments = "`"$inputPath`" -S `"$ModelDetailScriptPath`" `"$DependencyScriptPath`""
                Outputs = @(
                    [IO.Path]::GetFullPath((Join-Path $OutputFolderPath ($databaseName + '.csv')))
                    [IO.Path]::GetFullPath((Join-Path $OutputFolderPath ($databaseName + '_MD.csv')))
                )
            })
        }
        catch {
            Write-ImpactIQContinuingError -Record $_ -Context "Model detail '$($dataset.DatasetName)'"
        }
    }

    Invoke-ImpactIQTabularEditorJobs -Jobs @($jobs.ToArray()) -TabularEditorPath $TabularEditorPath -WorkingDirectory $BaseFolderPath -Context 'Model detail'
}

# Function to Export Using REST API and Stored Token
function Export-ReportUsingAPI {
    param (
        [string]$GroupId,
        [string]$ReportId,
        [string]$OutFilePath,
        [string]$DownloadType
    )

    $accessToken = Get-CurrentAccessToken
    $apiUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/groups/$GroupId/reports/$ReportId/Export?downloadType=$DownloadType"

    try {
        Invoke-RestMethod -Uri $apiUrl `
            -Headers @{ Authorization = "Bearer $accessToken" } `
            -Method GET `
            -OutFile $OutFilePath
        # Treat as success only if a non-empty file was actually written
        if ((Test-Path $OutFilePath) -and ((Get-Item $OutFilePath).Length -gt 0)) {
            return $true
        }
        return $false
    } catch {
        return $false
    }
}

# Function to Export a Report's PBIR Definition via the Fabric Definition API,
# then zip the returned parts and rename the archive to .pbix.
# NOTE: A zipped PBIR definition renamed to .pbix is NOT an openable Power BI
# Desktop file by design - it is produced this way intentionally.
# Used ONLY as a fallback when the standard LiveConnect export fails.
# Written for Windows PowerShell 5.1 compatibility.
function Export-ReportDefinitionAsPbix {
    param (
        [string]$WorkspaceId,
        [string]$ReportId,
        [string]$OutFilePath
    )

    Add-Type -AssemblyName System.IO.Compression.FileSystem -ErrorAction SilentlyContinue

    $accessToken = Get-FabricAccessToken
    $headers = @{ Authorization = "Bearer $accessToken"; 'Content-Type' = 'application/json' }
    $defUrl  = "$($global:PowerBIEndpoints.FabricApiPrefix)/v1/workspaces/$WorkspaceId/reports/$ReportId/getDefinition"

    try {
        $resp = Invoke-WebRequest -Uri $defUrl -Method POST -Headers $headers -Body '{}' -UseBasicParsing -ErrorAction Stop
    } catch {
        Write-Warning "getDefinition failed for report $ReportId : $($_.Exception.Message)"
        return $false
    }

    $definition = $null
    $statusCode = [int]$resp.StatusCode

    if ($statusCode -eq 200) {
        $definition = $resp.Content | ConvertFrom-Json
    }
    elseif ($statusCode -eq 202) {
        # Long-running operation: poll the operation until it completes, then fetch the result
        $opUrl = $resp.Headers['Location']
        if ($opUrl -is [array]) { $opUrl = $opUrl[0] }
        $retry = 2
        if ($resp.Headers['Retry-After']) { $retry = [int]($resp.Headers['Retry-After'] | Select-Object -First 1) }

        while ($true) {
            Start-Sleep -Seconds $retry
            try {
                $op = Invoke-WebRequest -Uri $opUrl -Method GET -Headers $headers -UseBasicParsing -ErrorAction Stop
            } catch {
                Write-Warning "Polling definition op failed for report $ReportId : $($_.Exception.Message)"
                return $false
            }
            $status = ($op.Content | ConvertFrom-Json).status
            if ($status -eq 'Succeeded') { break }
            if ($status -eq 'Failed')    { Write-Warning "Definition operation failed for report $ReportId"; return $false }
            if ($op.Headers['Retry-After']) { $retry = [int]($op.Headers['Retry-After'] | Select-Object -First 1) }
        }

        $resultUrl = ($opUrl.TrimEnd('/')) + '/result'
        try {
            $result = Invoke-WebRequest -Uri $resultUrl -Method GET -Headers $headers -UseBasicParsing -ErrorAction Stop
        } catch {
            Write-Warning "Fetching definition result failed for report $ReportId : $($_.Exception.Message)"
            return $false
        }
        $definition = $result.Content | ConvertFrom-Json
    }
    else {
        Write-Warning "Unexpected status $statusCode from getDefinition for report $ReportId"
        return $false
    }

    if (-not $definition.definition.parts) {
        Write-Warning "No definition parts returned for report $ReportId"
        return $false
    }

    # Stage the parts into a temp folder.
    # NOTE: getDefinition returns part paths WITHOUT a "Report/" prefix
    # (e.g. "definition/report.json"), but the PBIX/PBIR layout expected by the
    # Report Detail extractor is "Report/definition/...". Nest everything under
    # a "Report" folder so the zipped .pbix matches that structure.
    $stageDir = Join-Path $env:TEMP ("ReportDef_" + $ReportId)
    if (Test-Path $stageDir) { Remove-Item $stageDir -Recurse -Force }
    New-Item -ItemType Directory -Path $stageDir -Force | Out-Null
    $reportRoot = Join-Path $stageDir 'Report'
    New-Item -ItemType Directory -Path $reportRoot -Force | Out-Null

    foreach ($part in $definition.definition.parts) {
        $partPath = Join-Path $reportRoot $part.path
        $partDir  = Split-Path $partPath -Parent
        if (-not (Test-Path $partDir)) { New-Item -ItemType Directory -Path $partDir -Force | Out-Null }
        if ($part.payloadType -eq 'InlineBase64') {
            [System.IO.File]::WriteAllBytes($partPath, [System.Convert]::FromBase64String($part.payload))
        } else {
            [System.IO.File]::WriteAllText($partPath, [string]$part.payload)
        }
    }

    # If getDefinition returned the legacy (non-PBIR) layout, there is NO
    # "definition" folder - instead a root-level "report.json" carries the
    # report layout. Rename it to "Layout" (no extension) AND re-encode it as
    # UTF-16 LE (no BOM) to match the classic PBIX layout the downstream
    # extractor expects. getDefinition returns report.json as UTF-8, but a real
    # PBIX "Report/Layout" is UTF-16 LE; without this conversion the parser
    # reads garbage and fails.
    $defFolder = Join-Path $reportRoot 'definition'
    if (-not (Test-Path $defFolder)) {
        $reportJson = Join-Path $reportRoot 'report.json'
        if (Test-Path $reportJson) {
            $layoutPath = Join-Path $reportRoot 'Layout'
            if (Test-Path $layoutPath) { Remove-Item $layoutPath -Force }
            $layoutText = [System.IO.File]::ReadAllText($reportJson, [System.Text.Encoding]::UTF8)
            $utf16NoBom = New-Object System.Text.UnicodeEncoding($false, $false)  # LE, no BOM
            [System.IO.File]::WriteAllText($layoutPath, $layoutText, $utf16NoBom)
            Remove-Item $reportJson -Force
        }
    }

    # getDefinition does NOT return a "Connections" file, but the downstream
    # Report Detail extractor expects one at the ROOT (alongside the "Report"
    # folder) to populate the Connections sheet + ModelID/ReportID. Synthesize
    # it from definition.pbir's datasetReference so the layout matches a PBIX.
    $pbirPath = Join-Path $reportRoot 'definition.pbir'
    if (Test-Path $pbirPath) {
        try {
            $pbir  = Get-Content -Path $pbirPath -Raw | ConvertFrom-Json
            $dsRef = $pbir.datasetReference
            $connString = $null
            $datasetId  = ''

            if ($dsRef.byConnection -and $dsRef.byConnection.connectionString) {
                $connString = $dsRef.byConnection.connectionString
                $m = [regex]::Match($connString, 'semanticmodelid=([0-9a-fA-F-]+)')
                if ($m.Success) { $datasetId = $m.Groups[1].Value }
            }
            elseif ($dsRef.byPath -and $dsRef.byPath.path) {
                $connString = 'byPath:' + $dsRef.byPath.path
            }

            if ($connString) {
                $connObj = [pscustomobject]@{
                    Version     = '3.0'
                    Connections = @(
                        [pscustomobject]@{
                            Name             = 'EntityDataSource'
                            ConnectionString = $connString
                            ConnectionType   = 'pbiServiceLive'
                        }
                    )
                    RemoteArtifacts = @(
                        [pscustomobject]@{
                            DatasetId = $datasetId
                            ReportId  = $ReportId
                        }
                    )
                }
                # Root-level file literally named "Connections" (no extension) to match PBIX
                $connJson = $connObj | ConvertTo-Json -Depth 6 -Compress
                [System.IO.File]::WriteAllText((Join-Path $stageDir 'Connections'), $connJson)
            }
        } catch {
            Write-Warning "Could not synthesize Connections file for report $ReportId : $($_.Exception.Message)"
        }
    }

    # Zip the staged folder, then rename the archive to .pbix
    if (Test-Path $OutFilePath) { Remove-Item $OutFilePath -Force }
    $tempZip = Join-Path $env:TEMP ("ReportDef_" + $ReportId + ".zip")
    if (Test-Path $tempZip) { Remove-Item $tempZip -Force }

    [System.IO.Compression.ZipFile]::CreateFromDirectory($stageDir, $tempZip)
    Move-Item -Path $tempZip -Destination $OutFilePath -Force

    Remove-Item $stageDir -Recurse -Force -ErrorAction SilentlyContinue
    return $true
}

	# Function: get first free drive letter (Z down to D)
	function Get-FreeDriveLetter {
	    $used = Get-PSDrive -PSProvider FileSystem | ForEach-Object { $_.Name }
	    foreach ($letter in [char[]](90..68)) {  # Z..Y..X.. down to D
	        if ($used -notcontains $letter) { return "${letter}:" }
	    }
	    throw "No free drive letters available."
	}

	foreach ($workspace in $workspacesInfo) {
	    $workspaceName = $workspace.WorkspaceName
	    $workspaceId = $workspace.WorkspaceId
	
	    # Clean up workspace name
	    $cleanWorkspaceName = $workspaceName -replace '\[', '(' -replace '\]', ')'
	    $cleanWorkspaceName = $cleanWorkspaceName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
	    $cleanWorkspaceName = $cleanWorkspaceName.TrimStart()
	
	    # Fetch reports from the existing list
	    $reports = $reportsInfo | Where-Object { $_.WorkspaceId -eq $workspaceId }
	
	    foreach ($report in $reports) {
	        $reportName = $report.ReportName
	        $reportId = $report.ReportId
	
	        # Clean up report name
	        $cleanReportName = $reportName -replace '\[', '(' -replace '\]', ')'
	        $cleanReportName = $cleanReportName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
	        $cleanReportName = $cleanReportName.TrimStart()
	
	        # Determine file extension
	        $fileExtension = if ($report.WebUrl -like "*/rdlreports/*") { "rdl" } else { "pbix" }
	        $filename = "$cleanWorkspaceName ~ $cleanReportName.$fileExtension"
	        $filepath = Join-Path -Path $newDateFolder -ChildPath $filename
	        $extractFolder = Join-Path -Path $tempExtractFolder -ChildPath "$cleanWorkspaceName ~ $cleanReportName"
	
	        # Delete file if it already exists
	        if (Test-Path $filepath) {
	            Remove-Item $filepath -Force
	        }
	
	        Write-Host "Exporting $cleanWorkspaceName ~ $cleanReportName"
	
	        # Export based on Workspace Type
	        if ($workspace.WorkspaceIsOnDedicatedCapacity -eq $false) {
	            # Pro Workspace → IncludeModel (real .pbix, used for model extraction below)
	            $includeModelExported = Export-ReportUsingAPI -GroupId $workspaceId -ReportId $reportId -OutFilePath $filepath -DownloadType "IncludeModel"
	            if (-not $includeModelExported) {
	                Write-Warning "IncludeModel export failed for $cleanWorkspaceName ~ $cleanReportName; falling back to getDefinition."
	                Export-ReportDefinitionAsPbix -WorkspaceId $workspaceId -ReportId $reportId -OutFilePath $filepath | Out-Null
	            }
	        } elseif ($fileExtension -eq 'pbix') {
	            # Non-Pro PBIX report (Premium / Fabric / etc.).
	            # If the report carries a sensitivity label (only visible via the Fabric
	            # List Items API), prefer getDefinition FIRST so the protected definition is
	            # captured directly, falling back to LiveConnect only if that fails.
	            # Otherwise keep the default LiveConnect-first behaviour with getDefinition as
	            # the fallback (a zipped PBIR definition renamed to .pbix - intentional).
	            if ($global:ReportsWithSensitivityLabel -and $global:ReportsWithSensitivityLabel.ContainsKey($reportId)) {
	                $defExported = Export-ReportDefinitionAsPbix -WorkspaceId $workspaceId -ReportId $reportId -OutFilePath $filepath
	                if (-not $defExported) {
	                    Write-Warning "getDefinition export failed for $cleanWorkspaceName ~ $cleanReportName; falling back to LiveConnect."
	                    $liveExported = Export-ReportUsingAPI -GroupId $workspaceId -ReportId $reportId -OutFilePath $filepath -DownloadType "LiveConnect"
	                    if (-not $liveExported) {
	                        Write-Warning "LiveConnect export also failed for $cleanWorkspaceName ~ $cleanReportName."
	                    }
	                }
	            } else {
	                # No sensitivity label → LiveConnect first, getDefinition fallback.
	                $liveExported = Export-ReportUsingAPI -GroupId $workspaceId -ReportId $reportId -OutFilePath $filepath -DownloadType "LiveConnect"
	                if (-not $liveExported) {
	                    Write-Warning "LiveConnect export failed for $cleanWorkspaceName ~ $cleanReportName; falling back to getDefinition."
	                    $defExported = Export-ReportDefinitionAsPbix -WorkspaceId $workspaceId -ReportId $reportId -OutFilePath $filepath
	                    if (-not $defExported) {
	                        Write-Warning "getDefinition export also failed for $cleanWorkspaceName ~ $cleanReportName."
	                    }
	                }
	            }
	        } else {
	            # Non-Pro paginated (RDL) report → LiveConnect export
	            $null = Export-ReportUsingAPI -GroupId $workspaceId -ReportId $reportId -OutFilePath $filepath -DownloadType "LiveConnect"
	        }
			
		# Only process model extraction for Pro Workspaces
		if ($workspace.WorkspaceIsOnDedicatedCapacity -eq $false) {
		    Write-Host "Extracting $cleanWorkspaceName ~ $cleanReportName"
		
		    $driveLetter = Get-FreeDriveLetter
		
		    # ✅ Make sure the extract folder exists before subst
		    if (-not (Test-Path $extractFolder)) {
		        New-Item -ItemType Directory -Path $extractFolder -Force | Out-Null
		    }
		
		    if (Test-Path $driveLetter) { Invoke-ImpactIQNativeCommand -FilePath 'subst.exe' -ArgumentList @($driveLetter, '/D') -DiscardOutput }
		
		    Invoke-ImpactIQNativeCommand -FilePath 'subst.exe' -ArgumentList @($driveLetter, $extractFolder) -DiscardOutput
		
		    if (-not (Test-Path $driveLetter)) {
		        Write-Error "Failed to map $extractFolder to $driveLetter"
		        continue
		    }
		
		    try {
		        # Extract into subst drive
				Write-Host "If available, extracting Model from $cleanWorkspaceName ~ $cleanReportName"
		        Invoke-ImpactIQNativeCommand -FilePath $pbiToolsPath -ArgumentList @('extract', $filepath, '-extractFolder', $driveLetter, '-modelSerialization', 'Raw') -DiscardOutput
		
		        Invoke-ImpactIQNativeCommand -FilePath $pbiToolsPath -ArgumentList @('generate-bim', $driveLetter, '-transforms', 'RemovePBIDataSourceVersion') -DiscardOutput
		
		        $bimFilesGenerated = Get-ChildItem -Path $extractFolder -Filter *.bim
		        foreach ($bimFile in $bimFilesGenerated) {
		            $newBimName = "$cleanWorkspaceName ~ $cleanReportName.bim"
		            $newBimPath = Join-Path -Path $tempExtractFolder -ChildPath $newBimName
		
		            Move-Item -Path $bimFile.FullName -Destination $newBimPath -Force
		        }
		    }
		    finally {
		        # ✅ Always clean up subst drive, even if extract/generate fails
		        Invoke-ImpactIQNativeCommand -FilePath 'subst.exe' -ArgumentList @($driveLetter, '/D') -DiscardOutput
		    }
		}
	}

function Export-FabricDataflow {
    param (
        [string]$WorkspaceId,
        [string]$DataflowId,
        [string]$WorkspaceName,
        [string]$DataflowName,
        [string]$OutputPath,
        [string]$AccessToken
    )
    
    # Clean up names
    $cleanWorkspaceName = $WorkspaceName -replace '\[', '(' -replace '\]', ')'
    $cleanWorkspaceName = $cleanWorkspaceName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
    $cleanWorkspaceName = $cleanWorkspaceName.TrimStart()
    
    $cleanDataflowName = $DataflowName -replace '\[', '(' -replace '\]', ')'
    $cleanDataflowName = $cleanDataflowName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
    $cleanDataflowName = $cleanDataflowName.TrimStart()
    
    $fabricApiBaseUri = "$($global:PowerBIEndpoints.FabricApiPrefix)/v1/workspaces"
    $endpoint = "$fabricApiBaseUri/$WorkspaceId/dataflows/$DataflowId/getDefinition"
    
    $headers = @{
        "Authorization" = "Bearer $AccessToken"
        "Content-Type"  = "application/json"
    }
    
    try {
        $response = Invoke-RestMethod -Uri $endpoint -Method POST -Headers $headers -Body "{}"
        
        if (-not $response.definition.parts) {
            return $null
        }
        
        # Create temp directory for extraction
        $tempDir = Join-Path $env:TEMP "FabricDataflow_$DataflowId"
        if (Test-Path $tempDir) {
            Remove-Item -Path $tempDir -Recurse -Force
        }
        New-Item -Path $tempDir -ItemType Directory -Force | Out-Null
        
        # Extract files
        foreach ($part in $response.definition.parts) {
            $fileName = $part.path
            $filePath = Join-Path $tempDir $fileName
            $payloadType = $part.payloadType
            $payload = $part.payload
            
            if ($payloadType -eq "InlineBase64") {
                $bytes = [System.Convert]::FromBase64String($payload)
                $fileDir = Split-Path $filePath
                if (-not (Test-Path $fileDir)) {
                    New-Item -ItemType Directory -Path $fileDir -Force | Out-Null
                }
                [System.IO.File]::WriteAllBytes($filePath, $bytes)
            }
        }
        
        # Find the .pq file
        $pqFile = Get-ChildItem -Path $tempDir -Filter "*.pq" -Recurse | Select-Object -First 1
        
        if ($pqFile) {
            # Read the .pq file content
            $pqContent = Get-Content -Path $pqFile.FullName -Raw -Encoding UTF8
            
            # Save as .pqt
            $outputFileName = "$cleanWorkspaceName ~ $cleanDataflowName.pq"
            $outputFilePath = Join-Path $OutputPath $outputFileName
            $pqContent | Out-File -FilePath $outputFilePath -Encoding UTF8
                        
            # Clean up temp directory
            Remove-Item -Path $tempDir -Recurse -Force
            
            return @{
                Content = $pqContent
                FilePath = $outputFilePath
            }
        } else {
            Remove-Item -Path $tempDir -Recurse -Force
            return $null
        }
        
    } catch {
        return $null
    }
}

# Function to parse .pq content and extract queries
function Parse-FabricDataflowContent {
    param (
        [string]$Content,
        [string]$DataflowId,
        [string]$DataflowName,
        [string]$WorkspaceName,
        [datetime]$ReportDate
    )
    
    $queries = @()
    
    # Clean up names
    $cleanWorkspaceName = $WorkspaceName -replace '\[', '(' -replace '\]', ')'
    $cleanWorkspaceName = $cleanWorkspaceName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
    $cleanWorkspaceName = $cleanWorkspaceName.TrimStart()
    
    $cleanDataflowName = $DataflowName -replace '\[', '(' -replace '\]', ')'
    $cleanDataflowName = $cleanDataflowName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
    $cleanDataflowName = $cleanDataflowName.TrimStart()
    
    # Split content by "section Section1;" to get the queries section
    $sections = $Content -split 'section Section1;', 2
    
    if ($sections.Count -gt 1) {
        $queriesSection = $sections[1]
        
        # Use regex to find all queries:
        # Supports both shared QueryName = ... and shared #"Query Name With Spaces" = ...
        $pattern = '(?s)(?:\[[^\]]*\]\s*)?shared\s+(?:#"(.*?)"|([A-Za-z_]\w*))\s*=\s*(.*?)(?=(?:\[[^\]]*\]\s*)?shared\s+(?:#"(?:.*?)"|[A-Za-z_]\w*)\s*=|$)'
        $matches = [regex]::Matches($queriesSection, $pattern)
        
        foreach ($match in $matches) {
            # Group 1 = hash-quoted name, Group 2 = unquoted name, Group 3 = expression
            if ($match.Groups[1].Success) {
                $queryName = $match.Groups[1].Value
            } else {
                $queryName = $match.Groups[2].Value
            }
            $queryExpression = $match.Groups[3].Value.Trim()

            # Remove trailing semicolons
            $queryExpression = $queryExpression -replace ';\s*$', ''
            $queryExpression = $queryExpression.Trim()

            # Skip if empty or just whitespace
            if ([string]::IsNullOrWhiteSpace($queryName) -or [string]::IsNullOrWhiteSpace($queryExpression)) {
                continue
            }

            $queries += [PSCustomObject]@{
                "Dataflow ID" = $DataflowId
                "Dataflow Name" = $DataflowName
                "Query Name" = $queryName
                "Query" = $queryExpression
                "Report Date" = $ReportDate
                "Workspace Name - Dataflow Name" = "$cleanWorkspaceName ~ $cleanDataflowName"
            }
        }
    }
    
    return $queries
}

    $workspaceName = $workspace.WorkspaceName
    $workspaceId = $workspace.WorkspaceId

    # Set the Power BI REST API URL for the dataflow details
    $dataflowDetailsUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/groups/$workspaceId/dataflows"

    # Get the list of dataflows in the workspace
    $dataflowsResponse = Invoke-ImpactIQRestMethod -Url $dataflowDetailsUrl -Method Get

    # Parse the JSON response
    $dataflows = $dataflowsResponse | ConvertFrom-Json

    # Check if the response is valid and contains dataflows
    if ($dataflows -and $dataflows.value) {
        
        # Iterate through the Gen1 dataflows
        foreach ($dataflow in $dataflows.value) {
            $dataflowId = $dataflow.objectId
            $dataflowName = $dataflow.name

            # Clean up workspace name
            $cleanWorkspaceName = $workspaceName -replace '\[', '(' -replace '\]', ')'
            $cleanWorkspaceName = $cleanWorkspaceName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
            $cleanWorkspaceName = $cleanWorkspaceName.TrimStart()

            # Clean up dataflow name
            $cleanDataFlowName = $dataflowName -replace '\[', '(' -replace '\]', ')'
            $cleanDataFlowName = $cleanDataFlowName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
            $cleanDataFlowName = $cleanDataFlowName.TrimStart()
            
            # Define output file path specific to the dataflow (Gen1)
            $dataflowOutputFilePath = Join-Path -Path $baseOutputFilePath -ChildPath "$cleanWorkspaceName ~ $cleanDataFlowName.txt"
            
            # Set the Power BI REST API URL for the specific dataflow
            $apiUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/groups/$workspaceId/dataflows/$dataflowId"
            
            # Get the dataflow
            $response = Invoke-ImpactIQRestMethod -Url $apiUrl -Method Get
            
            # Save the original response to text file for backup
            $response | Out-File -FilePath $dataflowOutputFilePath -Encoding UTF8
            
            # Parse the JSON response to extract document content
            try {
                $dataflowJson = $response | ConvertFrom-Json
                
                # Check if the JSON has the expected structure
                if (-not $dataflowJson.'pbi:mashup' -or -not $dataflowJson.'pbi:mashup'.document) {
                    continue
                }
                
                # Extract the document content (Power Query code)
                $documentContent = $dataflowJson.'pbi:mashup'.document
                
                # The document contains escaped content - unescape it
                # Replace escaped newlines and quotes
                $documentContent = $documentContent -replace '\\r\\n', "`n" -replace '\\n', "`n"
                $documentContent = $documentContent -replace '\\"', '"'
                
                # Split by "section Section1;" to get the queries section
                $sections = $documentContent -split 'section Section1;', 2
                
                if ($sections.Count -lt 2) {
                    continue
                }
                
                $queriesSection = $sections[1]
                
                # Use regex to extract queries (similar to Gen2 parsing for consistency)
                # Supports both shared QueryName = ... and shared #"Query Name With Spaces" = ...
                $queryPattern = '(?s)(?:\[[^\]]*\]\s*)?shared\s+(?:#"(.*?)"|([A-Za-z_]\w*))\s*=\s*(.*?)(?=(?:\[[^\]]*\]\s*)?shared\s+(?:#"(?:.*?)"|[A-Za-z_]\w*)\s*=|$)'
                $queryMatches = [regex]::Matches($queriesSection, $queryPattern)
                
                $data = @()
                foreach ($match in $queryMatches) {
                    # Group 1 = hash-quoted name, Group 2 = unquoted name, Group 3 = expression
                    if ($match.Groups[1].Success) {
                        $queryName = $match.Groups[1].Value
                    } else {
                        $queryName = $match.Groups[2].Value
                    }
                    $queryExpression = $match.Groups[3].Value.Trim()
                    
                    # Remove trailing semicolon if present
                    $queryExpression = $queryExpression -replace ';\s*$', ''
                    $queryExpression = $queryExpression.Trim()
                    
                    if ([string]::IsNullOrWhiteSpace($queryName) -or [string]::IsNullOrWhiteSpace($queryExpression)) {
                        continue
                    }
                    
                    $data += [PSCustomObject]@{
                        "Dataflow ID" = $dataflowId
                        "Dataflow Name" = $dataflowName
                        "Query Name" = $queryName
                        "Query" = $queryExpression
                        "Report Date" = $currentDate
                        "Workspace Name - Dataflow Name" = "$cleanWorkspaceName ~ $cleanDataFlowName"
                    }
                }
                
                # Fill the combined DataTable with data
                foreach ($item in $data) {
                    $row = $combinedDataTable.NewRow()
                    $row["Dataflow ID"] = $item."Dataflow ID"
                    $row["Dataflow Name"] = $item."Dataflow Name"
                    $row["Query Name"] = $item."Query Name"
                    $row["Query"] = $item.Query
                    $row["Report Date" ] = $item."Report Date"
                    $row["Workspace Name - Dataflow Name"] = $item."Workspace Name - Dataflow Name"
                    $combinedDataTable.Rows.Add($row)
                }
                
            } catch {
                continue
            }
        }
    } else {
    }
}


# Loop through all workspaces to fetch Gen2 dataflows (Fabric Items API)
foreach ($workspace in $workspacesInfo) {
    $workspaceName = $workspace.WorkspaceName
    $workspaceId = $workspace.WorkspaceId
    
    # Set the Fabric API URL for items
    $fabricItemsUrl = "$($global:PowerBIEndpoints.FabricApiPrefix)/v1/workspaces/$workspaceId/items"
    
    try {
        # Get the list of items in the workspace
        $fabricItemsResponse = Invoke-ImpactIQRestMethod -Method GET -Url $fabricItemsUrl | ConvertFrom-Json
        
        # Filter for Dataflow items (Gen2 dataflows)
        $gen2Dataflows = $fabricItemsResponse.value | Where-Object { $_.type -eq 'Dataflow' }
        
        if ($gen2Dataflows -and $gen2Dataflows.Count -gt 0) {
            
            # Iterate through the Gen2 dataflows
            foreach ($dataflow in $gen2Dataflows) {
                $dataflowId = $dataflow.id
                $dataflowName = $dataflow.displayName
                
                
                # Get current access token
                $currentAccessToken = Get-FabricAccessToken
                
                # Download and extract Fabric dataflow
                $fabricResult = Export-FabricDataflow -WorkspaceId $workspaceId `
                                                      -DataflowId $dataflowId `
                                                      -WorkspaceName $workspaceName `
                                                      -DataflowName $dataflowName `
                                                      -OutputPath $baseOutputFilePath `
                                                      -AccessToken $currentAccessToken
                
                if ($fabricResult) {
                    # Parse the .pq content
                    $parsedQueries = Parse-FabricDataflowContent -Content $fabricResult.Content `
                                                                  -DataflowId $dataflowId `
                                                                  -DataflowName $dataflowName `
                                                                  -WorkspaceName $workspaceName `
                                                                  -ReportDate $currentDate
                    
                    # Add parsed queries to combined DataTable
                    foreach ($query in $parsedQueries) {
                        $row = $combinedDataTable.NewRow()
                        $row["Dataflow ID"] = $query."Dataflow ID"
                        $row["Dataflow Name"] = $query."Dataflow Name"
                        $row["Query Name"] = $query."Query Name"
                        $row["Query"] = $query.Query
                        $row["Report Date"] = $query."Report Date"
                        $row["Workspace Name - Dataflow Name"] = $query."Workspace Name - Dataflow Name"
                        $combinedDataTable.Rows.Add($row)
                    }
                }
            }
        } else {
        }
    } catch {
    }
}

# Check if the combined DataTable has any rows, if not add a dummy row with headers only
if ($combinedDataTable.Rows.Count -eq 0) {
    $row = $combinedDataTable.NewRow()
    foreach ($header in $headers) {
        $row[$header] = ""
    }
    $combinedDataTable.Rows.Add($row)
}



# Export the combined DataTable to an Excel file
$combinedDataTable | Export-Excel -Path $combinedExcelOutputPath -AutoSize
Write-Host "Data exported to $combinedExcelOutputPath"

# Copy the Excel file to the base folder for easy access
$fileName = "Dataflow Detail.xlsx"
$sourceFilePath = Join-Path -Path $dataflow_new_date_folder -ChildPath $fileName
$destinationFilePath = Join-Path -Path $baseFolderPath -ChildPath $fileName

# Check if the source file exists
if (Test-Path -Path $sourceFilePath) {
    # Remove the destination file if it already exists
    if (Test-Path -Path $destinationFilePath) {
        Remove-Item -Path $destinationFilePath -Force
    }
    
    # Copy the source file to the destination
    Copy-Item -Path $sourceFilePath -Destination $destinationFilePath
} else {
    Write-Warning "Source file not found: $sourceFilePath"
}

Write-Output "Dataflow Backup and Detail Extraction Process Completed. Excel output saved to $combinedExcelOutputPath"

}
else {
    $dataflowDetailWorksheets = [ordered]@{
        Sheet1 = @('Dataflow ID', 'Dataflow Name', 'Query Name', 'Query', 'Report Date', 'Workspace Name - Dataflow Name')
    }
    Export-EmptyDetailWorkbook -Path (Join-Path $baseFolderPath 'Dataflow Detail.xlsx') -Worksheets $dataflowDetailWorksheets -ErrorAction Stop
}

function Get-DataAgentDetailWorksheetHeaders {
    [ordered]@{
        'Data Agents' = @('Data Agent ID', 'Data Agent Name', 'Description', 'Workspace ID', 'Workspace Name', 'Schema Version', 'AI Instructions', 'Connected Source Count', 'Definition Retrieved', 'Sensitivity Label ID', 'Report Date', 'Workspace Name - Data Agent Name')
        'Data Agent Sources' = @('Data Agent ID', 'Data Agent Name', 'Workspace ID', 'Workspace Name', 'Stage', 'Source Type', 'Source Type (Raw)', 'Source Display Name', 'Connected Item ID', 'Connected Item Name', 'Source Workspace ID', 'Source Workspace Name', 'Source Instructions', 'Source Description', 'Selected Top-Level Elements', 'Report Date', 'Workspace Name - Data Agent Name')
    }
}

function Export-FabricDataAgentDetail {
    param(
        [string]$BaseFolderPath,
        [string]$ReportDate,
        [AllowEmptyCollection()][object[]]$Workspaces,
        [hashtable]$DatasetNames
    )

    $ErrorActionPreference = 'Stop'
    Write-Output "Data Agent Detail Extraction Process Started"
    $agentsInfo = @()
    $sourcesInfo = @()
    $definitionToken = if ($global:RunMode -eq 'Agents') { $script:DataAgentDefinitionToken } else { $null }
    $tokenAttempted = [bool]$definitionToken
    $sourceTypes = @{
        semantic_model = "Semantic Model"
        lakehouse = "Lakehouse"
        lakehouse_tables = "Lakehouse (Tables)"
        data_warehouse = "Data Warehouse"
        kusto = "KQL Database"
        graph = "Graph (GQL)"
        mirrored_database = "Mirrored Database"
        mirrored_azure_databricks = "Mirrored Azure Databricks"
        unknown = "Unknown"
    }

    $inventory = @()
    if ($global:RunMode -ne 'Agents') {
        $inventoryWorkspaces = @($Workspaces | Where-Object { $_.WorkspaceId -ne "My Workspace" } |
            ForEach-Object { [PSCustomObject]@{ id = $_.WorkspaceId; name = $_.WorkspaceName } })
        $inventory = @(Get-FabricDataAgentInventory -Workspaces $inventoryWorkspaces)
    }

    foreach ($workspace in $Workspaces) {
        $workspaceId = $workspace.WorkspaceId
        $workspaceName = $workspace.WorkspaceName
        if ($workspaceId -eq "My Workspace") { continue }
        $agents = @()
        if ($global:RunMode -eq 'Agents') {
            $agents = @($script:SelectedDataAgents.Keys | Where-Object { $_.StartsWith("$workspaceId|") } |
                ForEach-Object { $script:SelectedDataAgents[$_] })
        }
        else {
            $agents = @($inventory | Where-Object { $_.WorkspaceId -eq $workspaceId } | ForEach-Object { $_.Agent })
        }

        foreach ($agent in $agents) {
            if (-not $tokenAttempted) {
                $tokenAttempted = $true
                try { $definitionToken = Get-FabricDataAgentToken }
                catch {
                    Write-Host "[WARN] Fabric authentication failed for Data Agent definitions: $($_.Exception.Message). Agents will still be listed." -ForegroundColor Yellow
                }
            }
            $definition = $null
            $agentKey = "$workspaceId|$($agent.id)"
            if ($global:RunMode -eq 'Agents' -and $script:DataAgentDefinitions.ContainsKey($agentKey)) {
                $definition = $script:DataAgentDefinitions[$agentKey]
            }
            elseif ($definitionToken) {
                $definition = Get-FabricDataAgentDefinition -WorkspaceId $workspaceId -DataAgentId $agent.id -AccessToken $definitionToken
            }
            $cleanWorkspaceName = ($workspaceName -replace '\[', '(' -replace '\]', ')' -replace '[^a-zA-Z0-9\(\)&,.-]', ' ').TrimStart()
            $cleanAgentName = ($agent.displayName -replace '\[', '(' -replace '\]', ')' -replace '[^a-zA-Z0-9\(\)&,.-]', ' ').TrimStart()
            $combinedName = "$cleanWorkspaceName ~ $cleanAgentName"
            $schemaVersion = ""
            $aiInstructions = ""
            $sourceCount = 0
            $definitionRetrieved = $false

            if ($definition) {
                $definitionRetrieved = $true
                foreach ($part in $definition.parts) {
                    try {
                        if ($part.payloadType -ne 'InlineBase64') {
                            throw "Unsupported definition payload type '$($part.payloadType)'."
                        }
                        $relativePath = $part.path -replace '\\', '/'
                        if ($relativePath -notmatch '^Files/Config/(data_agent\.json|[^/]+/stage_config\.json|[^/]+/(?:.*/)?datasource\.json)$') {
                            continue
                        }
                        $partJson = [Text.Encoding]::UTF8.GetString([Convert]::FromBase64String($part.payload)) | ConvertFrom-Json
                        if ($relativePath -eq 'Files/Config/data_agent.json') {
                            $schemaVersion = $partJson.'$schema'
                        }
                        elseif ($relativePath -eq 'Files/Config/draft/stage_config.json') {
                            $aiInstructions = $partJson.aiInstructions
                        }
                        elseif ($relativePath.EndsWith('/datasource.json')) {
                            $stage = $relativePath.Split('/')[2]
                            $sourceType = [string]$partJson.type
                            $friendlyType = if ($sourceTypes.ContainsKey($sourceType)) { $sourceTypes[$sourceType] } else { $sourceType }
                            $resolvedName = $partJson.displayName
                            if ($partJson.artifactId -and $DatasetNames.ContainsKey($partJson.artifactId)) {
                                $resolvedName = $DatasetNames[$partJson.artifactId]
                            }
                            $sourceWorkspaceName = ($Workspaces | Where-Object { $_.WorkspaceId -eq $partJson.workspaceId } | Select-Object -First 1).WorkspaceName
                            if (-not $sourceWorkspaceName) { $sourceWorkspaceName = "Unknown Workspace" }
                            $sourcesInfo += [PSCustomObject][ordered]@{
                                "Data Agent ID" = $agent.id
                                "Data Agent Name" = $agent.displayName
                                "Workspace ID" = $workspaceId
                                "Workspace Name" = $workspaceName
                                "Stage" = $stage
                                "Source Type" = $friendlyType
                                "Source Type (Raw)" = $sourceType
                                "Source Display Name" = $partJson.displayName
                                "Connected Item ID" = $partJson.artifactId
                                "Connected Item Name" = $resolvedName
                                "Source Workspace ID" = $partJson.workspaceId
                                "Source Workspace Name" = $sourceWorkspaceName
                                "Source Instructions" = $partJson.dataSourceInstructions
                                "Source Description" = $partJson.userDescription
                                "Selected Top-Level Elements" = @($partJson.elements | Where-Object { $null -ne $_ }).Count
                                "Report Date" = $ReportDate
                                "Workspace Name - Data Agent Name" = $combinedName
                            }
                            if ($stage -eq 'draft') { $sourceCount++ }
                        }
                    }
                    catch {
                        $definitionRetrieved = $false
                        Write-Host "[WARN] Data Agent '$($agent.displayName)' part '$($part.path)' could not be extracted: $($_.Exception.Message)" -ForegroundColor Yellow
                    }
                }
            }

            $agentsInfo += [PSCustomObject][ordered]@{
                "Data Agent ID" = $agent.id
                "Data Agent Name" = $agent.displayName
                "Description" = $agent.description
                "Workspace ID" = $workspaceId
                "Workspace Name" = $workspaceName
                "Schema Version" = $schemaVersion
                "AI Instructions" = $aiInstructions
                "Connected Source Count" = $sourceCount
                "Definition Retrieved" = $definitionRetrieved
                "Sensitivity Label ID" = $agent.sensitivityLabel.id
                "Report Date" = $ReportDate
                "Workspace Name - Data Agent Name" = $combinedName
            }
        }
    }

    Write-Host "[INFO] Data Agent extraction complete. Agents found: $($agentsInfo.Count); connected sources: $($sourcesInfo.Count)." -ForegroundColor Cyan
    $worksheetHeaders = Get-DataAgentDetailWorksheetHeaders
    if ($agentsInfo.Count -eq 0) {
        $emptyAgent = [ordered]@{}
        foreach ($header in $worksheetHeaders['Data Agents']) { $emptyAgent[$header] = '' }
        $agentsInfo += [PSCustomObject]$emptyAgent
    }
    if ($sourcesInfo.Count -eq 0) {
        $emptySource = [ordered]@{}
        foreach ($header in $worksheetHeaders['Data Agent Sources']) { $emptySource[$header] = '' }
        $sourcesInfo += [PSCustomObject]$emptySource
    }
    $outputPath = Join-Path $BaseFolderPath "Data Agent Detail.xlsx"
    if (Test-Path -LiteralPath $outputPath) { Remove-Item -LiteralPath $outputPath -Force }
    $agentsInfo | Export-Excel -Path $outputPath -WorksheetName "Data Agents" -AutoSize -ErrorAction Stop
    $sourcesInfo | Export-Excel -Path $outputPath -WorksheetName "Data Agent Sources" -AutoSize -Append -ErrorAction Stop
    Write-Output "Data Agent Detail Extraction Process Completed. Excel output saved to $outputPath"
}
