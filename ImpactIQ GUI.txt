# --- Remembered settings (the backup folder) ---
function Get-GuiSettingsPath {
    if ($env:IMPACTIQ_SETTINGS_PATH) { return $env:IMPACTIQ_SETTINGS_PATH }
    return Join-Path $env:LOCALAPPDATA 'ImpactIQ\settings.json'
}

function Read-GuiSettings {
    try {
        $path = Get-GuiSettingsPath
        if (Test-Path -LiteralPath $path -PathType Leaf) { return (Get-Content -LiteralPath $path -Raw | ConvertFrom-Json) }
    }
    catch { }
    return $null
}

function Save-GuiSettings {
    param([string]$BackupFolder)
    try {
        $path = Get-GuiSettingsPath
        New-Item -ItemType Directory -Path (Split-Path -Parent $path) -Force | Out-Null
        @{ BackupFolder = $BackupFolder } | ConvertTo-Json | Set-Content -LiteralPath $path -Encoding UTF8
    }
    catch { }
}

# --- What is happening: friendly wording and stages taken from the script's own progress lines ---
function Get-GuiActivity {
    param([string]$Line)
    $text = $Line.Trim()
    if ($text -match '^\[STEP\] (?<step>.+)$') { return $Matches['step'] }
    if ($text -match '^\*\*Exporting (?<model>.+)\*\*$') { return "Backing up model: $($Matches['model'])" }
    if ($text -match '^Workspace \[(?<ws>.+?)\] (?<phase>.+?) extraction process started') { return "Reading $($Matches['phase']) - $($Matches['ws'])" }
    switch -Regex ($text) {
        '^Model Backup Process Started' { return 'Backing up models' }
        '^Report Backup Process Started' { return 'Backing up reports' }
        '^Report Detail Extraction Process Started' { return 'Analyzing report visuals' }
        '^Model Detail Extraction Process Started' { return 'Analyzing models' }
        '^Measure Dependency Detail Extraction Process Started' { return 'Analyzing measure dependencies' }
        '^Dataflow Backup and Detail Extraction Process Started' { return 'Backing up dataflows' }
        '^Data Agent Detail Extraction Process Started' { return 'Reading data agents' }
        '\[INFO\] Fabric App query extraction started' { return 'Reading Fabric apps' }
        '^Extraction processing finished' { return 'Wrapping up' }
    }
    return $null
}

function Get-GuiStages {
    return @(
        @{ Name = 'Set up'; Pattern = '\[STEP\] (Preparing PowerShell|Checking Tabular Editor|Waiting for selection: Choose your cloud)' },
        @{ Name = 'Sign in'; Pattern = '\[STEP\] (Sign in to Microsoft|Waiting for selection: Choose your Fabric organization)' },
        @{ Name = 'Choose scope'; Pattern = '\[STEP\] (Loading your workspaces|Waiting for selection: (?!Choose your cloud|Choose your Fabric))' },
        @{ Name = 'Collect'; Pattern = '\[STEP\] Extracting Power BI and Fabric details' },
        @{ Name = 'Back up'; Pattern = 'Model Backup Process Started|Report Backup Process Started' },
        @{ Name = 'Analyze'; Pattern = 'Report Detail Extraction Process Started|Model Detail Extraction Process Started|Measure Dependency Detail Extraction Process Started|Dataflow Backup and Detail Extraction Process Started|Data Agent Detail Extraction Process Started|Fabric App query extraction started|Extraction processing finished' }
    )
}

# Stages only move forward, so a later repeat of an earlier marker cannot send the tracker backwards.
function Get-GuiStageIndex {
    param([string]$Text, [int]$Current)
    $stages = Get-GuiStages
    $best = $Current
    for ($i = $Current + 1; $i -lt $stages.Count; $i++) {
        if ($Text -match $stages[$i].Pattern) { $best = $i }
    }
    return $best
}

function New-GuiBrush {
    param([string]$Color)
    return [Windows.Media.BrushConverter]::new().ConvertFromString($Color)
}

function Show-GuiStages {
    $stages = Get-GuiStages
    $controls.StageStrip.Children.Clear()
    for ($i = 0; $i -lt $stages.Count; $i++) {
        $chip = New-Object Windows.Controls.Border
        $chip.CornerRadius = New-Object Windows.CornerRadius 12
        $chip.Padding = New-Object Windows.Thickness 10, 4, 10, 4
        $chip.Margin = New-Object Windows.Thickness 0, 0, 8, 8
        $label = New-Object Windows.Controls.TextBlock
        $label.FontSize = 12
        if ($i -lt $state.Stage) {
            $chip.Background = New-GuiBrush '#0F4D47'; $label.Foreground = New-GuiBrush '#9BE7DD'
            $label.Text = ([string][char]0x2713) + ' ' + $stages[$i].Name
        }
        elseif ($i -eq $state.Stage) {
            $chip.Background = New-GuiBrush '#087F75'; $label.Foreground = New-GuiBrush '#FFFFFF'
            $label.FontWeight = [Windows.FontWeights]::SemiBold; $label.Text = $stages[$i].Name
        }
        else {
            $chip.Background = New-GuiBrush '#1A2639'; $label.Foreground = New-GuiBrush '#6F819C'; $label.Text = $stages[$i].Name
        }
        $chip.Child = $label
        [void]$controls.StageStrip.Children.Add($chip)
    }
}

# --- Finished summary: counts of what was backed up today ---
function Get-GuiRunSummary {
    param([string]$Folder, [int]$Workspaces = 0, [int]$Apps = 0, [int]$Agents = 0, [string]$Date = (Get-Date -Format 'yyyy-MM-dd'))
    $parts = @()
    if ($Workspaces -gt 0) { $parts += '{0} workspace{1}' -f $Workspaces, $(if ($Workspaces -eq 1) { '' } else { 's' }) }
    $models = @(Get-ChildItem -LiteralPath (Join-Path $Folder "Model Backups\$Date") -Filter '*.bim' -File -ErrorAction SilentlyContinue).Count
    $reports = @(Get-ChildItem -LiteralPath (Join-Path $Folder "Report Backups\$Date") -File -ErrorAction SilentlyContinue).Count
    if ($models -gt 0) { $parts += '{0} model{1}' -f $models, $(if ($models -eq 1) { '' } else { 's' }) }
    if ($reports -gt 0) { $parts += '{0} report{1}' -f $reports, $(if ($reports -eq 1) { '' } else { 's' }) }
    if ($Apps -gt 0) { $parts += '{0} Fabric app{1}' -f $Apps, $(if ($Apps -eq 1) { '' } else { 's' }) }
    if ($Agents -gt 0) { $parts += '{0} data agent{1}' -f $Agents, $(if ($Agents -eq 1) { '' } else { 's' }) }
    $saved = "Backups saved to $Folder"
    if ($parts.Count -eq 0) { return $saved }
    return (($parts -join '  -  ') + "`n" + $saved)
}

# --- Attention: flash the taskbar when the window is in the background and the user is needed (or the run is done) ---
function Invoke-GuiAttention {
    param([switch]$Sound)
    try {
        if ($window.IsActive) { return }
        $handle = (New-Object Windows.Interop.WindowInteropHelper $window).Handle
        if ($handle -eq [IntPtr]::Zero) { return }
        $info = New-Object ImpactIQ.NativeWindows+FLASHWINFO
        $info.cbSize = [uint32][Runtime.InteropServices.Marshal]::SizeOf($info)
        $info.hwnd = $handle
        $info.dwFlags = 15
        $info.uCount = 0
        $info.dwTimeout = 0
        [void][ImpactIQ.NativeWindows]::FlashWindowEx([ref]$info)
        if ($Sound) { [Media.SystemSounds]::Asterisk.Play() }
    }

}

# --- One window at a time: a second copy would run against the same backup folder ---
function Enter-GuiSingleInstance {
    param([string]$Name = 'Local\ImpactIQ.Gui')
    $created = $false
    $mutex = New-Object Threading.Mutex($true, $Name, [ref]$created)
    if (-not $created) { $mutex.Dispose(); return $false }
    $script:GuiMutex = $mutex
    return $true
}

function Show-LauncherError {
    param([string]$Message)

function Get-BackupFolder {
    $folder = $controls.Folder.Text.Trim()
    if ([string]::IsNullOrWhiteSpace($folder) -or -not [IO.Path]::IsPathRooted($folder)) {
        throw 'Choose an absolute backup folder path.'
    }
    $folder = [IO.Path]::GetFullPath($folder)
    if ($folder.Contains('"') -or $folder.Contains("`r") -or $folder.Contains("`n")) {
        throw 'The backup path contains invalid characters.'
    }
    return $folder
}

function ConvertTo-LauncherLiteral {
    param([string]$Value)
    return "'" + $Value.Replace("'", "''") + "'"
}

# The Microsoft sign-in window opens under the GUI (or another app) rather than in front, so the GUI raises new ones.
function Get-SignInWindowHandles {
    param([string]$ClassPattern = '^ApplicationFrameWindow$')
    $found = New-Object Collections.ArrayList
    $callback = [ImpactIQ.NativeWindows+EnumWindowsProc]{
        param($handle, $parameter)
        if ([ImpactIQ.NativeWindows]::IsWindowVisible($handle)) {
            $cloaked = 0
            [void][ImpactIQ.NativeWindows]::DwmGetWindowAttribute($handle, 14, [ref]$cloaked, 4)
            $className = New-Object Text.StringBuilder 128
            [void][ImpactIQ.NativeWindows]::GetClassName($handle, $className, 128)
            $rect = New-Object ImpactIQ.NativeWindows+RECT
            [void][ImpactIQ.NativeWindows]::GetWindowRect($handle, [ref]$rect)
            if ($cloaked -eq 0 -and $className.ToString() -match $ClassPattern -and
                ($rect.Right - $rect.Left) -ge 300 -and ($rect.Bottom - $rect.Top) -ge 300) {
                [void]$found.Add($handle.ToInt64())
            }
        }
        return $true
    }
    [void][ImpactIQ.NativeWindows]::EnumWindows($callback, [IntPtr]::Zero)
    return @($found)
}

function Start-SignInWatch {
    param([string]$ClassPattern = '^ApplicationFrameWindow$')
    $state.SignInBaseline = @{}
    foreach ($handle in (Get-SignInWindowHandles -ClassPattern $ClassPattern)) { $state.SignInBaseline[$handle] = $true }
    $state.SignInRaised = @{}
    $state.SignInWatch = $true
}

function Show-NewSignInWindows {
    param([string]$ClassPattern = '^ApplicationFrameWindow$')
    foreach ($handle in (Get-SignInWindowHandles -ClassPattern $ClassPattern)) {
        if ($state.SignInBaseline.ContainsKey($handle) -or $state.SignInRaised.ContainsKey($handle)) { continue }
        $state.SignInRaised[$handle] = $true
        $pointer = [IntPtr]$handle
        if ([ImpactIQ.NativeWindows]::IsIconic($pointer)) { [void][ImpactIQ.NativeWindows]::ShowWindow($pointer, 9) }
        [void][ImpactIQ.NativeWindows]::SetForegroundWindow($pointer)
    }
}

function Update-SelectionSummary {
    if ($null -eq $state.Request) { return }
    if ($state.Request.Kind -eq 'Choice') {
        $controls.SelectionCount.Text = 'Choose an option, then Continue.'
        $controls.Continue.IsEnabled = $null -ne $controls.ChoiceList.SelectedItem
    }
    else {
        $controls.SelectionCount.Text = '{0} checked / {1} total / {2} visible' -f
            $state.Checked.Count, @($state.Request.Items).Count, $controls.CheckedList.Items.Count
    }
}

function Update-SelectionFilter {
    if ($null -eq $state.Request -or $state.Request.Kind -eq 'Choice') { return }
    $controls.CheckedList.Items.Clear()
    $search = $controls.Search.Text
    foreach ($item in $state.DisplayItems) {
        if (-not [string]::IsNullOrWhiteSpace($search) -and $item.Display -notmatch [regex]::Escape($search)) { continue }
        $check = New-Object Windows.Controls.CheckBox
        $label = New-Object Windows.Controls.TextBlock
        $label.Text = $item.Display
        $label.TextWrapping = 'Wrap'
        $check.Content = $label
        $check.Tag = [string]$item.Id
        $check.Foreground = [Windows.Media.Brushes]::White
        $check.Margin = '3,5,3,5'
        $check.IsChecked = $state.Checked.ContainsKey([string]$item.Id)
        $check.Add_Checked({
            param($sender, $eventArgs)
            $state.Checked[[string]$sender.Tag] = $true
            Update-SelectionSummary
        })
        $check.Add_Unchecked({
            param($sender, $eventArgs)
            $state.Checked.Remove([string]$sender.Tag)
            Update-SelectionSummary
        })
        [void]$controls.CheckedList.Items.Add($check)
    }
    Update-SelectionSummary
}

function Show-GuiSelection {
    param($Request)
    if ($Request.Kind -notin @('Choice', 'Workspaces', 'Reports', 'Models') -or -not $Request.Id) {
        throw 'The worker requested an unsupported selection step.'
    }
    $state.Request = $Request
    $state.LastRequestId = $Request.Id
    $state.Checked = @{}
    $state.DisplayItems = @($Request.Items | Sort-Object Display)
    $controls.SelectionTitle.Text = $Request.Title
    $controls.SelectionDescription.Text = $Request.Description
    $controls.Search.Text = ''
    $controls.IncludeMy.IsChecked = $false
    $controls.IncludeMy.Visibility = if ($Request.Kind -eq 'Workspaces') { 'Visible' } else { 'Collapsed' }
    $choice = $Request.Kind -eq 'Choice'
    $controls.Search.Visibility = if ($choice) { 'Collapsed' } else { 'Visible' }
    $controls.ChoiceList.Visibility = if ($choice) { 'Visible' } else { 'Collapsed' }
    $controls.CheckedList.Visibility = if ($choice) { 'Collapsed' } else { 'Visible' }
    $controls.MultiActions.Visibility = if ($choice) { 'Collapsed' } else { 'Visible' }
    $controls.Back.Content = if ($Request.BackLabel) { $Request.BackLabel } else { 'Cancel' }
    $controls.Continue.IsEnabled = $true
    if ($choice) {
        $controls.ChoiceList.Items.Clear()
        foreach ($item in @($Request.Items)) { [void]$controls.ChoiceList.Items.Add($item) }
        $controls.ChoiceList.SelectedIndex = if ($Request.NoDefault) { -1 } else { 0 }
    }
    else { Update-SelectionFilter }
    Update-SelectionSummary
    $controls.SelectionDeadline.Text = ''
    $controls.SelectionPanel.Visibility = 'Visible'
    [Windows.Controls.Grid]::SetColumn($controls.LogPanel, 1)
    [Windows.Controls.Grid]::SetColumnSpan($controls.LogPanel, 1)
    $controls.Status.Text = 'Your selection is needed'
    $controls.Detail.Text = $Request.Title
    $controls.Progress.IsIndeterminate = $false
    if ($window.WindowState -eq 'Minimized') { $window.WindowState = 'Normal' }
    [void]$window.Activate()
    if ($choice) { [void]$controls.ChoiceList.Focus() } else { [void]$controls.Search.Focus() }
    Invoke-GuiAttention
}

function Submit-GuiSelection {
    param([ValidateSet('OK', 'Cancel', 'Timeout')][string]$Action)
    if ($null -eq $state.Request) { throw 'There is no active selection to submit.' }
    $response = @{
        Id = $state.Request.Id
        Action = $Action
        SelectedIds = @($state.Checked.Keys)
        IncludeMyWorkspace = [bool]$controls.IncludeMy.IsChecked
        Value = if ($state.Request.Kind -eq 'Choice') { $controls.ChoiceList.SelectedItem.Id } else { $null }
    }
    $temporaryPath = Join-Path $state.BridgePath 'response.tmp'
    $response | ConvertTo-Json -Depth 4 | Set-Content -LiteralPath $temporaryPath -Encoding UTF8
    Move-Item -LiteralPath $temporaryPath -Destination (Join-Path $state.BridgePath 'response.json') -Force
    $state.Request = $null
    $controls.SelectionPanel.Visibility = 'Collapsed'
    [Windows.Controls.Grid]::SetColumn($controls.LogPanel, 0)
    [Windows.Controls.Grid]::SetColumnSpan($controls.LogPanel, 2)
    $controls.Status.Text = 'Continuing analysis...'
    $controls.Detail.Text = 'Loading the next step. Watch this window for further selections.'
    $controls.Progress.IsIndeterminate = $true
}

$timer.Add_Tick({
    try {
        if ($null -ne $state.Stopwatch -and $state.Stopwatch.IsRunning) {
            $controls.LiveElapsed.Text = 'Elapsed ' + (Format-Elapsed $state.Stopwatch.Elapsed)
        }
        if ($null -eq $state.Reader -and (Test-Path -LiteralPath $state.LogPath -PathType Leaf)) {
            $stream = [IO.File]::Open($state.LogPath, [IO.FileMode]::Open, [IO.FileAccess]::Read, [IO.FileShare]::ReadWrite)
            $state.Reader = New-Object IO.StreamReader $stream
            $controls.OpenLog.IsEnabled = $true
        }
        if ($null -ne $state.Reader) {
            $text = $state.Reader.ReadToEnd()
            if ($text) {
                if ($text -match '(?i)\[ERROR\]|\[WARN(?:ING)?\]|WARNING:') { $state.Issues = $true }
                $controls.Log.AppendText($text)
                [void]$state.FullLog.Append($text)
                if ($controls.Log.Text.Length -gt 120000) {
                    $controls.Log.Text = $controls.Log.Text.Substring($controls.Log.Text.Length - 100000)
                }
                $controls.Log.ScrollToEnd()
                $latestActivity = $null
                foreach ($line in ($text -split "`r?`n")) {
                    $activity = Get-GuiActivity $line
                    if ($activity) { $latestActivity = $activity }
                }
                if ($latestActivity) { $controls.ActivityLine.Text = $latestActivity }
                $state.WorkspacesDone += [regex]::Matches($text, 'Report & Model metadata extraction process completed').Count
                $state.AppsDone += [regex]::Matches($text, '\[INFO\] Fabric App backup saved:').Count
                foreach ($agentMatch in [regex]::Matches($text, 'Data Agent extraction complete\. Agents found: (\d+)')) { $state.AgentsDone += [int]$agentMatch.Groups[1].Value }
                $newStage = Get-GuiStageIndex -Text $text -Current $state.Stage
                if ($newStage -ne $state.Stage) { $state.Stage = $newStage; Show-GuiStages }
                foreach ($step in [regex]::Matches($text, '\[STEP\] ([^\r\n]+)')) {
                    if ($step.Groups[1].Value -like 'Sign in to Microsoft*') {
                        Start-SignInWatch
                        $controls.Detail.Text = "A Microsoft sign-in window will open and come to the front. If you don't see it, check the taskbar."
                    }
                    elseif ($step.Groups[1].Value -notlike 'Waiting for selection*') { $state.SignInWatch = $false }
                }
                if ($null -eq $state.Request) {
                    $steps = [regex]::Matches($text, '\[STEP\] ([^\r\n]+)')
                    if ($steps.Count -gt 0) {
                        $controls.Status.Text = $steps[$steps.Count - 1].Groups[1].Value
                    }
                    elseif ($text -match '\[INFO\] Fetching (reports|models) for selection') {
                        $controls.Status.Text = "Loading $($Matches[1]) for selection..."
                    }
                }
            }
        }
        # Warnings and errors go only to the script's error log, which stays out of the live activity; it just sets the "notes" flag.
        try {
            if ($state.ErrorLogPath -and (Test-Path -LiteralPath $state.ErrorLogPath -PathType Leaf)) {
                $errorStream = [IO.File]::Open($state.ErrorLogPath, [IO.FileMode]::Open, [IO.FileAccess]::Read, [IO.FileShare]::ReadWrite)
                try {
                    if ($errorStream.Length -lt $state.ErrorLogOffset) { $state.ErrorLogOffset = 0 }
                    if ($errorStream.Length -gt $state.ErrorLogOffset) {
                        [void]$errorStream.Seek($state.ErrorLogOffset, [IO.SeekOrigin]::Begin)
                        $errorReader = New-Object IO.StreamReader($errorStream, [Text.Encoding]::UTF8)
                        $errorText = $errorReader.ReadToEnd()
                        $state.ErrorLogOffset = $errorStream.Position
                        if ($errorText -match '\[(WARN|ERROR)\]') { $state.Issues = $true }
                    }
                }
                finally { $errorStream.Dispose() }
            }
        }
        catch [IO.IOException] { }
        if ($state.SignInWatch -and -not $state.Process.HasExited) { Show-NewSignInWindows }
        $requestPath = Join-Path $state.BridgePath 'request.json'
        $responsePath = Join-Path $state.BridgePath 'response.json'
        if (-not $state.Process.HasExited -and $null -eq $state.Request -and
            -not (Test-Path -LiteralPath $responsePath) -and (Test-Path -LiteralPath $requestPath -PathType Leaf)) {
            $request = Get-Content -LiteralPath $requestPath -Raw | ConvertFrom-Json
            if ($request.Id -ne $state.LastRequestId) { Show-GuiSelection $request }
        }
        if ($null -ne $state.Request -and $state.Request.ExpiresUtc) {
            $remaining = [Math]::Max(0, [Math]::Ceiling(([DateTime]::Parse($state.Request.ExpiresUtc).ToUniversalTime() - [DateTime]::UtcNow).TotalSeconds))
            $defaultDescription = if ($state.Request.Kind -eq 'Workspaces') { 'includes My Workspace and retains checks' }
                elseif ($state.Request.Kind -eq 'Choice') { 'defaults to Public' } else { 'uses checked items' }
            $controls.SelectionDeadline.Text = "Auto-continue in ${remaining}s ($defaultDescription)."
            if ($remaining -eq 0) { Submit-GuiSelection -Action Timeout }
        }
        if ($state.Process.HasExited) {
            $timer.Stop()
            $state.SignInWatch = $false
            if ($null -ne $state.Reader) {
                $finalText = $state.Reader.ReadToEnd()
                if ($finalText) { $controls.Log.AppendText($finalText); [void]$state.FullLog.Append($finalText); $controls.Log.ScrollToEnd() }
                $state.Reader.Dispose(); $state.Reader = $null
            }
            $elapsedText = ''
            if ($null -ne $state.Stopwatch) { $state.Stopwatch.Stop(); $elapsedText = Format-Elapsed $state.Stopwatch.Elapsed }
            $elapsedSuffix = if ($elapsedText) { " Elapsed: $elapsedText." } else { '' }
            $controls.Progress.IsIndeterminate = $false
            $controls.Run.IsEnabled = $true
            $controls.Browse.IsEnabled = $true
            $controls.Folder.IsEnabled = $true
            $controls.Setup.Visibility = 'Visible'
            $controls.Intro.Visibility = 'Visible'
            $controls.Eyebrow.Visibility = if ($logoLoaded) { 'Collapsed' } else { 'Visible' }
            $controls.Header.Margin = '0,0,0,24'
            $controls.SelectionPanel.Visibility = 'Collapsed'
            $state.Request = $null
            [Windows.Controls.Grid]::SetColumn($controls.LogPanel, 0)
            [Windows.Controls.Grid]::SetColumnSpan($controls.LogPanel, 2)
            $controls.StopRun.Visibility = 'Collapsed'
            $endedWell = $false
            if ($state.StoppedByUser) {
                $controls.Status.Text = 'Analysis stopped'
                $controls.Detail.Text = 'You stopped the analysis. Backups already written are kept.' + $elapsedSuffix
            }
            elseif (Test-Path -LiteralPath $state.ResultPath -PathType Leaf) {
                $result = Get-Content -LiteralPath $state.ResultPath -Raw | ConvertFrom-Json
                if ($result.State -in @('Finished', 'FinishedWithIssues') -and $state.Process.ExitCode -eq 0) {
                    Show-GuiFinished -HadIssues ($result.State -eq 'FinishedWithIssues' -or $state.Issues) -Elapsed $elapsedText `
                        -Summary (Get-GuiRunSummary -Folder $state.Folder -Workspaces $state.WorkspacesDone -Apps $state.AppsDone -Agents $state.AgentsDone)
                    $endedWell = $true
                }
                else {
                    $controls.Status.Text = if ($result.State -eq 'Failed' -or $state.Process.ExitCode -ne 0) { 'Analysis failed' }
                        elseif ($result.State -eq 'Cancelled') { 'Analysis cancelled' }
                        else { 'Analysis stopped before completion' }
                    $controls.Detail.Text = $result.Message + $elapsedSuffix
                }
            }
            elseif ($state.Process.ExitCode -ne 0) {
                $controls.Status.Text = 'Analysis failed'
                $controls.Detail.Text = "Sign-in or setup did not complete. Review the live activity, 'View run detail', and 'Config\Error Log.txt' in your backup folder." + $elapsedSuffix
            }
            else {
                $controls.Status.Text = 'Analysis stopped before completion'
                $controls.Detail.Text = 'The run was cancelled, blocked, or exited early. Review the live activity and "View run detail"; no completion was confirmed.' + $elapsedSuffix
            }
            if (-not $endedWell) { Set-GuiDetails $true; Invoke-GuiAttention }
            if ($state.BridgePath -and (Test-Path -LiteralPath $state.BridgePath -PathType Container)) {
                Remove-Item -LiteralPath $state.BridgePath -Recurse -Force -ErrorAction SilentlyContinue
            }
            $state.Process.Dispose()
            $state.Process = $null
            if ($state.RunFolder -and (Test-Path -LiteralPath $state.RunFolder -PathType Container)) {
                Remove-Item -LiteralPath $state.RunFolder -Recurse -Force -ErrorAction SilentlyContinue
            }
        }
    }
    catch {
        $timer.Stop()
        if ($null -ne $state.Reader) { $state.Reader.Dispose(); $state.Reader = $null }
        Show-LauncherError ("Live monitoring failed. The analysis may still be running in the background. " + $_.Exception.Message)
    }
})
# Stop the running analysis (and the tools it started) after confirming.
function Stop-GuiRun {
    param([switch]$Confirmed)
    if ($null -eq $state.Process -or $state.Process.HasExited) { return }
    if (-not $Confirmed) {
        $answer = [Windows.MessageBox]::Show($window, 'Stop the analysis? Backups already written are kept.', 'Impact IQ', 'YesNo', 'Question', 'No')
        if ($answer -ne 'Yes') { return }
    }
    $state.StoppedByUser = $true
    $controls.StopRun.IsEnabled = $false
    $controls.Status.Text = 'Stopping...'
    & (Join-Path $env:SystemRoot 'System32\taskkill.exe') /PID $state.Process.Id /T /F 2>&1 | Out-Null
}
$controls.StopRun.Add_Click({ Stop-GuiRun })
$controls.DetailsToggle.Add_Click({ Set-GuiDetails ($controls.Log.Visibility -ne 'Visible') })

# Keyboard and mouse shortcuts on the selection screens: Enter continues, Esc goes back, double-click picks a choice.
$window.Add_PreviewKeyDown({
    param($sender, $keyArgs)
    if ($null -eq $state.Request) { return }
    if ($keyArgs.Key -eq 'Return' -and $controls.Continue.IsEnabled -and $keyArgs.OriginalSource -isnot [Windows.Controls.Button]) {
        $keyArgs.Handled = $true
        $controls.Continue.RaiseEvent((New-Object Windows.RoutedEventArgs ([Windows.Controls.Button]::ClickEvent)))
    }
    elseif ($keyArgs.Key -eq 'Escape') {
        $keyArgs.Handled = $true
        $controls.Back.RaiseEvent((New-Object Windows.RoutedEventArgs ([Windows.Controls.Button]::ClickEvent)))
    }
})
$controls.ChoiceList.Add_PreviewMouseDoubleClick({
    param($sender, $clickArgs)
    $selected = $controls.ChoiceList.SelectedItem
    $target = $clickArgs.OriginalSource
    if ($target -is [Windows.FrameworkElement] -and $null -ne $selected -and $target.DataContext.Id -eq $selected.Id -and $controls.Continue.IsEnabled) {
        $controls.Continue.RaiseEvent((New-Object Windows.RoutedEventArgs ([Windows.Controls.Button]::ClickEvent)))
    }
})

        $dark = 1

        $captionColor = 0x00271810
        [void][ImpactIQ.NativeWindows]::DwmSetWindowAttribute($handle, 35, [ref]$captionColor, 4)

})

$window.Add_Closing({
    param($sender, $eventArgs)
    if ($null -ne $state.Process -and -not $state.Process.HasExited) {
        $answer = [Windows.MessageBox]::Show($window,
            'Analysis is still running. Stop it and close Impact IQ? Backups already written are kept.',
            'Impact IQ', 'YesNo', 'Question', 'No')
        if ($answer -ne 'Yes') { $eventArgs.Cancel = $true; return }
        & (Join-Path $env:SystemRoot 'System32\taskkill.exe') /PID $state.Process.Id /T /F 2>&1 | Out-Null
        [void]$state.Process.WaitForExit(5000)
    }
    $timer.Stop()
    if ($null -ne $state.Reader) { $state.Reader.Dispose() }
    if ($null -ne $state.Process) { $state.Process.Dispose() }
    if ($state.RunFolder -and (Test-Path -LiteralPath $state.RunFolder -PathType Container)) {
        Remove-Item -LiteralPath $state.RunFolder -Recurse -Force -ErrorAction SilentlyContinue
    }
})
# A second copy would run against the same backup folder, so bring the open window forward and stop.
if ($env:IMPACTIQ_ALLOW_MULTIPLE -ne '1' -and -not (Enter-GuiSingleInstance)) {
    foreach ($other in [Diagnostics.Process]::GetProcessesByName('powershell')) {
        try {
            if ($other.Id -ne $PID -and $other.MainWindowTitle -eq 'Impact IQ' -and $other.MainWindowHandle -ne [IntPtr]::Zero) {
                if ([ImpactIQ.NativeWindows]::IsIconic($other.MainWindowHandle)) { [void][ImpactIQ.NativeWindows]::ShowWindow($other.MainWindowHandle, 9) }
                [void][ImpactIQ.NativeWindows]::SetForegroundWindow($other.MainWindowHandle)
            }
        }
        catch { }
    }
    $window.Close()
    return
}
